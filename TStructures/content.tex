\section{Bounded derived category of constructible complexes of \texorpdfstring{$\ell$}{ell}-adic sheaves}

\TODO{}
From here, let $X$ be a scheme over a field $k$ of characteristic $p \geq 0$ and let $\ell \neq p$ be a prime number. There is a derived category $\Dbc(X) = \Dbc(X,\Qellbar)$ of bounded complexes of constructible sheaves with $\Qellbar$-coefficients. There is an associated six-functor formalism:
\begin{enumerate}
    \item There are bifunctors $\Dbc(X) \to \Dbc(X) \to \Dbc(X)$ written
    $$\hlin{(M,N) \mapsto \RHom(M,N), \quad (M,N) \mapsto M \otimes N.}$$
    \item Given a $k$-morphism $f: X \to Y$, there are functors 
    $$\hlin{f^*: \Dbc(Y) \to \Dbc(X)}$$
    and 
    $$\hlin{Rf_*: \Dbc(X) \to \Dbc(Y)}.$$
    \item If $f$ is $k$-compactifiable (e.g. by virtue of $Y$ being quasi-compact and quasi-separated and $f$ being separated and of finite type), then there are functors \hl{$Rf_!: \Dbc(X) \to \Dbc(Y)$} and \hl{$f^!: \Dbc(Y) \to \Dbc(X)$}.
\end{enumerate}

These satisfy the following adjunctions (the last adjunction requires $f$ to be $k$-compactifiable for $Rf_!$ to be defined):
\begin{align*}
    \Hom(M \otimes N, P) &\cong \Hom(R, \RHom(N,P)) \\
    \Hom(f^* M, N) &\cong \Hom(M, Rf_* N) \\
    \Hom(Rf_! M, N) &\cong \Hom(M, f^! N). \\
\end{align*}


For any object $K \in \Dbc(X)$, there are the cohomology functors $K \mapsto \mathscr{H}^i(K)$. Note that each $\mathscr{H}^i(K)$ is a constructible sheaf with $\Qellbar$-coefficients.

\section{Exactness properties for the perverse \texorpdfstring{$t$}{t}-structure under direct images and inverse images}

We discuss some conditions under which the derived functors $Rf_*, Rf_!, f^*, f^!$ (or variants thereof) preserve perversity.

\input{../_definitions/definition_exact_functor_between_triangulated_categories.tex}

\input{../_definitions/definition_t_left_exact_right_exact_functors_between_triangulated_categories_with_t_structures.tex}

\input{../_concepts/lemma_derived_pushforward_of_affine_preserves_semiperversity_and_derived_proper_pushforward_of_quasi_finite_perverse_semiperversity.tex}

\begin{proposition}
    Let $f: X \to Y $ be a morphism of finite type schemes over a field $k$ of characteristic distinct  from $\ell$. 
\begin{enumerate}
    \item If $f$ is quasi-finite and affine, then $R f_* M, Rf_! M \in \Perv(Y)$ for all $M \in \Perv(X)$ {\cite[Corollary III.6.2]{kiehl_weissauer_wcps}}. 
    \item If $f$ is equidimensional of relative dimension $d$ and smooth, then 
    $$f^{*}[d](M) = f^{!}[-d](-d)(M) \in \Perv(X)$$
    for all $M \in \Perv(Y)$ {\cite[Theorem III.7.2]{kiehl_weissauer_wcps}}. 
\end{enumerate}
\end{proposition}
