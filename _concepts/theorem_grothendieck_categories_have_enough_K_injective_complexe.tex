\begin{theorem}[{\cite[Theorem 3.13.]{serpe_rucgc}}] \label{theorem:grothendieck_categories_have_enough_K_injective_complexe}
    Let $\calA$ be a \CrefAndHyperrefIfExist{definition:grothendiecks_additional_axioms_for_abelian_categories}{Grothendieck category}. 
    
    For every \CrefAndHyperrefIfExist{definition:chain_complex_of_objects_in_an_additive_category}{chain complex} $A^\bullet$ in $\calA$, there is a \CrefAndHyperrefIfExist{definition:left_right_resolution_of_a_chain_complex_in_an_abelian_category}{right resolution} $A^\bullet \to I^\bullet$  such that  $I^\bullet$ is \CrefAndHyperrefIfExist{definition:K_injective_projective_complex_in_an_abelian_category}{$K$-injective}.
\end{theorem}
