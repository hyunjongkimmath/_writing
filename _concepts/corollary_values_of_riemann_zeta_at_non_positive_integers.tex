\begin{corollary}[Values of the Riemann zeta function at non-positive integers] \label{corollary:values_of_riemann_zeta_at_non_positive_integers}

For every $n \in \bbN_0$, the \CrefAndHyperrefIfExist{definition:riemann_zeta_function}{Riemann zeta function} takes the rational value
$$
\boxed{ \zeta(-n) = -\frac{B_{n+1}}{n+1}, }
$$
where $B_{n+1}$ are the \CrefAndHyperrefIfExist{definition:bernoulli_numbers}{Bernoulli numbers} (with $B_1 = -1/2$, $B_{2k+1} = 0$ for $k \ge 1$).

In particular, $\zeta(0) = -1/2$, $\zeta(-1) = -1/12$, and $\zeta(-2n) = 0$ for $n \ge 1$ (the trivial zeros, \Cref{corollary:trivial_zeros_of_riemann_zeta_function}).
\end{corollary}

\begin{proof}

    We use the functional equation from \Cref{corollary:analytic_continuation_and_functional_equation_for_riemann_zeta_function}:
    $$\zeta(s) = 2^s \pi^{s-1} \sin\!\left(\frac{\pi s}{2}\right) \Gamma(1-s)\, \zeta(1-s).$$

    Setting $s = -n$ for $n \in \bbN_0$, we obtain
    $$\zeta(-n) = 2^{-n} \pi^{-n-1} \sin\!\left(\frac{-\pi n}{2}\right) \Gamma(1+n)\, \zeta(1+n).$$

    Since $\Gamma(1+n) = n!$, we analyze by parity of $n$.

    \textbf{Case 1: $n$ is even.} Write $n = 2m$ for some $m \in \bbN_0$. Then $\sin(-\pi m) = 0$. For $m > 0$ (i.e., $n \geq 2$), the factor $\zeta(1+n) = \zeta(2m+1)$ is finite and nonzero, so $\zeta(-2m) = 0$. For $m = 0$ (i.e., $n = 0$), we have $\sin(0) = 0$ but $\Gamma(1)\,\zeta(1)$ has a pole. Instead evaluate directly: the functional equation near $s=0$ gives
    $$\zeta(s) = 2^s \pi^{s-1} \left(\frac{-\pi s}{2} + O(s^3)\right) \Gamma(1-s)\, \zeta(1-s).$$
    As $s \to 0$, we have $\Gamma(1-s) \to 1$ and $(1-s)\zeta(1-s) \to -1$. Hence
    $$\lim_{s \to 0} \frac{\zeta(s)}{s} = 2^0 \pi^{-1} \cdot \left(-\frac{\pi}{2}\right) \cdot 1 \cdot (-1) = \frac{1}{2},$$
    so $\zeta(0) = -B_1/1 = -(-1/2)/1 = -1/2$, as claimed.

    \textbf{Case 2: $n$ is odd.} Write $n = 2m-1$ for some $m \geq 1$. Then $\sin\left(\frac{-\pi(2m-1)}{2}\right) = (-1)^{m+1}$. We have
    $$\zeta(-(2m-1)) = 2^{-(2m-1)} \pi^{-2m} (-1)^{m+1} (2m-1)! \, \zeta(2m).$$

    By the formula for $\zeta$ at even positive integers\CrefIfExists{corollary:values_of_riemann_zeta_at_even_positive_integers},
    $$\zeta(2m) = \frac{(-1)^{m+1}(2\pi)^{2m}}{2(2m)!} B_{2m}.$$

    Substituting this in:
    $$\zeta(-(2m-1)) = 2^{-(2m-1)} \pi^{-2m} (-1)^{m+1} (2m-1)! \cdot \frac{(-1)^{m+1}(2\pi)^{2m}}{2(2m)!} B_{2m}.$$

    Simplifying: $(2m-1)!/(2m)! = 1/(2m)$, and $2^{-(2m-1)} \cdot 2^{2m}/2 = 1$. Also $\pi^{-2m} \cdot \pi^{2m} = 1$ and $(-1)^{m+1} \cdot (-1)^{m+1} = 1$. Hence
    $$\zeta(-(2m-1)) = \frac{B_{2m}}{2m}.$$

    Since $n = 2m-1$, we have $n+1 = 2m$ and $B_{n+1} = B_{2m}$, so $\zeta(-n) = -\frac{-B_{n+1}}{n+1}$. We check the sign: our expression gives $\zeta(-(2m-1)) = \frac{B_{2m}}{2m} = -\frac{B_{n+1}}{n+1}$ only if $B_{2m}/(2m) = -B_{n+1}/(n+1)$, which holds since $n+1 = 2m$. Thus $\zeta(-n) = -B_{n+1}/(n+1)$ in all cases.
\end{proof}
