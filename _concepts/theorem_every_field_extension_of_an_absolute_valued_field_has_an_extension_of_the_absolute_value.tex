\begin{theorem} \label{theorem:every_field_extension_of_an_absolute_valued_field_has_an_extension_of_the_absolute_value}
    Let $(K, |\cdot|_K)$ be an an \CrefAndHyperrefIfExist{definition:absolute_valued_field}{absolute valued field}. Let $L$ be any \CrefAndHyperrefIfExist{definition:extension_of_a_field}{field extension} of $K$.

    There exists an absolute value on $L$ \CrefAndHyperrefIfExist{definition:extension_of_absolute_valued_fields}{extending} the absolute value on $K$. If $L$ is \CrefAndHyperrefIfExist{definition:algebraic_element_minimal_polynomial_of_an_algebraic_element_algebraic_field_extension_transcendental_element_field_extension}{algebraic} over $K$, then this extension is unique.
\end{theorem}
