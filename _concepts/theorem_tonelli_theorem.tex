\begin{theorem}[Tonelli's theorem (nonnegative case)] \label{theorem:tonelli_theorem}
Let $(X, \mathcal{A}, \mu)$ and $(Y, \mathcal{B}, \nu)$ be two \CrefAndHyperrefIfExist{definition:finite_sigma_finite_measure_space_for_a_measure_on_a_measurable_space}{$\sigma$-finite measure spaces}. Let $\mathcal{A} \otimes \mathcal{B}$ be the \CrefAndHyperrefIfExist{definition:product_sigma_algebra_and_product_measure_for_two_measure_spaces}{product $\sigma$-algebra} and $\mu \times \nu$ the corresponding \CrefAndHyperrefIfExist{definition:product_sigma_algebra_and_product_measure_for_two_measure_spaces}{product measure}.

Let $f : X \times Y \to [0,\infty]$ be a \CrefAndHyperrefIfExist{definition:measurable_function_between_measure_spaces}{measurable function} with respect to $\mathcal{A} \otimes \mathcal{B}$. Then
$$\int_{X \times Y} f\,d(\mu \times \nu) = \int_X \left(\int_Y f(x,y)\,d\nu(y)\right) d\mu(x) = \int_Y \left(\int_X f(x,y)\,d\mu(x)\right) d\nu(y),$$
\Cref{definition:lebesgue_integral_of_measurable_functions_into_the_real_line_extended_by_infinities}
where all three integrals may be $+\infty$. In particular, $f$ is $(\mu \times \nu)$-integrable if and only if either iterated integral of $f$ is finite.
\end{theorem}

We show that the three integrals coincide for every nonnegative $\mathcal{A} \otimes \mathcal{B}$-measurable function $f : X \times Y \to [0,\infty]$. The proof proceeds in four stages: first for indicator functions of measurable rectangles, then for indicator functions of arbitrary measurable sets in $\mathcal{A} \otimes \mathcal{B}$, next for simple functions, and finally for general nonnegative measurable functions via approximation by an increasing sequence of simple functions.

\begin{enumerate}
    \item Let $E \in \mathcal{A}$ and $F \in \mathcal{B}$. We show that the equality holds for the indicator function $\mathbf{1}_{E \times F} : X \times Y \to [0,\infty]$. The integral of $\mathbf{1}_{E \times F}$ with respect to the product measure $\mu \times \nu$ equals
    $$\int_{X \times Y} \mathbf{1}_{E \times F}\,d(\mu \times \nu) = (\mu \times \nu)(E \times F) = \mu(E)\nu(F),$$
    \Cref{definition:product_sigma_algebra_and_product_measure_for_two_measure_spaces}
    by the definition of the product measure on measurable rectangles. For the inner iterated integral, we have $\mathbf{1}_{E \times F}(x,y) = \mathbf{1}_E(x)\mathbf{1}_F(y)$ for all $(x,y) \in X \times Y$, so
    $$\int_Y \mathbf{1}_{E \times F}(x,y)\,d\nu(y) = \mathbf{1}_E(x) \int_Y \mathbf{1}_F\,d\nu = \mathbf{1}_E(x)\nu(F).$$
    Integrating with respect to $\mu$ yields
    $$\int_X \left(\int_Y \mathbf{1}_{E \times F}(x,y)\,d\nu(y)\right) d\mu(x) = \int_X \mathbf{1}_E(x)\nu(F)\,d\mu(x) = \nu(F)\mu(E).$$
    By symmetry, the iterated integral in the reverse order also equals $\mu(E)\nu(F)$. Hence all three integrals coincide for indicator functions of measurable rectangles.

    \item We extend the equality to arbitrary sets in $\mathcal{A} \otimes \mathcal{B}$ using a monotone class argument. Define
    $$\Lambda = \left\{ S \in \mathcal{A} \otimes \mathcal{B} : \int_{X \times Y} \mathbf{1}_S\,d(\mu \times \nu) = \int_X \left(\int_Y \mathbf{1}_S(x,y)\,d\nu(y)\right) d\mu(x) \right\}.$$
    We show that $\Lambda$ is a monotone class. Suppose that $S_1 \subseteq S_2 \subseteq \cdots$ are sets in $\Lambda$. Then $\mathbf{1}_{S_n}$ increases pointwise to $\mathbf{1}_{\bigcup_{n=1}^\infty S_n}$. By the monotone convergence theorem \Cref{theorem:monotone_convergence_theorem_for_the_limit_of_lebesgue_integrals_of_an_increasing_sesquence_of_functions_and_the_lebesgue_integral_of_the_limit_of_the_functions}, we may pass to the limit under each integral, so $\bigcup_{n=1}^\infty S_n \in \Lambda$. Similarly, if $S_1 \supseteq S_2 \supseteq \cdots$ are sets in $\Lambda$, then $\mathbf{1}_{S_n}$ decreases pointwise to $\mathbf{1}_{\bigcap_{n=1}^\infty S_n}$, and the same limit passage shows $\bigcap_{n=1}^\infty S_n \in \Lambda$ because at least one of the integrals is finite for indicator functions of subsets of $X \times Y$ when the measure spaces are $\sigma$-finite. The collection of measurable rectangles $\mathcal{R} = \{E \times F : E \in \mathcal{A}, F \in \mathcal{B}\}$ is a $\pi$-system that generates $\mathcal{A} \otimes \mathcal{B}$, and $\mathcal{R} \subseteq \Lambda$ by the previous step. By the monotone class lemma for sets \TODO{AI generated, reference needed: monotone class lemma stating that a monotone class containing a pi-system contains the sigma-algebra generated by that pi-system}, $\Lambda$ contains $\sigma(\mathcal{R}) = \mathcal{A} \otimes \mathcal{B}$. Hence the equality holds for all indicator functions $\mathbf{1}_S$ with $S \in \mathcal{A} \otimes \mathcal{B}$.

    \item Let $f : X \times Y \to [0,\infty]$ be a nonnegative \CrefAndHyperrefIfExist{definition:simple_function_on_a_measure_space}{simple function} with respect to $\mathcal{A} \otimes \mathcal{B}$, so $f = \sum_{i=1}^n c_i \mathbf{1}_{S_i}$ for some $c_i \geq 0$ and disjoint sets $S_i \in \mathcal{A} \otimes \mathcal{B}$. By linearity of the Lebesgue integral \Cref{theorem:linearity_of_lebesgue_integral}, we have
    $$\int_{X \times Y} f\,d(\mu \times \nu) = \sum_{i=1}^n c_i \int_{X \times Y} \mathbf{1}_{S_i}\,d(\mu \times \nu).$$
    Similarly, for each $x \in X$, the function $y \mapsto f(x,y)$ is a simple function on $Y$, and
    $$\int_Y f(x,y)\,d\nu(y) = \sum_{i=1}^n c_i \mathbf{1}_{S_i}(x,y),$$
    so integrating with respect to $\mu$ gives
    $$\int_X \left(\int_Y f(x,y)\,d\nu(y)\right) d\mu(x) = \sum_{i=1}^n c_i \int_X \left(\int_Y \mathbf{1}_{S_i}(x,y)\,d\nu(y)\right) d\mu(x).$$
    Since the equality holds for each $\mathbf{1}_{S_i}$ by step 2, it follows that all three integrals coincide for $f$.

    \item Let $f : X \times Y \to [0,\infty]$ be an arbitrary nonnegative $\mathcal{A} \otimes \mathcal{B}$-measurable function. There exists an increasing sequence of nonnegative simple functions $(s_n)_{n=1}^\infty$ such that $s_n(x,y)$ increases to $f(x,y)$ for all $(x,y) \in X \times Y$. For each $n$ and each $x \in X$, the function $y \mapsto s_n(x,y)$ is a nonnegative simple function on $Y$ increasing to $y \mapsto f(x,y)$. By the monotone convergence theorem applied to the inner integral, $\int_Y s_n(x,y)\,d\nu(y)$ increases to $\int_Y f(x,y)\,d\nu(y)$ for each $x$. Applying the monotone convergence theorem again to the outer integral, we obtain
    $$\lim_{n \to \infty} \int_X \left(\int_Y s_n(x,y)\,d\nu(y)\right) d\mu(x) = \int_X \left(\int_Y f(x,y)\,d\nu(y)\right) d\mu(x).$$
    Similarly, applying the monotone convergence theorem to the integral over $X \times Y$, we have
    $$\lim_{n \to \infty} \int_{X \times Y} s_n\,d(\mu \times \nu) = \int_{X \times Y} f\,d(\mu \times \nu).$$
    Since the equality holds for each $s_n$ by step 3, it follows that
    $$\int_{X \times Y} f\,d(\mu \times \nu) = \int_X \left(\int_Y f(x,y)\,d\nu(y)\right) d\mu(x).$$
    By symmetry of the product measure with respect to $X$ and $Y$, the same argument shows that this common value also equals $\int_Y \left(\int_X f(x,y)\,d\mu(x)\right) d\nu(y)$.

The final conclusion follows: if $f$ takes only finite values almost everywhere and any one of the three integrals is finite, then all three are finite, and $f$ is $(\mu \times \nu)$-integrable if and only if either iterated integral is finite.
\end{enumerate}
\end{proof}
