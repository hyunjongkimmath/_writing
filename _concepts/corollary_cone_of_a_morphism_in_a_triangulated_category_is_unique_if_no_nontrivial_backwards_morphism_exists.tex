\begin{corollary}[{\cite[Corollary 1.1.10.]{beilinson_berstein_deligne_atlesI}}] \label{corollary:cone_of_a_morphism_in_a_triangulated_category_is_unique_if_no_nontrivial_backwards_morphism_exists} 
Let $\mathcal{T}$ be a \CrefAndHyperrefIfExist{definition:triangulated_category}{triangulated category}. Let $X \xrightarrow{u} Y \xrightarrow{v} Z \xrightarrow{d} X[1]$ be a \CrefAndHyperrefIfExist{definition:triangulated_category}{distinguished triangle} in $\mathcal{T}$. If $\text{Hom}_{\mathcal{T}}(X, Z[-1]) = 0$, then:
\begin{itemize}
    \item[(i)] the \CrefAndHyperrefIfExist{definition:cone_of_a_morphism_in_a_triangulated_category}{cone of $u$} is unique up to unique isomorphism;
    \item[(ii)] $d$ is the unique morphism $x: Z \to X[1]$ such that the triangle $X \xrightarrow{u} Y \xrightarrow{v} Z \xrightarrow{x} X[1]$ is distinguished.
\end{itemize}
\end{corollary}
