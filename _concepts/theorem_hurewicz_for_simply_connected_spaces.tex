\begin{theorem}[Hurewicz, see e.g. {\cite[Theorem 2.14]{hatcher}}] \label{theorem:hurewicz}
Let $X$ be a path-connected \CrefAndHyperrefIfExist{definition:topological_space}{topological space} that is \CrefAndHyperrefIfExist{definition:n_connected_topological_space}{$(n-1)$-connected} for some $n \geq 2$. Then the \hldef{$n$th Hurewicz homomorphism}
\[
\psi: \pi_n(X,x_0) \to H_n(X;\ZZ)
\]
is an isomorphism, where $\psi([\alpha]) = \alpha_*([id_{\sphere{n}}])$ sends the class of a sphere map to its image on the fundamental class. Moreover, if $n = 1$ and $X$ is path-connected, then $\pi_1(X,x_0)^{\mathrm{ab}} \cong H_1(X;\ZZ)$.
\end{theorem}

\begin{corollary} \label{corollary:hurewicz_for_spheres_and_products}
For the $n$-dimensional sphere $S^n$, the Hurewicz homomorphism $\pi_n(S^n,x_0) \to H_n(S^n;\ZZ)$ is an isomorphism sending the identity class to a generator of $H_n(S^n;\ZZ) \cong \ZZ$. Consequently, $\pi_n(S^n) \cong \ZZ$ for all $n \geq 1$.
\end{corollary}
