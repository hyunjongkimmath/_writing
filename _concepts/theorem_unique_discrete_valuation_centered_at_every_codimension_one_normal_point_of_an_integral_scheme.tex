\begin{theorem} \label{theorem:unique_discrete_valuation_centered_at_every_codimension_one_normal_point_of_an_integral_scheme}
    Let $X$ be an \CrefAndHyperrefIfExist{definition:integral_scheme}{integral scheme}.
    If $x$ is a \CrefAndHyperrefIfExist{definition:normal_and_factorial_scheme}{normal} pointof $X$ of \CrefAndHyperrefIfExist{definition:codimension_of_an_irreducible_closed_subset_in_a_topological_space}{codimension} $1$, then there is a unique \CrefAndHyperrefIfExist{definition:discrete_valuation_on_a_field}{discrete valuation} $v$ \CrefAndHyperrefIfExist{definition:centered_at_for_a_prime_of_an_integral_domain_point_of_an_integral_scheme}{centered at} $x$.
\end{theorem}
