\begin{proposition}[See {\cite[Tag 03E7]{stacks-project}}]  \label{proposition:list_of_some_properties_of_schemes_that_are_local_for_the_etale_topology_over_a_scheme}
    Let $S$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme}.
    The following properties are properties of schemes that are \CrefAndHyperrefIfExist{definition:stable_local_class_of_objects_property_of_objects_on_a_locally_small_site_whose_representable_presheaves_are_sheaves}{local} for the \CrefAndHyperrefIfExist{definition:big_etale_site_of_a_scheme}{big \'etale topology} on \CrefAndHyperrefIfExist{definition:scheme_over_a_scheme}{$\mathit{Sch}/S$}:
    \begin{enumerate}
        \item \CrefAndHyperrefIfExist{definition:locally_noetherian_and_noetherian_scheme}{locally Noetherian}
        \item Jacobson
        \item Cohen-Macaulay
        \item Gorenstein
        \item reduced
        \item normal
        \item regular
        \item Nagata
    \end{enumerate}
\end{proposition}
