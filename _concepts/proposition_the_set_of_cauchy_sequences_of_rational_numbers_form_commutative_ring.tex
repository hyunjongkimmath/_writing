\begin{proposition} \label{proposition:the_set_of_cauchy_sequences_of_rational_numbers_form_commutative_ring}
    \TODO{I feel like a statement like this could be stated more generally for cauchy sequences on an extended metric space}
    Let $\mathcal{C}_\bbQ$ be the set of \CrefAndHyperrefIfExist{definition:cauchy_sequence_in_rational_numbers}{Cauchy sequences in $\bbQ$}. Define addition and multiplication on $\mathcal{C}_\bbQ$ termwise:
    $$(a_n) + (b_n) = (a_n + b_n), \qquad (a_n) \cdot (b_n) = (a_n b_n).$$
    Then $(\mathcal{C}_\bbQ, +, \cdot)$ is a \CrefAndHyperrefIfExist{definition:commutative_ring}{commutative ring with unity}, where:
    \begin{enumerate}
        \item The zero element is the constant sequence $(0)$.
        \item The unity is the constant sequence $(1)$.
        \item The additive inverse is given termwise: $-(a_n) = (-a_n)$.
    \end{enumerate}
\end{proposition}

\begin{proof}
    \TODO{AI generated, verify}
    We first verify that $\mathcal{C}_\bbQ$ is closed under the termwise operations. Let $(a_n), (b_n) \in \mathcal{C}_\bbQ$.

    \begin{itemize}
        \item \textbf{Closure under addition.} Given $\varepsilon \in \bbQ_{>0}$, choose $N_a, N_b \in \bbN$ such that for all $m, n \ge N_a$ we have $|a_m - a_n| < \varepsilon/2$, and for all $m, n \ge N_b$ we have $|b_m - b_n| < \varepsilon/2$. Let $N = \max\{N_a, N_b\}$. For any $m, n \ge N$, the triangle inequality (\Cref{definition:absolute_value_on_rational_numbers}) gives
        $$|(a_m + b_m) - (a_n + b_n)| \le |a_m - a_n| + |b_m - b_n| < \varepsilon/2 + \varepsilon/2 = \varepsilon.$$
        Hence $(a_n + b_n)$ is a Cauchy sequence.

        \item \textbf{Closure under multiplication.} A Cauchy sequence in $\bbQ$ is bounded: choose $N_0$ such that $|a_m - a_n| < 1$ for all $m, n \ge N_0$; then for all $n \ge N_0$ we have $|a_n| \le |a_{N_0}| + 1$ by the triangle inequality, and taking $M_a = \max\{|a_1|, \dots, |a_{N_0-1}|, |a_{N_0}| + 1\}$ gives $|a_n| \le M_a$ for all $n$. Similarly, let $M_b$ be a bound for $(b_n)$.

        Given $\varepsilon \in \bbQ_{>0}$, choose $N_a, N_b \in \bbN$ such that $|a_m - a_n| < \varepsilon/(2M_b + 1)$ for all $m, n \ge N_a$ and $|b_m - b_n| < \varepsilon/(2M_a + 1)$ for all $m, n \ge N_b$. Let $N = \max\{N_a, N_b\}$. For any $m, n \ge N$,
        \begin{align*}
            |a_m b_m - a_n b_n|
            &\le |a_m b_m - a_m b_n| + |a_m b_n - a_n b_n| \\
            &\le |a_m|\,|b_m - b_n| + |b_n|\,|a_m - a_n| \\
            &\le M_a\cdot\frac{\varepsilon}{2M_a + 1} + M_b\cdot\frac{\varepsilon}{2M_b + 1} \\
            &< \varepsilon/2 + \varepsilon/2 = \varepsilon,
        \end{align*}
        where we have used the multiplicativity and triangle inequality of $|\,\cdot\,|$ (\Cref{definition:absolute_value_on_rational_numbers}). Hence $(a_n b_n)$ is a Cauchy sequence.
    \end{itemize}

    Now we verify the ring axioms. The set $\mathcal{C}_\bbQ$ is a subset of $\bbQ^\bbN$ (the set of all sequences of rational numbers), and $\bbQ^\bbN$ is a commutative ring under termwise addition and multiplication because $\bbQ$ is a field (\CrefIfExists{definition:field_of_rational_numbers}). Since $\mathcal{C}_\bbQ$ is closed under addition, multiplication, additive inverses (if $(a_n)$ is Cauchy then $(-a_n)$ is Cauchy with the same modulus of Cauchyness), and contains the constant sequences $(0)$ and $(1)$, it is a \CrefIfExists{definition:subring_of_a_ring}{subring} of $\bbQ^\bbN$ and hence inherits the commutative ring structure. The commutativity of multiplication follows from commutativity in $\bbQ$.

    Concretely:
    \begin{enumerate}
        \item The constant sequence $(0)$ is Cauchy (given any $\varepsilon$, all terms differ by $0$), and $(a_n) + (0) = (a_n)$.
        \item The constant sequence $(1)$ is Cauchy and $(a_n) \cdot (1) = (a_n)$.
        \item For any $(a_n) \in \mathcal{C}_\bbQ$, the termwise negation $(-a_n)$ is Cauchy because $|(-a_m) - (-a_n)| = |a_m - a_n|$, and $(a_n) + (-a_n) = (0)$. Hence $-(a_n) = (-a_n)$ is the additive inverse.
    \end{enumerate}

    Thus $(\mathcal{C}_\bbQ, +, \cdot)$ is a commutative ring with unity, as claimed.
\end{proof}
