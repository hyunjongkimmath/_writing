\begin{proposition}[{\cite[1.3. Proposition]{spaltenstein_ruc}}]  \label{proposition:complex_in_abelian_category_is_K_projective_K_injective_iff_shifts_are}
Let $\calA$ be an \CrefAndHyperrefIfExist{definition:abelian_category}{abelian category}. 
A complex $A^\bullet$ of objects in $\calA$ is $K$-projective (resp. $K$-injective) if and only if \CrefAndHyperrefIfExist{definition:translation_of_chain_cochain_complex_in_preadditive_category}{$A^\bullet[n]$} is as well, where $n$ is any integer.
\end{proposition}
