\begin{proposition} \label{proposition:exact_sequence_of_inertia_and_decomposition_groups_for_finite_galois_cover_of_schemes}
    Let $Y \to X$ be a \CrefAndHyperrefIfExist{definition:morphism_of_schemes}{morphism of schemes} where $X$ is \CrefAndHyperrefIfExist{definition:disconnected_and_connected_topological_spaces}{connected}. Let $\overline{y}: \operatorname{Spec} \Omega \to Y$ be a \CrefAndHyperrefIfExist{definition:geometric_point_of_a_scheme}{geometric point} of $Y$ where $\Omega$ is a \CrefAndHyperrefIfExist{definition:separably_closed_field}{separably closed field}. Let $y \in Y$ be the image of $\overline{y}$. Let $\overline{x}$ and $x$ be the images of $\overline{y}$ and $y$ in $X$. 

    \begin{enumerate}
        \item 
        There is a left exact sequence
        $$1 \to I_{\overline{y}} \to D_{\overline{y}} \to \operatorname{Aut}(\kappa(y) / \kappa(x))$$
        \CrefIfExists{definition:inertia_group_of_a_morphism_of_schemes_at_a_geometric_point}\CrefIfExists{definition:decomposition_group_of_a_morphism_of_schemes_at_a_geometric_point}\CrefIfExists{definition:automorphism_group_of_a_field_extension} \CrefIfExists{definition:residue_field_of_a_ring_at_a_point}

        \item If additionally $Y \to X$ is a finite \CrefAndHyperrefIfExist{definition:galois_morphism_of_schemes}{Galois cover} and $X$ is \CrefAndHyperrefIfExist{definition:geometric_property_of_a_scheme_over_a_field}{geometrically} \CrefAndHyperrefIfExist{definition:unibranch_geometircally_unibranch_scheme}{unibranch}, then the left exact sequence is a short exact sequence
        $$1 \to I_{\overline{y}} \to D_{\overline{y}} \to \operatorname{Gal}(\kappa(y) / \kappa(x)) \to 1$$
        \CrefIfExists{definition:galois_extension_of_fields}\CrefIfExists{definition:residue_field_of_a_scheme_at_a_point}
    \end{enumerate}

\end{proposition}
