\begin{lemma} \label{lemma:bounded_below_spectral_sequence_in_abelian_category_converges_to_any_graded_object_that_it_abuts_to}
    Let $\calA$ be an \CrefAndHyperrefIfExist{definition:abelian_category}{abelian category}.
    \begin{enumerate}
        \item Let $E$ be a \CrefAndHyperrefIfExist{definition:bounded_complexes_on_an_additive_category_and_homologically_bounded_objects_on_an_abelian_category}{bounded below} \CrefAndHyperrefIfExist{definition:spectral_sequence_in_an_abelian_category}{homological spectral sequence}. If $E$ \CrefAndHyperrefIfExist{definition:abuts_to_approaches_a_filtration_for_a_spectral_sequence_in_an_abelian_category}{abuts to} a \CrefAndHyperrefIfExist{definition:graded_object_in_an_abelian_category}{graded object} $H$ in $\calA$, then $E$ \CrefAndHyperrefIfExist{definition:converges_to_a_graded_object_for_a_spectral_sequence_in_an_abelian_category}{converges} to $H$.

        \item Let $E$ be a bounded below cohomological spectral sequence. If $E$ abuts to a graded object $H$ in $\calA$, then $E$ converges to $H$.
    \end{enumerate}
\end{lemma}
