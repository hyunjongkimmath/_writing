\begin{proposition}[{\cite[Proposition 1.3.7]{beilinson_berstein_deligne_atlesI}}] \label{proposition:non_degenerate_t_structures_have_conservative_cohomology_functors}
    Let $D$ be a \CrefAndHyperrefIfExist{definition:triangulated_category}{triangulated category} and let $(\Dleqzero, \Dgeqzero)$ be a \CrefAndHyperrefIfExist{definition:t_structure_on_a_triangulated_category}{$t$-structure} on $D$.

    \TODO{conservative system of functors}
    If the $t$-structure $(\Dleqzero, \Dgeqzero)$ on $D$ is \CrefAndHyperrefIfExist{definition:non_degenerate_t_structure_on_a_triangulated_category}{non-degenerate}, then the system of functors $H^n$ is conservative, i.e. for a morphism $f: M \to N$ between objects of $D$, the induced maps $H^n(f): H^n(M) \to H^n(N)$ are all isomorphisms if and only if $f$ is an isomorphism. Moreover, an object $X$ in $D$ belongs to $\Dleqzero$ (resp. $\Dgeqzero$) if and only if the $H^n(X)$ are zero for $n > 0$ (resp. for $n < 0$).
\end{proposition}
