\begin{lemma}[{\cite[Chapter I, 2.1.1 Lemma]{knus_qhfr}}] \label{lemma:twisted_dual_module_of_a_module_over_a_ring_with_involution_can_be_defined_as_opposite_of_dual_or_dual_of_opposite}
    Let $A$ be a \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{ring with involution} denoted by $a \mapsto \overline{a}$. 

    For (left/right) \CrefAndHyperrefIfExist{definition:bimodule_over_rings_module_over_a_ring}{$A$-modules} $M$, the maps
    $$T_M: \overline{M^\vee} \to \overline{M}^\vee$$
    (\Cref{definition:dual_of_a_left_right_two_sided_module}) (\Cref{definition:opposite_module_of_a_module_over_a_ring_with_involution}) given by 
    $$f \mapsto \left( T_M(f): \overline{m} \mapsto \overline{f(m)} \right)s$$
    where $\overline{m} = m$ as an element over $\overline{M}$, defines a \CrefAndHyperrefIfExist{definition:natural_transformation_between_functors_between_categories}{natural isomorphism} 
    $$\overline{(-)^\vee} \to \left( \overline{-} \right)^\vee$$
    between the \CrefAndHyperrefIfExist{definition:functor_between_categories}{functors} $\overline{(-)^\vee}, \left( \overline{-} \right)^\vee$ from the \CrefAndHyperrefIfExist{definition:category_of_modules_and_bimodules_over_rings}{category of (left/right) $A$-modules} to itself. 

    In other words, the \CrefAndHyperrefIfExist{definition:twisted_dual_module_of_a_module_over_a_ring_with_involution}{twisted dual module} $M^*$ may be constructed either as $\overline{M^\vee}$ or as $\overline{M}^\vee$. 
\end{lemma}
