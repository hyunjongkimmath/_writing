\begin{theorem}[Proper base change for algebraic adic coefficients, cf. {\cite[Expos\'e XII Th\'eor\`eme 5.1]{SGA4_III}}, {\cite[Theorem I.6.1]{freitag_kiehl_ecwc}}, {\cite[Theorem 7.3.1]{fu_ect}}] \label{theorem:proper_base_change_theorem_for_derived_category_of_adic_sheaves_with_algebraic_coefficients_on_schemes}
    Let $S$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme}. Let
    \begin{center}
        \begin{tikzcd}
            X' \ar[r, "f'"] \ar[d,"g'"] & Y' \ar[d, "g"] \\
            X \ar[r, "f"] & Y
        \end{tikzcd}
    \end{center}
    be a Cartesian diagram of $S$-schemes such that $f$ is \CrefAndHyperrefIfExist{definition:proper_morphism_of_schemes}{proper}.

    Let $\ell$ be a prime invertible on $Y$. Let $R$ be the \CrefAndHyperrefIfExist{definition:integral_element_over_a_ring}{integral closure} of $\bbZ_\ell$ in a finite extension $E$ of $\Qell$, and let $F/E$ be an algebraic extension.

    For any $\mathcal{K} \in \Dbc(X, F)$ (\Cref{definition:six_functor_formalism_on_derived_category_of_bdd_constr_complexes_with_algebraic_coefficients_on_a_noetherian_scheme}), the canonical base change morphism
    $$g^* Rf_* \mathcal{K} \to Rf_*' {g'}^* \mathcal{K}$$
    is an isomorphism in $\Dbc(Y', F)$.

    The isomorphism follows from \Cref{theorem:proper_base_change_theorem_for_derived_category_of_adic_sheaves_with_rational_coefficients_on_schemes}: any object of $\Dbc(X, F)$ is represented by an object of $\Dbc(X, L')$ for some finite subextension $L'/E$ inside $F$, and the result descends to that level.
\end{theorem}
