\begin{theorem}[Properties of rational exponentiation] \label{theorem:properties_of_rational_exponentiation}

    Let $a,b > 0$ and let $r,s \in \mathbb{Q}$.

    \begin{enumerate}
        \item (Additivity of exponents) $a^r \cdot a^s = a^{r+s}$ in the sense of \CrefAndHyperrefIfExist{definition:rational_exponentiation_of_positive_reals}.
        \item (Power of a power) $\left(a^r\right)^s = a^{rs}$.
        \item (Monotonicity in the base) If $0 < a \leq b$, then $a^r \leq b^r$ for all $r > 0$.
        \item (Monotonicity in the exponent) If $a > 1$ and $r < s$, then $a^r < a^s$. If $0 < a < 1$ and $r < s$, then $a^r > a^s$.
    \end{enumerate}
\end{theorem}

\begin{proof}

We write $r = p/q$ and $s = u/v$ with $p,u \in \mathbb{Z}$, $q,v \in \mathbb{Z}_{>0}$, and $\gcd(p,q) = \gcd(u,v) = 1$.

\begin{enumerate}
    \item To prove $a^{p/q} \cdot a^{u/v} = a^{(pv+qu)/(qv)}$, we compute by raising both sides to the power $qv$:
    $$\left(a^{p/q} \cdot a^{u/v}\right)^{qv} = \left(\left(a^{p/q}\right)^q\right)^v \cdot \left(\left(a^{u/v}\right)^v\right)^q.$$
    By the definition of rational exponentiation (\CrefAndHyperrefIfExist{definition:rational_exponentiation_of_positive_reals}), we have $\left(a^{p/q}\right)^q = a^p$ and $\left(a^{u/v}\right)^v = a^u$. Hence
    $$\left(a^{p/q} \cdot a^{u/v}\right)^{qv} = (a^p)^v \cdot (a^u)^q = a^{pv} \cdot a^{qu} = a^{pv+qu},$$
    where the last equality uses integer exponent laws (\CrefAndHyperrefIfExist{lemma:multiplicativity_of_powers_in_a_field}). On the other hand, by definition $a^{(pv+qu)/(qv)}$ is the unique positive real whose $(qv)$th power equals $a^{pv+qu}$. Since both sides are positive and have the same $(qv)$th power, they must be equal.

    \item We compute $\left(a^r\right)^s = (a^{p/q})^{u/v}$. By definition of rational exponentiation, this is the unique positive real whose $v$th power equals $\left(a^{p/q}\right)^u$. Using part 1 repeatedly:
    $$\left(\left(a^{p/q}\right)^{u/v}\right)^{qv} = \left(\left(a^{p/q}\right)^u\right)^v = \left(a^{pu/q}\right)^v.$$
    By definition, $a^{pu/q}$ is the unique positive real whose $q$th power equals $a^{pu}$. Raising to the $v$th power: $(a^{pu/q})^v$ has $(qv)$th power equal to $(a^{pu})^v = a^{puv}$. Similarly, $a^{rs} = a^{(pv)/(qv)}$ also has $(qv)$th power equal to $a^{p v \cdot u / v} = a^{pu}$ raised to the appropriate power. Both are positive reals with matching powers, so they coincide by uniqueness in the definition of rational exponentiation.

    \item If $r > 0$, write $r = p/q$ with $p,q > 0$. Then for any $x > 0$, the function $x \mapsto x^r$ is increasing on $(0,\infty)$ because if $a \leq b$, then by \CrefAndHyperrefIfExist{lemma:monotonicity_of_integer_powers_of_positive_reals} (applied to the integer power $p$), we have $a^p \leq b^p$. Taking positive $q$th roots preserves order, so $(a^p)^{1/q} \leq (b^p)^{1/q}$, which means $a^{p/q} \leq b^{p/q}$.

    \item Suppose $a > 1$ and $r < s$, so $s - r = t > 0$. Then
    $$a^s = a^{r+t} = a^r \cdot a^t$$
    by part 1. Since $a > 1$ and $t > 0$, we have $a^t > 1$ (because $a^0 = 1 < a^t$ by the monotonicity of integer powers extended to rationals). Hence $a^s = a^r \cdot a^t > a^r$.

    For $0 < a < 1$, we have $a^{-1} > 1$, so writing $a^r = (a^{-1})^{-r}$ and applying the previous case:
    $$a^r = (a^{-1})^{-r} > (a^{-1})^{-s} = a^s$$
    since $-r > -s$ and base $a^{-1} > 1$.
\end{enumerate}
\end{proof}
