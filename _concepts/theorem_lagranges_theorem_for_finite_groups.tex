\begin{theorem}[Lagrange's Theorem] \label{theorem:lagranges_theorem_for_finite_groups}
Let $G$ be a \CrefAndHyperrefIfExist{definition:order_of_a_group}{finite group} and let $H \leq G$ be a \CrefAndHyperrefIfExist{definition:subgroup_of_a_group}{subgroup}. Then the \CrefAndHyperrefIfExist{definition:order_of_a_group}{order} $|H|$ of $H$ divides the order $|G|$ of $G$. More precisely,
$$|G| = |H| \cdot [G : H],$$
where $[G : H]$ is the \CrefAndHyperrefIfExist{definition:index_of_a_subgroup_in_a_group}{index} of $H$ in $G$.
\end{theorem}
