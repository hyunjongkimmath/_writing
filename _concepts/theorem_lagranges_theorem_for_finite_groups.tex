\begin{theorem}[Lagrange's Theorem] \label{theorem:lagranges_theorem_for_finite_groups}
    Let $G$ be a \CrefAndHyperrefIfExist{definition:order_of_a_group}{finite group} and let $H \leq G$ be a \CrefAndHyperrefIfExist{definition:subgroup_of_a_group}{subgroup}. Then the \CrefAndHyperrefIfExist{definition:order_of_a_group}{order} $|H|$ of $H$ divides the order $|G|$ of $G$. More precisely,
    $$|G| = |H| \cdot [G : H],$$
    where $[G : H]$ is the \CrefAndHyperrefIfExist{definition:index_of_a_subgroup_in_a_group}{index} of $H$ in $G$.
\end{theorem}
\begin{proof}
    Since $G$ is finite, the left cosets $\{gH : g \in G\}$ form a finite collection. By \Cref{lemma:left_cosets_of_subgroup_partition_the_group}, these cosets partition $G$. By \Cref{lemma:all_cosets_of_subgroup_have_equal_cardinality}, each coset $gH$ has the same cardinality as $H$, namely $|H|$.

    The index $[G : H]$ is by \CrefIfExists{definition:index_of_a_subgroup_in_a_group} the number of distinct left cosets. Since the cosets are pairwise disjoint and their union is $G$, we have
    $$|G| = \sum_{gH \in G/H} |gH| = \sum_{gH \in G/H} |H| = [G : H] \cdot |H|,$$
    where the sum runs over the distinct left cosets. This establishes the equality $|G| = |H| \cdot [G : H]$, from which it follows that $|H|$ divides $|G|$.
\end{proof}
