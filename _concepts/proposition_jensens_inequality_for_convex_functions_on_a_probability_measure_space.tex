\begin{proposition}[Jensen's inequality (measure-theoretic)] \label{proposition:jensens_inequality_for_convex_functions_on_a_probability_measure_space}
Let $(\Omega, \mathcal{F}, P)$ be a \CrefAndHyperrefIfExist{definition:probability_measure_space_for_a_measure_on_a_measurable_space}{probability measure space}. Let $\varphi : I \to \mathbb{R}$ be a convex function defined on an interval $I \subseteq \mathbb{R}$. If $X : \Omega \to I$ is a measurable function such that $X \in L^1(\Omega, \mathcal{F}, P)$ and $\varphi \circ X \in L^1(\Omega, \mathcal{F}, P)$, then
$$\varphi\left(\int_\Omega X\,dP\right) \leq \int_\Omega (\varphi \circ X)\,dP.$$
\end{proposition}
