\begin{theorem} \label{theorem:category_of_modules_over_an_essentially_small_ringed_site_has_enough_K_injectives}
    Let $(\calC, J, \calO)$ be a \CrefAndHyperrefIfExist{definition:ringed_site}{ringed site} where the underlying site $(\calC, J)$ is \CrefAndHyperrefIfExist{definition:essentially_small_site}{essentially small}. In particular, $\Mod_\calO$ is an \CrefAndHyperrefIfExist{definition:abelian_category}{abelian category} and in fact is a \CrefAndHyperrefIfExist{definition:grothendiecks_additional_axioms_for_abelian_categories}{Grothendieck category} (\Cref{theorem:examples_of_grothendieck_categories}).

    For every \CrefAndHyperrefIfExist{definition:chain_complex_of_objects_in_an_additive_category}{chain complex} $A^\bullet$ in \CrefAndHyperref{definition:module_over_a_sheaf_of_rings_on_a_site}{$\Mod_\calO$} (resp. $_{\calO} \Mod$), there is a \CrefAndHyperrefIfExist{definition:left_right_resolution_of_a_chain_complex_in_an_abelian_category}{right resolution} $A^\bullet \to I^\bullet$  such that  $I^\bullet$ is \CrefAndHyperrefIfExist{definition:K_injective_projective_complex_in_an_abelian_category}{$K$-injective}.
\end{theorem}
\begin{proof}
    This follows from \Cref{theorem:grothendieck_categories_have_enough_K_injective_complexe}.
\end{proof}
