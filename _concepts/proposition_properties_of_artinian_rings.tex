\begin{proposition} \label{proposition:properties_of_artinian_rings}
Let $R$ be a \CrefAndHyperrefIfExist{definition:left_right_artinian_ring}{left/right Artinian ring}. Then:
\begin{enumerate}
    \item $R$ is also a \CrefAndHyperrefIfExist{definition:locally_noetherian_and_noetherian_scheme}{left/right Noetherian ring} (Hopkins-Levitzki Theorem).
    \item The \CrefAndHyperrefIfExist{definition:jacobson_radical_of_a_ring}{Jacobson radical} \hl{$J(R)$} is a \CrefAndHyperrefIfExist{definition:nilpotent_ideal_of_a_ring}{nilpotent ideal}.
    \item $R/J(R)$ is a \CrefAndHyperrefIfExist{definition:semisimple_ring}{semisimple ring}.
\end{enumerate}
\end{proposition}
