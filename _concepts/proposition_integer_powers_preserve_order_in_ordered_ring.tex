\begin{proposition} \label{proposition:integer_powers_preserve_order_in_ordered_ring}

    Let $R$ be an \CrefAndHyperrefIfExist{definition:ordered_ring}{ordered ring}. For any $a,b \in R$ and any integer $n \ge 1$:
    \begin{enumerate}
        \item If $0 \le a \le b$, then $0 \le a^n \le b^n$.
        \item If $b \le a \le 0$, then $a^n \le b^n$ for odd $n$, and $b^n \le a^n$ for even $n$.
        \item If $a \le 0 \le b$, then $a^n \le b^n$ for odd $n$, and $|a|^n \le b^n$ for even $n$ (where $|a| = -a$).
    \end{enumerate}
    For $n = 0$, we have $a^0 = 1 = b^0$ (the multiplicative identity), so $a^0 = b^0$.
\end{proposition>

    \begin{enumerate}

        \item If $b \le a \le 0$, write $b = -x$, $a = -y$ with $0 \le y \le x$. Then $a^n = (-y)^n$, $b^n = (-x)^n$. For odd $n$, $(-y)^n = -y^n \le -x^n = (-x)^n$ since $y^n \le x^n$ by part (1) and negation reverses order. For even $n$, $(-y)^n = y^n \le x^n = (-x)^n$.
        \item If $a \le 0 \le b$, then for odd $n$, $a^n \le 0 \le b^n$ so $a^n \le b^n$. For even $n$, $a^n = (-a)^n \ge 0$ and $b^n \ge 0$, with $(-a)^n \le b^n$ by part (1) if $-a \le b$ (i.e., $|a| \le b$).
    \end{enumerate}
