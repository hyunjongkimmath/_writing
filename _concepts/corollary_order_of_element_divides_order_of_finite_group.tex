\begin{corollary} \label{corollary:order_of_element_divides_order_of_finite_group}
    Let $G$ be a \CrefAndHyperrefIfExist{definition:order_of_a_group}{finite group} and let $g \in G$. Then the \CrefAndHyperrefIfExist{definition:order_of_a_group_element}{order} $\operatorname{ord}(g)$ of $g$ divides the \CrefAndHyperrefIfExist{definition:order_of_a_group}{order} $|G|$ of $G$.
\end{corollary}
\begin{proof}
    Let $\langle g \rangle$ be the subgroup of $G$ generated by $g$. By \Cref{proposition:order_of_cyclic_subgroup_generated_by_element_equals_order_of_element}, we have $|\langle g \rangle| = \operatorname{ord}(g)$. Since $\langle g \rangle \leq G$ and $G$ is finite, \Cref{theorem:lagranges_theorem_for_finite_groups} implies that $|\langle g \rangle|$ divides $|G|$. Therefore $\operatorname{ord}(g)$ divides $|G|$.
\end{proof}
