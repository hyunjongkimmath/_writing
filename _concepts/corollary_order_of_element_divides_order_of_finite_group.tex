\begin{corollary} \label{corollary:order_of_element_divides_order_of_finite_group}
Let $G$ be a \CrefAndHyperrefIfExist{definition:order_of_a_group}{finite group} and let $g \in G$. Then the \CrefAndHyperrefIfExist{definition:order_of_a_group_element}{order} $\operatorname{ord}(g)$ of $g$ divides the \CrefAndHyperrefIfExist{definition:order_of_a_group}{order} $|G|$ of $G$.
\end{corollary}
