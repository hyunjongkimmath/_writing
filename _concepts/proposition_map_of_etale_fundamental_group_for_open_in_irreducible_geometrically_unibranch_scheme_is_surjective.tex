\begin{proposition}[{\cite[Lemma 0BQI]{stacks-project}}] \label{proposition:map_of_etale_fundamental_group_for_open_in_irreducible_geometrically_unibranch_scheme_is_surjective}
    Let $X$ be an \CrefAndHyperrefIfExist{definition:irreducible_topological_space}{irreducible}, \CrefAndHyperrefIfExist{definition:unibranch_geometircally_unibranch_scheme}{geometrically unibranch scheme}. For any nonempty open $U \subseteq X$ and \CrefAndHyperrefIfExist{definition:geometric_point_of_a_scheme}{geometric point} $\overline{u}$ of $U$, the canonical map
    $$\pioneet(U, \overline{u}) \to \pioneet(X, \overline{u})$$
    is surjective.

    Letting $\eta$ be the \CrefAndHyperrefIfExist{definition:generic_point_of_a_topological_space}{generic point} $X$ and letting $\overline{\eta}$ be a \CrefAndHyperrefIfExist{definition:geometric_point_of_a_scheme}{geometric point} of $X$ with image $\eta$, the canonical map 
    $$\pioneet(U, \overline{\eta}) \to \pioneet(X, \overline{\eta})$$
    is surjective as well.
\end{proposition}
