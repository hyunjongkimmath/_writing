\begin{theorem} \label{theorem:neron_models_of_abelian_varieties_exist_over_dvrs_dedekind_schemes_of_finite_type_over_a_field_and_integer_rings_of_number_fields}
    Let $S$ be an \CrefAndHyperrefIfExist{definition:integral_scheme}{integral scheme}, let $\eta$ be its \CrefAndHyperrefIfExist{definition:generic_point_of_a_topological_space}{generic point}, and let $K = \mathcal{O}_{S, \eta}$ be the \CrefAndHyperrefIfExist{definition:function_field_over_a_field}{function field} of $S$. Let $A_K$ be an \CrefAndHyperref{definition:abelian_variety_over_a_field}{abelian variety over $K$}.

    Under any of the following assumptions, a \CrefAndHyperrefIfExist{definition:neron_model_of_an_abelian_variety_over_the_function_field_of_a_dedekind_scheme}{N\'eron model} of $A_K$ over $S$ exists:
    \begin{enumerate}
        \item $S = \Spec R$ where $R$ is a \CrefAndHyperrefIfExist{definition:discrete_valuation_ring}{DVR}
        \item $S$ is a \CrefAndHyperrefIfExist{definition:dedekind_scheme}{Dedekind scheme} of \CrefAndHyperrefIfExist{definition:finite_type_morphism_of_schemes}{finite type} over a \CrefAndHyperrefIfExist{definition:field}{field}
        \item $S= \Spec R$ where $R$ is the \CrefAndHyperrefIfExist{definition:ring_of_integers_of_a_global_or_local_ring}{ring of integers} of a \CrefAndHyperrefIfExist{global_field}{number field}.
    \end{enumerate}

\end{theorem}
