\begin{proposition}[{\cite[Expos\'e III, Proposition 2.2.]{SGA4_I}}] \label{proposition:functor_between_sites_is_cocontinuous_if_and_only_if_precomposition_functor_sends_sheaves_to_sheaves}
Let $u: \mathcal{C} \to \mathcal{D}$ be a functor between sites $(\mathcal{C}, J)$ and $(\mathcal{D}, K)$. The functor $u$ is cocontinuous if and only if the precomposition/\CrefAndHyperrefIfExist{definition:inverse_image_of_sheaves_along_a_cocontinuous_functor_of_sites}{inverse image functor} $u^*: \widehat{\mathcal{D}} \to \widehat{\mathcal{C}}$, defined by $P \mapsto P \circ u^{\operatorname{op}}$ for any presheaf $P$ on $\mathcal{D}$, restricts to a functor $u^*: \operatorname{Sh}(\mathcal{D}, K; \Sets) \to \operatorname{Sh}(\mathcal{C}, J; \Sets)$ between the categories of sheaves.
\end{proposition}
