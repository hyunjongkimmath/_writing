\begin{proposition}[Reflex Field Characterization via Tori and the Reflex Norm] \label{proposition:reflex_field_characterization_via_tori_and_reflex_norm}
    Let $(E, \Phi)$ be a \CrefAndHyperrefIfExist{definition:cm_type_and_reflex_field_of_a_cm_type}{CM type}, and let $E^*(\Phi)$ be the \CrefAndHyperrefIfExist{definition:cm_type_and_reflex_field_of_a_cm_type}{reflex field}. Let $\mathbb{G}_m = \mathbb{G}_{m,\mathbb{Q}}$ and for any \CrefAndHyperrefIfExist{definition:number_field}{number field} $L$, let $T_L = \operatorname{Res}_{L/\mathbb{Q}}(\mathbb{G}_m)$ be the associated algebraic torus over $\mathbb{Q}$ whose character group is the permutation module $X(T_L) \cong \operatorname{Hom}(L, \overline{\mathbb{Q}})$.

    The reflex field $E^*(\Phi)$ is the unique number field (up to $\mathbb{Q}$-isomorphism) for which there exists a morphism of algebraic tori over $\mathbb{Q}$, called the \CrefAndHyperrefIfExist{definition:reflex_norm_for_a_cm_field}{reflex norm},
    $$N_{(E,\Phi)}: T_{E^*(\Phi)} \to T_E,$$
    whose dual map on character groups
    $$N_{(E,\Phi)}^*: X(T_E) \to X(T_{E^*(\Phi)})$$
    sends each $\tau \in \operatorname{Hom}(E, \overline{\mathbb{Q}})$ to
    $$N_{(E,\Phi)}^*(\tau) = h_\rho \cdot \rho,$$
    where $\rho = \tau|_{E^*(\Phi)} \in \operatorname{Hom}(E^*(\Phi), \overline{\mathbb{Q}})$ and $h_\rho = |\{\sigma \in \Phi : \sigma|_{E^*(\Phi)} = \rho\}|$.
\end{proposition}
