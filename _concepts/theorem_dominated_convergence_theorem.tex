\begin{theorem}[Dominated convergence theorem] \label{theorem:dominated_convergence_theorem}
Let $(X, \mathcal{A}, \mu)$ be a \CrefAndHyperrefIfExist{definition:measure_on_a_measurable_space_and_measure_space}{measure space}.

Let $(f_n)_{n \geq 1}$ be a sequence of \CrefAndHyperrefIfExist{definition:measurable_function_between_measure_spaces}{measurable functions} $f_n : X \to \overline{\mathbb{R}}$ (or $\mathbb{C}$) and let $f : X \to \overline{\mathbb{R}}$ (or $\mathbb{C}$) be a \CrefAndHyperrefIfExist{definition:function_of_sets}{function}. Suppose that:
\begin{enumerate}
    \item $f_n \to f$ \CrefAndHyperrefIfExist{definition:almost_sure_convergence_of_a_sequence_of_measurable_functions_from_a_measure_space_to_a_metric_space_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity}{almost everywhere} as $n \to \infty$, and
    \TODO{dominated by...}
    \item there exists an \CrefAndHyperrefIfExist{definition:lebesgue_integral_of_measurable_functions_into_the}{integrable function} $g : X \to [0,\infty]$ such that $|f_n(x)| \leq g(x)$ for all $n \geq 1$ and all $x \in X$ (i.e. $(f_n)$ is \hldef{dominated} by an integrable function).
\end{enumerate}

Then each $f_n$ (and $f$) is integrable, and
$$\lim_{n \to \infty} \int_X f_n\,d\mu = \int_X f\,d\mu.$$
Equivalently, $\|f_n - f\|_1 \to 0$, i.e. $f_n \to f$ in $L^1(\mu)$.
\end{theorem}
