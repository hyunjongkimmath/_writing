\begin{proposition} \label{proposition:equivalent_formulations_of_solvable_groups}
Let $G$ be a \CrefAndHyperrefIfExist{definition:group}{group}. The following are equivalent:

\begin{enumerate}
    \item $G$ is a \CrefAndHyperref{definition:solvable_group_via_derived_series}{solvable group}.

    \item There exists a finite chain of \CrefAndHyperrefIfExist{definition:normal_subgroup_of_a_group}{normal subgroups}
    $$1 = G_0 \triangleleft G_1 \triangleleft \cdots \triangleleft G_k = G$$
    such that each \CrefAndHyperrefIfExist{definition:quotient_group_of_a_group_by_a_normal_subgroup}{quotient group} $G_{i+1}/G_i$ is an \CrefAndHyperrefIfExist{definition:abelian_group}{abelian group}.
\end{enumerate}
\end{proposition}
