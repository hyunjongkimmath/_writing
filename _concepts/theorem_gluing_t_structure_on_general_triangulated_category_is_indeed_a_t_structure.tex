\begin{theorem}[{\cite[Th\'eor\`eme 1.4.10., Remarque 1.4.11.]{beilinson_berstein_deligne_atlesI}}] \label{theorem:gluing_t_structure_on_general_triangulated_category_is_indeed_a_t_structure}
    Assume \Cref{context:general_gluing_framework_for_t_structures_on_triangulated_categories}. Let $(D_U^{\leq 0}, D_U^{\geq 0})$ and $(D_F^{\leq 0}, D_F^{\geq 0})$ be \CrefAndHyperrefIfExist{definition:t_structure_on_a_triangulated_category}{$t$-structures} on $D_U$ and $D_F$ respectively.
    The pair $(D^{\leq 0}, D^{\geq 0})$ is a \CrefAndHyperrefIfExist{definition:t_structure_on_a_triangulated_category}{$t$-structure} on $D$. Moreover, $(D^{\leq 0}, D^{\geq 0})$ is \CrefAndHyperrefIfExist{definition:non_degenerate_t_structure_on_a_triangulated_category}{non-degenerate} if and only if $(D_U^{\leq 0}, D_U^{\geq 0})$ and $(D_F^{\leq 0}, D_F^{\geq 0})$ both are. 
\end{theorem}
