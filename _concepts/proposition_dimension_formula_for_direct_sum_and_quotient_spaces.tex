\begin{proposition} \label{proposition:dimension_formula_for_direct_sum_and_quotient_spaces}
    Let $F$ be a \CrefAndHyperrefIfExist{definition:field}{field}.

    \textbf{(Direct sum.)} If $V_1, \dots, V_n$ are finite-dimensional $F$-vector spaces, then
    $$\hlin{\dim_F(V_1 \oplus \cdots \oplus V_n) = \sum_{i=1}^n \dim_F(V_i)}.$$

    \textbf{(Quotient.)} If $V$ is a finite-dimensional $F$-vector space and $W$ is a subspace, then
    $$\hlin{\dim_F(V/W) = \dim_F(V) - \dim_F(W)}.$$
