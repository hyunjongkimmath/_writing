\begin{lemma} \label{lemma:image_coimage_of_a_morphism_in_an_abelian_category_coincides_with_kernel_of_cokernel_cokernel_of_kernel}
    Let $\calA$ be a \CrefAndHyperrefIfExist{definition:abelian_category}{abelian category}. 
    For any morphism $f$ in $\calA$, the \CrefAndHyperrefIfExist{definition:image_coimage_of_a_morphism_in_a_category}{image and coimage} of $f$ both exist, and there are isomorphisms
    $$\ker(\coker f) \cong \operatorname{im}(f)$$ 
    $$\coker(\ker f) \cong \operatorname{coim}(f).$$ 
    (\Cref{definition:kernel_and_cokernel_of_a_morphism_in_a_pointed_category}) In particular, the image and coimage of $f$ are isomorphic.
\end{lemma}
