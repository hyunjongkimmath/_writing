\begin{proposition} \label{proposition:adjugate_formula_for_determinant_over_a_commutative_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:commutative_ring}{commutative ring}, and let $A = (a_{ij}) \in M_n(R)$ be an $n \times n$ matrix. Let $\operatorname{adj}(A) \in M_n(R)$ be the \hldef{classical adjugate of $A$}\TODO{AI generated, reference needed: definition of classical adjugate matrix}, whose $(i,j)$-entry is $(-1)^{i+j}$ times the determinant of the submatrix obtained by deleting row $j$ and column $i$ from $A$. Then
    $$\operatorname{adj}(A) \cdot A = A \cdot \operatorname{adj}(A) = \det(A) \, I_n.$$
\end{proposition}
