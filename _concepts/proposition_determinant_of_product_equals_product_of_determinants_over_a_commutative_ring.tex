\begin{proposition} \label{proposition:determinant_of_product_equals_product_of_determinants_over_a_commutative_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:commutative_ring}{commutative ring}, and let $A, B \in M_n(R)$ be two $n \times n$ matrices. Then
    $$\det(AB) = \det(A)\,\det(B).$$
\end{proposition}
