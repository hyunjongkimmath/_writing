\begin{proposition} \label{proposition:order_of_cyclic_subgroup_generated_by_element_equals_order_of_element}
    Let $G$ be a \CrefAndHyperrefIfExist{definition:group}{group} and let $g \in G$. Let $\langle g \rangle$ be the \CrefAndHyperrefIfExist{definition:subgroup_of_a_group_generated_by_elements}{subgroup generated by} $\{g\}$, and let $\operatorname{ord}(g)$ be the \CrefAndHyperrefIfExist{definition:order_of_a_group_element}{order of $g$}.

    \begin{enumerate}
        \item If $\operatorname{ord}(g)$ is finite, then $\langle g \rangle$ is a \CrefAndHyperrefIfExist{definition:cyclic_group}{cyclic group} of \CrefAndHyperrefIfExist{definition:order_of_a_group}{order} $\operatorname{ord}(g)$. In particular,
        $$|\langle g \rangle| = \operatorname{ord}(g).$$

        \item If $\operatorname{ord}(g)$ is infinite, then $\langle g \rangle$ is an infinite cyclic group, and the map
        $$\varphi : \bbZ \to \langle g \rangle, \quad n \mapsto g^n$$
        is a \CrefAndHyperrefIfExist{definition:injective_surjective_bijective_map_of_sets}{bijection}. In particular, $|\langle g \rangle|$ is \CrefAndHyperrefIfExist{definition:countable_finite_uncountable_sets}{(countably) infinite}.
    \end{enumerate}
\end{proposition}
\begin{proof}
    By \Cref{definition:subgroup_of_a_group_generated_by_elements}, the subgroup $\langle g \rangle$ consists of all finite products of $g$ and $g^{-1}$, which simplifies to powers of $g$: $\langle g \rangle = \{g^n : n \in \bbZ\}$ by \CrefIfExists{definition:power_of_a_group_element_by_an_integer_multiple_of_an_abelian_group_element_by_an_integer}. Define
    $$\varphi : \bbZ \to \langle g \rangle, \quad n \mapsto g^n.$$
    This map is surjective by construction, since every element of $\langle g \rangle$ is a power of $g$.

    We analyze injectivity in the two cases.
    \begin{enumerate}
        \item Suppose that $\operatorname{ord}(g) = m < \infty$. In other words, $m$ is the smallest positive integer with $g^m = e$. We claim that $\varphi(n) = \varphi(k)$ if and only if $n \equiv k \pmod{m}$\CrefIfExists{definition:congruence_modulo_n}. Indeed, $\varphi(n) = \varphi(k)$ means $g^n = g^k$, which is equivalent to $g^{n-k} = e$ by the cancellation law in $G$. By the \CrefAndHyperrefIfExist{theorem:division_algorithm_for_integers_euclidean_algorithm_for_integers}{division algorithm}, write $n - k = qm + r$ with $0 \leq r < m$. Then $e = g^{qm+r} = (g^m)^q g^r = g^r$. Since $m$ is the smallest positive integer with $g^m = e$ and $0 \leq r < m$, we must have $g^{n-k} = e$ is equivalent to $r = 0$. In other words, $\varphi(n) = \varphi(k)$ if and only if $n-k = qm$, i.e. $n \equiv k \pmod{m}$.

        It follows that $\varphi$ induces a bijection from the set of residue classes $\{0, 1, \dots, m-1\}$ onto $\langle g \rangle$. Therefore $|\langle g \rangle| = m = \operatorname{ord}(g)$\CrefIfExists{number_of_residue_classes_modulo_m}.

        \item Suppose that $\operatorname{ord}(g)$ is infinite. In other words, no positive power of $g$ equals $e$. If $\varphi(n) = \varphi(k)$ for some $n, k \in \bbZ$, then $g^n = g^k$, so $g^{n-k} = e$. If $n - k > 0$, this contradicts the assumption that no positive power of $g$ is the identity. Similarly, if $n - k < 0$, then $g^{-(n-k)} = e$ with $-(n-k) > 0$, again a contradiction. Hence $n - k = 0$, so $n = k$. This shows $\varphi$ is injective.

        Since $\varphi$ is both injective and surjective, it is a bijection, so $|\langle g \rangle|$ is \CrefAndHyperrefIfExist{definition:countable_finite_uncountable_sets}{countably infinite}.
    \end{enumerate}
\end{proof}
