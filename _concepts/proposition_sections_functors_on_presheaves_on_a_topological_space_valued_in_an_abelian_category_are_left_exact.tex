\begin{proposition} \label{proposition:sections_functors_on_presheaves_on_a_topological_space_valued_in_an_abelian_category_are_left_exact}
    Let $X$ be a \CrefAndHyperrefIfExist{definition:topological_space}{topological space} and let $\calA$ be an \CrefAndHyperrefIfExist{definition:abelian_category}{abelian category}.
    \begin{enumerate}
        \item The \CrefAndHyperrefIfExist{definition:functor_between_categories}{functor}
        $$\Gamma: \mathrm{PSh}(\mathcal{C}, \calA) \to \calA.$$
        \CrefIfExists{definition:presheaf_on_a_topological_space}\CrefIfExists{definition:sections_of_a_presheaf_on_a_category_valued_in_a_category}
        is \CrefAndHyperrefIfExist{definition:exact_functor_between_abelian_categories}{exact}.

        \item 
        Assume that a \CrefAndHyperrefIfExist{definition:sheafification_of_a_presheaf_on_a_topological_space}{sheafification functor}
        $$a: \PreShv(X, \calA) \to \Shv(X, \calA) $$
        exists; in particular, $\Shv(X, \calA)$ is an abelian category (\Cref{proposition:circumstances_under_which_categories_of_sheaves_and_presheaves_on_a_topological_space_valued_in_an_abelian_category_are_abelian}).

        The functor
        $$\Gamma: \mathrm{Sh}(\mathcal{C}, \calA) \to \calA.$$
        \CrefIfExists{definition:sheaf_on_a_topological_space_valued_in_a_category_with_a_terminal_object} is \CrefAndHyperrefIfExist{definition:exact_functor_between_abelian_categories}{left exact}.
    \end{enumerate}
\end{proposition}
