\begin{theorem} \label{theorem:nice_enough_topological_spaces_are_paracompact}
    The following are \CrefAndHyperrefIfExist{definition:paracompact_topological_space}{paracompact} topological spaces:
    \begin{enumerate}
        \item \CrefAndHyperrefIfExist{definition:extended_metric_on_a_set}{Metric spaces} (Stone's theorem)
        \item \CrefAndHyperrefIfExist{definition:topological_manifold}{Topological manifolds}
        \item \CrefAndHyperrefIfExist{definition:compact_topological_space}{Compact} \CrefAndHyperrefIfExist{definition:separation_axioms_of_topology}{Hausdorff} spaces
        \item \CrefAndHyperrefIfExist{definition:CW_complex}{CW-complexes}
    \end{enumerate}
\end{theorem}
