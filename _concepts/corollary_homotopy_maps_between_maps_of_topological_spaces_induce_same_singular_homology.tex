\begin{corollary} \label{corollary:homotopy_maps_between_maps_of_topological_spaces_induce_same_singular_homology}
Let $(X, A)$ and $(Y, B)$ be pairs of topological spaces, and let $G$ be an abelian group. If $f \simeq g: (X, A) \to (Y, B)$ are homotopic maps of pairs, then the \CrefAndHyperrefIfExist{proposition:functoriality_of_singular_chain_complex_of_topological_spaces_with_coefficients_in_abelian_groups}{induced homomorphisms} on \CrefAndHyperrefIfExist{definition:singular_homology_and_relative_singular_homology_of_topological_space_with_coefficients_in_an_abelian_group}{singular homology groups} are equal:
$$ f_* = g_*: H_n(X, A; G) \to H_n(Y, B; G) $$
Similarly, the \CrefAndHyperrefIfExist{proposition:functoriality_of_singular_cohomology_of_topological_spaces_with_coefficients_in_abelian_groups}{induced homomorphisms} on \CrefAndHyperrefIfExist{definition:singular_cohomology_and_relative_singular_homology_of_topological_space_with_coefficients_in_an_abelian_group}{singular cohomology groups} are equal:
$$ f^* = g^*: H^n(Y, B; G) \to H^n(X, A; G) $$
\end{corollary}
