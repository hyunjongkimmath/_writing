\begin{theorem} [{\cite[Chapter I, 7.4.1 Theorem]{knus_qhfr}}]  \label{theorem:cats_varepsilon_herm_spaces_over_inv_ring_and_of_varepsilon_herm_space_over_a_ring_that_admits_a_symm_alg_with_inv_are_equivalent}
Let $R$ be a \CrefAndHyperrefIfExist{definition:commutative_ring}{commutative ring}. Let $A$ be a \CrefAndHyperrefIfExist{definition:symmetric_algebra_over_a_ring_with_involution}{symmetric $R$-algebra with involution} that is \CrefAndHyperrefIfExist{definition:finitely_generated_modules_over_rings}{finitely generated} and \CrefAndHyperrefIfExist{definition:projective_bimodule_over_rings}{projective} as an \CrefAndHyperrefIfExist{definition:bimodule_over_rings_module_over_a_ring}{$R$-module}. Let $\varepsilon \in A$ be an element satisfying $\varepsilon \cdot \overline{\varepsilon} = 1$.

The \CrefAndHyperrefIfExist{definition:scharlau_transfer_functor_for_hermitian_spaces_on_involution_rings}{Scharlau transfer functors} 
\begin{align*}
        T: \mathfrak{Herm}^{\varepsilon}(A) \to \mathfrak{Herm}^\varepsilon(R) \\ 
        T: \mathfrak{Herm}_+^{\varepsilon}(A) \to \mathfrak{Herm}_+^\varepsilon(R)
\end{align*}
give rise to \CrefAndHyperrefIfExist{definition:isomorphism_of_categories}{isomorphisms of categories}
\begin{align*}
    \mathfrak{H}^\varepsilon(A) \xrightarrow{\sim} \mathfrak{H}^\varepsilon(R;A) \\
    \mathfrak{H}_+^\varepsilon(A) \xrightarrow{\sim} \mathfrak{H}_+^\varepsilon(R;A)
\end{align*}
(\Cref{definition:category_of_varepsilon_hermitian_spaces_over_a_ring_with_involution}) (\Cref{definition:category_of_varepsilon_hermitian_space_over_a_ring_that_admits_a_symmetric_algebra_with_involution}) 
which preserve orthogonal sums and hyperbolic spaces.

\end{theorem}
