\begin{proposition} \label{proposition:logarithm_base_b_is_inverse_of_b_to_the_x}

    Let $b \in \mathbb{R}$ with $b > 0$ and $b \neq 1$. The logarithm with base $b$, $\log_b : (0, \infty) \to \mathbb{R}$, is the inverse of the function $f: \mathbb{R} \to (0, \infty)$ given by $f(x) = b^x$. Specifically,
    $$\log_b(b^x) = x \quad \text{for all } x \in \mathbb{R},$$
    and
    $$b^{\log_b(x)} = x \quad \text{for all } x > 0.$$
    Consequently, $\log_b$ is \CrefAndHyperrefIfExist{definition:injective_surjective_bijective_map_of_sets}{bijective} from $(0, \infty)$ onto $\mathbb{R}$, and $x \mapsto b^x$ is bijective from $\mathbb{R}$ onto $(0, \infty)$.
\end{proposition}
\begin{proof}
    \TODO{AI generated, verify}
    We prove the two identities separately.

    \begin{enumerate}
        \item Let $x \in \mathbb{R}$. By the identity relating real exponentiation to $\exp$ and $\ln$ (see \CrefAndHyperrefIfExist{proposition:real_exponentiation_via_exp_and_ln}{real exponentiation via exp and ln}), we have $b^x = \exp(x \ln(b))$. Applying $\log_b$:
        $$\log_b(b^x) = \frac{\ln(\exp(x \ln(b)))}{\ln(b)}.$$
        Since $\ln$ and $\exp$ are mutual inverses (see \CrefAndHyperrefIfExist{proposition:exponential_and_logarithm_are_mutual_inverses}{exponential and logarithm are mutual inverses}), we have $\ln(\exp(y)) = y$ for all $y \in \mathbb{R}$. Taking $y = x \ln(b)$:
        $$\log_b(b^x) = \frac{x \ln(b)}{\ln(b)} = x.$$

        \item Let $x > 0$. We compute:
        $$b^{\log_b(x)} = \exp(\log_b(x) \cdot \ln(b))$$
        by the identity $a^y = \exp(y \ln(a))$ for positive reals. Substituting $\log_b(x) = \frac{\ln(x)}{\ln(b)}$:
        $$\exp\left( \frac{\ln(x)}{\ln(b)} \cdot \ln(b) \right) = \exp(\ln(x)).$$
        Again using that $\exp(\ln(x)) = x$ for $x > 0$ (from \CrefAndHyperrefIfExist{proposition:exponential_and_logarithm_are_mutual_inverses}):
        $$b^{\log_b(x)} = x.$$
    \end{enumerate}

    The bijectivity follows immediately from the mutual inverse property.
\end{proof}
