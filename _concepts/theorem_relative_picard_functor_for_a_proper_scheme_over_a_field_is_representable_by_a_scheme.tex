\begin{theorem}[Mumford, unpublished, see {\cite[Section 8.2, Theorem 2]{bosch_lutkebohmert_raynaud_nm}}] \label{theorem:relative_picard_functor_for_a_proper_scheme_over_a_field_is_representable_by_a_scheme}
    Let $f: X \longrightarrow S$ be \CrefAndHyperrefIfExist{definition:flat_morphism_of_schemes}{flat}, \CrefAndHyperrefIfExist{definition:projective_variety_over_a_field}{projective}, and \CrefAndHyperrefIfExist{definition:locally_of_finite_presentation_finite_presentation_morphism_of_schemes}{finitely presented} with \CrefAndHyperrefIfExist{definition:geometric_property_of_a_scheme_over_a_field}{geometrically reduced} fibres. Assume that the \CrefAndHyperrefIfExist{definition:irreducible_component_of_a_topological_space}{irreducible components} of the fibres of $f$ are \CrefAndHyperrefIfExist{definition:geometric_property_of_a_scheme_over_a_field}{geometrically} \CrefAndHyperrefIfExist{definition:irreducible_topological_space}{irreducible}. 

    The \CrefAndHyperrefIfExist{definition:big_fppf_site_of_a_scheme}{fppf}-\CrefAndHyperrefIfExist{definition:sheaf_on_a_site}{sheaf} 
    $$\Pic_{X/S}: (\Sch/S)_{\mathrm{fppf}}^\circ \to \Sets$$
    (\Cref{definition:relative_picard_functor_of_a_scheme_over_a_scheme}) is \CrefAndHyperrefIfExist{definition:representable_functor_on_a_locally_small_category}{representable by} by a (not necessarily \CrefAndHyperrefIfExist{definition:quasi_separated_morphism_of_schemes}{separated}) $S$-scheme which is \CrefAndHyperrefIfExist{definition:locally_of_finite_presentation_finite_presentation_morphism_of_schemes}{locally of finite presentation} over $S$.
\end{theorem}
