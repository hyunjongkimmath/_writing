\begin{corollary}[Dimension formula] \label{corollary:dimension_formula_for_nondegenerate_quadratic_forms_on_vector_spcaes}
Let $F$ be a \CrefAndHyperrefIfExist{definition:field}{field} and $q: M \to F$ a \CrefAndHyperrefIfExist{definition:nondegenerate_quadratic_form_on_a_module_over_a_commutative_ring}{non-degenerate} \CrefAndHyperrefIfExist{definition:quadratic_formm_on_a_module_over_a_commutative_ring}{quadratic form}.
We have $\dim M = \dim M_a + 2k$, where $M_a$ is the anisotropic part and $k$ is the Witt index. In particular, $\dim M \equiv \dim M_a \pmod{2}$.
\end{corollary}
