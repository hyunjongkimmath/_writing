\begin{proposition} \label{proposition:derivative_of_tangent_is_secant_squared}
The \CrefAndHyperrefIfExist{definition:tangent_function_on_real_line}{tangent function} $\tan$ is \CrefAndHyperrefIfExist{definition:derivative_of_a_real_valued_function_on_a_real_interval}{differentiable} at every point $x$ in its domain, and
\TODO{$\sec^2(x) = 1 + \tan^2(x)$ identity}
\TODO{$1 = \sin^2(x) + \cos^2(x)$ identity}
$$\frac{d}{dx} \tan(x) = \sec^2(x) = 1 + \tan^2(x).$$
\CrefIfExists{definition:secant_cosecant_and_cotangent_functions_on_real_line}
\end{proposition}
\begin{proof}
    By the \CrefAndHyperrefIfExist{proposition:derivative_rules_for_sum_product_quotient_and_chain}{quotient rule},
    \begin{align*}
        \frac{d}{dx} \frac{\sin(x)}{\cos(x)} &= \frac{\cos(x) \cdot \frac{d}{dx} \sin(x) - \sin(x) \cdot \frac{d}{dx} \cos(x)}{\cos^2(x)} \\
        &= \frac{\cos(x) \cdot \cos(x) - \sin(x) \cdot (-\sin(x))}{\cos^2(x)} \\
        &= \frac{\cos^2(x) + \sin^2(x)}{\cos^2(x)} \\
        &= \frac{1}{\cos^2(x)} \\
        &= \sec^2(x).
    \end{align*}
    \CrefIfExists{proposition:derivatives_of_sine_and_cosine}
\end{proof}
