\begin{theorem} \label{theorem:classification_of_places_of_global_fields}
    \begin{enumerate}
        \item Let $K$ be a \CrefAndHyperrefIfExist{definition:number_field}{number field} with \CrefAndHyperrefIfExist{definition:ring_of_integers_of_a_global_or_local_ring}{ring of integers} $\calO_K$. 
        \begin{enumerate}
            \item The \CrefAndHyperrefIfExist{definition:archimedean_absolute_value_on_a_field}{nonarchimedean} \CrefAndHyperrefIfExist{definition:absolute_value_on_a_ring}{absolute values} on $K$ are all \CrefAndHyperrefIfExist{definition:equivalent_absolute_values_on_a_field}{equivalent to} \CrefAndHyperrefIfExist{definition:p_adic_valuation_and_absolute_value_on_a_number_field}{$\mathfrak{p}$-adic absolute values} for nonzero \CrefAndHyperrefIfExist{definition:prime_and_maximal_ideal_of_a_ring}{prime ideals} $\mathfrak{p}$ of $\calO_K$.

            \item The archimedean absolute values on $K$ are equivalent to absolute values induced by an embedding $\sigma: K \hookrightarrow \bbC$ (among these may be embeddings whose images lie in $\bbR$).
        \end{enumerate}

        \item Let $K$ be a \CrefAndHyperrefIfExist{definition:global_function_field}{global function field} with \CrefAndHyperrefIfExist{definition:ring_of_integers_of_a_global_or_local_ring}{ring of integers} $\calO_K$. 

        All \CrefAndHyperrefIfExist{definition:absolute_value_on_a_ring}{absolute values} on $K$ are \CrefAndHyperrefIfExist{definition:archimedean_absolute_value_on_a_field}{non-archimedean}.

    \end{enumerate}    
\end{theorem}
