\begin{proposition} \label{proposition:group_homology_cohomology_for_a_module_of_an_infinite_cyclic_group}
    Let $\mathbb{Z}$ be the infinite cyclic \CrefAndHyperrefIfExist{definition:group}{group} generated by $\sigma$. Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{(not necessarily commutative) ring}. Let $A$ be a \CrefAndHyperrefIfExist{definition:left_right_module_over_a_group_ring_of_a_group_over_a_ring}{left $R[\mathbb{Z}]$-module}. We have
    \begin{align*}
        H_n(\mathbb{Z}, A) &\cong \begin{cases}
            A_{\mathbb{Z}} = A/(\sigma-1)A & n = 0 \\
            A^{\mathbb{Z}} = \ker(\sigma-1) & n = 1 \\
            0 & n \geq 2
        \end{cases} \\
        H^n(\mathbb{Z}, A) &\cong \begin{cases}
            A^{\mathbb{Z}} = \ker(\sigma-1) & n = 0 \\
            A_{\mathbb{Z}} = A/(\sigma-1)A & n = 1 \\
            0 & n \geq 2
        \end{cases}
    \end{align*}
    (\Cref{definition:group_cohomology_homology_of_a_module_over_a_group_ring_over_a_ring}) (\Cref{definition:invariant_subgroup_of_a_group_acting_on_an_abelian_group}) (\Cref{definition:coinvariant_quotient_group_of_a_group_acting_on_an_abelian_group})
\end{proposition}

    By \Cref{corollary:group_homology_cohomology_are_just_tor_and_ext_functors_by_trivial_module_over_group_ring},
    \begin{align*}
        H_n(\mathbb{Z}, A) &\cong \Tor_n^{R[\mathbb{Z}]}(R, A), \\
        H^n(\mathbb{Z}, A) &\cong \Ext^n_{R[\mathbb{Z}]}(R, A).
    \end{align*}

    The sequence
    $$0 \to R[\mathbb{Z}] \xrightarrow{\sigma-1} R[\mathbb{Z}] \xrightarrow{\varepsilon} R \to 0$$
    is a \CrefAndHyperrefIfExist{definition:left_right_resolution_of_an_object_by_a_class_of_objects_in_an_abelian_category}{projective resolution} of the \CrefAndHyperrefIfExist{definition:trivial_G_module_of_a_group_over_a_ring}{trivial $R[\mathbb{Z}]$-module} $R$ by \CrefAndHyperrefIfExist{definition:free_bimodule_over_rings}{free} $R[\mathbb{Z}]$-\CrefAndHyperrefIfExist{definition:module_of_a_cosmos_between_categories_enriched_in_the_cosmos}{modules} (\Cref{lemma:group_ring_of_infinite_cyclic_group_has_length_one_free_resolution}).

    \textbf{Homology.} Tensoring the resolution (without the $R$ term) with $A$ over $R[\mathbb{Z}]$ gives the complex
    $$0 \to A \xrightarrow{(\sigma-1)\otimes \operatorname{id}_A} A \to 0$$
    with the rightmost $A$ in degree $0$. Its homology yields:
    \begin{itemize}
        \item $H_0(\mathbb{Z}, A) \cong A/\operatorname{im}((\sigma-1)\otimes \operatorname{id}_A) = A/(\sigma-1)A = A_{\mathbb{Z}}$,
        \item $H_1(\mathbb{Z}, A) \cong \ker((\sigma-1)\otimes \operatorname{id}_A) = \ker(\sigma-1) = A^{\mathbb{Z}}$,
        \item $H_n(\mathbb{Z}, A) = 0$ for $n \geq 2$ because the resolution has length $1$.
    \end{itemize}

    \textbf{Cohomology.} Applying $\Hom_{R[\mathbb{Z}]}(-, A)$ to the resolution (without the $R$ term) gives the complex
    $$0 \to \Hom_{R[\mathbb{Z}]}(R[\mathbb{Z}], A) \xrightarrow{(\sigma-1)^*} \Hom_{R[\mathbb{Z}]}(R[\mathbb{Z}], A) \to 0$$
    Identifying $\Hom_{R[\mathbb{Z}]}(R[\mathbb{Z}], A)$ with $A$ via $f \mapsto f(1)$, the map $(\sigma-1)^*$ becomes left multiplication by $\sigma-1$ on $A$. With the rightmost copy of $A$ in degree $0$, the cohomology yields:
    \begin{itemize}
        \item $H^0(\mathbb{Z}, A) \cong \ker(\sigma-1) = A^{\mathbb{Z}}$,
        \item $H^1(\mathbb{Z}, A) \cong A/\operatorname{im}(\sigma-1) = A/(\sigma-1)A = A_{\mathbb{Z}}$,
        \item $H^n(\mathbb{Z}, A) = 0$ for $n \geq 2$ because the resolution has length $1$.
    \end{itemize}
\end{proof}
