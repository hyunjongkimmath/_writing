\begin{proposition}[Projection formula for six functor formalism with algebraic adic coefficients] \label{proposition:projection_formula_for_six_functor_formalism_on_derived_category_of_bdd_constr_complexes_with_algebraic_coefficients}
Let $S$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme}. Let $f: X \to Y$ be a \CrefAndHyperrefIfExist{definition:compactification_of_a_morphism_of_schemes}{compactifiable $S$-morphism} between $S$-schemes where $Y$ is \CrefAndHyperrefIfExist{definition:quasi_compact_morphism_of_schemes}{quasi-compact} and \CrefAndHyperrefIfExist{definition:quasi_separated_morphism_of_schemes_and_separated_morphism_of_schemes}{quasi-separated} over $S$ and where $X$ and $Y$ are \CrefAndHyperrefIfExist{definition:locally_noetherian_and_noetherian_scheme}{noetherian schemes}.
Let $(R, \lambda)$ be a \CrefAndHyperrefIfExist{definition:complete_ring_with_respect_to_an_ideal}{complete} \CrefAndHyperrefIfExist{definition:discrete_valuation_ring}{DVR} such that $R/\lambda^n$ is finite for all $n \geq 1$, let $E$ be the \CrefAndHyperrefIfExist{definition:field_of_fractions_of_an_integral_domain}{fraction field} of $R$, and $F$ be an \CrefAndHyperrefIfExist{definition:algebraic_element_minimal_polynomial_of_an_algebraic_element_algebraic_field_extension_transcendental_element_field_extension}{algebraic extension} of $E$. 

For $K \in \Dbc(X,F)$ and $L \in \Dbc(Y,F)$ (\Cref{definition:six_functor_formalism_on_derived_category_of_bdd_constr_complexes_with_algebraic_coefficients_on_a_noetherian_scheme}), there is a canonical isomorphism in $\Dbc(Y,F)$:
$$Rf_!(K \otimes_F^L Lf^* L) \cong Rf_! K \otimes_F^L L$$
\CrefIfExists{definition:derived_direct_image_functor_with_proper_or_compact_support_for_compactifiable_morphisms_of_schemes_over_a_scheme}\CrefIfExists{definition:derived_tensor_product_on_bdd_constructible_complexes_with_algebraic_coefficients_on_a_noetherian_scheme}\TODO{do total inverse image for algebraic coefficients}\CrefIfExists{definition:total_left_derived_inverse_image_functor_for_morphism_of_ringed_sites}

In particular, if $f$ is \CrefAndHyperrefIfExist{definition:proper_morphism_of_schemes}{proper}, then $Rf_! = Rf_*$ and hence the projection formula takes the form
$$Rf_*(K \otimes_F^L Lf^* L) \cong Rf_* K \otimes_F^L L.$$
\end{proposition}
