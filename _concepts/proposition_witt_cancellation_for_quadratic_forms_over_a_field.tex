\begin{proposition}[Witt cancellation] \label{proposition:witt_cancellation_for_quadratic_forms_over_a_field}
    Let $F$ be a \CrefAndHyperrefIfExist{definition:field}{field}, and let $(M_1, q_1)$ and $(M_2, q_2)$ be finite-dimensional quadratic modules over $F$. If there exists an isometry between $(M_1, q_1) \oplus (H, q_H)$ and $(M_2, q_2) \oplus (H, q_H)$ for some \CrefAndHyperrefIfExist{definition:hyperbolic_plane_over_a_commutative_ring}{hyperbolic plane} $(H, q_H)$, then there exists an isometry between $(M_1, q_1)$ and $(M_2, q_2)$.
\end{proposition}
