\begin{theorem}[{\cite[Main Theorem]{tate_eavff}}] \label{theorem:tates_isogeny_theorem_for_abelian_varieties_over_finite_fields}
    Let $k$ be a \CrefAndHyperrefIfExist{definition:finite_field}{finite field} and let $A, B$ be \CrefAndHyperrefIfExist{definition:abelian_variety_over_a_field}{abelian varieties} over $k$. Let $\ell \neq \Char k$ be a \CrefAndHyperrefIfExist{definition:prime_number}{prime number}.
    The \CrefAndHyperrefIfExist{definition:ell_adic_representation_map_from_base_change_of_isogenies_of_abelian_varieties_to_hom_set_of_tate_modules}{canonical map} 
    $$\mathbb{Z}_\ell \otimes_{\mathbb{Z}} \Hom(A, B) \to \Hom_{\bbZ_\ell}(T_\ell A, T_\ell B)$$
    is a bijection.
\end{theorem}
