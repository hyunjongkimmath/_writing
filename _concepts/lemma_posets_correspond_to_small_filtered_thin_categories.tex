\begin{lemma} \label{lemma:posets_correspond_to_small_filtered_thin_categories}
    Let $(P, \leq)$ be a nonempty \CrefAndHyperrefIfExist{definition:partially_ordered_set}{poset}. 
    \begin{enumerate}
        \item Regarding $P$ as a category whose objects are the elements of $P$ and such that there is a unique arrow $a \to b$ if and only if $a \leq b$, the category is filtered. 
        \item Every nonempty \CrefAndHyperrefIfExist{definition:locally_small_category}{small} \CrefAndHyperrefIfExist{definition:thin_category}{thin} \CrefAndHyperrefIfExist{definition:filtered_cofiltered_category}{filtered category} corresponds to a poset in this way.
        \item Moreover, the poset $P$ is \CrefAndHyperrefIfExist{definition:partially_ordered_set}{directed} if and only if the category is filtered. The poset $P$ is \CrefAndHyperrefIfExist{definition:partially_ordered_set}{codirected} if and only if the category is cofiltered.
    \end{enumerate}
\end{lemma}
