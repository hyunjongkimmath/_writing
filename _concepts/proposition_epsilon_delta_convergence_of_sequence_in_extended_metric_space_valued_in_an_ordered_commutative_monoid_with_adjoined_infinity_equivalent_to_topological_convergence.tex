\begin{proposition} \label{proposition:epsilon_delta_convergence_of_sequence_in_extended_metric_space_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity_equivalent_to_topological_convergence}
    Let $T$ be an \CrefAndHyperrefIfExist{definition:ordered_commutative_monoid}{ordered commutative monoid}, let $T^\infty$ be the \CrefAndHyperrefIfExist{definition:ordered_monoid_with_adjoined_infinity}{ordered monoid with adjoined infinity} obtained from $T$, and let $(X,d)$ be an \CrefAndHyperrefIfExist{definition:extended_metric_on_a_set_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity}{extended metric space valued in $T^\infty$}. Let $\mathcal{T}_d$ be the \CrefAndHyperrefIfExist{definition:topology_induced_by_an_extended_metric_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity_and_open_ball_for_an_extended_metric}{topology induced by $d$} on $X$. Let $(x_n)_{n=1}^\infty$ be a sequence in $X$, and let $x \in X$.

    The sequence $(x_n)$ converges to $x$ in the sense of \Cref{definition:convergence_and_limit_of_sequence_in_extended_metric_space_valued_in_ordered_commutative_monoid_with_adjoined_infinity} if and only if it converges to $x$ in the topological sense of \Cref{definition:convergence_and_limit_of_sequence_in_a_topological_space} with respect to $\mathcal{T}_d$.
\end{proposition}

\begin{proof}
    Let $(X,d)$ be an extended metric space valued in $T^\infty$, let $\mathcal{T}_d$ be the topology induced by $d$, let $(x_n)_{n=1}^\infty$ be a sequence in $X$, and let $x \in X$. We prove the two implications separately.

    Suppose that $(x_n)$ converges to $x$ in the sense of \Cref{definition:convergence_and_limit_of_sequence_in_extended_metric_space_valued_in_ordered_commutative_monoid_with_adjoined_infinity}. Let $U$ be an open neighborhood of $x$ with respect to $\mathcal{T}_d$. By \Cref{definition:topology_induced_by_an_extended_metric_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity_and_open_ball_for_an_extended_metric}, since $U$ is open and $x \in U$, there exists $\varepsilon \in T_{>0}$ such that the open ball $B(x,\varepsilon) = \{y \in X : d(x,y) < \varepsilon\}$ satisfies $B(x,\varepsilon) \subseteq U$. Since $(x_n)$ converges to $x$ in the sense of \Cref{definition:convergence_and_limit_of_sequence_in_extended_metric_space_valued_in_ordered_commutative_monoid_with_adjoined_infinity}, there exists $N \in \mathbb{N}$ such that for all $n \geq N$, $d(x_n, x) < \varepsilon$. For each such $n$, we have $x_n \in B(x,\varepsilon)$ because $d(x_n,x) < \varepsilon$. Since $B(x,\varepsilon) \subseteq U$, it follows that $x_n \in U$ for all $n \geq N$. Therefore, $(x_n)$ converges to $x$ in the topological sense of \Cref{definition:convergence_and_limit_of_sequence_in_a_topological_space}.

    Conversely, suppose that $(x_n)$ converges to $x$ in the topological sense of \Cref{definition:convergence_and_limit_of_sequence_in_a_topological_space} with respect to $\mathcal{T}_d$. Let $\varepsilon \in T_{>0}$. The open ball $B(x,\varepsilon) = \{y \in X : d(x,y) < \varepsilon\}$ is an open set in $\mathcal{T}_d$ by \Cref{definition:topology_induced_by_an_extended_metric_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity_and_open_ball_for_an_extended_metric}, and it contains $x$. By topological convergence, there exists $N \in \mathbb{N}$ such that for all $n \geq N$, $x_n \in B(x,\varepsilon)$. For each such $n$, we have $d(x_n, x) < \varepsilon$ because $x_n \in B(x,\varepsilon)$. Therefore, $(x_n)$ converges to $x$ in the $\varepsilon$-$\delta$ sense of \Cref{definition:convergence_and_limit_of_sequence_in_extended_metric_space_valued_in_ordered_commutative_monoid_with_adjoined_infinity}.
\end{proof}
