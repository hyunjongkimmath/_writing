\begin{lemma} \label{lemma:identity_element_of_a_binary_operation_on_a_set_is_unique_if_exists}
    % EDITOR APPROVED
    Let $S$ be a \CrefAndHyperrefIfExist{definition:zermelo_fraenkel_set_theory}{set}. Let $\cdot: S \times S \to S$ be a \CrefAndHyperrefIfExist{definition:n_ary_operation_on_a_set}{binary operator} on $S$. Assuming that there exists an \CrefAndHyperrefIfExist{definition:left_right_identity_element_of_a_binary_operation_on_a_set}{identity element} of the binary operator $\cdot$, it is unique. More precisely, if $e_1$ and $e_2$ are two identity elements of $\cdot$, then $e_1 = e_2$.  
\end{lemma}
\begin{proof}
    % EDITOR APPROVED
    Suppose that $e_1$ and $E_2$ are identity elements of $\cdot$. Since $e_1$ is in particular a left identity element of $\cdot$, we have that $e_1 \cdot e_2 = e_2$.  
    Since $e_2$ is a right identity element of $\cdot$, we have that $e_1 \cdot e_2 = e_1$. Therefore, $e_1 = e_2$ as desired.
\end{proof}
