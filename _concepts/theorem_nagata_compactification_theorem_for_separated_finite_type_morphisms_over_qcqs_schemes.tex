\begin{theorem}[Nagata's compactification theorem, cf. {\cite[Tag 0F41]{stacks_project}}] \label{theorem:nagata_compactification_theorem_for_separated_finite_type_morphisms_over_qcqs_schemes}
Let $S$ be a \CrefAndHyperrefIfExist{definition:quasi_compact_and_quasi_separated_scheme}{quasi-compact and quasi-separated scheme} and let $f: X \to S$ be a \CrefAndHyperrefIfExist{definition:separated_morphism_of_schemes}{separated} and of \CrefAndHyperrefIfExist{definition:finite_type_morphism_of_schemes}{finite type} morphism of schemes. Then, $f$ admits a \CrefAndHyperrefIfExist{definition:compactification_of_a_morphism_of_schemes}{compactification}
$$
\begin{tikzcd}
X \ar[rr, "f"] \ar[dr, "j"'] & & S \\
& \bar{X} \ar[ur, "\bar{f}"'] &.
\end{tikzcd}
$$
\end{theorem}
