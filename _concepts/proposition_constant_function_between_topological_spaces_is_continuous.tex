\begin{proposition} \label{proposition:constant_function_between_topological_spaces_is_continuous}
    Let $X$ and $Y$ be \CrefAndHyperrefIfExist{definition:topological_space}{topological spaces} and let $c \in Y$. The \CrefAndHyperrefIfExist{definition:constant_function_on_a_set}{constant function} $f \equiv c : X \to Y$ is \CrefAndHyperrefIfExist{definition:continuous_map_of_topological_spaces}{continuous}.
\end{proposition}
\begin{proof}
    Let $V \subseteq Y$ be an open set. If $c \in V$, then $f^{-1}(V) = X$, which is open in $X$. If $c \notin V$, then $f^{-1}(V) = \emptyset$, which is also open in $X$. In both cases, the preimage of an open set is open, so $f$ is continuous.
\end{proof}
