\begin{lemma} \label{lemma:zeroth_homotopy_set_of_a_topological_space_at_any_point_corresponds_to_path_components}
    Let $X$ be a topological space. Let $x_0 \in X$. 
    The set \CrefAndHyperrefIfExist{definition:homotopy_groups_of_a_pointed_topological_space}{$\pi_0(X,x_0)$} bijectively corresponds to the set of \CrefAndHyperrefIfExist{definition:path_connected_topological_space}{path connected} components of $X$.  
\end{lemma}
