\begin{theorem}[Universal coefficient theorem for homology, see e.g. {\cite[Theorem 3.6.1, 3.6.2, Exercise 3.6.2]{weibel}}] \label{theorem:universal_coefficient_theorem_for_homology_for_modules_over_a_ring}
Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{(not necessarily commutative) ring}. Let $P$ be a \CrefAndHyperrefIfExist{definition:chain_complex_of_objects_in_an_additive_category}{chain complex} of \CrefAndHyperrefIfExist{definition:flat_left_right_module_over_a_ring}{flat right $R$ modules} such that each \CrefAndHyperrefIfExist{definition:submodule_of_a_module_over_a_ring}{submodule} $d\left(P_{n}\right)$ of $P_{n-1}$ is also flat. Then for every $n$ and every \CrefAndHyperrefIfExist{definition:module_of_a_ring}{left $R$-module} $M$, there is an \CrefAndHyperrefIfExist{definition:short_exact_sequence_in_an_additive_category}{exact sequence}
$$ 0 \rightarrow H_{n}(P) \otimes_{R} M \rightarrow H_{n}\left(P \otimes_{R} M\right) \rightarrow \operatorname{Tor}_{1}^{R}\left(H_{n-1}(P), M\right) \rightarrow 0 $$
\CrefIfExists{definition:tor_functors_of_bimodules_of_rings} \CrefIfExists{definition:homology_and_cohomology_objects_for_a_chain_complex_in_an_additive_category}

\end{theorem}
