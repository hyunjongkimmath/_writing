\begin{theorem} \label{theorem:cauchy_integral_theorem_for_contour_integrals_of_holomorphic_functions_on_simply_connected_domains}
Let $U \subset \mathbb{C}$ be a \CrefAndHyperrefIfExist{definition:simply_connected_space}{simply connected} open subset (a simply connected domain) and let $f : U \to \mathbb{C}$ be a \CrefAndHyperrefIfExist{definition:holomorphic_function_from_a_subset_of_Cn_to_Cm}{holomorphic function} on $U$. Then for every closed \CrefAndHyperrefIfExist{definition:piecewise_continuously_differentiable_path_from_a_closed_real_interval_to_the_complex_plane}{piecewise continuously differentiable path} $\gamma$ in $U$,
$\hlin{\int_{\gamma} f(z)\,dz = 0}$.
(\Cref{definition:contour_integral_of_a_complex_valued_function_along_a_piecewise_continuously_differentiable_path})
\end{theorem}
