\begin{lemma}[{\cite[Chapter I, 2.3]{knus_qhfr}}] \label{lemma:ad_of_conj_trans_of_sesq_form_over_module_on_a_involution_ring_is_composition_of_canonical_map_and_twisted_dual_of_adjoint}
    Let $(A, \sigma)$ be a \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{ring with involution} where $\sigma$ is also denoted by $\overline{\cdot}$, let $\varepsilon \in Z(A)$ be an element such that $\varepsilon \cdot \overline{\varepsilon} = 1$, let $M$ be a \CrefAndHyperrefIfExist{definition:bimodule_over_rings_module_over_a_ring}{right $A$-module}, and let $b: M \times M \to A$ be a \CrefAndHyperrefIfExist{definition:sesquilinear_form_on_a_module_over_a_ring_with_involution}{sesquilinear form} on $M$.

    We have
    $$h_{b^*} = (h_b)^* \circ \operatorname{can}_M$$
    where \CrefAndHyperrefIfExist{definition:conjugate_transpose_sesquilinear_form_of_a_sesquilinear_form_on_a_module_over_a_ring_with_involution}{$b^*$} is the conjugate transpose of $b$, \CrefAndHyperrefIfExist{definition:adjoint_map_of_a_sesquilinear_form_on_a_module_over_a_ring_with_involution}{$h_b$} is the adjoint of $b$, \CrefAndHyperrefIfExist{definition:canonical_morphism_from_a_module_over_an_involution_ring_to_its_double_dual}{$\operatorname{can}_M$} is the canonical map from $M$ to its double twisted dual, and $h_b^*$ is the map 
    $$h_b^*: M^{**} \to M^{*}$$
    induced from $h_b: M \to M^{*}$ by the contravariance of the \CrefAndHyperrefIfExist{definition:twisted_dual_module_of_a_module_over_a_ring_with_involution}{twisted dual} functor $(-)^*$. 
\end{lemma}
