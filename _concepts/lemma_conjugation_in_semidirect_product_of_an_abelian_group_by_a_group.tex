\begin{lemma} \label{lemma:conjugation_in_semidirect_product_of_an_abelian_group_by_a_group}
    Let $G$ be a group and let $A$ be a $\bbZ[G]$-module. Let $(a_1, g_1), (a_2,g_2) \in A \rtimes G$ be elements.
    The conjugattion
    $$(a_1, g_1) \cdot (a_2,g_2) \cdot (a_1,g_1)^{-1}$$
    equals
    $$((1-g_1g_2g_1^{-1}) \cdot a_1 + g_1 \cdot a_2, g_1g_2g_1^{-1}).$$
\end{lemma}
\begin{proof}
    This holds by direct computation.
\end{proof}
