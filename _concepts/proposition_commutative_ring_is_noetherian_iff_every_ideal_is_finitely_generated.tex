\begin{proposition} \label{proposition:commutative_ring_is_noetherian_iff_every_ideal_is_finitely_generated}
If $R$ is a \CrefAndHyperrefIfExist{definition:commutative_ring}{commutative ring}, then $R$ is \CrefAndHyperrefIfExist{definition:noetherian_ring}{Noetherian} if and only if every \CrefAndHyperrefIfExist{definition:ideal_of_a_ring}{ideal} is \CrefAndHyperrefIfExist{definition:finitely_generated_modules_over_rings}{finitely generated}.
\end{proposition}
