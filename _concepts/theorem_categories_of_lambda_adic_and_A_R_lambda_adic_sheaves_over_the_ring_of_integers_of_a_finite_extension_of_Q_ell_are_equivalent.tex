\begin{theorem}{See e.g. {\cite[10.1]{fu_ect}}} \label{theorem:categories_of_lambda_adic_and_A_R_lambda_adic_sheaves_over_the_ring_of_integers_of_a_finite_extension_of_Q_ell_are_equivalent}
    Let $X$ be a noetherian scheme. Let $\ell$ be a prime number that is invertible on $X$. Let $R$ be the integral closure of $\bbZ_\ell$ in a finite extension $E$ of $\Qell$. Note that $R$ is a complete discrete valuation ring and let $\lambda$ be the maximal ideal of $R$. The category of \CrefAndHyperrefIfExist{definition:lambda_adic_sheaf_on_the_small_etale_site_of_a_scheme_for_an_ideal_of_a_commutative_ring}{$\lambda$-adic sheaves} is equivalent to the category of \CrefAndHyperrefIfExist{definition:lambda_adic_sheaf_on_the_small_etale_site_of_a_scheme_for_an_ideal_of_a_commutative_ring}{A-R $\lambda$-adic sheaves}.
\end{theorem}
