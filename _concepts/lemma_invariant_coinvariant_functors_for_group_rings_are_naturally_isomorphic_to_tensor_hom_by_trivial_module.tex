\begin{lemma}[cf. {\cite[Lemma 6.1.1]{weibel}}] \label{lemma:invariant_coinvariant_functors_for_group_rings_are_naturally_isomorphic_to_tensor_hom_by_trivial_module}
    Let $G$ be a \CrefAndHyperrefIfExist{definition:group}{group} and $R$ a \CrefAndHyperrefIfExist{definition:ring}{not necessarily commutative ring}. Write $R[G]$ for the \CrefAndHyperrefIfExist{definition:group_ring_of_a_group_over_a_ring}{group ring} and let $A$ be a \CrefAndHyperrefIfExist{definition:left_right_module_over_a_group_ring_of_a_group_over_a_ring}{left $R[G]$-module}. 

    Consider $R$ as a \CrefAndHyperrefIfExist{definition:trivial_G_module_of_a_group_over_a_ring}{trivial $G$-module over $R$}; one may in fact obtain a \CrefAndHyperrefIfExist{definition:bimodule_over_rings_module_over_a_ring}{two-sided $R[G]$ module structure} in doing so. We have \CrefAndHyperrefIfExist{definition:natural_transformation_between_functors_between_categories}{natural isomorphisms}
    $$A_G \cong R \otimes_{R[G]} A$$
    (\Cref{definition:coinvariant_quotient_group_of_a_group_acting_on_an_abelian_group}) \CrefIfExists{definition:tensor_product_of_bimodules_of_rings}
    of left $R$-modules by considering $R$ as an $R$-$R[G]$-bimodule and 
    $$A^G \cong \Hom_{R[G]}(R, A)$$
    (\Cref{definition:invariant_subgroup_of_a_group_acting_on_an_abelian_group}) \CrefIfExists{definition:hom_of_left_right_bi_modules_of_rings}
    of left $R$-modules by considering $R$ as an $R[G]$-$R$-bimodule.
\end{lemma}

\begin{proof}
    \begin{enumerate}
        \item To show $A_G \cong R \otimes_{R[G]} A$, consider the $R$-linear map 
        $\psi: A \to R \otimes_{R[G]} A$ defined by $a \mapsto 1 \otimes a$. 
        For any $g \in G$ and $a \in A$, the tensor relation over $R[G]$ gives:
        $$1 \otimes (ga) = (1 \cdot g) \otimes a = 1 \otimes a$$
        since $R$ is a trivial right $R[G]$-module. Thus, $1 \otimes (ga - a) = 0$, so $\psi$ factors through the quotient $A_G = A / \langle ga - a \rangle$. 
        The inverse is given by the map $R \otimes_{R[G]} A \to A_G$ sending $r \otimes a \mapsto r[a]$, which is well-defined by the same calculation. 
        This is clearly natural in $A$.

        \item To show $A^G \cong \Hom_{R[G]}(R, A)$, consider the evaluation map:
        $$\text{ev}_1: \Hom_{R[G]}(R, A) \to A, \quad f \mapsto f(1)$$
        Since $f$ is $R[G]$-linear and $R$ is a trivial left $R[G]$-module, we have:
        $$f(1) = f(g \cdot 1) = g \cdot f(1)$$
        for all $g \in G$. This implies $\text{ev}_1(f) \in A^G$. 
        Conversely, any $a \in A^G$ defines an $R[G]$-linear map $f_a: R \to A$ via $f_a(r) = ra$. This is $G$-equivariant because $f_a(g \cdot r) = f_a(r) = ra$ and $g \cdot f_a(r) = g(ra) = r(ga) = ra$ (since $ga=a$). 
        The maps $f \mapsto f(1)$ and $a \mapsto f_a$ are inverse $R$-module isomorphisms and are natural in $A$.
    \end{enumerate}
\end{proof}
