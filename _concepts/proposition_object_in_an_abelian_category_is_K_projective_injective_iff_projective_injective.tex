\begin{proposition}[{\cite[1.2. Proposition]{spaltenstein_ruc}}] \label{proposition:object_in_an_abelian_category_is_K_projective_injective_iff_projective_injective}
    Let $\calA$ be an \CrefAndHyperrefIfExist{definition:abelian_category}{abelian category} and let $A^\bullet \in \mathbf{Ch}(\calA)$ be a \CrefAndHyperrefIfExist{definition:chain_complex_of_objects_in_a_preadditive_category}{complex} such that $A^i = 0$ for all $i \neq 0$. 
    The \CrefAndHyperrefIfExist{definition:chain_complex_of_objects_in_a_preadditive_category}{complex} $A^\bullet$ is $K$-\CrefAndHyperrefIfExist{definition:injective_and_projective_objects_in_a_category}{projective} (resp. $K$-\CrefAndHyperrefIfExist{definition:injective_and_projective_objects_in_a_category}{injective}) if and only if $A^0$ is a projective (resp. injective) object of $\mathcal{A}$.
\end{proposition}
