\begin{theorem} \label{theorem:fiber_functor_on_a_galois_category_induces_equivalence_of_categories_to_category_of_finite_groups_with_continuous_action_by_fundamental_group}
Let $(\mathcal{C}, F)$ be a \CrefAndHyperrefIfExist{definition:fiber_functor_for_a_functor_from_a_galois_category_to_the_category_of_finite_sets}{Galois category}. Let $G = \pi(\calC, F)$ be the \CrefAndHyperrefIfExist{definition:fundamental_group_of_a_galois_category}{fundamental group} of $(\calC, F)$. 

Let $\text{F-}G\textbf{-Set}$ denote the category of finite sets equipped with a continuous action of $G$. The functor $F$ induces an \CrefAndHyperrefIfExist{definition:equivalence_of_categories}{equivalence of categories}
$$ \Phi: \mathcal{C} \xrightarrow{\sim} \text{F-}G\textbf{-Set} $$
defined by mapping an object $X \in \mathcal{C}$ to the set $F(X)$ with the natural $G$-action.
\end{theorem}
