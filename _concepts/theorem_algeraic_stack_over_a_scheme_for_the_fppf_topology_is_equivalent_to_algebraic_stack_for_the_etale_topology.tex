\begin{theorem}[See {\cite[Tag 076V]{stacks-project}}] \label{theorem:algeraic_stack_over_a_scheme_for_the_fppf_topology_is_equivalent_to_algebraic_stack_for_the_etale_topology}
    Let $S$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme}. 
    Every \CrefAndHyperrefIfExist{definition:artin_stack_over_a_scheme_for_a_grothendieck_topology_on_the_category_of_S_schemes}{algebraic stack} for the \CrefAndHyperrefIfExist{definition:big_etale_site_of_a_scheme}{(big) \'etale topology} on \CrefAndHyperrefIfExist{definition:scheme_over_a_scheme}{$\mathit{Sch}/S$} is an algebraic stack for the \CrefAndHyperrefIfExist{definition:big_fppf_site_of_a_scheme}{(big) \'etale topology} on $\mathit{Sch}/S$.
\end{theorem}
