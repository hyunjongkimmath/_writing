\begin{theorem} \label{theorem:artin_wedderburn_theorem_semisimple_ring_classification.tex}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:semisimple_ring}{semisimple ring}. 

    \TODO{matrix ring}
    It is \CrefAndHyperrefIfExist{definition:ring_homomorphism}{isomorphic} to the \CrefAndHyperrefIfExist{definition:products_of_rings}{product} of finitely many $n_i \times n_i$ matrix rings over \CrefAndHyperrefIfExist{definition:division_ring}{division rings} $D_i$ for some integers $n_i$, both of which are uniquely determined up to \CrefAndHyperrefIfExist{definition:symmetric_group_on_a_set}{permutation} of the index $i$. 

    In particular, $R$ must be an \CrefAndHyperrefIfExist{definition:left_right_artinian_ring}{Artinian ring}.
\end{theorem}
