\begin{theorem}[Maschke's Theorem] \label{theorem:maschkes_theorem_equiv_conditions_on_semisimplicity_of_group_algebra_of_finite_group_over_field}

Let $G$ be a finite group and let $k$ be a field. The following are equivalent:
\begin{enumerate}
    \item[(i)] $\operatorname{char}(k)$ does not divide $|G|$ (or $\operatorname{char}(k) = 0$).
    \item[(ii)] Every \CrefAndHyperrefIfExist{definition:representation_of_a_group_on_a_module_over_a_not_necessarily_commutative_ring}{representation of $G$ on a left $k$-module} is a \CrefAndHyperref{definition:semisimple_representation_of_a_group_on_a_left_module_over_a_ring}{semisimple representation}.
    \item[(iii)] The \CrefAndHyperrefIfExist{definition:group_ring_of_a_group_over_a_ring}{group algebra} $k[G]$ is a \CrefAndHyperrefIfExist{definition:semisimple_ring}{semisimple ring}.
\end{enumerate}
Equivalently, under condition (i), every \CrefAndHyperrefIfExist{definition:subrepresentation_of_a_representation_of_a_group_on_a_left_module_on_a_ring}{subrepresentation} of a representation of $G$ on a left $k$-module admits a complement.
\end{theorem}
