\begin{theorem}[{\cite[Corollary 1.3.]{kilicer_streng_cmcnopcg2}}] \label{theorem:kilicer_streng_cmcnopcg2_classification_of_genus_2_curves_with_CM_defined_over_reflex_field_for_non_biquadratic_quartic_cm_field}
    The set 
    $$\{\bbQ\text{-isomorphism classes of smooth projective  geometrically irreducible algebraic curves} C/\overline{\bbQ} : C \text{ has CM by } \calO_K \text{ for some non-biquadratic quartic CM field } K \text{ and } K \text{ is the reflex field of } \Jac(C)  \} / \sim$$
    is of cardinality $125$. 
\end{theorem}
