\begin{corollary} \label{corollary:eulers_number_equals_series_of_reciprocals_of_integer_factorials}
    $$e = \sum_{n=0}^\infty \frac{1}{n!}$$
    \CrefIfExists{definition:euler_number_e_as_limit}\CrefIfExists{definition:factorial_of_a_nonnegative_integer}
\end{corollary}
\begin{proof}
    Recall that\CrefIfExists{proposition:real_exponential_function_taylor_series} 
    $$\exp(x) = \sum_{n=0}^\infty \frac{1}{n!} x^n.$$
    for all $x \in \bbR$. Letting $x = 1$, we have 
    $$\exp(1) = \sum_{n=0}^\infty \frac{1}{n!}.$$
    By definition, $\exp(1) = e^1 = e$\CrefIfExists{definition:real_exponential_function}, so the desired equality holds.
\end{proof}
