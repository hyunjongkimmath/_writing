\begin{lemma} \label{lemma:class_equation_for_finite_groups}
Let $G$ be a \CrefAndHyperrefIfExist{definition:countable_finite_uncountable_sets}{finite} \CrefAndHyperrefIfExist{definition:group}{group}. Let $Z(G)$ denote the \CrefAndHyperrefIfExist{definition:center_of_a_group}{center of $G$}. Then
$$|G| = |Z(G)| + \sum_{i} [G : C_G(g_i)],$$
where the sum runs over a set of representatives $\{g_i\}$ for the non-central conjugacy classes of $G$, and $C_G(g_i)$ denotes the \CrefAndHyperrefIfExist{definition:centralizer_of_a_subgroup_of_a_group}{centralizer} of $g_i$ in $G$.
\end{lemma}
