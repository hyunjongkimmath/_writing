\begin{proposition}[Nakayama's lemma for graded modules]  \label{proposition:nakayamas_lemma_for_finitely_generated_graded_modules_over_a_non_negatively_graded_ring}

Let $R = \bigoplus_{d \geq 0} R_d$ be a \hldef{non-negatively graded ring} with $R_0$ a field, and let $M = \bigoplus_{d \geq 0} M_d$ be a \hldef{finitely generated graded right $R$-module}, meaning $M$ is generated as an $R$-module by finitely many homogeneous elements. Let $R_{>0} = \bigoplus_{d > 0} R_d$.

\begin{enumerate}
    \item If $R_{>0}M = M$, then $M = 0$. Equivalently, if $M/R_{>0}M = 0$, then $M = 0$.

    \item The quotient $M/R_{>0}M$ is a finite-dimensional right vector space over $R_0$. If $x_1 + R_{>0}M, \ldots, x_n + R_{>0}M \in M/R_{>0}M$ form a basis as an $R_0$-vector space, then homogeneous lifts $x_1, \ldots, x_n \in M$ generate $M$ as an $R$-module.

    \item The minimum number of homogeneous generators of $M$ equals $\dim_{R_0}(M/R_{>0}M)$.
\end{enumerate}
\end{proposition}
