\begin{theorem}[Integration by parts for Riemann integrals] \label{theorem:integration_by_parts_for_riemann_integrals_on_closed_intervals}
    Let $[a, b] \subseteq \mathbb{R}$\CrefIfExists{definition:field_of_real_numbers} be a closed bounded interval with $a < b$. Let $f, g : [a, b] \to \mathbb{R}$ be \CrefAndHyperrefIfExist{definition:function_of_sets}{functions}.

    Suppose that:
    \begin{enumerate}
        \item Both $f$ and $g$ are \CrefAndHyperrefIfExist{definition:derivative_of_a_real_valued_function_on_a_real_interval}{differentiable} at each point of $(a, b)$;
        \item The derivatives $f'$ and $g'$ are \CrefAndHyperrefIfExist{definition:lower_upper_riemann_sum_integral_of_a_bounded_function_on_a_closed_interval_to_R_and_riemann_integrable_function}{Riemann integrable on $[a, b]$}.
    \end{enumerate}
    Then
    $$\int_a^b f(x)\,g'(x)\,dx = \Bigl[f(x)\,g(x)\Bigr]_{x=a}^{x=b} - \int_a^b f'(x)\,g(x)\,dx.$$
\end{theorem}
