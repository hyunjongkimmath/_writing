\begin{proposition} \label{proposition:ell_adic_tate_module_for_a_cm_abelian_variety_whose_cm_field_is_maximal_degree_is_locally_free_rank_1_module_over_rational_endomorphism_algebra}
    Let $A/K$ be a \CrefAndHyperrefIfExist{definition:complex_multiplication_of_an_abelian_variety_over_a_field}{CM abelian variety} of \CrefAndHyperrefIfExist{definition:dimension_of_a_scheme}{dimension} $g$ over a \CrefAndHyperrefIfExist{definition:field}{field} with complex multiplication by a \CrefAndHyperrefIfExist{definition:cm_field}{CM field} $E$ and assume that $E$ is of dimension $2g$, i.e. $\operatorname{End}_K(A) \otimes \bbQ = E$\CrefIfExists{definition:endomorphism_algebra_up_to_isogeny_of_an_abelian_variety_over_a_field}. Let $\ell$ be a prime number. 
    % Write $\calO_E$ for the ring of integers of $E$. 
    \begin{enumerate}
        \item The \CrefAndHyperrefIfExist{definition:ell_adic_tate_module_of_an_abelian_variety_over_a_field}{rational $\ell$-adic Tate module} $V_\ell(A)$ is a \CrefAndHyperrefIfExist{definition:free_bimodule_over_rings}{free} $E \otimes_\bbQ \bbQ_\ell$-module of rank $1$. 

        \item The \CrefAndHyperrefIfExist{definition:ell_adic_tate_module_of_an_abelian_variety_over_a_field}{integral $\ell$-adic Tate module} $T_\ell(A)$ is a locally free $\End_K(A) \otimes \bbZ_\ell$-module of rank $1$.

        In particular, if $\End_K(A) \cong \mathcal{O}_E$ where $\mathcal{O}_E$ is the ring of integers of $E$, then $T_\ell(A)$ is a free $\calO_E \otimes \bbZ_\ell$-module of rank $1$.

    \end{enumerate}
\end{proposition}
