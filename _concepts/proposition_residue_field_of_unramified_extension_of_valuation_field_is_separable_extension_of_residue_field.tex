\begin{proposition} \label{proposition:residue_field_of_unramified_extension_of_valuation_field_is_separable_extension_of_residue_field}
    Let $(K,v)$ be a \CrefAndHyperrefIfExist{definition:valuation_on_a_field}{valuation field}. 
    If $(L,w)$ is a \CrefAndHyperrefIfExist{definition:extension_of_a_valuation_of_a_field_to_an_extension_field}{extension valuation field} of $(K,v)$, then the natural \CrefAndHyperrefIfExist{definition:field_homomorphism}{field embedding} $\kappa(v) \hookrightarrow \kappa(w)$ of \CrefAndHyperrefIfExist{definition:valuation_ring_of_a_nonarchimedean_valued_field}{residue fields} is a \CrefAndHyperrefIfExist{definition:separable_field_extension_general}{separable extension}. If $L/K$ is additionally \CrefAndHyperrefIfExist{definition:algebraic_element_minimal_polynomial_of_an_algebraic_element_algebraic_field_extension_transcendental_element_field_extension}{algebraic}, then $\kappa(w)/\kappa(v)$ is algebraic.
\end{proposition}
