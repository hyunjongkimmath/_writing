\begin{lemma} \label{lemma:change_of_base_for_logarithms}
    Let $a, b \in \mathbb{R}$ with $a > 0$, $b > 0$, and $a \neq 1$, $b \neq 1$. For all $x > 0$, the change-of-base formula holds:
    $$\log_b(x) = \frac{\log_a(x)}{\log_a(b)}.$$
    In particular, $\log_b(x) = \frac{\ln(x)}{\ln(b)}$\CrefIfExists{definition:natural_logarithm_as_inverse_of_exponential}.
\end{lemma}

    Applying the same definition to $\log_a(x)$ and $\log_a(b)$:
    $$\log_a(x) = \frac{\ln(x)}{\ln(a)}, \quad \log_a(b) = \frac{\ln(b)}{\ln(a)}.$$
    Therefore,
    $$\frac{\log_a(x)}{\log_a(b)} = \frac{\ln(x)/\ln(a)}{\ln(b)/\ln(a)} = \frac{\ln(x)}{\ln(b)} = \log_b(x).$$
