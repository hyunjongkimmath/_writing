\begin{proposition} \label{proposition:integral_scheme_has_a_valuation_centered_at_every_point_ideal}
    Let $X$ be an \CrefAndHyperrefIfExist{definition:integral_scheme}{integral scheme}. For any $x \in X$, there is at least one \CrefAndHyperrefIfExist{definition:valuation_on_a_field}{valuation} $v$ \CrefAndHyperrefIfExist{definition:centered_at_for_a_prime_of_an_integral_domain_point_of_an_integral_scheme}{centered at} $x$.
\end{proposition}
