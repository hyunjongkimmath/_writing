\begin{theorem}[Cohomological Künneth Formula over General Rings] \label{theorem:cohomological_kunneth_formula_for_complexes_of_modules_over_rings}
Let $R$ be any \CrefAndHyperrefIfExist{definition:ring}{ring}. Let $P$ and $Q$ be \CrefAndHyperrefIfExist{definition:chain_complex_of_objects_in_an_additive_category}{chain complexes} of \CrefAndHyperrefIfExist{definition:projective_bimodule_over_rings}{projective $R$-modules}. Assume that $P$ is a \CrefAndHyperrefIfExist{definition:chain_complex_of_objects_in_an_additive_category}{complex} of \CrefAndHyperrefIfExist{definition:finitely_generated_modules_over_rings}{finitely generated} \CrefAndHyperrefIfExist{definition:projective_bimodule_over_rings}{projective} \CrefAndHyperrefIfExist{definition:module_of_a_ring}{modules}.

If we assume that the \CrefAndHyperrefIfExist{definition:boundary_cycle_coboundary_cocyble_of_a_chain_cochain_complex}{boundaries} $d(P_n)$ are projective $R$-modules for all $n$, and that the \CrefAndHyperrefIfExist{definition:tor_functors_of_bimodules_of_rings}{higher Tor groups} $\operatorname{Tor}_k^R(H^i(P), H^j(Q)) = 0$ for all $k \geq 2$, then there is a natural \CrefAndHyperrefIfExist{definition:short_exact_sequence_in_an_additive_category}{short exact sequence}:
$$ 0 \rightarrow \bigoplus_{i+j=n} H^i(P) \otimes_R H^j(Q) \rightarrow H^n(P \otimes_R Q) \rightarrow \bigoplus_{i+j=n+1} \operatorname{Tor}_1^R(H^i(P), H^{j-1}(Q)) \rightarrow 0 $$

If $R$ is a \CrefAndHyperrefIfExist{definition:hereditary_ring}{hereditary ring}, the conditions on $d(P_n)$ and $\operatorname{Tor}_k$ are automatically satisfied, and the sequence \CrefAndHyperrefIfExist{definition:split_exact_sequence}{splits} non-canonically.
\end{theorem}
