\begin{proposition} \label{proposition:profinite_set_is_same_as_compact_hausdorff_totally_disconnected_set_and_stone_space}
Let $X$ be a \CrefAndHyperrefIfExist{definition:category_of_topological_spaces}{topological space}. The following conditions are equivalent:
\begin{itemize}
        \item $X$ is a \CrefAndHyperrefIfExist{definition:profinite_set}{profinite set}.
        \item $X$ is \CrefAndHyperrefIfExist{definition:compact_topological_space}{compact}, \CrefAndHyperrefIfExist{definition:separation_axioms_of_topology}{Hausdorff}, and \CrefAndHyperrefIfExist{definition:totally_disconnected_topological_space}{totally disconnected}.
        \item $X$ is a \CrefAndHyperrefIfExist{definition:stone_space}{Stone space}.
\end{itemize}
\end{proposition}
