\begin{lemma}[cf. {\cite[Lemma 3.4.1]{weibel}} for a discussion for modules] \label{lemma:if_first_ext_functor_in_an_abelian_category_is_zero_then_extensions_are_split}
    Let $\calA$ be an \CrefAndHyperrefIfExist{definition:abelian_category}{abelian category} with \CrefAndHyperrefIfExist{definition:has_enough_injectives_or_projectives_for_an_abelian_category}{enough injectives} or with \CrefAndHyperrefIfExist{definition:has_enough_injectives_or_projectives_for_an_abelian_category}{enough projectives}. Let $A,B \in \calA$.

    If \CrefAndHyperrefIfExist{definition:ext_functor_on_a_general_abelian_category}{$\Ext^1(A,B) = 0$}, then every \CrefAndHyperrefIfExist{definition:extension_of_objects_in_an_additive_category}{extension} of $A$ by $B$ is \CrefAndHyperrefIfExist{definition:extension_of_objects_in_an_additive_category}{split}.
\end{lemma}
