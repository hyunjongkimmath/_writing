\begin{definition} \label{theorem:category_of_unramified_extensions_of_a_valuation_field_is_equivalent_to_category_of_separable_extensions_residue_field}
    Let $(K,v)$ be a \CrefAndHyperrefIfExist{definition:henselian_field_with_respect_to_a_valuation}{Henselian field}. 
    The category of \CrefAndHyperrefIfExist{definition:degree_of_a_field_extension}{finite} \CrefAndHyperrefIfExist{definition:unramified_extension_of_valuation_fields}{unramified extensions} of $(K,v)$ is equivalent to the category of finite \CrefAndHyperrefIfExist{definition:separable_algebraic_field_extension}{separable} extensions of the \CrefAndHyperrefIfExist{definition:residue_field_of_a_ring_at_a_point}{residue field} of $(K,v)$.
\end{definition}
