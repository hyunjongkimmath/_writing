\begin{lemma}  \label{lemma:etale_fundamental_group_of_the_spectrum_of_a_field_is_absolute_galois_group}
    Let $k$ be a \CrefAndHyperrefIfExist{definition:field}{field}. 
    For any \CrefAndHyperrefIfExist{definition:geometric_point_of_a_scheme}{geometric point} $\overline{x}: \operatorname{Spec} \Omega \to \operatorname{Spec} k$\CrefIfExists{definition:affine_scheme}, which corresponds to a \CrefAndHyperrefIfExist{definition:field_homomorphism}{field embedding} $k \to \Omega$, the \CrefAndHyperrefIfExist{definition:etale_fundamental_group_of_a_connected_scheme}{\'etale fundamental group} $\pi_1(\operatorname{Spec} k, \overline{x})$ is naturally isomorphic to the \CrefAndHyperrefIfExist{definition:absolute_galois_group_of_a_field}{absolute Galois group} $\Gal(k^{\text{sep}} / k)$ where $k^{\text{sep}}$ is the \CrefAndHyperrefIfExist{definition:separable_closure_of_a_field_in_a_field_extension}{separable closure} of $k$ in $\Omega$. 
\end{lemma}
