\begin{proposition} \label{proposition:finite_etale_algebra_iff_finitely_presented_separable_flat}

    Let $R$ be a \CrefAndHyperref{definition:commutative_ring}{commutative ring}, and let $A$ be a commutative \CrefAndHyperref{definition:algebra_of_a_ring}{$R$-algebra}. The following are equivalent:

    \begin{enumerate}
        \item $A$ is a finite \CrefAndHyperrefIfExist{definition:etale_algebra_over_a_commutative_ring}{étale $R$-algebra}; that is, $A$ is étale over $R$ and finitely generated as an $R$-module.
        \item $A$ is \CrefAndHyperrefIfExist{definition:finitely_presented_algebra_over_a_not_necessarily_commutative_ring}{finitely presented} as an $R$-algebra, \CrefAndHyperrefIfExist{definition:separable_algebra_over_a_commutative_ring}{separable} over $R$, and \CrefAndHyperrefIfExist{definition:flat_module_over_a_ring}{flat} as an $R$-module.
    \end{enumerate}
\end{proposition}
