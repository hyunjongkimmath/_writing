\begin{theorem} \label{theorem:ring_is_left_artinian_if_and_only_if_left_noetherian_and_has_krull_dimension_zero}
A \CrefAndHyperrefIfExist{definition:ring}{ring} $R$ is \CrefAndHyperrefIfExist{definition:left_right_artinian_ring}{left Artinian} if and only if it is \CrefAndHyperrefIfExist{definition:noetherian_ring}{left Noetherian} and has \CrefAndHyperrefIfExist{definition:krull_dimension_of_a_commutative_ring}{Krull dimension} zero.
\end{theorem}
