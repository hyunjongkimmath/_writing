\begin{theorem} \label{theorem:adjunction_of_inverse_image_and_direct_image_for_modules_on_ringed_sites_under_ringed_site_morphism}
    Let $f: (\mathcal{C}, J, \mathcal{O}_{\mathcal{C}}) \to (\mathcal{D}, K), \mathcal{O}_{\mathcal{D}})$ be a \CrefAndHyperrefIfExist{definition:morphism_of_ringed_sites}{morphism of ringed sites} with \CrefAndHyperrefIfExist{definition:locally_small_category}{small} \CrefAndHyperrefIfExist{definition:grothendieck_topology_on_a_category_site_covering_sieve_topologically_generating_family}{topological generating families}.

        The \CrefAndHyperrefIfExist{definition:inverse_image_of_a_module_on_a_ringed_site_via_a_morphism_of_ringed_sites}{inverse image} and \CrefAndHyperrefIfExist{definition:direct_image_of_a_module_on_a_ringed_site_via_a_morphism_of_ringed_sites}{direct image} functors 
        $$f^* : \Mod(\calO_\calD) \to \Mod(\calO_\calC)$$
        $$f_* : \Mod(\calO_\calC) \to \Mod(\calO_\calD)$$
        are \CrefAndHyperrefIfExist{definition:adjoint_functors_between_categories_unit_counit_of_adjoint_functors}{adjoint}, i.e. for any 
        \[
        \calF \in \Mod(\calO_\calC), \quad \mathcal{G} \in \Mod(\calO_\calD),
        \]
        there is a natural isomorphism
        $$\Hom_{\calO_\calC}(f^*\calG, \calF) \cong \Hom_{\calO_\calD}(\calG, f_* \calF).$$
\end{theorem}
