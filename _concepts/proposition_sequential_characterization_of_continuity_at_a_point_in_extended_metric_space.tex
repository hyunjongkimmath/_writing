\begin{proposition}[Sequential characterization of continuity at a point in an extended metric space] \label{proposition:sequential_characterization_of_continuity_at_a_point_in_extended_metric_space}
    Let $T$ be an \CrefAndHyperrefIfExist{definition:ordered_semiring}{ordered semiring}, let $T^\infty$ be the \CrefAndHyperrefIfExist{definition:ordered_semiring_with_adjoined_infinity}{ordered semiring with adjoined infinity} obtained from $T$, and let $(X, d_X)$ and $(Y, d_Y)$ be \CrefAndHyperrefIfExist{definition:extended_metric_on_a_set_valued_in_an_ordered_semiring_with_adjoined_infinity}{extended metric spaces valued in $T^\infty$}. Let $S \subseteq X$ and let $f : S \to Y$ be a \CrefAndHyperrefIfExist{definition:function_of_sets}{function}. Let $x \in S$.

    The following hold:
    \begin{enumerate}
        \item If $f$ is \CrefAndHyperrefIfExist{definition:continuity_at_a_point_in_an_extended_metric_space}{continuous at $x$} (in the $\varepsilon$-$\delta$ sense), then for every sequence $(x_n)_{n=1}^\infty$ in $S$ such that $x_n \to x$ \CrefIfExists{definition:convergence_and_limit_of_sequence_in_extended_metric_space}, we have $f(x_n) \to f(x)$ \CrefIfExists{definition:convergence_and_limit_of_sequence_in_extended_metric_space}.

        \item If there exists a sequence $(\delta_n)_{n=1}^\infty$ in $T_{>0}$ such that for every $\varepsilon \in T_{>0}$ there is some $N \in \mathbb{N}$ with $\delta_N < \varepsilon$ (e.g., if $\mathbb{Q}_{\geq 0} \hookrightarrow T$ and $T$ is \CrefAndHyperrefIfExist{definition:archimedean_ordered_semiring_dense_at_zero_ordered_semiring}{Archimedean}), then the converse also holds: if $f(x_n) \to f(x)$ for every sequence $x_n \to x$, then $f$ is continuous at $x$.
    \end{enumerate}
\end{proposition}

\begin{proof}
    We prove the two implications separately.

    \begin{enumerate}
        \item \textbf{(1).} Assume $f$ is continuous at $x$ in the $\varepsilon$-$\delta$ sense as in \Cref{definition:continuity_at_a_point_in_an_extended_metric_space}. Let $(x_n)_{n=1}^\infty$ be a sequence in $S$ such that $x_n \to x$. We show that $f(x_n) \to f(x)$.

        Let $\varepsilon \in T_{>0}$ be arbitrary. By the $\varepsilon$-$\delta$ definition of continuity at $x$, there exists $\delta \in T_{>0}$ such that for all $y \in S$ with $d_X(y, x) < \delta$, we have $d_Y(f(y), f(x)) < \varepsilon$.

        Since $x_n \to x$ in the sense of \Cref{definition:convergence_and_limit_of_sequence_in_extended_metric_space}, there exists $N \in \mathbb{N}$ such that for all $n \geq N$, $d_X(x_n, x) < \delta$. For each such $n$, we have $d_Y(f(x_n), f(x)) < \varepsilon$ by the continuity condition. Therefore, $f(x_n) \to f(x)$ in the sense of \Cref{definition:convergence_and_limit_of_sequence_in_extended_metric_space}.

        \item \textbf{(2).} We prove the contrapositive under the assumption that a sequence $(\delta_n)_{n=1}^\infty$ in $T_{>0}$ exists with $\delta_n$ probing arbitrarily small distances: for every $\varepsilon \in T_{>0}$ there is some $N$ with $\delta_N < \varepsilon$.

        Assume $f$ is \emph{not} continuous at $x$ in the $\varepsilon$-$\delta$ sense. Then there exists $\varepsilon_0 \in T_{>0}$ such that for every $\delta \in T_{>0}$, there is some $y \in S$ with $d_X(y, x) < \delta$ but $d_Y(f(y), f(x)) \geq \varepsilon_0$.

        For each $n \in \mathbb{N}$, apply this with $\delta = \delta_n$. There exists $x_n \in S$ such that $d_X(x_n, x) < \delta_n$ and $d_Y(f(x_n), f(x)) \geq \varepsilon_0$.

        We claim that $x_n \to x$. Let $\varepsilon \in T_{>0}$ be arbitrary. By assumption, there exists $N \in \mathbb{N}$ such that $\delta_N < \varepsilon$. Since $(\delta_n)$ is a sequence in $T_{>0}$ with the probing property, for all sufficiently large $n$, we have $\delta_n < \varepsilon$, hence $d_X(x_n, x) < \delta_n < \varepsilon$. Thus $x_n \to x$ in the sense of \Cref{definition:convergence_and_limit_of_sequence_in_extended_metric_space}.

        On the other hand, $d_Y(f(x_n), f(x)) \geq \varepsilon_0 > 0_T$ for all $n$, so $f(x_n) \not\to f(x)$ in the sense of \Cref{definition:convergence_and_limit_of_sequence_in_extended_metric_space}. This establishes the contrapositive: if condition (2) holds, then condition (1) must hold.
    \end{enumerate}
\end{proof}
