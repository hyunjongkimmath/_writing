\begin{proposition} \label{proposition:the_field_of_real_numbers_is_a_totally_ordered_field}
    Let $\bbR$ be the \CrefAndHyperrefIfExist{definition:real_numbers_as_cauchy_completion_of_rational_numbers}{field of real numbers} equipped with the \CrefAndHyperrefIfExist{definition:standard_order_on_real_numbers}{standard order}. Then:
    \begin{enumerate}
        \item $(\bbR, \le)$ is a \CrefAndHyperrefIfExist{definition:total_order_linear_order_on_a_set}{totally ordered set}.
        \item The order is compatible with addition: for all $x, y, z \in \bbR$, if $x \le y$, then $x + z \le y + z$.
        \item The order is compatible with multiplication by non-negative elements: for all $x, y \in \bbR$, if $x \ge 0$ and $y \ge 0$, then $xy \ge 0$.
        \item The order extends the \CrefAndHyperrefIfExist{definition:standard_order_on_field_of_rational_numbers}{standard order on $\bbQ$} under the canonical embedding $\iota: \bbQ \hookrightarrow \bbR$: if $q_1 \le q_2$ in $\bbQ$, then $[q_1] \le [q_2]$ in $\bbR$.
    \end{enumerate}
    Equivalently, $(\bbR, +, \cdot, \le)$ is a \CrefAndHyperrefIfExist{definition:ordered_field_complete_ordered_field}{totally ordered field}.
\end{proposition}

    We verify each claim in turn.

    \begin{enumerate}
        \item We show that $\le$ is a total order on $\bbR$. Recall from \Cref{definition:standard_order_on_real_numbers} that $[a_n] \le [b_n]$ if and only if $[b_n] - [a_n] \in \bbR_{\ge 0}$, where $\bbR_{\ge 0} = \bbR_{>0} \cup \{[0]\}$ and $[c_n] \in \bbR_{>0}$ precisely when there exist $\varepsilon \in \bbQ_{>0}$ and $N \in \bbN$ such that $c_n \ge \varepsilon$ for all $n \ge N$.

        The relation is reflexive because $[a_n] - [a_n] = [0] \in \bbR_{\ge 0}$. For antisymmetry, suppose $[a_n] \le [b_n]$ and $[b_n] \le [a_n]$. Then $[b_n] - [a_n] \in \bbR_{\ge 0}$ and $[a_n] - [b_n] \in \bbR_{\ge 0}$. Let $(d_n) = (b_n - a_n)$ be a representative of $[b_n] - [a_n]$. If $[b_n] - [a_n] \in \bbR_{>0}$, then there exist $\varepsilon > 0$ and $N$ such that $d_n \ge \varepsilon$ for all $n \ge N$. Then $[a_n] - [b_n] = -[b_n] - [a_n]$ would be represented by $(-d_n)$, and $(-d_n) \le -\varepsilon$ for $n \ge N$, so $[a_n] - [b_n] \notin \bbR_{\ge 0}$, a contradiction. Therefore $[b_n] - [a_n] \notin \bbR_{>0}$, which forces $[b_n] - [a_n] = [0]$, i.e.\ $[a_n] = [b_n]$.

        For transitivity, suppose $[a_n] \le [b_n]$ and $[b_n] \le [c_n]$. Then $[b_n] - [a_n], [c_n] - [b_n] \in \bbR_{\ge 0}$. Their sum is $[c_n] - [a_n]$. The set $\bbR_{\ge 0}$ is closed under addition: if both summands are $[0]$, the sum is $[0]$; if at least one is in $\bbR_{>0}$ with representatives eventually $\ge \varepsilon_1$ and $\ge \varepsilon_2$, then the sum is eventually $\ge \min\{\varepsilon_1, \varepsilon_2\} > 0$. Hence $[c_n] - [a_n] \in \bbR_{\ge 0}$, so $[a_n] \le [c_n]$.

        For totality, let $[a_n], [b_n] \in \bbR$ and consider $[d_n] = [b_n] - [a_n]$. Choose a representative $(d_n) \in \mathcal{C}_\bbQ$ of this class. Since $\mathcal{N}_\bbQ$ is a maximal ideal (\Cref{proposition:the_set_of_null_sequences_of_rational_numbers_form_an_ideal_in_the_ring_of_cauchy_sequences_of_rational_numbers}) and in particular $(d_n)$ is Cauchy, either $(d_n) \in \mathcal{N}_\bbQ$, in which case $[d_n] = [0]$ and $[a_n] = [b_n]$; or $(d_n) \notin \mathcal{N}_\bbQ$. In the latter case, by the argument used in the proof of \Cref{proposition:the_set_of_null_sequences_of_rational_numbers_form_an_ideal_in_the_ring_of_cauchy_sequences_of_rational_numbers}, $(d_n)$ is eventually bounded away from $0$: there exist $\varepsilon > 0$ and $N$ such that $|d_n| \ge \varepsilon$ for all $n \ge N$. Since $|d_n| = d_n$ or $-d_n$ by \Cref{definition:absolute_value_on_rational_numbers}, either $d_n \ge \varepsilon$ for all $n \ge N$ or $-d_n \ge \varepsilon$ (i.e.\ $d_n \le -\varepsilon$) for all $n \ge N$. In the first case $[d_n] \in \bbR_{>0}$, so $[a_n] \le [b_n]$; in the second case $-[d_n] \in \bbR_{>0}$, so $[b_n] \le [a_n]$. Thus every pair of elements is comparable.

        \item Suppose $[a_n] \le [b_n]$ and let $[c_n] \in \bbR$. Then $[b_n] - [a_n] \in \bbR_{\ge 0}$, so
        $$([b_n] + [c_n]) - ([a_n] + [c_n]) = [b_n] - [a_n] \in \bbR_{\ge 0},$$
        which gives $[a_n] + [c_n] \le [b_n] + [c_n]$.

        \item Suppose $[a_n] \ge [0]$ and $[b_n] \ge [0]$, i.e.\ $[a_n], [b_n] \in \bbR_{\ge 0}$. If either is $[0]$, then $[a_n][b_n] = [0] \in \bbR_{\ge 0}$. Otherwise, $[a_n], [b_n] \in \bbR_{>0}$, so there exist $\varepsilon_1, \varepsilon_2 \in \bbQ_{>0}$ and $N_1, N_2 \in \bbN$ such that $a_n \ge \varepsilon_1$ for all $n \ge N_1$ and $b_n \ge \varepsilon_2$ for all $n \ge N_2$. Then for $n \ge \max\{N_1, N_2\}$, we have $a_n b_n \ge \varepsilon_1 \varepsilon_2 > 0$ by the compatibility of the order on $\bbQ$ with multiplication (\Cref{proposition:the_field_of_rational_numbers_is_a_totally_ordered_field}). Hence $[a_n][b_n] \in \bbR_{>0} \subseteq \bbR_{\ge 0}$, so $[a_n][b_n] \ge [0]$.

        \item Let $q_1 \le q_2$ in $\bbQ$. Write $q_2 - q_1 = \frac{a}{b}$ with $a \ge 0$ in $\bbZ$ and $b > 0$ in $\bbZ$ (\Cref{definition:standard_order_on_field_of_rational_numbers}). The constant sequences $(q_1)$, $(q_2)$ are Cauchy, and their difference $(q_2 - q_1)$ is the constant sequence with value $\frac{a}{b}$. Since $\frac{a}{b} \ge 0$ in $\bbQ$, the constant value is $\ge 0$. If $\frac{a}{b} = 0$, then $[q_2] - [q_1] = [0]$, so $\iota(q_1) = \iota(q_2)$. Otherwise $\frac{a}{b} > 0$, so the constant sequence $(q_2 - q_1)$ is eventually $\ge \frac{a}{b} > 0$, which places $[q_2] - [q_1]$ in $\bbR_{>0}$ by \Cref{definition:standard_order_on_real_numbers}. Hence $\iota(q_1) \le \iota(q_2)$.
    \end{enumerate}

    The four parts together show that $(\bbR, +, \cdot, \le)$ is an \Cref{definition:ordered_field_complete_ordered_field}{ordered field} whose order extends that of $\bbQ$.
\end{proof}
