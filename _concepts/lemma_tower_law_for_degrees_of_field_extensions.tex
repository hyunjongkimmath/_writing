\begin{lemma}[Tower Law] \label{lemma:tower_law_for_degrees_of_field_extensions}
Let $K \subseteq F \subseteq L$ be a tower of \CrefAndHyperrefIfExist{definition:extension_of_a_field}{field extensions}. Then
\[
[L:K] = [L:F] \cdot [F:K],
\]
where the product is interpreted in the extended sense: if either $[L:F]$ or $[F:K]$ is infinite, then $[L:K]$ is infinite.
\end{lemma}
