\begin{theorem}[Tychonoff's theorem] \label{theorem:tychonoffs_theorem_for_the_compactness_of_a_small_product_of_compact_topological_spaces}
Let $I$ be an index set and let $\{X_i\}_{i \in I}$ be a collection of \CrefAndHyperrefIfExist{definition:compact_topological_space}{compact spaces}. 

The \CrefAndHyperrefIfExist{definition:box_topology_on_a_cartesian_product_of_topological_spaces}{product space} $X = \prod_{i \in I} X_i$ is compact in the product \CrefAndHyperrefIfExist{definition:topological_space}{topology}.
\end{theorem}
