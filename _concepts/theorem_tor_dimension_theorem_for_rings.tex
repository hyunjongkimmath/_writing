\begin{theorem} \label{theorem:tor_dimension_theorem_for_rings}
(Tor Dimension Theorem) For any \CrefAndHyperrefIfExist{definition:ring}{not necessarily commutative ring} $R$, the Tor-dimension is symmetric and relates to global dimension as follows:
\begin{enumerate}
    \item $\operatorname{wdim}(R) = \sup \{ \operatorname{fd}_R(M) \mid M \in {}_R\mathsf{Mod} \} = \sup \{ \operatorname{fd}_R(N) \mid N \in \mathsf{Mod}_R \}$. (\Cref{definition:tor_dimension_of_a_ring})
    \item $\operatorname{wdim}(R) \leq \min(\operatorname{lgldim}(R), \operatorname{rgldim}(R))$. (\Cref{definition:left_right_global_dimension_of_a_ring})
    \item If $R$ is left and right Noetherian, then $\operatorname{wdim}(R) = \operatorname{lgldim}(R) = \operatorname{rgldim}(R)$.
\end{enumerate}
\end{theorem}
