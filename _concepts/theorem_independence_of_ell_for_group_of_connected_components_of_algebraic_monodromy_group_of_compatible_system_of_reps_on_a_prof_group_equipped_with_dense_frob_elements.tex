\begin{theorem}[{\cite[Proposition (6.14)]{larsen_pink}}, cf. {\cite[2.2.3]{serre_resume_des_cours_1984_1985}}] \label{theorem:independence_of_ell_for_group_of_connected_components_of_algebraic_monodromy_group_of_compatible_system_of_reps_on_a_prof_group_equipped_with_dense_frob_elements}
    Let $G$ be a \CrefAndHyperrefIfExist{definition:profinite_group}{profinite group} and $\{ \rho_\ell: G \to \mathrm{GL}_n(\mathbb{Q}_\ell) \}_{\ell \in L}$ be a \CrefAndHyperref{definition:compatible_system_of_representations_on_a_profinite_group_equipped_with_dense_frobenius_elements}{compatible system} of \CrefAndHyperrefIfExist{definition:continuous_representation_of_a_topological_group_on_a_topological_module_over_a_topological_ring}{continuous representations} indexed by a set of rational primes $L$. 

    The group of connected components $G_\ell / G_\ell^\circ$ \CrefIfExists{definition:identity_component_of_a_group_scheme_over_a_scheme} of the \CrefAndHyperref{definition:algebraic_monodromy_group_at_a_prime_for_a_compatible_system_of_reps_on_a_prof_group_with_dense_frob_elements}{algebraic monodromy group} $G_\ell$ at $\ell$ is independent of $\ell$. 
\end{theorem}
