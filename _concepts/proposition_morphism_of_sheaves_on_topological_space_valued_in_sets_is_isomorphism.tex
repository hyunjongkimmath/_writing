\begin{proposition} \label{proposition:morphism_of_sheaves_on_topological_space_valued_in_sets_is_isomorphism}
    Let $X$ be a \CrefAndHyperrefIfExist{definition:topological_space}{topological space}. 
    A morphism $f: \calF \to \calG$ in $\Sh(X, \Sets)$ is an isomorphism (resp. \CrefAndHyperrefIfExist{definition:monomorphism_and_epimorphism_in_categories}{monomorphism}, resp. \CrefAndHyperrefIfExist{definition:monomorphism_and_epimorphism_in_categories}{epimorphism}) if and only if all of the \CrefAndHyperrefIfExist{definition:stalk_of_a_presheaf_on_a_topological_space_at_a_point}{stalk} morphisms $f: \calF_x \to \calG_x$ are \CrefAndHyperrefIfExist{definition:injective_surjective_bijective_map_of_sets}{bijections} (resp. \CrefAndHyperrefIfExist{definition:injective_surjective_bijective_map_of_sets}{injections}, resp. \CrefAndHyperrefIfExist{definition:injective_surjective_bijective_map_of_sets}{surjections}). In particular, \CrefAndHyperrefIfExist{definition:category_of_opens_of_a_topological_space}{$\mathbf{Open}(X)$} is a \CrefAndHyperrefIfExist{definition:grothendieck_topology_on_a_category_site_covering_sieve_topologically_generating_family}{site} \CrefAndHyperrefIfExist{definition:site_with_enough_points}{with enough points}.
\end{proposition}
