\begin{lemma}[Determinant lemma for modules]  \label{lemma:determinant_lemma_for_modules_over_a_ring}
	Let $R$ be a \CrefAndHyperrefIfExist{definition:commutative_ring}{commutative ring}, $M$ a finitely generated right $R$-module, $I$ an ideal of $R$, and $\varphi \colon M \to M$ an $R$-linear endomorphism such that $\varphi(M) \subseteq MI$. Let $n$ be the number of generators of $M$.

	Then there exist elements $a_1, \ldots, a_{n-1} \in I$ and $a_n \in R$ such that
	$$\varphi^n + a_1 \varphi^{n-1} + a_2 \varphi^{n-2} + \cdots + a_{n-1} \varphi + (a_n - 1) \operatorname{id}_M = 0$$
	as an endomorphism of $M$.

	In particular, if $I \subseteq J(R)$\CrefIfExists{definition:jacobson_radical_of_a_ring}, then $\varphi$ is invertible.
\end{lemma}

	Suppose that $M$ is generated by $n$ elements $m_1, \ldots, m_n$ as a right $R$-module. Since $\varphi(M) \subseteq MI$, for each $j = 1, \ldots, n$ there exist coefficients $b_{ji} \in I$ such that
	$$\varphi(m_j) = \sum_{i=1}^n m_i b_{ji}.$$

	Rearranging gives
	$$\sum_{i=1}^n m_i (\delta_{ij}\varphi - b_{ji}) = 0 \quad \text{for each } j = 1, \ldots, n,$$
	where $\delta_{ij}$ is the Kronecker delta and we interpret $\delta_{ij}\varphi$ as scalar multiplication by $\varphi$.

	Let $A = (\delta_{ij}\varphi - b_{ji}) \in M_n(\operatorname{End}_R(M))$ be the matrix whose entries are endomorphisms of $M$. Since $R$ is commutative and $\varphi$ is $R$-linear, each entry $\delta_{ij}\varphi - b_{ji}$ commutes with every other entry: multiplication by scalars from $I$ commute with each other because $R$ is commutative, and $\varphi$ commutes with right-multiplication by any element of $R$ because $\varphi$ is $R$-linear.

	Let $C = \operatorname{adj}(A) \in M_n(\operatorname{End}_R(M))$ be the classical adjugate matrix\CrefIfExists{proposition:adjugate_formula_for_determinant_over_a_commutative_ring} of $A$. By the adjugate formula, $C A = (\det A) I_n$, where $\det(A)$ is computed in the commutative ring $\operatorname{End}_R(M)^{\text{comm}}$ generated by these commuting endomorphisms.

	Applying the relation $CA$ to the column vector $(m_1, \ldots, m_n)^\top$, each entry vanishes because $\sum_i (\delta_{ij}\varphi - b_{ji}) m_j = 0$. Hence
	$$\det(A) \cdot m_k = 0 \quad \text{for all } k = 1, \ldots, n.$$

	Since the $m_k$ generate $M$, we have $\det(\delta_{ij}\varphi - b_{ji}) = 0$ as an endomorphism of $M$.

	Expanding the determinant by the Leibniz formula\CrefIfExists{definition:determinant_of_a_matrix_over_a_ring} yields a polynomial in $\varphi$:
	$$\varphi^n + c_1 \varphi^{n-1} + \cdots + c_{n-1} \varphi + c_n = 0,$$
	where each coefficient $c_k$ is a sum of products involving exactly $k$ entries from the matrix $(b_{ji})$, hence lies in $I^k \subseteq I$ for $k = 1, \ldots, n$. Setting $a_k = c_k$ for $k = 1, \ldots, n-1$ and noting that $c_n \in I$, we rewrite the relation as
	$$\varphi^n + a_1 \varphi^{n-1} + \cdots + a_{n-1} \varphi + (a_n - 1) \operatorname{id}_M = 0,$$
	with $a_1, \ldots, a_{n-1} \in I$ and $a_n = c_n + 1_R \in R$.

	Now suppose that $I \subseteq J(R)$. Rewrite the equation as
	$$\varphi \circ (\varphi^{n-1} + a_1 \varphi^{n-2} + \cdots + a_{n-1} \operatorname{id}_M) = (1_R - c_n) \operatorname{id}_M.$$
	Since $c_n \in I \subseteq J(R)$, the element $1_R - c_n$ is a unit in $R$. Hence we can multiply by $(1_R - c_n)^{-1}$ to obtain
	$$\varphi \circ (\varphi^{n-1} + a_1 \varphi^{n-2} + \cdots + a_{n-1} \operatorname{id}_M) \cdot (1_R - c_n)^{-1} = \operatorname{id}_M.$$
	Thus $\varphi$ has a right inverse. By the same polynomial relation, we also have
	$$(\varphi^{n-1} + a_1 \varphi^{n-2} + \cdots + a_{n-1} \operatorname{id}_M) \cdot (1_R - c_n)^{-1} \circ \varphi = \operatorname{id}_M,$$
	so $\varphi$ also has a left inverse. Therefore, $\varphi$ is invertible.
\end{proof}
