\begin{proposition} \label{proposition:adjunction_of_extension_by_zero_and_inverse_image_via_open_immersion_for_etale_sheaves_of_abelian_groups_on_scheme}
    Let $j: U \hookrightarrow X$ be an \CrefAndHyperrefIfExist{definition:open_and_closed_immersions_of_schemes}{open immersion} of \CrefAndHyperrefIfExist{definition:scheme}{schemes}.

    \begin{enumerate}
        \item
        There is an \CrefAndHyperrefIfExist{definition:adjoint_functors_between_categories_unit_counit_of_adjoint_functors}{adjunction} $j_! \dashv j^*$ (\Cref{definition:extension_by_zero_of_a_sheaf_on_the_small_etale_site_on_a_scheme_under_open_immersion}) (\Cref{definition:inverse_image_and_direct_image_of_sheaves_on_schemes_for_various_topologies}) between the \CrefAndHyperrefIfExist{definition:functor_between_categories}{functors}
        $$j_!: \Sh(U_{\et}, \Ab) \to \Sh(X_{\et}, \Ab)$$
        \CrefIfExists{definition:sheaf_on_a_site}
        and
        $$j^*: \Sh(X_{\et}, \Ab) \to \Sh(U_{\et}, \Ab).$$

        \item
        Let $\Lambda$ be a \CrefAndHyperrefIfExist{definition:ring}{ring}. Denote by $\Sh(X_{\et}, \Mod(\Lambda))$ and $\Sh(U_{\et}, \Mod(\Lambda))$ the categories of \CrefAndHyperrefIfExist{definition:module_over_a_sheaf_of_rings_on_a_site}{modules} over the \CrefAndHyperrefIfExist{definition:constant_sheaf_on_a_site_with_sheafification}{constant sheaves} $\underline{\Lambda}$ on $X_{\et}$ and $U_{\et}$ respectively. The functors $j_!$ and $j^*$ restrict to
        $$j_!: \Sh(U_{\et}, \Mod(\Lambda)) \to \Sh(X_{\et}, \Mod(\Lambda))$$
        and
        $$j^*: \Sh(X_{\et}, \Mod(\Lambda)) \to \Sh(U_{\et}, \Mod(\Lambda)),$$
        forming an adjunction $j_! \dashv j^*$ between these categories.
    \end{enumerate}
\end{proposition}
