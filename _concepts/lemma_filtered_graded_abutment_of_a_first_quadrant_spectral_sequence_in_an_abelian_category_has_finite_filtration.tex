\begin{lemma} \label{lemma:filtered_graded_abutment_of_a_first_quadrant_spectral_sequence_in_an_abelian_category_has_finite_filtration}
    Let $\mathcal{A}$ be an \CrefAndHyperrefIfExist{definition:abelian_category}{abelian category}.
    \begin{enumerate}
        \item 
        Let $E = \{E^r\}_{r \ge r_0}$ be a \CrefAndHyperrefIfExist{definition:first_quadrant_spectral_sequence}{first quadrant} \CrefAndHyperrefIfExist{definition:spectral_sequence_in_an_abelian_category}{homological spectral sequence}. Suppose that $E$ \CrefAndHyperrefIfExist{definition:abuts_to_approaches_a_filtration_for_a_spectral_sequence_in_an_abelian_category}{abuts} to a \CrefAndHyperrefIfExist{definition:filtered_graded_object_in_a_category}{filtered graded object} $H$ in $\calA$ where the \CrefAndHyperrefIfExist{definition:filtration_on_an_object_of_a_category}{filtration} is increasing. The filtration on $H_n$ is finite. More specfically, the filration can be expressed in the form
        $$0 = F_{-1} H_n \subseteq F_0 H_n \subseteq F_1 H_n \subseteq \cdots F_n H_n = H_n.$$

        \item 
        Let $E = \{E_r\}_{r \ge r_0}$ be a \CrefAndHyperrefIfExist{definition:first_quadrant_spectral_sequence}{first quadrant} \CrefAndHyperrefIfExist{definition:spectral_sequence_in_an_abelian_category}{cohomological spectral sequence}. Suppose that $E$ \CrefAndHyperrefIfExist{definition:abuts_to_approaches_a_filtration_for_a_spectral_sequence_in_an_abelian_category}{abuts} to a \CrefAndHyperrefIfExist{definition:filtered_graded_object_in_a_category}{filtered graded object} $H$ in $\calA$ where the \CrefAndHyperrefIfExist{definition:filtration_on_an_object_of_a_category}{filtration} is decreasing. The filtration on $H^n$ is finite. More specifically, the filtration can be expressed in the form
        $$H^n = F^0 H^n \supseteq F^1 H^n \supseteq F^2 H^n \supseteq \cdots \supseteq F^n H^n \supseteq F^{n+1} H^n = 0.$$
    \end{enumerate}
\end{lemma}
