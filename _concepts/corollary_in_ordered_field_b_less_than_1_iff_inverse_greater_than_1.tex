\begin{corollary}[Order reversal by inversion at 1 in an ordered field] \label{corollary:in_ordered_field_b_less_than_1_iff_inverse_greater_than_1}
    Let $F$ be an \CrefAndHyperrefIfExist{definition:ordered_field_complete_ordered_field}{ordered field}. For any $b \in F^\times = F \setminus \{0\}$ with $b > 0$,
    $$0 < b < 1 \iff b^{-1} > 1.$$
\end{corollary}
