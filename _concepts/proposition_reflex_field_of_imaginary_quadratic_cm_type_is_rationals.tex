\begin{proposition}[Reflex Field of an Imaginary Quadratic CM Type is $\Q$] \label{proposition:reflex_field_of_imaginary_quadratic_cm_type_is_rationals}
    Let $E$ be an imaginary quadratic field with unique \CrefAndHyperrefIfExist{definition:cm_type_and_reflex_field_of_a_cm_type}{CM type} $\Phi = \{\tau\}$ (up to complex conjugation). Then the \CrefAndHyperrefIfExist{definition:cm_type_and_reflex_field_of_a_cm_type}{reflex field} is $E^*(\Phi) = \mathbb{Q}$.

    The \CrefAndHyperrefIfExist{definition:reflex_norm_for_a_cm_field}{reflex norm} $N_{(E,\Phi)}: T_{\mathbb{Q}} \to T_E$ coincides with the morphism of tori induced by the field norm $N_{E/\mathbb{Q}}: E^\times \to \mathbb{Q}^\times$.
\end{proposition}
