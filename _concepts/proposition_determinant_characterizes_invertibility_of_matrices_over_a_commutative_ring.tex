\begin{proposition} \label{proposition:determinant_characterizes_invertibility_of_matrices_over_a_commutative_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:commutative_ring}{commutative ring}, and let $A \in M_n(R)$ be an $n \times n$ matrix. Then $A$ is invertible in $M_n(R)$ if and only if $\det(A)$ is a unit in $R$. In that case,
    $$A^{-1} = \det(A)^{-1} \operatorname{adj}(A).$$
\end{proposition}
