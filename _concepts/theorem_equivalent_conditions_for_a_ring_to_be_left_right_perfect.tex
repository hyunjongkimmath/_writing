\begin{theorem} \label{theorem:equivalent_conditions_for_a_ring_to_be_left_right_perfect}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{ring}.
    The following conditions are equivalent:
    \begin{enumerate}
        \item $R$ is \CrefAndHyperrefIfExist{definition:perfect_ring}{left perfect}, i.e. every \CrefAndHyperrefIfExist{definition:flat_left_right_module_over_a_ring}{flat} \CrefAndHyperrefIfExist{definition:bimodule_over_rings_module_over_a_ring}{left $R$-module} is \CrefAndHyperrefIfExist{definition:projective_bimodule_over_rings}{projective}.
        \item Every left $R$-module has a projective cover.
        \item $R/J(R)$ is a \CrefAndHyperrefIfExist{definition:semisimple_ring}{semisimple ring} and $J(R)$ is left T-nilpotent.
        \item $R$ satisfies the \CrefAndHyperrefIfExist{definition:ascending_descending_chain_condition_on_a_partially_ordered_set}{descending chain condition} on \CrefAndHyperrefIfExist{definition:ideal_generated_by_a_subset_of_a_ring}{principal} \CrefAndHyperrefIfExist{definition:ideal_of_a_ring}{right ideals}.
        \item Every descending chain of principal right ideals $a_1 R \supseteq a_1 a_2 R \supseteq a_1 a_2 a_3 R \supseteq \dots$ terminates.
    \end{enumerate}
\end{theorem}
