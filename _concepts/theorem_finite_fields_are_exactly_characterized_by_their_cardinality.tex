\begin{theorem} \label{theorem:finite_fields_are_exactly_characterized_by_their_cardinality}
    \begin{enumerate}
        \item All \CrefAndHyperrefIfExist{definition:finite_field}{finite fields} have \CrefAndHyperrefIfExist{definition:cardinal_number_and_cardinality_of_a_set}{cardinality} equal to the \CrefAndHyperrefIfExist{definition:power_of_an_element_of_a_ring_by_a_nonnegative_integer}{power} of a \CrefAndHyperrefIfExist{definition:prime_number}{prime number}.

        \item For every $q = p^k$ for $p$ a prime number and $k \geq 0$ an integer,
        \begin{enumerate}
            \item any two fields of cardinality $q$ are \CrefAndHyperrefIfExist{definition:field_homomorphism}{isomorphic as fields}.

            \item there is a field of cardinality exactly $q$.
        \end{enumerate}
    \end{enumerate}
\end{theorem}

    We prove each part in turn.

    \begin{enumerate}
        \item Let $F$ be a finite field. By \Cref{proposition:characteristic_of_a_field_is_zero_or_a_prime}, $\operatorname{char}(F) = p$ for some prime number $p$, or $\operatorname{char}(F) = 0$. The characteristic cannot be $0$, because then the unique ring homomorphism $\mathbb{Z} \to F$ has trivial kernel by \Cref{definition:characteristic_of_a_ring}, so its image is isomorphic to $\mathbb{Z}$, which is infinite, contradicting that $F$ is finite. Hence $\operatorname{char}(F) = p$.

            The prime subfield of $F$ is isomorphic to the finite field $\mathbb{F}_p$ by \Cref{proposition:prime_subfield_of_a_field_of_characteristic_p_is_isomorphic_to_finite_field_F_p}. Viewing $F$ as a \CrefAndHyperrefIfExist{definition:vector_space_over_a_field}{vector space} over its prime subfield $\mathbb{F}_p$ (which is valid by \Cref{lemma:extension_of_a_field_is_a_vector_space}), we have that $F$ is finite-dimensional because $F$ is finite. Let $k = \dim_{\mathbb{F}_p} F$. By \Cref{definition:basis_and_dimension_of_a_vector_space_over_a_field}, every element of $F$ can be expressed uniquely as a linear combination of $k$ basis vectors with coefficients in $\mathbb{F}_p$. Since each coefficient has $p$ possible values, there are exactly $p^k$ such linear combinations; hence $|F| = p^k$, which is the power of a prime number.

        \item Fix a prime number $p$ and an integer $k \geq 1$, and set $q = p^k$. We prove uniqueness and existence separately.

        \begin{enumerate}
            \item Suppose that $F_1$ and $F_2$ are two fields of cardinality $q$. By part (1), both have characteristic $p$, so their prime subfields are isomorphic to $\mathbb{F}_p$. Every element of $F_i$ satisfies $x^q - x = 0$ in $F_i[x]$, because the multiplicative group $F_i^\times$ has order $q-1$ and thus $a^{q-1} = 1$ for all nonzero $a \in F_i$ by \Cref{corollary:order_of_element_divides_order_of_finite_group}; hence $a^q = a$ for all $a \in F_i$.

            Each $F_i$ is therefore the \CrefAndHyperrefIfExist{definition:splitting_field_of_polynomials_over_a_field_normal_field_extension}{splitting field} of the polynomial $x^q - x \in \mathbb{F}_p[x]$ over $\mathbb{F}_p$, since every element is a root and the field is generated by those roots. By \Cref{theorem:existence_and_uniqueness_of_splitting_fields}, any two splitting fields of the same polynomial over the same base field are isomorphic; hence $F_1$ and $F_2$ are isomorphic as fields.

            \item The polynomial $x^q - x \in \mathbb{F}_p[x]$ has a splitting field $L$ over $\mathbb{F}_p$, by \Cref{theorem:existence_and_uniqueness_of_splitting_fields}. We show that $|L| = q$. First, the formal derivative of $x^q - x$ in $\mathbb{F}_p[x]$ is $(x^q - x)' = qx^{q-1} - 1 = -1$, because $q = p^k \equiv 0 \pmod{p}$; hence $\gcd(x^q - x, (x^q - x)') = \gcd(x^q - x, -1) = 1$ in $\mathbb{F}_p[x]$. By \Cref{definition:separable_polynomial_over_a_field}, $x^q - x$ is a separable polynomial over $\mathbb{F}_p$, and therefore by \Cref{lemma:separable_polynomial_has_distinct_roots_in_any_splitting_field} it has exactly $q$ distinct roots in $L$.

            Let $S = \{\alpha \in L : \alpha^q - \alpha = 0\}$ be the set of these $q$ distinct roots. We show that $S$ is a subfield of $L$. For all $\alpha, \beta \in S$, we have $(\alpha + \beta)^q = \alpha^q + \beta^q = \alpha + \beta$ by \Cref{lemma:freshmans_dream_identity_in_characteristic_p}, so $\alpha + \beta \in S$. Similarly, $(\alpha\beta)^q = \alpha^q \beta^q = \alpha\beta$ by the same lemma, so $\alpha\beta \in S$. The same lemma also gives $(-\alpha)^q = -\alpha^q = -\alpha$, so $-\alpha \in S$. For nonzero $\alpha \in S$, we have $(\alpha^{-1})^q = (\alpha^q)^{-1} = \alpha^{-1}$ because $\alpha \neq 0$ and inversion commutes with exponentiation in a field, so $\alpha^{-1} \in S$. Hence $S$ is a subfield of $L$.

            By definition of the splitting field, $L$ is generated over $\mathbb{F}_p$ by the roots of $x^q - x$, which are exactly the elements of $S$; hence $L = S$ and $|L| = q$.
        \end{enumerate}
    \end{enumerate}
    This completes the proof of all three parts.
\end{proof}
