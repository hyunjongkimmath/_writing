\begin{theorem} \label{theorem:finite_fields_are_exactly_characterized_by_their_cardinality}

    \begin{enumerate}
        \item All \CrefAndHyperrefIfExist{definition:finite_field}{finite fields} have \CrefAndHyperrefIfExist{definition:cardinal_number_and_cardinality_of_a_set}{cardinality} equal to the \CrefAndHyperrefIfExist{definition:power_of_an_element_of_a_ring_by_a_nonnegative_integer}{power} of a \CrefAndHyperrefIfExist{definition:prime_number}{prime number}.

        \item For every $q = p^k$ for $p$ a prime number and $k \geq 0$ an integer, 
        \begin{enumerate}
            \item any two fields of cardinality $q$ are \CrefAndHyperrefIfExist{definition:field_homomorphism}{isomorphic as fields}. 

            \item there is a field of cardinality exactly $q$.
        \end{enumerate}
    \end{enumerate}
\end{theorem}
