\begin{lemma} \label{lemma:image_of_integral_ell_adic_representation_of_abelian_variety_over_a_subfield_of_complex_numbers_is_contained_in_the_integral_ell_adic_points_of_mumford_tate_group}
    Let $A$ be an \CrefAndHyperrefIfExist{definition:abelian_variety_over_a_field}{abelian variety} over a subfield $K$ of $\bbC$, and let $\ell$ be a \CrefAndHyperrefIfExist{definition:prime_number}{prime number}. Let $T_\ell(A)$ be the \CrefAndHyperrefIfExist{definition:ell_adic_tate_group_of_a_commutative_group_scheme_over_a_field}{$\ell$-adic Tate group} of $A$, and let $\mathrm{MT}(A)/\bbQ$ be the \CrefAndHyperrefIfExist{definition:mumford_tate_group_of_an_abelian_variety_over_the_complex_numbers}{Mumford-Tate group} of $A$. Let
    $$\rho_{A,\ell}: G_K \to \Aut(T_\ell(A))$$
    be the \CrefAndHyperrefIfExist{definition:ell_adic_tate_group_of_a_commutative_group_scheme_over_a_field}{rational $\ell$-adic Galois representation}. 

    The image of $\rho_{A,\ell}$ is contained $$\mathrm{MT}(A)(\Q_ell) \cap \Aut(T_\ell(A))$$
    where $\mathrm{MT}(A)(\Q_ell)$ is considered as a subgroup of $\Aut(V_\ell(A))$ via the singular homology comparison isomorphism
    $$H_1(A(\mathbb{C}), \mathbb{Q}) \otimes_\mathbb{Q} \mathbb{Q}_\ell \xrightarrow{\sim} V_\ell(A)$$
\end{lemma}
