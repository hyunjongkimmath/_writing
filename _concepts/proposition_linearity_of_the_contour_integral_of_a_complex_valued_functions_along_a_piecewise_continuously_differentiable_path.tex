\begin{proposition}[Linearity of the contour integral] \label{proposition:linearity_of_the_contour_integral_of_a_complex_valued_functions_along_a_piecewise_continuously_differentiable_path}

Let $\gamma : [a,b] \to \mathbb{C}$ be a \CrefAndHyperrefIfExist{definition:piecewise_continuously_differentiable_path_from_a_closed_real_interval_to_the_complex_plane}{piecewise continuously differentiable path} and let $f, g : \gamma([a,b]) \to \mathbb{C}$ be \CrefAndHyperrefIfExist{definition:continuous_map_of_topological_spaces}{continuous} functions on the image of $\gamma$. Let $c \in \mathbb{C}$.

\begin{enumerate}
    \item (Sum rule) $\displaystyle\int_{\gamma} \bigl(f(z) + g(z)\bigr)\,dz = \int_{\gamma} f(z)\,dz + \int_{\gamma} g(z)\,dz.$
    \item (Scalar multiple rule) $\displaystyle\int_{\gamma} c \cdot f(z)\,dz = c \int_{\gamma} f(z)\,dz.$
\end{enumerate}
\end{proposition}

\begin{proof}
    Both statements follow directly from the \CrefAndHyperrefIfExist{definition:contour_integral_of_a_complex_valued_function_along_a_piecewise_continuously_differentiable_path}{definition of the contour integral} and the linearity of the \CrefAndHyperrefIfExist{theorem:linearity_of_riemann_integral}{Riemann integral}. Indeed,
    \begin{align*}
        \int_{\gamma} \bigl(f(z) + g(z)\bigr)\,dz &= \int_a^b \bigl(f(\gamma(t)) + g(\gamma(t))\bigr)\,\gamma'(t)\,dt \\
        &= \int_a^b f(\gamma(t))\gamma'(t)\,dt + \int_a^b g(\gamma(t))\gamma'(t)\,dt \\
        &= \int_{\gamma} f(z)\,dz + \int_{\gamma} g(z)\,dz,
    \end{align*}
    and similarly
    \begin{align*}
        \int_{\gamma} c \cdot f(z)\,dz &= \int_a^b c \cdot f(\gamma(t))\,\gamma'(t)\,dt \\
        &= c \int_a^b f(\gamma(t))\,\gamma'(t)\,dt = c \int_{\gamma} f(z)\,dz,
    \end{align*}
    where in both cases the Riemann integral of a complex-valued function is understood componentwise and the linearity of the Riemann integral (\Cref{theorem:linearity_of_riemann_integral}) applies to the real and imaginary parts separately.
\end{proof}
