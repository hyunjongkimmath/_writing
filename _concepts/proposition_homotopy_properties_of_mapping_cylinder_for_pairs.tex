\begin{proposition} \label{proposition:homotopy_properties_of_mapping_cylinder_for_pairs}
Let $f: (X, A) \to (Y, B)$ be a \CrefAndHyperrefIfExist{definition:map_of_pairs}{map of pairs}. 

\begin{enumerate}
    \item The pair $(M_f, M_{f|_A})$ \CrefAndHyperrefIfExist{definition:deformation_retraction}{deformation retracts} onto the subspace $(Y, B)$ via the map induced by $H_t(x, s) = (x, (1-t)s + t)$ and $H_t(y) = y$.
    
    \item Every map of pairs $f: (X, A) \to (Y, B)$ factors as $f = r \circ i$, where $i: (X, A) \hookrightarrow (M_f, M_{f|_A})$ is a \CrefAndHyperrefIfExist{definition:cofibration_of_pairs}{cofibration} and $r: (M_f, M_{f|_A}) \to (Y, B)$ is a \CrefAndHyperrefIfExist{definition:homotopy_equivalence_of_pairs}{homotopy equivalence}.
\end{enumerate}
\end{proposition}
