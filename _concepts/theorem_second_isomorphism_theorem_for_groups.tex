\begin{theorem}[Second Isomorphism Theorem for Groups (Diamond Isomorphism Theorem)] \label{theorem:second_isomorphism_theorem_for_groups}
    Let $G$ be a \CrefAndHyperrefIfExist{definition:group}{group}. Let $H$ be a \CrefAndHyperrefIfExist{definition:subgroup_of_a_group}{subgroup} of $G$, and let $N$ be a \CrefAndHyperrefIfExist{definition:normal_subgroup_of_a_group}{normal subgroup} of $G$.

    Then $HN = \{hn : h \in H, n \in N\}$ is a \CrefAndHyperrefIfExist{definition:subgroup_of_a_group}{subgroup} of $G$, $H \cap N$ is a \CrefAndHyperrefIfExist{definition:normal_subgroup_of_a_group}{normal subgroup} of $H$, and there is a \CrefAndHyperrefIfExist{definition:group_homomorphism}{group isomoprhism}
    $$\psi : H/(H \cap N) \xrightarrow{\,\sim\,} HN/N$$
    \CrefIfExists{definition:quotient_of_a_group_by_a_normal_subgroup}
    given by $\psi(h(H \cap N)) = hN$ for all $h \in H$.
\end{theorem}
\begin{proof}
    Let $e$ denote the identity element of $G$.

    \begin{enumerate}
        \item We show that the set $HN = \{hn : h \in H, n \in N\}$ is a subgroup of $G$. First, $e = ee \in HN$. Second, if $h_1 n_1, h_2 n_2 \in HN$, then we rewrite the product as $n_1 h_2 = h_2 (h_2^{-1} n_1 h_2)$, so
        $$h_1 n_1 h_2 n_2 = h_1 h_2 (h_2^{-1} n_1 h_2) n_2.$$
        Since $N \trianglelefteq G$, we have $h_2^{-1} n_1 h_2 \in N$, so the product lies in $HN$. Third, if $hn \in HN$, then we rewrite $(hn)^{-1}$ as $n^{-1} h^{-1} = h^{-1} (h n^{-1} h^{-1})$, so
        $$(hn)^{-1} = h^{-1} (h n^{-1} h^{-1}).$$
        Since $N \trianglelefteq G$, we have $h n^{-1} h^{-1} \in N$, which implies that $(hn)^{-1} \in HN$. Therefore, $HN$ is indeed a subgroup of $G$.

        \item We show that the intersection $H \cap N$ is a normal subgroup of $H$. It is a subgroup since both $H$ and $N$ are subgroups\CrefIfExists{lemma:intersection_of_subgroups_of_a_group_is_a_subgroup}. To show that $H \cap N$ is in fact a normal subgroup of $H$, let $h_0 \in H$ and $x \in H \cap N$. Then $h_0 x h_0^{-1} \in N$ because $x \in N$ and $N \trianglelefteq G$, and also $h_0 x h_0^{-1} \in H$ because $x \in H$ and $H$ is a group. Hence $h_0 x h_0^{-1} \in H \cap N$, so $H \cap N \trianglelefteq H$.

        \item The quotient groups $H/(H \cap N)$ and $HN/N$ are well-defined because $H \cap N \trianglelefteq H$ by part 2 and $N \trianglelefteq HN$ since $N \trianglelefteq G$ and $HN$ is a subgroup of $G$ containing $N$ as a consequence of \Cref{proposition:intersection_of_normal_subgroup_and_subgroup_is_normal_subgroup_of_subgroup}.

        \item Define $\psi : H/(H \cap N) \to HN/N$ by $\psi(h(H \cap N)) = hN$. We show that it is a group isomorphism. To show that $\psi$ is well defined, suppose that $h_1 (H \cap N) = h_2 (H \cap N)$ for some $h_1, h_2 \in H$. Then $h_2^{-1} h_1 \in H \cap N$, so in particular $h_2^{-1} h_1 \in N$, which implies that $h_1 N = h_2 N$\CrefIfExists{lemma:cosets_are_equal_iff_product_of_elements_is_in_subgroup}. Hence $\psi(h_1(H \cap N)) = \psi(h_2(H \cap N))$.

        The map $\psi$ is a group homomorphism because for any $h_1, h_2 \in H$,
        $$\psi\bigl((h_1 (H \cap N))(h_2 (H \cap N))\bigr) = \psi\bigl((h_1 h_2)(H \cap N)\bigr) = h_1 h_2 N = (h_1 N)(h_2 N) = \psi(h_1(H \cap N)) \cdot \psi(h_2(H \cap N)).$$
        \CrefIfExists{definition:quotient_of_a_group_by_a_normal_subgroup}

        We show that $\psi$ is injective. Suppose that $\psi(h(H \cap N)) = N$, the identity in $HN/N$. Then $hN = N$, so $h \in N$. Since $h \in H$ as well, we have that $h \in H \cap N$, which means $h(H \cap N) = H \cap N$, the identity in $H/(H \cap N)$. Hence $\operatorname{ker}(\psi)$ is trivial, so $\psi$ is injective by \Cref{proposition:basic_facts_about_kernel_and_image_of_group_homomorphisms}\eqref{item:group_hom_is_injective_iff_kernel_is_trivial}.

        We show that $\psi$ is surjective. Given any element $xN \in HN/N$, we have $x = hn$ for some $h \in H, n \in N$ so $xN = hnN = hN = \psi(h(H \cap N))$.  Therefore, $\psi$ is a group isomorphism $H/(H \cap N) \cong HN/N$.
    \end{enumerate}

\end{proof}
