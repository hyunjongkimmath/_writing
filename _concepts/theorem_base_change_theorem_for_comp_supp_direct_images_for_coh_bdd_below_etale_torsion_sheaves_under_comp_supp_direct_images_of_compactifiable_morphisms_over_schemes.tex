\begin{theorem}[Base change theorem for direct images with proper/compact support, see {\cite[Expos\'e XVII 5.2]{SGA4_III}}, {\cite[Theorem 7.4.4]{fu_ect}}, see also {\cite[Theorem I.8.7]{freitag_kiehl_ecwc}}] \label{theorem:base_change_theorem_for_comp_supp_direct_images_for_coh_bdd_below_etale_torsion_sheaves_under_comp_supp_direct_images_of_compactifiable_morphisms_over_schemes}
Let $S$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme}. Let
\begin{center}
    \begin{tikzcd}
        X' \ar[r, "f'"] \ar[d,"g'"] & Y' \ar[d, "g"] \\
        X \ar[r, "f"] & Y
    \end{tikzcd}
\end{center}
be a \CrefAndHyperrefIfExist{definition:cartesian_product_of_two_objects_in_a_category_over_an_object}{Cartesian diagram} of $S$-morphisms between $S$-schemes such that $f$ is \CrefAndHyperrefIfExist{definition:compactification_of_a_morphism_of_schemes}{$S$-compactifiable} and $Y'$ is \CrefAndHyperrefIfExist{definition:quasi_compact_morphism_of_schemes}{quasi-compact} and \CrefAndHyperrefIfExist{definition:quasi_separated_morphism_of_schemes}{(absolutely) quasi-separated}.

There is a canonical isomorphism $g^* Rf_{!} K \cong Rf_!' {g'}^* K$\CrefIfExists{definition:six_functor_formalism_on_derived_category_of_etale_sheaves_with_torsion_coefficients_on_a_noetherian_scheme}\CrefIfExists{definition:derived_direct_image_functor_with_proper_or_compact_support_for_compactifiable_morphisms_of_schemes_over_a_scheme} for any $K \in D_\text{tor}^+(X_{\'et})$\CrefIfExists{notation:derived_category_of_sheaves_of_abelian_groups_on_small_site_of_a_scehem}\CrefIfExists{definition:small_etale_site_of_a_scheme}.
\end{theorem}
