\begin{proposition} \label{proposition:artinian_involution_ring_is_semisimple_involution_ring_iff_semisimple_as_ordinary_ring}
    Let $(R,\sigma)$ be an \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{involution ring} with $R$ (left/right) \CrefAndHyperrefIfExist{definition:left_right_artinian_ring}{Artinian}. 
    The involution ring $(R,\sigma)$ is a \CrefAndHyperref{definition:semisimple_involution_ring}{semisimple involution ring} if and only if the ring $R$ is \CrefAndHyperref{definition:semisimple_ring}{semisimple as an ordinary ring}.
\end{proposition}
