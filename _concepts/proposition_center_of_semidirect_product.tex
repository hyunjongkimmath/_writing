\begin{proposition} \label{proposition:center_of_semidirect_product}
    Let $N$ and $H$ be \CrefAndHyperrefIfExist{definition:group}{groups}, and let $\varphi: H \to \operatorname{Aut}(N)$ be a \CrefAndHyperrefIfExist{definition:group_homomorphism}{homomorphism}. Write the action of $h \in H$ on $n \in N$ as $h \cdot n = \varphi(h)(n)$. Let $Z(N)$ and $Z(H)$ denote the \CrefAndHyperrefIfExist{definition:center_of_a_group}{centers} of $N$ and $H$, respectively, and let
    $$N^H = \{ n \in N : h \cdot n = n \text{ for all } h \in H \}$$
    be the set of $H$-fixed points in $N$. The \CrefAndHyperrefIfExist{definition:center_of_a_group}{center} of the \CrefAndHyperrefIfExist{definition:semidirect_product_of_groups_external_and_internal}{external semidirect product} $N \rtimes_\varphi H$ is
    $$Z(N \rtimes_\varphi H) = \bigl( Z(N) \cap N^H \bigr) \times \bigl( Z(H) \cap \ker(\varphi) \bigr).$$
\end{proposition}
