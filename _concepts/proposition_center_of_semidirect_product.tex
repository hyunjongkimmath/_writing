Let $Z(N)$ and $Z(H)$ denote the \CrefAndHyperrefIfExist{definition:center_of_a_group}{centers} of $N$ and $H$, respectively, and let
    $$N^H = \{ n \in N : h \cdot n = n \text{ for all } h \in H \}$$

    $$Z(N \rtimes_\varphi H) = \bigl( Z(N) \cap N^H \bigr) \times \bigl( Z(H) \cap \ker(\varphi) \bigr).$$
\end{proposition}
