\begin{theorem} \label{theorem:cech_cohom_for_a_sheaf_of_ab_gps_agrees_with_sheaf_cohom_on_a_paracomp_haus_space_qcqs_scheme_or_loc_contr_semi_loc_simp_conn_space}
    \TODO{there might be more circumstances under which these are true.}
    Let $X$ be a \CrefAndHyperrefIfExist{definition:topological_space}{topological space}. There is a \CrefAndHyperrefIfExist{definition:natural_transformation_between_functors_between_categories}{natural isomorphism} 
    $$\check{H}^p(X, \calF) \cong H^p(X,\calF)$$
    for all $p \geq 0$ where $\check{H}^p(X, \calF)$ is the \CrefAndHyperrefIfExist{definition:cech_cohomology_of_a_sheaf_of_abelian_groups_on_a_topological_space}{$p$th \v{C}ech cohomology group of $\calF$} and $H^p(X,\calF)$ is the \CrefAndHyperrefIfExist{definition:sheaf_cohomology_objects_of_a_sheaf_on_a_top_space_in_an_ab_catwith_enough_injs_and_sheafification}{$p$th sheaf cohomology group of $\calF$} under any of the following circumstances:
    \begin{enumerate}
        \item $X$ is a \CrefAndHyperrefIfExist{definition:paracompact_topological_space}{paracompact} \CrefAndHyperrefIfExist{definition:separation_axioms_of_topology}{Hausdorff space}
        \item $X$ is a \CrefAndHyperrefIfExist{definition:quasi_compact_scheme}{quasi-compact} and \CrefAndHyperrefIfExist{definition:quasi_separated_morphism_of_schemes_and_separated_morphism_of_schemes}{quasi-separated} \CrefAndHyperrefIfExist{definition:scheme}{scheme} and $\calF$ is a quasi-coherent sheaf.
        \item $X$ is a \CrefAndHyperrefIfExist{definition:locally_contractible_topological_space}{locally contractible} and \CrefAndHyperrefIfExist{definition:semi_locally_simply_connected_space}{semi-locally simply connected} \CrefAndHyperrefIfExist{definition:topological_space}{topological space} and $\calF$ is a \CrefAndHyperrefIfExist{definition:locally_constant_sheaf_on_a_topological_space_valued_in_category_admitting_sheafification}{locally constant sheaf}.
    \end{enumerate}
\end{theorem}
