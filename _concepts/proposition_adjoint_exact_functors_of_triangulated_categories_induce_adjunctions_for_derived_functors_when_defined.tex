\begin{proposition}[{\cite[Tag 0FND]{stacks-project}}] \label{proposition:adjoint_exact_functors_of_triangulated_categories_induce_adjunctions_for_derived_functors_when_defined}
Let $F : \mathcal{D} \to \mathcal{D}'$ and $G : \mathcal{D}' \to \mathcal{D}$ be \CrefAndHyperrefIfExist{definition:exact_functor_between_triangulated_categories}{exact functors of triangulated categories}. Assume $F$ is \CrefAndHyperrefIfExist{definition:adjoint_functors_between_categories_unit_counit_of_adjoint_functors}{right adjoint} to $G$. Let $S$ and $S'$ respectively be multiplicative \CrefAndHyperrefIfExist{definition:saturated_multiplicative_system_in_a_category}{systems} for $\mathcal{D}$ and $\mathcal{D}'$ \CrefAndHyperrefIfExist{definition:compatible_with_triangulation_for_a_multiplicative_system_of_a_triangulated_category}{compatible with the triangulated structures}. Let $K \in \text{Ob}(\mathcal{D})$ and $M \in \text{Ob}(\mathcal{D}')$. If \CrefAndHyperrefIfExist{definition:right_left_derived_functor_of_a_exact_mixed_variant_multifunctor_between_triangulated_categories_relative_to_saturated_multiplicative_systems}{$RF$} is \CrefAndHyperrefIfExist{definition:defined_at_objects_derivable_and_adapted_objects_for_the_derived_functors_of_an_exact_multifunctor_between_triangulated_categories}{defined} at $K$ and $LG$ is defined at $M$, then there is a canonical isomorphism 
\[
\text{Hom}_{\mathcal{D}'[(S')^{-1}]}(M, RF(K)) = \text{Hom}_{\mathcal{D}[S^{-1}]}(LG(M), K)
\]
which is \CrefAndHyperrefIfExist{definition:functor_between_categories}{functorial} in both variables on the \CrefAndHyperrefIfExist{definition:triangulated_category}{triangulated subcategories} of $\mathcal{D}[S^{-1}]$ and $\mathcal{D}'[(S')^{-1}]$ where $RF$ and $LG$ are defined.
\end{proposition}
