\begin{lemma} \label{lemma:series_of_1_over_n_factorial_converges}
    The \CrefAndHyperrefIfExist{definition:infinite_series_and_convergence_of_infinite_series_in_a_topological_abelian_semigroup}{series}
    $$\sum_{n=0}^\infty \frac{1}{n!}$$
    \CrefIfExists{definition:factorial_of_a_nonnegative_integer}
    \CrefAndHyperrefIfExist{definition:convergence_and_limit_of_sequence_in_extended_metric_space_valued_in_ordered_commutative_monoid_with_adjoined_infinity}{converges} in $\bbR$.
\end{lemma}
\begin{proof}
    It suffices to show that the series $\sum_{n=4}^\infty \frac{1}{n!}$ converges.
    By \CrefAndHyperrefIfExist{lemma:factorial_lower_bound_by_power_of_two_for_n_ge_4}{$n! \geq 2^n$ for all $n \geq 4$}, we have $\frac{1}{n!} \leq \frac{1}{2^n}$ for all $n \geq 4$.
    Note that for all integers $m \geq 4$, we have
    \begin{align*}
    \sum_{n=4}^m \frac{1}{n!} &\leq \sum_{n=4}^m \frac{1}{2^n} = \frac{\frac{1}{2^4} - \frac{1}{2^{m+1}}}{1-\frac{1}{2}} = \frac{\frac{1}{2^4} - \frac{1}{2^{m+1}}}{\frac{1}{2}} \leq \frac{\frac{1}{2^4}}{\frac{1}{2}} = \frac{1}{8}.
    \end{align*}
    \CrefIfExists{proposition:sum_formula_for_a_finite_geometric_series_over_a_commutative_ring}
    The sequence of partial sums $\left(\sum_{n=4}^m \frac{1}{n!}\right)_{m \geq 4}$ is monotone increasing and bounded above by $\frac{1}{8}$. By the \CrefAndHyperrefIfExist{corollary:monotone_convergence_theorem_for_sequences_of_real_numbers}{monotone convergence theorem for sequences of real numbers}, the sequence of partial sums converges, hence the series $\sum_{n=4}^\infty \frac{1}{n!}$ converges.
\end{proof}
