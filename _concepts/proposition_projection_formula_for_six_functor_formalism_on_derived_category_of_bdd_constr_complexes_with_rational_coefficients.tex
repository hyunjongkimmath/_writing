\begin{proposition}[Projection formula for six functor formalism with rational adic coefficients] \label{proposition:projection_formula_for_six_functor_formalism_on_derived_category_of_bdd_constr_complexes_with_rational_coefficients}
Let $S$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme}. Let $f: X \to Y$ be a \CrefAndHyperrefIfExist{definition:compactification_of_a_morphism_of_schemes}{compactifiable $S$-morphism} between $S$-schemes where $Y$ is \CrefAndHyperrefIfExist{definition:quasi_compact_morphism_of_schemes}{quasi-compact} and \CrefAndHyperrefIfExist{definition:quasi_separated_morphism_of_schemes_and_separated_morphism_of_schemes}{quasi-separated} over $S$ and where $X$ and $Y$ are \CrefAndHyperrefIfExist{definition:locally_noetherian_and_noetherian_scheme}{noetherian schemes}.
Let $(R, \lambda)$ be a \CrefAndHyperrefIfExist{definition:complete_ring_with_respect_to_an_ideal}{complete} \CrefAndHyperrefIfExist{definition:discrete_valuation_ring}{DVR} such that $R/\lambda^n$ is finite for all $n \geq 1$, and let $E$ be the \CrefAndHyperrefIfExist{definition:field_of_fractions_of_an_integral_domain}{fraction field} of $R$. 

For $K \in \Dbc(X,E)$ and $L \in \Dbc(Y,E)$ (\Cref{definition:derived_category_of_cohomologically_bounded_constructible_sheaves_with_rational_adic_coefficients_on_a_noetherian_scheme}), there is a canonical isomorphism in $\Dbc(Y,E)$:
$$Rf_!(K \otimes_E^L Lf^* L) \cong Rf_! K \otimes_E^L L$$
\CrefIfExists{definition:derived_direct_image_functor_on_bounded_constructible_complexes_with_rational_coefficients_on_a_noetherian_scheme}\CrefIfExists{definition:derived_tensor_product_on_bdd_constructible_complexes_with_rational_coefficients_on_a_noetherian_scheme}\TODO{derived inverse image functor for rational coefficients}\CrefIfExists{definition:inverse_image_functor_on_etale_sheaves_with_rational_adic_coefficients_on_a_noetherian_scheme}

In particular, if $f$ is \CrefAndHyperrefIfExist{definition:proper_morphism_of_schemes}{proper}, then $Rf_! = Rf_*$ and hence the projection formula takes the form
$$Rf_*(K \otimes_E^L Lf^* L) \cong Rf_* K \otimes_E^L L.$$
\end{proposition}
