\begin{theorem}[Isomorphism between derivations and homs from the augmentation ideal, see {\cite[Chapter VI, Theorem 5.1]{hilton_stammbach_cha}}] \label{theorem:isomorphism_between_derivations_and_homs_from_augmentation_ideal}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{ring}, let $\Gamma$ be a \CrefAndHyperrefIfExist{definition:group}{group}, and let $A$ be an abelian \CrefAndHyperrefIfExist{definition:left_right_module_over_a_group_ring_of_a_group_over_a_ring}{$R[\Gamma]$-module}. Let $I_\Gamma$ be the \CrefAndHyperrefIfExist{definition:augmentation_map_of_a_group_ring_to_its_base_ring}{augmentation ideal} of $R[\Gamma]$.

    There is a natural isomorphism of $R$-modules
    $$\Phi: \Der_R(\Gamma, A) \xrightarrow{\cong} \Hom_{R[\Gamma]}(I_\Gamma, A),$$
    given by the assignment
    $$(\Phi(d))(y - 1) = d(y), \quad y \in \Gamma.$$
    The inverse sends $\varphi \in \Hom_{R[\Gamma]}(I_\Gamma, A)$ to the derivation $d_\varphi: \Gamma \to A$ defined by $d_\varphi(y) = \varphi(y - 1)$.
\end{theorem}
