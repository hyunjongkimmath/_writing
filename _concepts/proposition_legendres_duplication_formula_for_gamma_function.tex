\begin{proposition}[Legendre's duplication formula for the Gamma function] \label{proposition:legendres_duplication_formula_for_gamma_function}

	For all complex numbers $z \in \mathbb{C}$ with $\Re(z) > 0$, the \CrefAndHyperrefIfExist{definition:gamma_function_on_the_complex_numbers}{Gamma function} $\Gamma(z)$ satisfies Legendre's duplication formula:
	$$
	\boxed{\Gamma(z) \, \Gamma\!\left( z + \frac{1}{2} \right) = 2^{1 - 2z} \sqrt{\pi} \, \Gamma(2z).}
	$$
\end{proposition}

	$$B(z,z) = \int_{-1}^1 \left(\frac{1+u}{2}\right)^{z-1} \left(\frac{1-u}{2}\right)^{z-1} \cdot \frac{du}{2}.$$

	$$\left(\frac{1+u}{2}\right)^{z-1} \left(\frac{1-u}{2}\right)^{z-1} = 2^{-2(z-1)} (1+u)(1-u))^{z-1} = 2^{-2z+2}(1-u^2)^{z-1}.$$

	$$B(z,z) = \frac{1}{2} \cdot 2^{-2z+2} \int_{-1}^1 (1-u^2)^{z-1}\,du = 2^{-2z+1} \int_0^1 (1-u^2)^{z-1}\,du,$$

	$$B(z,z) = 2^{-2z+1} \cdot \frac{1}{2} \int_0^1 (1-v)^{z-1} v^{-1/2}\,dv = 2^{1-2z} B(1/2, z).$$

	Substituting the Gamma function expressions for both Beta functions:
	$$\frac{\Gamma(z)^2}{\Gamma(2z)} = 2^{1-2z} \frac{\sqrt{\pi}\,\Gamma(z)}{\Gamma(z+1/2)}.$$

	Canceling $\Gamma(z)$ from both sides (which is nonzero for $\Re(z) > 0$):
	$$\frac{\Gamma(z)}{\Gamma(2z)} = \frac{2^{1-2z}\sqrt{\pi}}{\Gamma(z+1/2)}.$$

	Rearranging gives Legendre's duplication formula:
	$$\Gamma(z)\Gamma\!\left(z+\frac{1}{2}\right) = 2^{1-2z}\sqrt{\pi}\,\Gamma(2z).$$

	The formula holds for $\Re(z) > 0$ by the above computation. By analytic continuation, it extends to all $z \in \mathbb{C}$ except at the poles of the Gamma function.
\end{proof}
