\begin{proposition} \label{proposition:fiber_bundles_with_paracompact_base_is_a_hurewicz_fibration}
Let $\pi: E \to B$ be a \CrefAndHyperrefIfExist{definition:fiber_bundle_of_topological_spaces}{fiber bundle} with fiber $F$. 
If $B$ is a \CrefAndHyperrefIfExist{definition:paracompact_topological_space}{paracompact space}, then $\pi$ is a \CrefAndHyperrefIfExist{definition:hurewicz_fibration_of_topological_spaces}{Hurewicz fibration}.
\end{proposition}
