\begin{lemma}[Holomorphic function defined by a parameter integral with uniform convergence] \label{lemma:holomorphic_function_defined_by_parameter_integral_with_uniform_convergence}

Let $U \subseteq \mathbb{C}$ be an open subset, let $(X, \mathcal{A}, \mu)$ be a \CrefAndHyperrefIfExist{definition:measure_on_a_measurable_space_and_measure_space}{measure space}, and let $f : U \times X \to \mathbb{C}$ be a function satisfying:
\begin{enumerate}
    \item For every $s \in U$, the function $x \mapsto f(s, x)$ is \CrefAndHyperrefIfExist{definition:vector_valued_lebesgue_integral_into_normed_vector_space}{integrable} with respect to $\mu$ (as a function into $\mathbb{C}$).
    \item For $\mu$-almost every $x \in X$, the function $s \mapsto f(s, x)$ is \CrefAndHyperrefIfExist{definition:holomorphic_function_from_a_subset_of_Cn_to_Cm_and_derivative_of_a_holomorphic_function_at_a_point}{holomorphic} on $U$.
    \item The integral $F(s) = \int_X f(s, x) \, d\mu(x)$ converges uniformly on every \CrefAndHyperrefIfExist{definition:compact_subset_of_a_topological_space_alternate_definition}{compact subset} $K \subset U$; i.e., for every sequence $(s_n)$ in $K$ converging to $s_0 \in K$, the sequence of functions $F_n(s) = \int_X f(s, x) \, d\mu(x)$ converges uniformly on $K$ to $F(s)$.
\end{enumerate}
Then $F : U \to \mathbb{C}$ is holomorphic on $U$.
\end{lemma}
\begin{proof}

Let $K \subset U$ be compact. For each $n \in \mathbb{N}$, approximate the integral by a Riemann sum or a sequence of simple function integrals. More concretely, choose a sequence of measurable sets $X_{n,1}, \dots, X_{n,m_n}$ partitioning $X$ and points $x_{n,j} \in X_{n,j}$ such that the simple functions
$$f_n(s) = \sum_{j=1}^{m_n} f(s, x_{n,j}) \, \mu(X_{n,j})$$
converge uniformly on $K$ to $F(s)$ by the dominated convergence theorem and the uniform convergence hypothesis. Each $f_n$ is a finite linear combination of holomorphic functions (in $s$) times constants, hence holomorphic on $U$. By the \CrefAndHyperrefIfExist{theorem:the_weierstrass_convergence_theorem_stating_that_the_uniform_limit_on_compact_subsets_of_a_sequence_of_holomorphic_functions_is_itself_holomorphic_with_all_derivatives_converging_uniformly_on_compact_subsets}{Weierstrass convergence theorem}, the uniform limit $F$ of the sequence $(f_n)$ on compact subsets is holomorphic on $U$.
\end{proof}
