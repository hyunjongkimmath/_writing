\begin{lemma} \label{lemma:center_of_a_finite_p_group_is_nontrivial}
Let $G$ be a non-trivial \CrefAndHyperrefIfExist{definition:p_group}{finite $p$-group}. Then the \CrefAndHyperrefIfExist{definition:center_of_a_group}{center of $G$}, $Z(G)$, is non-trivial. In particular,
$$|Z(G)| \geq p.$$
\end{lemma}
