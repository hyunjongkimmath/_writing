\begin{proposition}[See {\cite[02WH]{stacks-project}}] \label{proposition:list_of_properties_of_morphisms_of_schemes_over_a_base_preserved_under_base_change_and_fpqc_local_on_the_base}
    Let $S$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme}. The following are properties for morphisms of $S$-schemes that are preserved under \CrefAndHyperrefIfExist{definition:extension_of_scalars_base_change_induction_for_modules_along_ring_homomorphisms}{base change} and \CrefAndHyperrefIfExist{definition:tau_local_property_of_a_morphism_of_schemes_over_a_scheme}{fpqc-local on the base}:

    \begin{itemize}
        \item \CrefAndHyperrefIfExist{definition:open_and_closed_immersions_of_schemes}{closed immersions} (\CrefIfExists{proposition:closed_immersion_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:open_and_closed_immersions_of_schemes}{open immersions} (\CrefIfExists{proposition:open_immersion_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:quasi_compact_immersion_of_schemes}{quasi-compact immersions} (\CrefIfExists{proposition:quasi_compact_immersion_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:quasi_compact_morphism_of_schemes}{quasi-compact} (\CrefIfExists{proposition:quasi_compact_morphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:universally_closed_morphism_of_schemes}{universally closed} (\CrefIfExists{proposition:universally_closed_morphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:quasi_separated_morphism_of_schemes}{quasi-separated} (\CrefIfExists{proposition:quasi_separated_morphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:quasi_separated_morphism_of_schemes}{separated} (\CrefIfExists{proposition:separated_morphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:monomorphism_and_epimorphism_in_categories}{monomorphism} (\CrefIfExists{proposition:monomorphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:injective_surjective_bijective_map_of_sets}{surjective} (\CrefIfExists{proposition:surjective_morphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:universally_injective_morphism_of_schemes}{universally injective} (\CrefIfExists{proposition:universally_injective_morphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:affine_morphism_of_schemes}{affine} (\CrefIfExists{proposition:affine_morphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:quasi_affine_morphism_of_schemes}{quasi-affine} (\CrefIfExists{proposition:quasi_affine_morphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:finite_type_morphism_of_schemes}{locally of finite type} (\CrefIfExists{proposition:locally_finite_type_morphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:finite_type_morphism_of_schemes}{finite type} (\CrefIfExists{proposition:finite_type_morphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:locally_quasifinite_morphism_of_schemes}{locally quasi-finite} (\CrefIfExists{proposition:locally_quasifinite_morphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:quasifinite_morphism_of_schemes}{quasi-finite} (\CrefIfExists{proposition:quasifinite_morphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:locally_of_finite_presentation_finite_presentation_morphism_of_schemes}{locally of finite presentation} (\CrefIfExists{proposition:locally_finite_presentation_morphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:locally_of_finite_presentation_finite_presentation_morphism_of_schemes}{finite presentation} (\CrefIfExists{proposition:finite_presentation_morphism_of_schemes_stable_under_base_change_and_composition})
        \item locally of finite type of \CrefAndHyperrefIfExist{definition:relative_dimension_for_a_morphism_of_schemes}{relative dimension} $d$ (\CrefIfExists{proposition:relative_dimension_morphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:universally_open_morphism_of_schemes}{universally open} (\CrefIfExists{proposition:universally_open_morphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:flat_morphism_of_schemes}{flat} (\CrefIfExists{proposition:flat_morphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:syntomic_morphism_of_schemes}{syntomic} (\CrefIfExists{proposition:syntomic_morphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:smooth_morphism_of_schemes}{smooth} (\CrefIfExists{proposition:smooth_morphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:unramified_morphism_of_schemes}{unramified} (\CrefIfExists{proposition:unramified_morphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:G_unramified_morphism_of_schemes}{$G$-unramified} (\CrefIfExists{proposition:G_unramified_morphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:etale_morphism_of_schemes}{\'etale} (\CrefIfExists{proposition:etale_morphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:proper_morphism_of_schemes}{proper} (\CrefIfExists{proposition:proper_morphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:finite_morphism_of_schemes}{finite} (\CrefIfExists{proposition:finite_morphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:integral_scheme}{integral} (\CrefIfExists{proposition:integral_morphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:finite_locally_free_morphism_of_schemes}{finite locally free} (\CrefIfExists{proposition:finite_locally_free_morphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:universally_submersive_morphism_of_schemes}{universally submersive} (\CrefIfExists{proposition:universally_submersive_morphism_of_schemes_stable_under_base_change_and_composition})
        \item \CrefAndHyperrefIfExist{definition:universal_homeomorphism_of_schemes}{universal homeomorphism} (\CrefIfExists{proposition:universal_homeomorphism_of_schemes_stable_under_base_change_and_composition})
    \end{itemize}

    In particular, these properties must be local on the base for the \CrefAndHyperrefIfExist{definition:big_fppf_site_of_a_scheme}{fppf}, syntomic, smooth, \CrefAndHyperrefIfExist{definition:big_etale_site_of_a_scheme}{\'etale}, and \CrefAndHyperrefIfExist{definition:big_zariski_site_of_a_scheme}{Zariski} topologies as well as per Remark \ref{remark:a_property_that_is_local_on_the_base_for_a_fine_topology_must_be_local_on_the_base_for_a_coarse_topology} because they are all \CrefAndHyperrefIfExist{definition:coarser_and_finer_grothendieck_topologies}{coarser} than the fpqc topology.

    As per \Cref{definition:property_for_a_representable_morphism_of_sheaves_coming_from_a_property_of_morphisms_of_schemes_preserved_by_base_change_and_base_local}, the above properties of morphisms of schemes induce properties of morphisms of \CrefAndHyperrefIfExist{definition:algebraic_space_on_a_site_on_Sch_S_that_preserves_under_base_change_and_has_tau_local_surjectivity_and_etaleness}{algebraic spaces}.

\end{proposition}
