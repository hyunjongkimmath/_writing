\begin{lemma} \label{lemma:1_is_least_element_of_natural_numbers}
    $1$ is the least element of \CrefAndHyperrefIfExist{definition:natural_numbers}{$\bbN$} (equipped with the \CrefAndHyperrefIfExist{definition:standard_order_on_natural_numbers}{standard order}), i.e. for all $n \in \bbN$ either $n = 1$ or $1 < n$. In particular, for all $n \in \bbN$, $1 \leq n$. 
\end{lemma}
\begin{proof}
    If $n \neq 1$, then \Cref{lemma:natural_numbers_are_either_1_or_are_successors} $n$ is $s(j)$ for some $j \in \bbN$, i.e. $n = j+1$ by \Cref{proposition:natural_numbers_addition_existence}. By the \CrefAndHyperrefIfExist{proposition:natural_numbers_addition_associative_commutative}{commutativity of addition}, $n = 1+j$ and hence $1 < n$. 
\end{proof}
