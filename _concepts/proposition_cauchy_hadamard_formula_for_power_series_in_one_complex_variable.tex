\begin{proposition}[Cauchy--Hadamard formula for power series in one complex variable] \label{proposition:cauchy_hadamard_formula_for_power_series_in_one_complex_variable}
    Let $\sum_{n=0}^\infty a_n (z - z_0)^n$ be a \CrefAndHyperrefIfExist{definition:power_series_in_one_complex_variable_and_its_radius_of_convergence_given_by_the_cauchy_hadamard_formula}{power series in one complex variable} with coefficients $a_n \in \mathbb{C}$. Its radius of convergence $R \in [0, \infty]$ is given by the Cauchy--Hadamard formula
    $$R = \frac{1}{\limsup_{n \to \infty} |a_n|^{1/n}},$$
    \CrefIfExists{definition:least_upper_bound_and_greatest_lower_bound_of_a_subset_of_an_ordered_set}
    with the conventions $1/0 = \infty$ and $1/\infty = 0$. More precisely, the series \CrefAndHyperrefIfExist{definition:absolute_convergence_of_a_series_in_module_over_a_commutative_ring_with_an_absolute_value_valued_in_an_ordered_semiring}{converges absolutely} for every $z$ with $|z - z_0| < R$ and \CrefAndHyperrefIfExist{definition:convergence_and_limit_of_sequence_in_extended_metric_space_valued_in_ordered_commutative_monoid_with_adjoined_infinity}{diverges} for every $z$ with $|z - z_0| > R$.
\end{proposition}

\begin{proof}
    \TODO{technically the root test is so far only written for $\bbR$; generalize it}
    Fix $z \in \mathbb{C}$ with $|z - z_0| = r$ and consider the series $\sum_{n=0}^\infty a_n (z - z_0)^n$. The sequence of nonnegative real numbers $|a_n (z - z_0)^n|$ satisfies
    $$\limsup_{n \to \infty} |a_n (z - z_0)^n|^{1/n} = r \cdot \limsup_{n \to \infty} |a_n|^{1/n}.$$
    If $r < R$, then $r \cdot \limsup_{n \to \infty} |a_n|^{1/n} < 1$, so \Cref{theorem:root_test_for_absolute_convergence_of_series} implies that $\sum_{n=0}^\infty |a_n (z - z_0)^n|$ converges absolutey. If $r > R$, then $r \cdot \limsup_{n \to \infty} |a_n|^{1/n} > 1$, so by the same theorem the series $\sum_{n=0}^\infty |a_n (z - z_0)^n|$ diverges, and consequently $\sum_{n=0}^\infty a_n (z - z_0)^n$ diverges.
\end{proof}
