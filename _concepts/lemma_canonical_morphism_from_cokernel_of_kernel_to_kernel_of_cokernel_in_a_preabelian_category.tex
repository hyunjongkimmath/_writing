\begin{lemma} \label{lemma:canonical_morphism_from_cokernel_of_kernel_to_kernel_of_cokernel_in_a_preabelian_category}
    Let $\calA$ be a \CrefAndHyperrefIfExist{definition:pre_abelian_category}{pre-abelian category}. 
    For any morphism $f$ in $\calA$, there is a canonical morphism
    $$\coker(\ker f) \to \ker(\coker f).$$
\end{lemma}
