\begin{theorem}[Comparison theorem for spectral sequences, see e.g. {\cite[5.2.12]{weibel}}] \label{theorem:comparison_theorem_of_spectral_sequences_in_an_abelian_category}
    Let $\calA$ be an \CrefAndHyperrefIfExist{definition:abelian_category}{abelian category}. 
    \begin{enumerate}
        \item Let $\left\{E_{p q}^{r}\right\}$ and $\left\{E_{p q}^{\prime r} \right\}$ be \CrefAndHyperrefIfExist{definition:spectral_sequence_in_an_abelian_category}{homological spectral sequences} in $\calA$ \CrefAndHyperrefIfExist{definition:converges_to_a_graded_object_for_a_spectral_sequence_in_an_abelian_category}{converging to} $H_{*}$ and $H_{*}^{\prime}$ respectively. 

        Suppose that $h: H_{*} \rightarrow H_{*}^{\prime}$ is a map \CrefAndHyperrefIfExist{definition:compatible_with_a_morphism_between_spectral_sequences_in_an_abelian_category_for_filtered_objects_that_the_spectral_sequences_weakly_converge_to}{compatible with} a \CrefAndHyperrefIfExist{definition:morphism_of_spectral_sequences_in_an_abelian_category}{morphism} $f: E \rightarrow E^{\prime}$ of spectral sequences. If $f^{r}: E_{p q}^{r} \cong E_{p q}^{\prime}$ is an isomorphism for all $p$ and $q$ and some $r$, and hence for $r=\infty$ by the \CrefAndHyperrefIfExist{lemma:mapping_lemma_for_spectral_sequences_in_abelian_categories}{mapping lemma}, then $h: H_{*} \rightarrow H_{*}^{\prime}$ is an isomorphism.

        \item Let $\left\{E_{r}^{p q}\right\}$ and $\left\{E_{r}^{\prime p q} \right\}$ be \CrefAndHyperrefIfExist{definition:spectral_sequence_in_an_abelian_category}{cohomological spectral sequences} in $\calA$ \CrefAndHyperrefIfExist{definition:converges_to_a_graded_object_for_a_spectral_sequence_in_an_abelian_category}{converging to} $H^{*}$ and $H^{\prime *}$ respectively. 

        Suppose that $h: H^{*} \rightarrow H^{\prime *}$ is a map \CrefAndHyperrefIfExist{definition:compatible_with_a_morphism_between_spectral_sequences_in_an_abelian_category_for_filtered_objects_that_the_spectral_sequences_weakly_converge_to}{compatible with} a \CrefAndHyperrefIfExist{definition:morphism_of_spectral_sequences_in_an_abelian_category}{morphism} $f: E \rightarrow E^{\prime}$ of spectral sequences. If $f_{r}: E_{r}^{p q} \cong E_{r}^{\prime p q}$ is an isomorphism for all $p$ and $q$ and some $r$, and hence for $r=\infty$ by the \CrefAndHyperrefIfExist{lemma:mapping_lemma_for_spectral_sequences_in_abelian_categories}{mapping lemma}, then $h: H^{*} \rightarrow H^{\prime *}$ is an isomorphism.

    \end{enumerate}
\end{theorem}
