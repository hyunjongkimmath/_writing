\begin{theorem}[Existence and Uniqueness of Splitting Fields] \label{theorem:existence_and_uniqueness_of_splitting_fields}
Let $K$ be a \CrefAndHyperrefIfExist{definition:field}{field} and let $f \in K[t]$ be a non-constant polynomial.

\begin{enumerate}
    \item \textbf{(Existence)} There exists a \CrefAndHyperrefIfExist{definition:splitting_field_of_polynomials_over_a_field_normal_field_extension}{splitting field} of $f$ over $K$.
    \item \textbf{(Uniqueness up to isomorphism)} Let $L_1$ and $L_2$ be two \CrefAndHyperrefIfExist{definition:splitting_field_of_polynomials_over_a_field_normal_field_extension}{splitting fields} of $f$ over $K$. Then there exists a $K$-isomorphism $L_1 \xrightarrow{\,\sim\,} L_2$.
\end{enumerate}
More generally, the same existence and uniqueness statements hold for any non-empty set $S \subseteq K[t]$ of polynomials.
\end{theorem}
