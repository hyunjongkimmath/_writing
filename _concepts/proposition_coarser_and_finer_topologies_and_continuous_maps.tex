\begin{proposition} \label{proposition:coarser_and_finer_topologies_and_continuous_maps}
Let $X$ be a set and let $f: (X, \tau_X) \to (Y, \tau_Y)$ be a \CrefAndHyperrefIfExist{definition:function_of_sets}{function} between \CrefAndHyperrefIfExist{definition:topological_space}{topological spaces}. 
\begin{itemize}
    \item If $\tau_X'$ is a \CrefAndHyperrefIfExist{definition:topological_space}{topology} on $X$ \CrefAndHyperrefIfExist{definition:coarser_finer_topologies_on_a_set}{finer} than $\tau_X$ and $f$ is \CrefAndHyperrefIfExist{definition:continuous_map_of_topological_spaces}{continuous}, then $f: (X, \tau_X') \to (Y, \tau_Y)$ is also continuous.
    \item If $\tau_Y'$ is a topology on $Y$ coarser than $\tau_Y$ and $f$ is continuous, then $f: (X, \tau_X) \to (Y, \tau_Y')$ is also continuous.
\end{itemize}
\end{proposition}
