\begin{lemma} \label{lemma:element_of_free_module_over_ring_is_unimodular_iff_it_is_part_of_some_basis}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{not necessarily commutative ring}. Let $M$ be a \CrefAndHyperrefIfExist{definition:free_bimodule_over_rings}{free left/right $R$-module}.
    An element $v \in M$ is \CrefAndHyperrefIfExist{definition:unimodular_element_of_a_module_over_a_ring}{unimodular} if and only if there exists a \CrefAndHyperrefIfExist{definition:free_bimodule_over_rings}{basis} of $M$ containing $v$.
\end{lemma}
