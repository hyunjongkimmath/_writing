\begin{proposition}[{\cite[Proposition 2.2.15.]{beilinson_berstein_deligne_atlesI}}] \label{proposition:projective_system_of_triangulated_categories_indexed_by_integers_with_exact_transition_functors_and_finite_Hom_sets_has_triangulated_projective_2_limit}
    Let $(D_n)_{n \in \bbN}$ be a \CrefAndHyperrefIfExist{definition:system_in_a_category_indexed_by_a_directed_poset}{projective system} of \CrefAndHyperrefIfExist{definition:triangulated_category}{triangulated categories} such that the transition \CrefAndHyperrefIfExist{definition:functor_between_categories}{functors} $T_{m,n}: D_n \to D_m$ (for $m \leq n$) are \CrefAndHyperrefIfExist{definition:exact_functor_between_abelian_categories}{exact}. 
    \TODO{projective $2$-limit}
    If $\Hom_{D_n}(K,L)$ is a \CrefAndHyperrefIfExist{definition:countable_finite_uncountable_sets}{finite set} for all $n \in \bbN$ and all $K,L \in D_n$, then the projective $2$-limit $D$ of $D_n$, equipped with the family of triangles whose images in the $D_n$ are \CrefAndHyperrefIfExist{definition:triangulated_category}{distinguished triangles}, is a \CrefAndHyperrefIfExist{definition:triangulated_category}{triangulated category}.
\end{proposition}
