\begin{lemma} \label{lemma:inverse_of_power_equals_power_of_inverse_in_a_group}
Let $G$ be a \CrefAndHyperrefIfExist{definition:group}{group} and let $g \in G$. For all integers $n \in \bbZ$,
$$(g^n)^{-1} = g^{-n} = (g^{-1})^n,$$
where $g^k$ denotes the \CrefAndHyperrefIfExist{definition:power_of_a_group_element_by_an_integer}{$k$th power of $g$} and $g^{-1}$ is the \CrefAndHyperrefIfExist{definition:group}{inverse} of $g$.
\end{lemma}
