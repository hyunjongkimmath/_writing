\begin{theorem}[Homological Künneth Formula for Complexes, see {\cite[Theorem 3.6.3]{weibel}}] \label{theorem:homological_kunneth_formula_for_complexes_of_modules_over_rings}
Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{ring}. Let $P$ be a \CrefAndHyperrefIfExist{definition:chain_complex_of_objects_in_an_additive_category}{chain complex} of \CrefAndHyperrefIfExist{definition:flat_left_right_module_over_a_ring}{flat} \CrefAndHyperrefIfExist{definition:module_of_a_ring}{right $R$-modules} such that each \CrefAndHyperrefIfExist{definition:submodule_of_a_module_over_a_ring}{submodule} of \CrefAndHyperrefIfExist{definition:boundary_cycle_coboundary_cocyble_of_a_chain_cochain_complex}{boundaries} $d(P_n) \subseteq P_{n-1}$ is also flat. Let $Q$ be any chain complex of \CrefAndHyperrefIfExist{definition:module_of_a_ring}{left $R$-modules}. Then there is a natural \CrefAndHyperrefIfExist{definition:short_exact_sequence_in_an_additive_category}{exact sequence} for every $n$:

$$ 0 \rightarrow \bigoplus_{i+j=n} H_i(P) \otimes_R H_j(Q) \rightarrow H_n(\Tot^{\bigoplus}(P \otimes_R Q)) \rightarrow \bigoplus_{i+j=n-1} \operatorname{Tor}_1^R(H_i(P), H_j(Q)) \rightarrow 0. $$
\CrefIfExists{definition:total_complexes_of_a_double_complex_of_objects_in_an_additive_category} \CrefIfExists{definition:double_complex_associated_to_biadditive_functor_and_chain_complexes} \CrefIfExists{definition:homology_and_cohomology_objects_for_a_chain_complex_in_an_additive_category} \CrefIfExists{definition:tor_functors_of_bimodules_of_rings}

In the case that $R$ is additionally a \CrefAndHyperrefIfExist{definition:hereditary_ring}{right hereditary ring} and $P$ is any \CrefAndHyperrefIfExist{definition:chain_complex_of_objects_in_an_additive_category}{complex} of \CrefAndHyperrefIfExist{definition:projective_bimodule_over_rings}{projective} right $R$-modules, the modules $d(P_n)$ are automatically projective and hence the above holds --- In fact, the above short exact sequence \CrefAndHyperrefIfExist{definition:split_exact_sequence_in_an_additive_category}{splits} (non-canonically) in this case.
\TextIfExists{theorem:cohomological_kunneth_formula_for_complexes_of_modules_over_rings}{See \Cref{theorem:cohomological_kunneth_formula_for_complexes_of_modules_over_rings} for a dual statement.}
\end{theorem}
