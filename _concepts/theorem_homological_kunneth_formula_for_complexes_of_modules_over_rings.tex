\begin{theorem}[Homological Künneth Formula for Complexes] \label{theorem:homological_kunneth_formula_for_complexes_of_modules_over_rings}
Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{ring}. Let $P$ be a \CrefAndHyperrefIfExist{definition:chain_complex_of_objects_in_an_additive_category}{chain complex} of \CrefAndHyperrefIfExist{definition:flat_left_right_module_over_a_ring}{flat} \CrefAndHyperrefIfExist{definition:module_of_a_ring}{right $R$-modules} such that each \CrefAndHyperrefIfExist{definition:submodule_of_a_module_over_a_ring}{submodule} of \CrefAndHyperrefIfExist{definition:boundary_cycle_coboundary_cocyble_of_a_chain_cochain_complex}{boundaries} $d(P_n) \subseteq P_{n-1}$ is also flat. Let $Q$ be any chain complex of \CrefAndHyperrefIfExist{definition:module_of_a_ring}{left $R$-modules}. Then there is a natural \CrefAndHyperrefIfExist{definition:short_exact_sequence_in_an_additive_category}{exact sequence} for every $n$:
$$ 0 \rightarrow \bigoplus_{i+j=n} H_i(P) \otimes_R H_j(Q) \rightarrow H_n(P \otimes_R Q) \rightarrow \bigoplus_{i+j=n-1} \operatorname{Tor}_1^R(H_i(P), H_j(Q)) \rightarrow 0 $$
Furthermore, if $R$ is a \CrefAndHyperrefIfExist{definition:hereditary_ring}{hereditary ring} and $P$ is a \CrefAndHyperrefIfExist{definition:chain_complex_of_objects_in_an_additive_category}{complex} of \CrefAndHyperrefIfExist{definition:projective_bimodule_over_rings}{projective} \CrefAndHyperrefIfExist{definition:module_of_a_ring}{modules}, this sequence \CrefAndHyperrefIfExist{definition:split_exact_sequence_in_an_additive_category}{splits} (non-canonically).
\end{theorem}
