\begin{proposition}[Isomorphism criterion for hyperbolic planes] \label{proposition:isomorphism_criterion_for_a_quadratic_module_over_a_commutative_ring_to_be_a_hyperbolic_plane}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:commutative_ring}{commutative ring}, and let $(H, q)$ be a quadratic module such that $H$ is a free $R$-module of rank 2. Then $(H, q)$ is a \CrefAndHyperrefIfExist{definition:hyperbolic_plane_over_a_commutative_ring}{hyperbolic plane} if and only if there exists an $R$-basis $\{u, v\}$ for $H$ such that
    $$q(r u + s v) = rs \quad \text{for all } r, s \in R.$$

\end{proposition}
