\begin{theorem} \label{theorem:ext_objects_for_general_abelian_categories_are_well_defined_even_if_some_combination_of_projectives_or_injectives_exist}
    Let $\calA$ be an \CrefAndHyperrefIfExist{definition:abelian_category}{abelian category}. Let $A_1,A_2 \in \calA$ be objects.
    \begin{enumerate}
        \item The objects $\Ext^n_\calA(A_1,A_2)$ computed via \CrefAndHyperrefIfExist{definition:left_right_resolution_of_a_class_of_objects_in_an_abelian_category}{injective resolutions} of $A_2$ are well defined. In other words, if $A_2 \to I^{i,\bullet}$ are injective resolutions of $A_2$ for $i = 1,2$, then there are \CrefAndHyperrefIfExist{definition:natural_transformation_between_functors_between_categories}{natural isomorphisms}
        $$H^n(\Hom_\calA(A_1, I^{1, \bullet})) \cong H^n(\Hom_\calA(A_1, I^{2, \bullet}))$$
        for all $n \geq 0$.

        \item Dually, the objects $\Ext^n_\calA(A_1,A_2)$ computed via \CrefAndHyperrefIfExist{definition:left_right_resolution_of_a_class_of_objects_in_an_abelian_category}{projective resolutions} of $A_1$ are well defined. In other words, if $P_{i,\bullet} \to A_1$ are projective resolutions of $A_2$ for $i = 1,2$, then there are natural isomorphisms
        $$H^n(\Hom_\calA(P_{1,\bullet}, A_2)) \cong H^n(\Hom_\calA(P_{2,\bullet}, A_2))$$
        for all $n \geq 0$.

        \item Suppose that an injective resolution $A_2 \to I^\bullet$ of $A_2$ and a projectve resolution $P_\bullet \to A_1$ exist. The two objects $\Ext^n_\calA(A_1,A_2)$ computed by each of these two resolutions are naturally isomorphic. In other words, there are natural isomorphisms
        $$H^n(\Hom_\calA(P_\bullet, A_2)) \cong H^n(\Hom_\calA(A_1, I^\bullet))$$
        for all $n \geq 0$.
    \end{enumerate}
\end{theorem}
\begin{proof}
    \begin{enumerate}
        \item This follows from \Cref{proposition:left_right_derived_objects_for_a_right_left_exact_functor_between_abelian_categories_are_well_defined}.
        \item This follows from \Cref{proposition:left_right_derived_objects_for_a_right_left_exact_functor_between_abelian_categories_are_well_defined}.
        \item This follows from \Cref{theorem:balancing_of_tor_on_abelian_categories_with_right_exact_bifunctor}; note that projective objects and injective objects in $\calA$ are \CrefAndHyperrefIfExist{definition:flat_object_in_an_abelian_category_with_respect_to_a_right_exact_monoidal_product_functor}{flat}/exact for the first and second inputs of $\Hom_\calA$ respectively by \Cref{lemma:injective_projective_objects_in_an_abelian_category_have_exact_Hom_functors}.
    \end{enumerate}
\end{proof}
