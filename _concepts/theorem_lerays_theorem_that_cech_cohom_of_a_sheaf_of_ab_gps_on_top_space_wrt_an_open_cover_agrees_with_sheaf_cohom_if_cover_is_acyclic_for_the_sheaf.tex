\begin{theorem}[Leray's theorem] \label{theorem:lerays_theorem_that_cech_cohom_of_a_sheaf_of_ab_gps_on_top_space_wrt_an_open_cover_agrees_with_sheaf_cohom_if_cover_is_acyclic_for_the_sheaf}
    Let $X$ be a \CrefAndHyperrefIfExist{definition:topological_space}{topological space}, let $\calF$ be a \CrefAndHyperrefIfExist{definition:sheaf_on_a_topological_space_valued_in_a_category_with_a_terminal_object}{sheaf of} \CrefAndHyperrefIfExist{definition:group}{abelian groups} on $X$, and let $\calU = \{U_i\}_{i \in I}$ be an \CrefAndHyperrefIfExist{definition:open_covering_of_a_topological_space}{open cover} of $X$.
    If $\calU$ is \CrefAndHyperrefIfExist{definition:acyclic_for_a_sheaf_of_abelian_groups_for_an_open_cover_of_a_topological_space}{acyclic for $\calF$}, then there is a \CrefAndHyperrefIfExist{definition:natural_transformation_between_functors_between_categories}{natural isomorphism}
    $$\check{H}^p(\calU, \calF) \cong H^p(X,\calF)$$
    for all $p \geq 0$ where $\check{H}^p(\calU, \calF)$ is the \CrefAndHyperrefIfExist{definition:cech_cohomology_of_a_sheaf_of_abelian_groups_on_a_topological_space}{$p$th \v{C}ech cohomology group of $\calF$ with respect to $\calU$} and $H^p(X,\calF)$ is the \CrefAndHyperrefIfExist{definition:sheaf_cohomology_objects_of_a_sheaf_on_a_top_space_in_an_ab_catwith_enough_injs_and_sheafification}{$p$th sheaf cohomology group of $\calF$}.
\end{theorem}
