\begin{proposition} \label{proposition:counit_map_of_adjunction_of_inverse_image_and_direct_image_of_etale_sheaves_of_abelian_groups_for_an_open_embedding_of_schemes_is_an_isomorphism}
    Let $j: U \hookrightarrow X$ be an \CrefAndHyperrefIfExist{definition:open_and_closed_immersions_of_schemes}{open embedding} of schemes. Let $\Lambda$ be a ring. 
    \CrefAndHyperrefIfExist{theorem:adjunction_of_inverse_image_and_direct_image_for_modules_on_ringed_sites_under_ringed_site_morphism}{Recall that} there is an \CrefAndHyperrefIfExist{definition:adjoint_functors_between_categories_unit_counit_of_adjoint_functors}{adjunction} $j^* \dashv j_*$ between the functors
    $$j^*: \Sh(X_{\et}, \Lambda) \Rightarrow \Sh(U_{\et}, \Lambda)$$
    (\Cref{definition:inverse_image_and_direct_image_of_sheaves_on_schemes_for_various_topologies}) (\Cref{definition:small_etale_site_of_a_scheme}) (\Cref{definition:module_over_a_sheaf_of_rings_on_a_site})
    and
    $$j_*: \Sh(U_{\et}, \Lambda) \Rightarrow \Sh(X_{\et}, \Lambda)$$
    The \CrefAndHyperrefIfExist{definition:adjoint_functors_between_categories_unit_counit_of_adjoint_functors}{counit} natural morphism
    $$j^* j_* \Rightarrow \id$$
    is in fact a \CrefAndHyperrefIfExist{definition:natural_transformation_between_functors_between_categories}{natural isomorphism}.
\end{proposition}
