\begin{proposition} \label{proposition:valued_field_is_henselian_iff_its_valuation_ring_is_henselian}
    Let $(K,v)$ be a \CrefAndHyperrefIfExist{definition:valuation_on_a_field}{valuation field}. It is \CrefAndHyperrefIfExist{definition:henselian_field_with_respect_to_a_valuation}{Henselian} if and only if its \CrefAndHyperrefIfExist{definition:valuation_ring_of_a_nonarchimedean_valued_field}{valuation ring} is a \CrefAndHyperrefIfExist{definition:henselian_ring}{Henselian (local) ring}.
\end{proposition}
