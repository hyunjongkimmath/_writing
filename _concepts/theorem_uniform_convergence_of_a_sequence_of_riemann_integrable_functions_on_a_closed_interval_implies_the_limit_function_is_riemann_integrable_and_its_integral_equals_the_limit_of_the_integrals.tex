\begin{theorem} \label{theorem:uniform_convergence_of_a_sequence_of_riemann_integrable_functions_on_a_closed_interval_implies_the_limit_function_is_riemann_integrable_and_its_integral_equals_the_limit_of_the_integrals}
Let $[a, b] \subset \mathbb{R}$ be a closed bounded interval and let $(f_n)_{n=1}^{\infty}$ be a sequence of \CrefAndHyperrefIfExist{definition:lower_upper_riemann_sum_integral_of_a_bounded_function_on_a_closed_interval_to_R_and_riemann_integrable_function}{Riemann integrable functions} $f_n : [a, b] \to \mathbb{R}$. Suppose that $(f_n)$ converges to a function $f : [a, b] \to \mathbb{R}$ \CrefAndHyperrefIfExist{definition:uniform_convergence_of_a_sequence_of_functions_from_a_set_to_an_extended_metric_space}{uniformly} on $[a, b]$. Then:
\begin{enumerate}
  \item The \CrefAndHyperrefIfExist{definition:pointwise_convergence_of_a_sequence_of_functions_from_a_set_to_an_extended_metric_space}{limit function} $f$ is Riemann integrable on $[a, b]$.
  \item The integrals converge to the integral of the limit:
  $\lim_{n \to \infty} \int_{a}^{b} f_n(x)\,dx = \int_{a}^{b} f(x)\,dx$.
\end{enumerate}
\end{theorem}
