\begin{enumerate}

  \item The integrals converge to the integral of the limit:

\end{enumerate}
\end{theorem}

$$|f(y) - f(z)| \leq |f(y) - f_N(y)| + |f_N(y) - f_N(z)| + |f_N(z) - f(z)| < 2\epsilon / (3(b-a)) + |f_N(y) - f_N(z)|.$$

$$U(f, P) - L(f, P) \leq U(f_N, P) - L(f_N, P) + 2\epsilon < \epsilon.$$
Therefore, $f$ is Riemann integrable on $[a,b]$.
