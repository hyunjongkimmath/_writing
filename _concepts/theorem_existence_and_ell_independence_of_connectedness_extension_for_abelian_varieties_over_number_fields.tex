\begin{theorem}[Serre; see {\cite[Introduction]{silverberg_zarhin_ceav}}, {\cite[Theorem~0.1]{larsen_pink_ccellgr}}] \label{theorem:existence_and_ell_independence_of_connectedness_extension_for_abelian_varieties_over_number_fields}
    Let $A/K$ be an abelian variety over a number field $K$. For each prime $\ell$, let $\rho_\ell : \Gal(\overline{K}/K) \to \mathrm{Aut}(T_\ell(A))$ be the $\ell$-adic Tate module representation, and let $G_\ell$ be the Zariski closure of its image.

    There exists a finite extension $K_{\mathrm{conn}}/K$, independent of $\ell$, such that for every prime $\ell$, the algebraic monodromy group $G_\ell|_{\Gal(\overline{K}/K_{\mathrm{conn}})}$ is connected. Equivalently, the \CrefAndHyperref{definition:connectedness_extension_for_a_compatible_system_of_representations}{connectedness extension} of $K$ for the compatible system $\{\rho_\ell\}$ associated to $A$ is a finite extension independent of $\ell$.
\end{theorem}
