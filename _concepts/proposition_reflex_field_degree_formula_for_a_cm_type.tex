\begin{proposition}[Reflex Field Degree Formula for a CM Type] \label{proposition:reflex_field_degree_formula_for_a_cm_type}
    Let $E$ be a \CrefAndHyperrefIfExist{definition:cm_field}{CM field} with maximal \CrefAndHyperrefIfExist{definition:totally_real_and_totally_imaginary_number_fields}{totally real} subfield $F$, and let $(E, \Phi)$ be a \CrefAndHyperrefIfExist{definition:cm_type_and_reflex_field_of_a_cm_type}{CM type}. Let $g = [E:F]$ and let $E^*(\Phi)$ be the \CrefAndHyperrefIfExist{definition:cm_type_and_reflex_field_of_a_cm_type}{reflex field} of $(E, \Phi)$.

    Let $\Gamma = \operatorname{Gal}(\overline{\mathbb{Q}}/\mathbb{Q})$ act on $\operatorname{Hom}(E, \overline{\mathbb{Q}})$ by right multiplication: $(\tau^\sigma)(x) = \sigma(\tau(\sigma^{-1}(x)))$ for $\sigma \in \Gamma$, $\tau \in \operatorname{Hom}(E, \overline{\mathbb{Q}})$, $x \in E$. Let $\Gamma_\Phi \subset \Gamma$ be the stabilizer of $\Phi$. Then
    $$[E^*(\Phi) : \mathbb{Q}] = [\Gamma : \Gamma_\Phi].$$

    Let $h(\Phi) := |\{\tau \in \Phi : \tau|_{E^*(\Phi)} = \rho\}|$ for any (equivalently, every) $\rho \in \operatorname{Hom}(E^*(\Phi), \overline{\mathbb{Q}})$. Then $h(\Phi)$ divides $2g$, and
    $$[E^*(\Phi) : \mathbb{Q}] \cdot h(\Phi) = 2g.$$
\end{proposition}
