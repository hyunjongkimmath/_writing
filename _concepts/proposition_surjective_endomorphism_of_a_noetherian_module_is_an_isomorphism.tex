\begin{proposition}[Surjective endomorphism of a Noetherian module is an isomorphism]  \label{proposition:surjective_endomorphism_of_a_noetherian_module_is_an_isomorphism}

Let $A$ be a \CrefAndHyperrefIfExist{definition:ring}{ring} and let $M$ be a left $A$-module.

\begin{enumerate}
    \item If $M$ is a \CrefAndHyperrefIfExist{definition:noetherian_artinian_module_over_a_ring}{Noetherian $A$-module} and $u \colon M \to M$ is a surjective $A$-linear endomorphism, then $u$ is an isomorphism.

    \item If $M$ is a \CrefAndHyperrefIfExist{definition:noetherian_artinian_module_over_a_ring}{Artinian $A$-module} and $u \colon M \to M$ is an injective $A$-linear endomorphism, then $u$ is an isomorphism.
\end{enumerate}

(cf.\ \cite[Chapter 6, Exercise 1]{atiyah_macdonald})
\end{proposition}
