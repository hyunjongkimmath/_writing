\begin{proposition} \label{proposition:simple_ring_is_semisimple_iff_artinian}
    A \CrefAndHyperrefIfExist{definition:simple_ring}{simple ring} is \CrefAndHyperrefIfExist{definition:semisimple_additive_category}{semisimple} if and only if it is \CrefAndHyperrefIfExist{definition:left_right_artinian_ring}{Artinian}. 
\end{proposition}
