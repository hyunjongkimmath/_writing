\begin{theorem}[Proper base change for integral adic coefficients, cf. {\cite[Expos\'e XII Th\'eor\`eme 5.1]{SGA4_III}}, {\cite[Theorem I.6.1]{freitag_kiehl_ecwc}}, {\cite[Theorem 7.3.1]{fu_ect}}] \label{theorem:proper_base_change_theorem_for_derived_category_of_adic_sheaves_with_integral_coefficients_on_schemes}
    Let $S$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme}. Let
    \begin{center}
        \begin{tikzcd}
            X' \ar[r, "f'"] \ar[d,"g'"] & Y' \ar[d, "g"] \\
            X \ar[r, "f"] & Y
        \end{tikzcd}
    \end{center}
    be a Cartesian diagram of $S$-schemes such that $f$ is \CrefAndHyperrefIfExist{definition:proper_morphism_of_schemes}{proper}.

    Let $\ell$ be a prime invertible on $Y$. Let $R$ be the \CrefAndHyperrefIfExist{definition:integral_element_over_a_ring}{integral closure} of $\bbZ_\ell$ in a finite extension $E$ of $\Qell$, and let $\lambda$ be a uniformizer of $R$.

    For any $K = (K_n) \in D_c^b(X, R)$ (\Cref{definition:derived_category_of_bounded_constructible_complexes_of_adic_sheaves_with_integral_coefficients_on_a_noetherian_scheme}), the canonical base change morphism
    $$g^* Rf_* K \to Rf_*' {g'}^* K$$
    is an isomorphism in $D_c^b(Y', R)$ (\Cref{definition:six_functor_formalism_on_derived_category_of_bdd_constr_complexes_of_adic_sheaves_with_integral_coefficients_on_a_noetherian_scheme}).

    The isomorphism follows levelwise from proper base change for each finite coefficient system $R/\lambda^{n+1}$ (cf.\ \cite[Theorem 7.3.1]{fu_ect}), since the functors $g^*$, $Rf_*$, ${g'}^*$, and $Rf'_*$ act levelwise on inverse systems (\Cref{definition:six_functor_formalism_on_derived_category_of_bdd_constr_complexes_of_adic_sheaves_with_integral_coefficients_on_a_noetherian_scheme}).
\end{theorem}
