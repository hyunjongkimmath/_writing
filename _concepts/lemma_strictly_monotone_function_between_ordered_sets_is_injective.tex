\begin{lemma}[Strictly monotone functions are injective] \label{lemma:strictly_monotone_function_between_ordered_sets_is_injective}
    Let $(X, \leq_X)$ and $(Y, \leq_Y)$ be \CrefAndHyperrefIfExist{definition:partially_ordered_set}{ordered sets}, and let $f : X \to Y$ be a function.

    If $f$ is \CrefAndHyperrefIfExist{definition:monotone_function_between_ordered_sets}{strictly increasing} or \CrefAndHyperrefIfExist{definition:monotone_function_between_ordered_sets}{strictly decreasing}, then $f$ is \CrefAndHyperrefIfExist{definition:injective_surjective_bijective_map_of_sets}{injective}.
\end{lemma}
