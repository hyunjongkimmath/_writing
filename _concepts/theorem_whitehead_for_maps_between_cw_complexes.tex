\begin{theorem}[Whitehead, see e.g. {\cite[Theorem 4.5]{hatcher}}] \label{theorem:whitehead}
Let $f: X \to Y$ be a \CrefAndHyperrefIfExist{definition:continuous_map_of_topological_spaces}{continuous map} between \CrefAndHyperrefIfExist{definition:CW_complex}{CW complexes}. If $f_*: \pi_n(X,x_0) \to \pi_n(Y,f(x_0))$ is an isomorphism for all $n \geq 1$ and all basepoints $x_0 \in X$, then $f$ is a \CrefAndHyperrefIfExist{definition:homotopy_equivalence_between_topological_spaces}{homotopy equivalence}.
\end{theorem}

A consequence of the same hypothesis, applied to mapping cylinders, yields:

\begin{corollary} \label{corollary:whitehead_weak_equivalence_of_cw_complexes_is_homotopy_equivalence}
Any \CrefAndHyperrefIfExist{definition:weak_homotopy_equivalence_of_topological_spaces}{weak homotopy equivalence} between CW complexes is a \CrefAndHyperrefIfExist{definition:homotopy_equivalence_between_topological_spaces}{homotopy equivalence}.
\end{corollary}
