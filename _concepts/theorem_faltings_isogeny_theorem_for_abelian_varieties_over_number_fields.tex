\begin{theorem}[{\cite[Korollary 1 to Satz 4]{faltings_eavz}}] \label{theorem:faltings_isogeny_theorem_for_abelian_varieties_over_number_fields}
    Let $K$ be a \CrefAndHyperrefIfExist{global_field}{number field} and let $A, B$ be \CrefAndHyperrefIfExist{definition:abelian_variety_over_a_field}{abelian varieties} over $K$. Let $\ell \neq \Char K$ be a \CrefAndHyperrefIfExist{definition:prime_number}{prime number}.
    The \CrefAndHyperrefIfExist{definition:ell_adic_representation_map_from_base_change_of_isogenies_of_abelian_varieties_to_hom_set_of_tate_modules}{canonical map} 
    $$\mathbb{Z}_\ell \otimes_{\mathbb{Z}} \Hom(A, B) \to \Hom_{\bbZ_\ell}(T_\ell A, T_\ell B)$$
    is a bijection.
\end{theorem}
