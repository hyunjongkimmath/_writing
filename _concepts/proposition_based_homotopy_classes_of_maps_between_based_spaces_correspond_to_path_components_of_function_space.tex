\begin{proposition} \label{proposition:based_homotopy_classes_of_maps_between_based_spaces_correspond_to_path_components_of_function_space}
    Let $(X, x_0)$ and $(Y, y_0)$ be \CrefAndHyperrefIfExist{definition:pointed_topological_space}{pointed spaces}. There is a canonical bijection
    $$[X, Y]_* \cong \pi_0(\operatorname{Map}_*(X, Y), f_0)$$
    (\Cref{definition:homotopy_groups_of_a_pointed_topological_space}) (\Cref{definition:compact_open_topology_on_the_set_of_continuous_maps_between_topological_spaces_and_function_space}) where $f_0: X \to Y$ is the constant map to $y_0$, and $\operatorname{Map}_*(X,Y)$ is equipped with the \CrefAndHyperrefIfExist{definition:compact_open_topology_on_the_set_of_continuous_maps_between_topological_spaces_and_function_space}{compact open topology}. 
\end{proposition}
