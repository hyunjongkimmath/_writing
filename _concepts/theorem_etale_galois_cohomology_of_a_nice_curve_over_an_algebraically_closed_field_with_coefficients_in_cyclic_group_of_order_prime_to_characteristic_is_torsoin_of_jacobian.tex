    Let $C/k$ be a \CrefAndHyperrefIfExist{definition:C_k_manifold}{smooth} \CrefAndHyperrefIfExist{definition:projective_morphism_of_schemes_projective_scheme_over_a_scheme}{projective} \CrefAndHyperrefIfExist{definition:algebraic_curve_over_a_field}{curve} over an \CrefAndHyperrefIfExist{definition:algebraically_closed_field}{algebraically closed field}. Let $n$ be an integer not divisible by $\Char k$. 
    There is a group isomorphism
    $$H_{\et}^1 (C, \mu_n) \cong \Jac(C)[n](k).$$

    $$H_{\et}^1 (C, \mu_n) \cong \Jac(C)[n](k).$$
\end{theorem}
