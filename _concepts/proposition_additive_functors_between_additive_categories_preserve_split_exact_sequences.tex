\begin{proposition}  \label{proposition:additive_functors_between_additive_categories_preserve_split_exact_sequences}
In an \CrefAndHyperrefIfExist{definition:additive_category_preadditive_category}{additive category} $\mathcal{A}$, any \CrefAndHyperrefIfExist{definition:split_exact_sequence_in_an_additive_category}{split exact sequence} $0 \to A \xrightarrow{f} B \xrightarrow{g} C \to 0$ is preserved by any \CrefAndHyperrefIfExist{definition:additive_functor_between_additive_categories}{additive functor} $F: \mathcal{A} \to \mathcal{B}$ to another additive category $\mathcal{B}$. That is, $0 \to F(A) \xrightarrow{F(f)} F(B) \xrightarrow{F(g)} F(C) \to 0$ is also a split exact sequence.
\end{proposition}
