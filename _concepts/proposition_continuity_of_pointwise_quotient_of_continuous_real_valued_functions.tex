\begin{proposition} \label{proposition:continuity_of_pointwise_quotient_of_continuous_real_valued_functions}
    Let $X$ be a \CrefAndHyperrefIfExist{definition:topological_space}{topological space} and let $\bbR$ be the \CrefAndHyperrefIfExist{definition:field_of_real_numbers}{field of real numbers} equipped with its \CrefAndHyperrefIfExist{definition:standard_absolute_value_on_real_numbers_standard_topology_on_real_numbers}{standard topology}. Let $f,g : X \to \bbR$ be \CrefAndHyperrefIfExist{definition:continuous_map_of_topological_spaces}{continuous} functions and suppose $g(x) \neq 0$ for all $x \in X$. Their pointwise quotient $f/g : X \to \bbR$, defined by $(f/g)(x) = f(x) / g(x)$, is continuous.
\end{proposition}
\begin{proof}
    Let $\operatorname{rec} : \bbR^\times \to \bbR^\times$ be the reciprocal function, which is continuous by \Cref{proposition:reciprocal_function_on_real_numbers_is_continuous}. Since $g$ takes values in $\bbR^\times$, the composition $\operatorname{rec} \circ g : X \to \bbR$ is continuous by \Cref{proposition:composition_of_continuous_functions_on_topological_spaces_is_continuous}. The product $f \cdot (\operatorname{rec} \circ g)$ is then continuous by \Cref{proposition:continuity_of_pointwise_product_of_continuous_real_valued_functions}, and this product is precisely $f/g$.
    \medskip
    \noindent\textit{Remark.} This is a special case of \Cref{proposition:continuity_of_pointwise_quotient_of_continuous_functions_on_topological_spaces_valued_in_topological_field}, because $\bbR$ equipped with its standard topology induced by the absolute value is a \CrefAndHyperrefIfExist{definition:topological_field}{topological field}.
\end{proof}
