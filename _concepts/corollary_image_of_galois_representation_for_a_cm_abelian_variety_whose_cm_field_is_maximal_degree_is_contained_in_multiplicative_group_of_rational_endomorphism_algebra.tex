\begin{definition} \label{corollary:image_of_galois_representation_for_a_cm_abelian_variety_whose_cm_field_is_maximal_degree_is_contained_in_multiplicative_group_of_rational_endomorphism_algebra}
    Let $A/K$ be a \CrefAndHyperrefIfExist{definition:complex_multiplication_of_an_abelian_variety_over_a_field}{CM abelian variety} of \CrefAndHyperrefIfExist{definition:dimension_of_a_scheme}{dimension} $g$ over a \CrefAndHyperrefIfExist{definition:field}{field} with complex multiplication by a \CrefAndHyperrefIfExist{definition:cm_field}{CM field} $E$ and assume that $E$ is of dimension $2g$, i.e. $\operatorname{End}_K(A) \otimes \bbQ = E$\CrefIfExists{definition:endomorphism_algebra_up_to_isogeny_of_an_abelian_variety_over_a_field}. Let $\ell$ be a prime number. 

    \begin{enumerate}
        \item Recalling that (\Cref{proposition:ell_adic_tate_module_for_a_cm_abelian_variety_whose_cm_field_is_maximal_degree_is_locally_free_rank_1_module_over_rational_endomorphism_algebra}) the \CrefAndHyperrefIfExist{definition:ell_adic_tate_module_of_an_abelian_variety_over_a_field}{integral $\ell$-adic Tate module} $T_\ell(A)$ is a locally free $\End_K(A) \otimes \bbZ_\ell$-module of rank $1$, one can identify $(\End_K(A) \otimes \bbZ_\ell)^\times$ as a subgroup of $\Aut(T_\ell(A))$ and the image of the integral $\ell$-adic Galois representation
        $$\rho_{A, \ell^\infty}: \Gal(\overline{K} / K ) \to \Aut(T_\ell(A))$$
        is contained in $(\End_K(A) \otimes \bbZ_\ell)^\times$.

        \item The image of the rational $\ell$-adic Galois representation
        $$\rho_{A, \ell^\infty}: \Gal(\overline{K} / K ) \to \Aut(V_\ell(A))$$
        is contained in $E^\times$.
    \end{enumerate}

\end{definition}
