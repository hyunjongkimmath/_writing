\begin{lemma} \label{lemma:kernel_cokernel_of_morphisms_in_a_pointed_category_are_equalizers_coequalizers}
    Let $\calC$ be a \CrefAndHyperrefIfExist{definition:pointed_category}{pointed category}. All \CrefAndHyperrefIfExist{definition:kernel_and_cokernel_of_a_morphism_in_a_pointed_category}{Kernels/cokernels} are to \CrefAndHyperrefIfExist{definition:equalizer_and_coequalizer_of_morphisms_in_a_category}{equalizers/coequalizers} in the following sense: given a morphism $f: X \to Y$ in $\calC$, $\ker f$ and $\coker f$ are respectively isomorphic to the equalizer and the coequalizer of $f$ and the \CrefAndHyperrefIfExist{definition:zero_morphism_in_a_pointed_category}{zero map} $0_{XY}: X \to Y$.
\end{lemma}
