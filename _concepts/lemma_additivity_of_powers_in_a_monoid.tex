\begin{lemma} \label{lemma:additivity_of_powers_in_a_monoid}
    Let $(M,\cdot)$ be a \CrefAndHyperrefIfExist{definition:monoid}{monoid} with identity element $e$, and let $a \in M$. For all nonnegative integers $m,n \in \bbZ_{\geq 0}$,
    $$a^m \cdot a^n = a^{m+n},$$
    where $a^k$ denotes the \CrefAndHyperrefIfExist{definition:power_of_a_monoid_element_by_a_nonnegative_integer}{$k$th power of $a$}.
\end{lemma}
\begin{proof}
    Fix $n \in \bbZ_{\geq 0}$ and induct on $m$. For the base case $m = 0$, we have by \CrefIfExists{definition:power_of_a_monoid_element_by_a_nonnegative_integer}:
    $$a^0 \cdot a^n = e \cdot a^n = a^n = a^{0+n},$$
    so the identity holds. Inductively assume that $m \in \bbZ_{\geq 0}$ satisfies $a^m \cdot a^n = a^{m+n}$. Then
    $$a^{m+1} \cdot a^n = (a \cdot a^m) \cdot a^n$$
    by the recursive definition of powers. By associativity of $\cdot$ from \CrefIfExists{definition:monoid}, this equals
    $$a \cdot (a^m \cdot a^n) = a \cdot a^{m+n}$$
    by the inductive hypothesis. Since $a \cdot a^{m+n} = a^{(m+1)+n}$ by definition of powers, the identity holds for $m+1$. This completes the induction.
\end{proof}
