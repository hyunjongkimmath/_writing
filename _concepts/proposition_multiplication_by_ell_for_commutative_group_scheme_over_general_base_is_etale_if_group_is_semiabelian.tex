\begin{proposition}[see {\cite[Chapter 7, Section 3, Lemma 2(a)]{bosch_lutkebohmert_raynaud_nm}}] \label{proposition:multiplication_by_ell_for_commutative_group_scheme_over_general_base_is_etale_if_group_is_semiabelian}
    Let $S$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme}. Let $G$ be a \CrefAndHyperrefIfExist{definition:semi_abelian_group_scheme_over_a_scheme}{semiabelian scheme} over $S$ that is \CrefAndHyperrefIfExist{definition:smooth_morphism_of_schemes}{smooth} and of \CrefAndHyperrefIfExist{definition:finite_type_morphism_of_schemes}{finite type}. Let $\ell$ be a positive integer.
    The the $\ell$-multiplication $[\ell]: G \longrightarrow G$ is \CrefAndHyperrefIfExist{definition:quasifinite_morphism_of_schemes}{quasifinite} and \CrefAndHyperrefIfExist{definition:flat_morphism_of_schemes}{flat}.  
    In particular, the \CrefAndHyperrefIfExist{definition:torsion_subgroup_scheme_of_a_group_scheme}{torsion subgroup scheme} $G[\ell]$ is quasifinite and flat over $S$ \TODO{quasifinite and flat morphisms are preserved under base change}
\end{proposition}
