\begin{lemma} \label{lemma:kernels_cokernels_in_pointed_categories_are_monomorphisms_epimorphisms}
    Let $\mathcal{C}$ be a \CrefAndHyperrefIfExist{definition:category}{(large)} \CrefAndHyperrefIfExist{definition:pointed_category}{pointed category}, i.e. a category with a \CrefAndHyperrefIfExist{definition:initial_final_zero_objects_of_a_category}{zero object} $0$. Let $f: X \to Y$ be a morphism.
    \begin{enumerate}
        \item The \CrefAndHyperrefIfExist{definition:kernel_and_cokernel_of_a_morphism_in_a_pointed_category}{kernel} of $f$ is a \CrefAndHyperrefIfExist{definition:monomorphism_and_epimorphism_in_categories}{monomorphism} $\ker(f) \to X$.
        \item The \CrefAndHyperrefIfExist{definition:kernel_and_cokernel_of_a_morphism_in_a_pointed_category}{cokernel} of $f$ is a \CrefAndHyperrefIfExist{definition:monomorphism_and_epimorphism_in_categories}{epimorphism} $Y \to \coker(f)$.
    \end{enumerate}
\end{lemma}
