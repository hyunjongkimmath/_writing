\begin{proposition}[see {\cite[Chapter 7, Section 3, Lemma 2(b)]{bosch_lutkebohmert_raynaud_nm}}] \label{proposition:multiplication_by_ell_for_commutative_group_scheme_over_general_base_is_etale_if_characteristic_of_residue_fields_is_different_from_ell}
    Let $S$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme}. Let $G$ be a \CrefAndHyperrefIfExist{definition:algebraic_group_scheme_over_a_scheme}{commutative $S$-group scheme} which is \CrefAndHyperrefIfExist{definition:smooth_at_a_point_for_a_finite_type_scheme_over_a_field}{smooth} and of \CrefAndHyperrefIfExist{definition:finite_type_morphism_of_schemes}{finite type}. Let $\ell$ be a positive integer.
    Suppose that $\Char k(s)$ does not divide $\ell$ for all $s \in S$. Then the $\ell$-multiplication $[\ell]: G \longrightarrow G$ is \CrefAndHyperrefIfExist{definition:etale_morphism_of_schemes}{\'etale}. 
    In particular, the \CrefAndHyperrefIfExist{definition:torsion_subgroup_scheme_of_a_group_scheme}{torsion subgroup scheme} $G[\ell]$ is \'etale over $S$ \TODO{etale morphisms are preserved under base change}
\end{proposition}
