\begin{lemma} \label{lemma:null_homotopic_morphism_of_chain_complexes_in_an_abelian_category_yields_zero_homology_maps}
    Let $\calA$ be an \CrefAndHyperref{definition:abelian_category}{abelian category}. Let $f_\bullet: C_\bullet \to D_\bullet$ (resp. $f^\bullet: C^\bullet \to D^\bullet$) be a \CrefAndHyperref{definition:chain_complex_of_objects_in_an_additive_category}{morphism of chain complexes} (resp. of cochain complexes). 
    If $f_\bullet$ is \CrefAndHyperref{definition:chain_homotopy_between_chain_maps_between_complexes}{null homotopic} (resp. $f^\bullet$ is null homotopic), then the induced maps 
    $$H_n(f_\bullet): H_n(C_\bullet) \to H_n(D_\bullet)$$
    of \CrefAndHyperref{definition:homology_and_cohomology_objects_for_a_chain_complex_in_an_additive_category}{homology} 
    (resp. 
    $$H^n(f^\bullet): H^n(C^\bullet) \to H^n(D^\bullet)$$
    of cohomology)
    are zero. 
\end{lemma}
