\begin{lemma} \label{lemma:polynomial_ring_over_an_integral_domain_is_an_integral_domain}
    Let $R$ be an \CrefAndHyperrefIfExist{definition:domain_integral_domain}{integral domain}. 
    The \CrefAndHyperrefIfExist{definition:polynomial_ring_over_a_commutative_ring}{polynomial ring} $R[x_1,\ldots,x_n]$ is an integral domain. 
\end{lemma}
