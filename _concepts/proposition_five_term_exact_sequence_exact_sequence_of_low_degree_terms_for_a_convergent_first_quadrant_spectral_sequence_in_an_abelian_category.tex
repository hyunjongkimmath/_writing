\begin{proposition} \label{proposition:five_term_exact_sequence_exact_sequence_of_low_degree_terms_for_a_convergent_first_quadrant_spectral_sequence_in_an_abelian_category}
Let $\mathcal{A}$ be an \CrefAndHyperrefIfExist{definition:abelian_category}{abelian category}. 
\begin{enumerate}
    \item Let $E^r_{p,q}$ be a \CrefAndHyperrefIfExist{definition:first_quadrant_spectral_sequence}{first quadrant} \CrefAndHyperrefIfExist{definition:spectral_sequence_in_an_abelian_category}{homological spectral sequence} in $\mathcal{A}$ starting at $r=2$. Suppose $E$ \CrefAndHyperrefIfExist{definition:converges_to_a_graded_object_for_a_spectral_sequence_in_an_abelian_category}{converges} to a \CrefAndHyperrefIfExist{definition:graded_object_in_a_category}{graded object} $H$ such that $E^2_{p,q} \Rightarrow H_{p+q}$. Then there exists an \CrefAndHyperrefIfExist{definition:acyclic_complex_of_objects_in_an_abelian_category}{exact sequence} in $\mathcal{A}$:
    $$E^2_{0,1} \to E^2_{2,0} \to H_2 \to E^2_{1,1} \to E^2_{0,2} \to 0$$
    
    \item Let $E_r^{p,q}$ be a first quadrant \CrefAndHyperrefIfExist{definition:spectral_sequence_in_an_abelian_category}{cohomological spectral sequence} in $\mathcal{A}$ starting at $r=2$. Suppose $E$ converges to a graded object $H$ such that $E_2^{p,q} \Rightarrow H^{p+q}$. Then there exists an exact sequence in $\mathcal{A}$:
    $$0 \to E_2^{1,0} \to H^1 \to E_2^{0,1} \to E_2^{2,0} \to H^2$$
\end{enumerate}
Either such exact sequence is often called the \hldef{five-term exact sequence} or \hldef{exact sequence of low-degree terms} of the spectral sequence.
\end{proposition}
