\begin{theorem} \label{theorem:the_weierstrass_convergence_theorem_stating_that_the_uniform_limit_on_compact_subsets_of_a_sequence_of_holomorphic_functions_is_itself_holomorphic_with_all_derivatives_converging_uniformly_on_compact_subsets}
Let $U \subset \mathbb{C}$ be an open set and let $(f_n)_{n=1}^{\infty}$ be a sequence of \CrefAndHyperrefIfExist{definition:holomorphic_function_from_a_subset_of_Cn_to_Cm}{holomorphic functions} $f_n : U \to \mathbb{C}$. Suppose that $(f_n)$ converges to a function $f : U \to \mathbb{C}$ \CrefAndHyperrefIfExist{definition:uniform_convergence_of_a_sequence_of_functions_from_a_set_to_an_extended_metric_space}{uniformly} on every \CrefAndHyperrefIfExist{definition:compact_topological_space}{compact} subset of $U$. Then:
\begin{enumerate}
  \item The \CrefAndHyperrefIfExist{definition:pointwise_convergence_of_a_sequence_of_functions_from_a_set_to_an_extended_metric_space}{limit function} $f$ is holomorphic on $U$.
  \item For each $k = 0, 1, 2, \ldots$, the sequence of \CrefAndHyperrefIfExist{definition:holomorphic_function_from_a_subset_of_Cn_to_Cm}{derivatives} $(f_n^{(k)})$ converges to $f^{(k)}$ uniformly on every compact subset of $U$.
\end{enumerate}
\end{theorem}
