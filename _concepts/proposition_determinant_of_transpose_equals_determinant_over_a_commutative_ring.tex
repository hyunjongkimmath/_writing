\begin{proposition} \label{proposition:determinant_of_transpose_equals_determinant_over_a_commutative_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:commutative_ring}{commutative ring}, and let $A = (a_{ij}) \in M_n(R)$ be an $n \times n$ matrix. Then
    $$\det(A^{\mathsf{T}}) = \det(A),$$
    where $A^{\mathsf{T}}$\CrefIfExists{definition:transpose_of_a_matrix_over_a_ring} is the \CrefAndHyperrefIfExist{definition:transpose_of_a_matrix_over_a_ring}{transpose of $A$}.
\end{proposition}
