\begin{proposition}[Density of $\mathbb{Q}$ in $\mathbb{R}$] \label{proposition:field_of_rational_numbers_is_dense_in_field_of_real_numbers}
    Let $\mathbb{R}$ be the \CrefAndHyperrefIfExist{definition:field_of_real_numbers}{field of real numbers} and let $\mathbb{Q}$ be the \CrefAndHyperrefIfExist{definition:field_of_rational_numbers}{field of rational numbers}. Via the canonical embedding $\mathbb{Q} \hookrightarrow \mathbb{R}$\CrefIfExists{proposition:the_field_of_rational_numbers_embeds_into_the_field_of_real_numbers}, identify $\mathbb{Q}$ as a subfield of $\mathbb{R}$.

    For every $x, y \in \mathbb{R}$ with $x < y$, there exists $q \in \mathbb{Q}$ such that $x < q < y$.
\end{proposition}
