\begin{lemma}[cf. {\cite[Section 2.1]{katz_rls}}, {\cite[Section 8.1]{katz_esde}, }]   \label{lemma:derived_pushforward_of_affine_preserves_semiperversity_and_derived_proper_pushforward_of_quasi_finite_perverse_semiperversity}

    Let $f: X \to Y $ be a morphism of finite type schemes over a field $k$ of characteristic distinct from $\ell$. 
    \begin{enumerate}
        \item (\cite[Th\'eor\`eme 4.1.1]{beilinson_berstein_deligne_atlesI}, cf. \cite[Expos\'e XIV, Th\'eor\`eme]{SGA4_III}, \cite[Theorem III.6.1]{kiehl_weissauer_wcps}) If $f: X \to Y$ is affine, then $Rf_*: \Dbc(X, \Qellbar) \to \Dbc(Y, \Qellbar)$ is right $t$-exact with respect to the perverse $t$-structures on $\Dbc(X, \Qellbar)$ and $\Dbc(Y, \Qellbar)$, i.e. restricts to a functor $\pDleqzero(X) \to \pDleqzero(Y)$, and $Rf_!: \Dbc(X, \Qellbar) \to \Dbc(Y, \Qellbar)$ is left $t$-exact. In particular, $Rf_*$ preserves semiperversity.
        
        \item (See {\cite[Proposition 2.2.5]{beilinson_berstein_deligne_atlesI}}) If $f: X \to Y$ is a quasi-finite, then $Rf_!$ and $f^*$ are right $t$-exact, and $f^!$ and $Rf_*$ are left $t$-exact. In particular, $Rf_!$ and $f^*$ preserve semiperversity.
        \end{enumerate}
\end{lemma}
