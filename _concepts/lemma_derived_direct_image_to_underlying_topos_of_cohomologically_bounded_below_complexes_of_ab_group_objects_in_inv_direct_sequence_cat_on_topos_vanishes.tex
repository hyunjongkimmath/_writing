\begin{lemma}[{\cite[Lemma 1.1.]{ekedahl_af}}] \label{lemma:derived_direct_image_to_underlying_topos_of_cohomologically_bounded_below_complexes_of_ab_group_objects_in_inv_direct_sequence_cat_on_topos_vanishes}
    Assume \Cref{context:ekedahl_af_ringed_topos_category_of_inverse_sequences} and \Cref{context:ekedahl_af_topos_morphisms_to_and_from_Pro_N_S_to_S}. Assume \Cref{context:ekedahl_af_nice_resolutions_exist_on_R_mod_m_n}

    Let $M$ be a \CrefAndHyperrefIfExist{definition:bounded_complexes_on_an_additive_category_and_homologically_bounded_objects_on_an_abelian_category}{cohomologically bounded below} \CrefAndHyperrefIfExist{definition:cochain_complex_of_objects_in_an_additive_category}{cochain complex} of \CrefAndHyperrefIfExist{definition:group_object_in_a_category_with_a_final_object}{abelian group objects} of \CrefAndHyperrefIfExist{definition:category_of_inverse_direct_sequences_of_objects_in_a_category}{$\mathrm{Pro}_{\bbN}(S)$}. Then, $R\pi_* M = 0$ (\Cref{context:ekedahl_af_topos_morphisms_to_and_from_Pro_N_S_to_S}) (\Cref{definition:total_right_derived_direct_image_functor_for_morphism_of_ringed_topoi}). 
\end{lemma}
