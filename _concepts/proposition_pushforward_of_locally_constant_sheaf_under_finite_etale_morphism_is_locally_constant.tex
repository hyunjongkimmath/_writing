\begin{proposition}[Pushforward of locally constant sheaf under finite \'etale morphism is locally constant]
\label{proposition:pushforward_of_locally_constant_sheaf_under_finite_etale_morphism_is_locally_constant}
Let $f: X \to Y$ be a \CrefAndHyperrefIfExist{definition:finite_etale_cover_of_a_scheme}{finite \'etale morphism of schemes}. Let $\mathcal{A}$ be a \CrefAndHyperrefIfExist{definition:category}{category} satisfying the following conditions:
\begin{enumerate}
    \item $\mathcal{A}$ has a \CrefAndHyperrefIfExist{definition:initial_final_zero_objects_of_a_category}{terminal object} (so that \CrefAndHyperrefIfExist{definition:constant_sheaf_on_a_site_with_sheafification}{constant sheaves} valued in $\mathcal{A}$ exist).
    \item $\mathcal{A}$ has finite \CrefAndHyperrefIfExist{definition:product_and_coproduct_of_objects_in_a_category}{coproducts} (so that finite direct sums of constant sheaves valued in $\mathcal{A}$ exist).
    \item A \CrefAndHyperrefIfExist{definition:sheafification_functor_on_a_site}{sheafification functor} exists for presheaves on \CrefAndHyperrefIfExist{definition:small_etale_site_of_a_scheme}{$X_{\et}$} and $Y_{\et}$ valued in $\mathcal{A}$ (e.g. $\mathcal{A}$ is \CrefAndHyperrefIfExist{definition:complete_and_cocomplete_category}{complete, cocomplete}, small \CrefAndHyperrefIfExist{definition:projective_and_inductive_limits_in_categories}{filtered colimits} in $\mathcal{A}$ are \CrefAndHyperrefIfExist{definition:left_right_exact_functor_between_categories}{exact}, and $\mathcal{A}$ satisfies the \CrefAndHyperrefIfExist{definition:ipc_property_of_a_category_that_admits_small_products_and_small_filtered_limits}{IPC-property}, see \Cref{theorem:sheafification_of_a_presheaf_of_sets_on_a_small_enough_site}).
\end{enumerate}
Let $\mathcal{F}$ be a \CrefAndHyperrefIfExist{definition:locally_constant_sheaf_on_a_site_with_sheafification}{locally constant sheaf} on $X_{\et}$ valued in $\mathcal{A}$. Then the \CrefAndHyperrefIfExist{definition:direct_image_of_a_sheaf_on_a_site_under_a_continuous_functor_of_sites_or_a_site_morphism}{direct image} $f_* \mathcal{F}$ is a locally constant sheaf on $Y_{\et}$ valued in $\mathcal{A}$.
