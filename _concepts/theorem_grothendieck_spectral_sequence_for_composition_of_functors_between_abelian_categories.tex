\begin{theorem}[Grothendieck Spectral Sequence, {\cite[5.8.3]{weibel}}] \label{theorem:grothendieck_spectral_sequence_for_composition_of_functors_between_abelian_categories}
    \begin{enumerate}
        \item Let $\calA, \calB, \calC$ be \CrefAndHyperrefIfExist{definition:abelian_category}{abelian categories} such that $\calA$ and $\calB$ \CrefAndHyperrefIfExist{definition:has_enough_injectives_or_projectives_for_an_abelian_category}{have enough injectives}. Let $G: \calA \to \calB$ and $F: \calB \to \calC$ be \CrefAndHyperrefIfExist{definition:exact_functor_between_abelian_categories}{left exact} \CrefAndHyperrefIfExist{definition:functor_between_categories}{functors}. Assume that $G$ sends \CrefAndHyperrefIfExist{definition:injective_and_projective_objects_in_a_category}{injective objects} of $\calA$ to \CrefAndHyperrefIfExist{definition:F_acyclic_object_for_a_left_or_right_functor_between_abelian_categories}{$F$-acyclic objects} of $\calB$. 

        There is a \CrefAndHyperrefIfExist{definition:converges_to_a_graded_object_for_a_spectral_sequence_in_an_abelian_category}{convergent} \CrefAndHyperrefIfExist{definition:first_quadrant_spectral_sequence}{first quadrant} \CrefAndHyperrefIfExist{definition:spectral_sequence_in_an_abelian_category}{cohomological spectral sequence} for each $A$ in $\calA$ starting in the second page:
        $$E_2^{p,q} = (R^p F) (R^q G)(A) \Rightarrow R^{p+q} (FG) (A).$$
        (\Cref{definition:left_right_derived_functors_of_a_right_left_exact_functor_between_abelian_categories_where_source_has_enough_projectives_injectives})

        The \CrefAndHyperrefIfExist{definition:homological_cohomological_base_fiber_edge_homomorphism_for_first_quadrant_convergent_spectral_sequence_in_an_abelian_category}{baes and fiber edge maps} in this spectral sequence are the natural maps 
        $$(R^p F) (GA) \to R^p(FG)A \quad \text{and} \quad R^q(FG)(A) \to F(R^q G(A))$$

        The \CrefAndHyperrefIfExist{proposition:five_term_exact_sequence_exact_sequence_of_low_degree_terms_for_a_convergent_first_quadrant_spectral_sequence_in_an_abelian_category}{exact sequence of low degree terms} is 
        $$0 \to (R^1 F)(GA) \to R^1(FG)(A) \to F(R^1 G(A)) \to (R^2 F)(GA) \to R^2(FG)A$$


        \item Let $\mathcal{A}, \mathcal{B}, \mathcal{C}$ be \CrefAndHyperrefIfExist{definition:abelian_category}{abelian categories} such that $\calA$ and $\calB$ \CrefAndHyperrefIfExist{definition:has_enough_injectives_or_projectives_for_an_abelian_category}{have enough projectives}. Let $G: \mathcal{A} \to \mathcal{B}$ and $F: \mathcal{B} \to \mathcal{C}$ be \CrefAndHyperrefIfExist{definition:exact_functor_between_abelian_categories}{right exact functors}. Assume that $G$ sends \CrefAndHyperrefIfExist{definition:injective_and_projective_objects_in_a_category}{projective objects} of $\mathcal{A}$ to \CrefAndHyperrefIfExist{definition:F_acyclic_object_for_a_left_or_right_functor_between_abelian_categories}{$F$-acyclic objects} of $\mathcal{B}$. 

        There is a \CrefAndHyperrefIfExist{definition:converges_to_a_graded_object_for_a_spectral_sequence_in_an_abelian_category}{convergent} \CrefAndHyperrefIfExist{definition:first_quadrant_spectral_sequence}{first quadrant} \CrefAndHyperrefIfExist{definition:spectral_sequence_in_an_abelian_category}{homological spectral sequence} for each $A$ in $\mathcal{A}$ starting in the second page:
        $$E^2_{p,q} = (L_p F) (L_q G)(A) \Rightarrow L_{p+q} (FG) (A).$$
        (\Cref{definition:left_right_derived_functors_of_a_right_left_exact_functor_between_abelian_categories_where_source_has_enough_projectives_injectives})

        The \CrefAndHyperrefIfExist{definition:homological_cohomological_base_fiber_edge_homomorphism_for_first_quadrant_convergent_spectral_sequence_in_an_abelian_category}{base and fiber edge maps} in this spectral sequence are the natural maps:
        $$(L_n F)(GA) \leftarrow L_n(FG)(A) \quad \text{and} \quad L_n(FG)(A) \to F(L_n G(A))$$

        The \CrefAndHyperrefIfExist{proposition:five_term_exact_sequence_exact_sequence_of_low_degree_terms_for_a_convergent_first_quadrant_spectral_sequence_in_an_abelian_category}{exact sequence of low degree terms} is:
        $$L_2(FG)A \to (L_2 F)(GA) \to F(L_1 G(A)) \to L_1(FG)(A) \to (L_1 F)(GA) \to 0$$

    \end{enumerate}

    
\end{theorem}
