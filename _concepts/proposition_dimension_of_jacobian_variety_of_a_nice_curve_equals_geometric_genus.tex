\begin{proposition}[see e.g. {\cite[Chapter III, Proposition 2.1.]{milne_av}}] \label{proposition:dimension_of_jacobian_variety_of_a_nice_curve_equals_geometric_genus}
    \TODO{tangent space}
    Let $X$ be a \CrefAndHyperrefIfExist{definition:smooth_morphism_of_schemes}{smooth}, \CrefAndHyperrefIfExist{definition:projective_variety_over_a_field}{projective} \CrefAndHyperrefIfExist{definition:algebraic_curve_over_a_field}{curve} whose \CrefAndHyperrefIfExist{definition:connected_component_of_a_point_of_a_topological_space}{connected components} are \CrefAndHyperrefIfExist{definition:geometrically_connected_scheme_over_a_field}{geometrically connected}.
    The tangent space to the \CrefAndHyperrefIfExist{definition:jacobian_of_a_curve_over_a_field}{Jacobian variety} $\Jac(X)$ at the identity point is canonically isomorphic to $H^1(X, \calO_X)$. Consequently, the dimension of $\Jac(X)$ equals the geometric genus of $X$.
\end{proposition}
