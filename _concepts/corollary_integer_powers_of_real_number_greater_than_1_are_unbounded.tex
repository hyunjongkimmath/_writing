\begin{corollary}[Integer powers of real numbers greater than 1 are unbounded] \label{corollary:integer_powers_of_real_number_greater_than_1_are_unbounded}
    Let $b \in \mathbb{R}$ with $b > 1$. For every $M > 0$, there exists an integer $n \in \mathbb{N}$ such that $b^n > M$.

    Equivalently, the set $\{ b^n : n \in \mathbb{N} \}$ is unbounded above in $\mathbb{R}$.
\end{corollary}
