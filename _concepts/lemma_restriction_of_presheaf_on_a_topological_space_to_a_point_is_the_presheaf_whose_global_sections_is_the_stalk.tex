\begin{lemma} \label{lemma:restriction_of_presheaf_on_a_topological_space_to_a_point_is_the_presheaf_whose_global_sections_is_the_stalk}
    \TODO{check for general categories}
    Let $X$ be a \CrefAndHyperrefIfExist{definition:topological_space}{topological space}. Let $i_x: \{x\} \hookrightarrow X$ be the inclusion of a point. Let $\calA$ be a \CrefAndHyperrefIfExist{definition:category}{category}.
    For any presheaf $\calF$ on $X$ valued in $\calA$, 
    \begin{enumerate}
        \item The \CrefAndHyperrefIfExist{definition:restriction_of_a_presheaf_sheaf_on_a_topological_space}{restriction} $\calF|_{\{x\}} = i_x^* \calF$ \CrefIfExists{definition:inverse_image_of_a_sheaf_on_a_topological_space} exists if and only if the \CrefAndHyperrefIfExist{definition:stalk_of_a_presheaf_on_a_topological_space_at_a_point}{stalk} $\calF_x$ does. 
        \item $\calF|_{\{x\}}$ is the presheaf on $\{x\}$ \CrefAndHyperref{lemma:categories_of_sheaves_presheaves_on_a_singleton_set_are_equivalent_to_value_category}{corresponding to} $\calF_x$.  
    \end{enumerate}
\end{lemma}
