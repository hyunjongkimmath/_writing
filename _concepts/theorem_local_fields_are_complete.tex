
\begin{theorem} \label{theorem:local_fields_are_complete}
    Let $K$ be a \CrefAndHyperrefIfExist{definition:local_field}{local field}. $K$ is \CrefAndHyperrefIfExist{definition:complete_extended_metric_space}{complete} under the metric \CrefAndHyperrefIfExist{definition:metric_induced_by_an_absolute_value_on_a_field}{induced by} its absolute value.
\end{theorem}
