\begin{lemma}[A power series converges uniformly on compact subsets of its disk of convergence] \label{lemma:power_series_in_one_complex_variable_converges_uniformly_and_absolutely_on_compact_subsets_of_its_disk_of_convergence}
    Let $\sum_{n=0}^\infty a_n (z - z_0)^n$ be a \CrefAndHyperrefIfExist{definition:power_series_in_one_complex_variable_and_its_radius_of_convergence_given_by_the_cauchy_hadamard_formula}{power series in one complex variable} with coefficients $a_n \in \mathbb{C}$ and with \CrefAndHyperrefIfExist{definition:power_series_in_one_complex_variable_and_its_radius_of_convergence_given_by_the_cauchy_hadamard_formula}{radius of convergence} $R \in (0, \infty]$. For every \CrefAndHyperrefIfExist{definition:compact_topological_space}{compact} subset $K \subset D(z_0, R)$, the series \CrefAndHyperrefIfExist{definition:uniform_convergence_of_a_series_of_functions_from_a_set_to_an_extended_metric_space_valued_in_ordered_commutative_monoid_with_adjoined_infinity}{converges uniformly} and \CrefAndHyperrefIfExist{definition:absolute_convergence_of_a_series_in_module_over_a_commutative_ring_with_an_absolute_value_valued_in_an_ordered_semiring}{absolutely} on $K$.
\end{lemma}

\begin{proof}
    By \Cref{proposition:power_series_in_one_complex_variable_on_closed_disks_inside_its_radius_of_convergence_converges_absolutely_and_uniformly}, the series \CrefAndHyperrefIfExist{definition:uniform_convergence_of_a_series_of_functions_from_a_set_to_an_extended_metric_space_valued_in_ordered_commutative_monoid_with_adjoined_infinity}{converges uniformly} and \CrefAndHyperrefIfExist{definition:absolute_convergence_of_a_series_in_module_over_a_commutative_ring_with_an_absolute_value_valued_in_an_ordered_semiring}{absolutely} on every closed disk $\overline{D}(z_0, r)$ with $r < R$. 

    \TODO{factor out and reference the general statement that $d(z) = |z-z_0|$ is a continuous function; is this true of general extended metric spaces valeud in ordered rings?}
    The function $d(z) = |z - z_0|$ is continuous on $K$ by \Cref{proposition:identity_function_is_continuous_between_topological_spaces} and the \CrefAndHyperrefIfExist{proposition:reverse_triangle_inequality_in_extended_metric_space}{reverse triangle inequality}\CrefIfExists{definition:absolute_value_on_a_ring_valued_in_an_ordered_semiring}. Since $K$ is compact, \Cref{proposition:continuous_real_valued_function_on_compact_topological_space_attains_maximum_and_minimum} implies that $d$ attains a maximum $r_K = \max_{z \in K} |z - z_0|$ on $K$. Because $K \subset D(z_0, R)$, we have $r_K < R$. Hence $K \subset \overline{D}(z_0, r_K)$, and the convergence on $\overline{D}(z_0, r_K)$ is uniform and absolute. Therefore the convergence is uniform and absolute on $K$.
\end{proof}
