\begin{lemma} \label{lemma:differential_graded_object_in_a_preadditive_category_is_chain_complex}
    Let $\calA$ be a \CrefAndHyperrefIfExist{definition:additive_category_preadditive_category}{preadditive category}. 
    \begin{enumerate}
        \item A \CrefAndHyperrefIfExist{definition:chain_complex_of_objects_in_a_preadditive_category}{chain complex} in $\calA$ is exactly a \CrefAndHyperrefIfExist{definition:differential_graded_object_in_a_category_with_a_zero_object}{differential graded object} of degree $-1$. 
        \item A \CrefAndHyperrefIfExist{definition:cochain_complex_of_objects_in_an_additive_category}{cochain complex} in $\calA$ is exactly a \CrefAndHyperrefIfExist{definition:differential_graded_object_in_a_category_with_a_zero_object}{differential graded object} of degree $1$. 
    \end{enumerate}
\end{lemma}
