\begin{proposition} \label{proposition:sylow_p_subgroup_is_normal_if_and_only_if_n_p_equals_1}
Let $G$ be a \CrefAndHyperrefIfExist{definition:countable_finite_uncountable_sets}{finite} \CrefAndHyperrefIfExist{definition:group}{group} and let $p$ be a prime number dividing the order of $G$. The following are equivalent:

\begin{enumerate}
    \item The number of \CrefAndHyperrefIfExist{definition:sylow_p_subgroup_of_a_finite_group}{Sylow $p$-subgroups} satisfies \hl{$n_p = 1$}.
    \item There exists a unique Sylow $p$-subgroup of $G$, which is therefore a \CrefAndHyperrefIfExist{definition:normal_subgroup_of_a_group}{normal subgroup} of $G$.
\end{enumerate}

% In this case, the unique Sylow $p$-subgroup is denoted by $P \in \operatorname{Syl}_p(G)$ and satisfies $P \trianglelefteq G$.
\end{proposition}
