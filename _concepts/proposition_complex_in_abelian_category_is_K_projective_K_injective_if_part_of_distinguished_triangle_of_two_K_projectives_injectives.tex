\begin{proposition}[{\cite[1.3. Proposition]{spaltenstein_ruc}}]  \label{proposition:complex_in_abelian_category_is_K_projective_K_injective_if_part_of_distinguished_triangle_of_two_K_projectives_injectives}
Let $\calA$ be an \CrefAndHyperrefIfExist{definition:abelian_category}{abelian category}. 
If two of the vertices of a \CrefAndHyperref{definition:triangulated_category}{distinguished triangle} of \CrefAndHyperrefIfExist{definition:homotopy_category_of_chain_complexes_of_an_additive_category}{$K(\calA)$}  are \CrefAndHyperrefIfExist{definition:K_injective_projective_complex_in_an_abelian_category}{$K$-projective} (resp. \CrefAndHyperrefIfExist{definition:K_injective_projective_complex_in_an_abelian_category}{$K$-injective}), then so is the third.
\end{proposition}
