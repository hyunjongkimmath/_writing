\begin{theorem}[Residue Theorem] \label{theorem:the_residue_theorem_relating_the_contour_integral_of_a_meromorphic_function_along_a_closed_piecewise_continuously_differentiable_path_to_the_sum_of_residues_at_isolated_singularities_enclosed_by_the_path}
    Let $U \subset \mathbb{C}$ be a \CrefAndHyperrefIfExist{definition:simply_connected_space}{simply connected} open subset and let $f : U \to \widehat{\mathbb{C}}$ be a \CrefAndHyperrefIfExist{definition:meromorphic_function_on_an_open_subset_of_the_complex_plane}{meromorphic function} on $U$. Let $\gamma : [a,b] \to U$ be a \CrefAndHyperrefIfExist{definition:path_and_loop_in_a_topological_space}{closed} \CrefAndHyperrefIfExist{definition:piecewise_continuously_differentiable_path_from_a_closed_real_interval_to_the_complex_plane}{piecewise continuously differentiable path} in $U$ and let $z_1, \ldots, z_k \in U \setminus \gamma([a,b])$ be the \CrefAndHyperrefIfExist{definition:isolated_singularity_of_a_holomorphic_function_from_an_open_subset_of_the_complex_plane_to_the_complex_plane_classified_as_removable_pole_or_essential}{poles} of $f$ that lie in the \CrefAndHyperrefIfExist{definition:winding_number_of_a_closed_path_in_the_complex_plane_around_a_point}{region enclosed by $\gamma$} in the sense that $n(\gamma, z_j) \neq 0$ for each $j$. Then
    $$\int_{\gamma} f(z)\,dz = 2\pi i \sum_{j=1}^{k} n(\gamma, z_j) \operatorname{Res}(f, z_j),$$
    \CrefIfExists{definition:contour_integral_of_a_complex_valued_function_along_a_piecewise_continuously_differentiable_path}
    where $n(\gamma, z_j)$ is the \CrefAndHyperrefIfExist{definition:winding_number_of_a_closed_path_in_the_complex_plane_around_a_point}{winding number of $\gamma$ around $z_j$} and $\operatorname{Res}(f, z_j)$ is the \CrefAndHyperrefIfExist{definition:residue_of_a_holomorphic_function_at_an_isolated_singularity_defined_as_the_coefficient_of_z_minus_z_0_to_the_negative_1_in_its_laurent_expansion}{residue of $f$ at $z_j$}.
\end{theorem}

\begin{proof}
    We proceed in three steps: the \CrefAndHyperrefIfExist{definition:laurent_series_expansion_of_a_holomorphic_function_on_an_annulus_centered_at_an_isolated_singularity}{Laurent expansion} of $f$ at each pole, the removal of all \CrefAndHyperrefIfExist{definition:laurent_series_expansion_of_a_holomorphic_function_on_an_annulus_centered_at_an_isolated_singularity}{principal parts}, and the explicit evaluation of the remaining contour integrals.

    \begin{enumerate}
        \item For each $j = 1,\ldots,k$, write $m_j$ for the order of the pole of $f$ at $z_j$. For each such $j$, there exists a punctured neighborhood $0 < |z - z_j| < R_j$ on which $f$ \CrefAndHyperrefIfExist{theorem:laurent_series_expansion_for_holomorphic_functions_on_annuli_in_the_complex_plane}{admits} a \CrefAndHyperrefIfExist{definition:laurent_series_expansion_of_a_holomorphic_function_on_an_annulus_centered_at_an_isolated_singularity}{Laurent expansion}
        $$f(z) = \sum_{n=-m_j}^{\infty} a_n^{(j)} (z - z_j)^n,$$
        with coefficients $a_n^{(j)} \in \mathbb{C}$. In particular, the coefficient $a_{-1}^{(j)}$ is the residue $\operatorname{Res}(f, z_j)$ by \CrefAndHyperrefIfExist{definition:residue_of_a_holomorphic_function_at_an_isolated_singularity_defined_as_the_coefficient_of_z_minus_z_0_to_the_negative_1_in_its_laurent_expansion}{definition}.

        The \CrefAndHyperrefIfExist{definition:laurent_series_expansion_of_a_holomorphic_function_on_an_annulus_centered_at_an_isolated_singularity}{principal part} of $f$ at $z_j$ is the finite sum
        $$P_j(z) := \sum_{n=-m_j}^{-1} a_n^{(j)} (z - z_j)^n,$$
        which is holomorphic on $\mathbb{C} \setminus \{z_j\}$.

        \item We next remove all principal parts to obtain a holomorphic function. Define $g : U \setminus \{z_1, \ldots, z_k\} \to \mathbb{C}$ by
        $$g(z) := f(z) - \sum_{j=1}^{k} P_j(z).$$
        Note that $g$ is holomorphic on $U \setminus \{z_1, \ldots, z_k\}$ because\CrefIfExists{proposition:proposition_sum_rule_product_rule_quotient_rule_chain_rule_for_complex_derivatives_of_holomorphic_functions_on_open_subsets_of_the_complex_plane} $f(z)$ and the $P_j(z)$ are.
        We show that $g$ extends to a holomorphic function on all of $U$.

        At each $z_j$, the Laurent expansion of $g$ in the punctured disk $0 < |z - z_j| < R_j$ is
        $$\sum_{n=-m_j}^{\infty} a_n^{(j)} (z - z_j)^n - \sum_{n=-m_j}^{-1} a_n^{(j)} (z - z_j)^n = \sum_{n=0}^{\infty} a_n^{(j)} (z - z_j)^n,$$
        The Laurent series of $g$ at $z_j$ is obtained by the termwise sum of the Laurent series of $f$ and the principal part $P_j$, by \Cref{proposition:termwise_algebra_for_absolutely_convergent_series}. This resulting power series $\sum_{n=0}^\infty a_n^{(j)} (z - z_j)^n$ has radius of convergence at least $R_j$, because it converges at every $z$ with $0 < |z - z_j| < R_j$ (it is a subseries of the absolutely convergent Laurent series of $f$). By \Cref{lemma:power_series_in_one_complex_variable_converges_uniformly_and_absolutely_on_compact_subsets_of_its_disk_of_convergence}, the partial sums $S_N^{(j)}(z) = \sum_{n=0}^N a_n^{(j)} (z - z_j)^n$ converge uniformly on every compact subset of $D(z_j, R_j)$.

        Each partial sum $S_N^{(j)}$ is a polynomial, hence an entire function by \Cref{proposition:complex_polynomials_are_entire_functions}, so $(S_N^{(j)})$ is a sequence of holomorphic functions converging uniformly on compact subsets of $D(z_j, R_j)$. By \Cref{theorem:the_weierstrass_convergence_theorem_stating_that_the_uniform_limit_on_compact_subsets_of_a_sequence_of_holomorphic_functions_is_itself_holomorphic_with_all_derivatives_converging_uniformly_on_compact_subsets}, the limit function
        $$h_j(z) := \sum_{n=0}^{\infty} a_n^{(j)} (z - z_j)^n$$
        is holomorphic on $D(z_j, R_j)$ and satisfies $h_j(z) = g(z)$ for all $z$ with $0 < |z - z_j| < R_j$. Therefore $h_j$ is a holomorphic extension of $g$ across $z_j$, which by \Cref{definition:isolated_singularity_of_a_holomorphic_function_from_an_open_subset_of_the_complex_plane_to_the_complex_plane_classified_as_removable_pole_or_essential} means that $z_j$ is a removable singularity of $g$. After removing all such singularities we obtain a holomorphic extension of $g$ to all of $U$.
        

        Because $U$ is \CrefAndHyperrefIfExist{definition:simply_connected_space}{simply connected} and $\gamma$ is a closed piecewise continuously differentiable path in $U$, \Cref{theorem:cauchy_integral_theorem_stating_that_the_contour_integral_of_a_holomorphic_function_along_any_closed_piecewise_continuously_differentiable_path_in_a_simply_connected_domain_vanishes} gives
        $$\int_{\gamma} g(z) \, dz = 0.$$

        \item We conclude by evaluating the contour integral of $f$. From the definition of $g$, we have
        $$f(z) = g(z) + \sum_{j=1}^{k} P_j(z)$$
        for all $z \in U \setminus \{z_1, \ldots, z_k\}$. Integrating termwise and using $\int_\gamma g = 0$, we obtain
        By \Cref{proposition:linearity_of_the_contour_integral_of_a_complex_valued_functions_along_a_piecewise_continuously_differentiable_path}, the contour integral splits as the sum of the individual integrals:
        \begin{align*}
        \int_{\gamma} f(z) \, dz &= \sum_{j=1}^{k} \int_{\gamma} P_j(z) \, dz \\
        &= \sum_{j=1}^{k} \sum_{n=-m_j}^{-1} a_n^{(j)} \int_{\gamma} (z - z_j)^n \, dz.
        \end{align*}

        For each $j$ and each integer $n$, \Cref{lemma:contour_integral_of_powers_of_z_minus_z0_around_closed_path_in_complex_plane} gives
        $$\int_{\gamma} (z - z_j)^n \, dz =
        \begin{cases}
        2\pi i \cdot n(\gamma, z_j), & n = -1,\\
        0,                           & n \neq -1.
        \end{cases}$$

        Therefore all terms with $n \neq -1$ vanish, and only the $n = -1$ terms, i.e.\ the residues, contribute:
        \begin{align*}
        \int_{\gamma} f(z) \, dz &= \sum_{j=1}^{k} a_{-1}^{(j)} \cdot 2\pi i \cdot n(\gamma, z_j) \\
        &= 2\pi i \sum_{j=1}^{k} n(\gamma, z_j) \operatorname{Res}(f, z_j)
        \end{align*}
        by the identification $a_{-1}^{(j)} = \operatorname{Res}(f, z_j)$ from Step 1.
    \end{enumerate}

    This completes the proof.
\end{proof}
