\begin{proposition}[{\cite[1.3.3(i)]{beilinson_berstein_deligne_atlesI}}] \label{proposition:adjunction_of_inclusion_functors_of_t_structures_on_triangulated_categories}

    Let $D$ be a \CrefAndHyperrefIfExist{definition:t_structure_on_a_triangulated_category}{triangulated category} and let $(\Dleqzero, \Dgeqzero)$ be a \CrefAndHyperrefIfExist{definition:t_structure_on_a_triangulated_category}{$t$-structure} on $D$.

    For all $n \in \bbZ$, the inclusion functor $\Dleqn \hookrightarrow D$ has a \CrefAndHyperrefIfExist{definition:adjoint_functors_between_categories_unit_counit_of_adjoint_functors}{right adjoint} \hl{$\tauleqn: D \to \Dleqn$} and the inclusion functor $\Dgeqn \subset D$ has a \CrefAndHyperrefIfExist{definition:adjoint_functors_between_categories_unit_counit_of_adjoint_functors}{left adjoint} \hl{$\taugeqn: D \to \Dgeqn$}.
\end{proposition}
