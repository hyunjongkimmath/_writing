\begin{proposition}[Jensen's inequality] \label{proposition:jensens_inequality_for_convex_functions_on_a_probability_space}
    Let $(\Omega, \mathcal{F}, P)$ be a \CrefAndHyperrefIfExist{definition:probability_space}{probability space}. Let $\varphi : I \to \mathbb{R}$ be a convex function defined on an interval $I \subseteq \mathbb{R}$. If $X : \Omega \to I$ is a random variable such that $\mathbb{E}[|X|] < \infty$ and $\mathbb{E}[|\varphi(X)|] < \infty$, then
    $$\varphi(\mathbb{E}[X]) \leq \mathbb{E}[\varphi(X)].$$
\end{proposition}
