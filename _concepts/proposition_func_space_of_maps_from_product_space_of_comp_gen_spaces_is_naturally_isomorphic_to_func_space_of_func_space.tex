\begin{proposition}[{\cite[Proposition Page 41]{may}}] \label{proposition:func_space_of_maps_from_product_space_of_comp_gen_spaces_is_naturally_isomorphic_to_func_space_of_func_space}
    For \CrefAndHyperrefIfExist{definition:compactly_generated_topological_space}{compactly generated} \CrefAndHyperrefIfExist{definition:topological_space}{topological spaces} $X,Y,Z$, the canonical bijection
    $$Z^{(X \times Y)} \cong (Z^Y)^X$$
    of \CrefAndHyperrefIfExist{definition:compact_open_topology_on_the_set_of_continuous_maps_between_topological_spaces_and_function_space}{function spaces (with the compact-open topology)} is a \CrefAndHyperrefIfExist{definition:homeomorphism_of_topological_spaces}{homeomorphism}.
\end{proposition}
