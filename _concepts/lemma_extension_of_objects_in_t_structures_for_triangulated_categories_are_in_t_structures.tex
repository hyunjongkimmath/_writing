Let $(X,Y,Z)$ be a distinguished triangle. then

\begin{enumerate}
    \item If $X,Z \in \Dleqn$, then $Y \in \Dleqn$ as well.
    \item If $Y \in \Dleqn$ and if $X \in \Dleqn[n+1]$, then $Z \in \Dleqn$. 
\end{enumerate}
\end{lemma}
