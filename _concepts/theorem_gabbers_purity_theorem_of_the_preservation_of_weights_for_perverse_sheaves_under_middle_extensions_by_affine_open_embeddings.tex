\begin{theorem}[Gabber's purity theorem, see {\cite[Corollaire 5.3.2.]{beilinson_berstein_deligne_atlesI}}] \label{}

    Let $X/\Fq$ be a finitely generated scheme over a finite field. Let $\ell \neq \characteristic \Fq$ be a prime. Let $j: U \hookrightarrow X$ be an \CrefAndHyperrefIfExist{definition:affine_morphism_of_schemes}{affine} \CrefAndHyperrefIfExist{definition:open_and_closed_immersions_of_schemes}{open embedding}. Let $M \in \Perv(U, \Qellbar)$ be a mixed \CrefAndHyperrefIfExist{definition:self_dual_middle_perverse_t_structure_on_bdd_constructible_derived_category_with_algebraic_coefficients}{perverse sheaf} of weight $\leq w$ (resp. $\geq w$). The \CrefAndHyperrefIfExist{definition:middle_extension_functor_for_perverse_sheaves_with_algebraic_coefficients_on_open_immersion}{intermediate extension} $j_{!*} U$ is mixed of weight $\leq w$ (resp. $\geq w$). In particular, if $M$ is pure of weight $w$, then so is $j_{!*} M$. 

\end{theorem}
