\begin{theorem}{\cite[Theorem Chapter 22 Page 176]{may}} \label{theorem:reduced_homology_theory_on_pointed_CW_complexes_defined_by_homotopy_groups_of_smash_product_with_a_prespectrum}
    Let $\{T_n\}$ be a \CrefAndHyperrefIfExist{definition:prespectrum_of_pointed_topological_spaces}{prespectrum} such that $T_n$ is \CrefAndHyperrefIfExist{definition:n_connected_topological_space}{$(n-1)$-connected} and of the \CrefAndHyperrefIfExist{definition:homotopy_equivalence_between_topological_spaces}{homotopy type} of a \CrefAndHyperrefIfExist{definition:CW_complex}{CW complex} for each $n$. Define
    $$\hlin{\tilde{E}_q(X) = \colim_n \pi_{q+n}(X \wedge T_n)}$$
    where the colimit is taken over the maps 
    $$ \pi_{q+n}\left(X \wedge T_{n}\right) \stackrel{\Sigma}{\rightarrow} \pi_{q+n+1}\left(\Sigma\left(X \wedge T_{n}\right)\right) \cong \pi_{q+n+1}\left(X \wedge \Sigma T_{n}\right) \stackrel{\text { id } \wedge \sigma}{\longrightarrow} \pi_{q+n+1}\left(X \wedge T_{n+1}\right).$$

\end{theorem}
