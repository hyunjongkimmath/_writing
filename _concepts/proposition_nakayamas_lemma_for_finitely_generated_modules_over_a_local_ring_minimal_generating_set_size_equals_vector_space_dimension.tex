\begin{proposition}[Nakayama's lemma for local rings: minimal generating sets]  \label{proposition:nakayamas_lemma_for_finitely_generated_modules_over_a_local_ring_minimal_generating_set_size_equals_vector_space_dimension}

	Let $(R, \mathfrak{m})$ be a \CrefAndHyperrefIfExist{definition:local_ring}{local ring} (not necessarily commutative) with unique maximal ideal $\mathfrak{m}$ and residue field $k = R/\mathfrak{m}$. Let $M$ be a \CrefAndHyperrefIfExist{definition:finitely_generated_modules_over_rings}{finitely generated} right $R$-module.

	\begin{enumerate}
	    \item The quotient $M/\mathfrak{m}M$ is a finite-dimensional right \CrefAndHyperrefIfExist{definition:vector_space_over_a_field}{vector space} over $k$. If $M$ can be generated by $n$ elements as an $R$-module, then $\dim_k(M/\mathfrak{m}M) \leq n$.

	    \item Conversely, if $x_1 + \mathfrak{m}M, \ldots, x_n + \mathfrak{m}M \in M/\mathfrak{m}M$ form a basis of $M/\mathfrak{m}M$ as a $k$-vector space, then $x_1, \ldots, x_n$ generate $M$ as an $R$-module.

	    \item The minimum number of generators of $M$ as an $R$-module equals $\dim_k(M/\mathfrak{m}M)$. Moreover, any subset of $M$ whose image in $M/\mathfrak{m}M$ is linearly independent over $k$ can be extended to a minimal generating set of $M$.
	\end{enumerate}
\end{proposition}

	\begin{enumerate}
	    \item Since $M$ is finitely generated as an $R$-module and $\mathfrak{m}$ is an ideal of $R$, the quotient $M/\mathfrak{m}M$ is annihilated by $\mathfrak{m}$. Hence $M/\mathfrak{m}M$ is a right module over $R/\mathfrak{m} = k$, i.e., a right vector space. Since $M$ is finitely generated, the images of any finite generating set span $M/\mathfrak{m}M$, so it has finite dimension. If $M$ can be generated by $n$ elements, their images span $M/\mathfrak{m}M$, which implies $\dim_k(M/\mathfrak{m}M) \leq n$.

	    \item Suppose that $\bar{x}_1, \ldots, \bar{x}_n$ form a basis of $M/\mathfrak{m}M$ over $k$. Let $N$ be the submodule generated by lifts $x_1, \ldots, x_n$. The images of $x_1, \ldots, x_n$ in $M/\mathfrak{m}M$ generate this quotient, so $N + \mathfrak{m}M = M$. Since $(R, \mathfrak{m})$ is a local ring, $\mathfrak{m} = J(R)$. Apply Nakayama's submodule lemma\CrefIfExists{lemma:nakayamas_submodule_lemma_for_finitely_generated_modules_over_a_ring} with $I = \mathfrak{m}$ to conclude that $N = M$. Therefore, $x_1, \ldots, x_n$ generate $M$.

	    \item Part (2) shows that a basis of $M/\mathfrak{m}M$ lifts to a generating set of the same size. Part (1) shows that any generating set has size at least $\dim_k(M/\mathfrak{m}M)$. Therefore, the minimum number of generators equals $\dim_k(M/\mathfrak{m}M)$.

	    For the extension statement, suppose $S \subseteq M$ is a subset whose image in $M/\mathfrak{m}M$ is linearly independent over $k$. Extend this to a basis of $M/\mathfrak{m}M$, and lift each new element to an element of $M$. By part (2), the resulting set generates $M$.
	\end{enumerate}
\end{proof}
