\begin{theorem}[Criterion of Néron--Ogg--Shafarevich for good reduction of abelian varieties] \label{theorem:criterion_of_neron_ogg_shafarevich_for_good_reduction_of_abelian_varieties}

    \TODO{generic point}
    Let $S$ be a \CrefAndHyperrefIfExist{definition:dedekind_scheme}{Dedekind scheme}, let $\eta$ be its generic point, and let $K = \mathcal{O}_{S, \eta}$ be the \CrefAndHyperrefIfExist{definition:function_field_over_a_field}{function field} of $S$. Let $A_K$ be an \CrefAndHyperrefIfExist{definition:abelian_variety_over_a_field}{abelian variety over $K$}, and let $s \in S$ be a \CrefAndHyperrefIfExist{definition:closed_point_of_a_topological_space}{closed point}. Fix a prime $\bar{\mathfrak{s}}$ of the \CrefAndHyperrefIfExist{definition:integral_element_over_a_ring}{integral closure} of $\mathcal{O}_{S, s}$ in a \CrefAndHyperrefIfExist{definition:separable_closure_of_a_field}{separable closure} $\bar{K}$ of $K$, and let $I_s \subset G_K = \Gal(\bar{K}/K)$ be the corresponding \CrefAndHyperrefIfExist{definition:inertia_group_in_the_absolute_galois_group_of_the_function_field_of_a_dedekind_scheme_at_a_closed_point}{inertia subgroup}. Let $\ell$ be a \CrefAndHyperrefIfExist{definition:prime_number}{prime number} distinct from the \CrefAndHyperrefIfExist{definition:characteristic_of_a_ring}{characteristic} of the \CrefAndHyperrefIfExist{definition:residue_field_of_a_scheme_at_a_point}{residue field} $k(s)$.

    The following conditions are equivalent:
    \begin{enumerate}
        \item $A_K$ has \CrefAndHyperrefIfExist{definition:good_reduction_of_an_abelian_variety_over_a_dedekind_scheme}{good reduction at $s$}.
        \item There exists an \CrefAndHyperrefIfExist{definition:abelian_scheme_over_a_scheme}{abelian scheme} $\mathcal{A}$ over $S$ whose \CrefAndHyperrefIfExist{definition:special_and_generic_fiber_of_a_scheme_over_a_dedekind_scheme}{generic fiber} is isomorphic to $A_K$. In this case, the \CrefAndHyperrefIfExist{definition:neron_model_of_an_abelian_variety_over_the_function_field_of_an_integral_scheme.tex}{N\'eron model} of $A_K$ over $S$ is an abelian scheme over $S$, and its special fiber $\mathcal{A}_s$ is an abelian variety over $k(s)$.

        \item The inertia group $I_s$ acts trivially on the $\ell^\nu$-torsion points $A_K[\ell^\nu](\bar{K})$ for some integer $\nu \geq 1$.

        \item The inertia group $I_s$ acts trivially on the full \CrefAndHyperrefIfExist{definition:ell_adic_tate_module_of_an_abelian_variety}{$\ell$-adic Tate module} $T_\ell(A_K)(\bar{K})$.
        \item The inertia group $I_s$ acts trivially on the rational $\ell$-adic Tate module $V_\ell(A_K) = T_\ell(A_K) \otimes_{\mathbb{Z}_\ell} \mathbb{Q}_\ell$.
        \item The Tate module $T_\ell(A_K)$ (equivalently, $V_\ell(A_K)$) is an \CrefAndHyperrefIfExist{definition:unramified_galois_representation_for_a_dedekind_scheme_at_a_closed_point}{unramified} continuous $G_K$-representation at $s$.
    \end{enumerate}

    Moreover, if $A_K$ has \CrefAndHyperrefIfExist{definition:good_reduction_of_an_abelian_variety_over_a_dedekind_scheme}{good reduction} at $s$ with abelian scheme model $\mathcal{A}$, then for every prime $\ell'$ distinct from $\operatorname{char}(k(s))$, the Tate module $T_{\ell'}(A_K)$ is unramified at $s$. Furthermore, for any such $\ell'$, the specialization map induces an isomorphism of $G_{k(s)}$-modules:
    $$\hlin{T_{\ell'}(A_K) \xrightarrow{\sim} T_{\ell'}(\mathcal{A}_s)}$$
    where the $G_{k(s)} = \Gal(\overline{k(s)}/k(s))$-action on the left is via the identification $G_{k(s)} \cong D_s / I_s$.
\end{theorem}
