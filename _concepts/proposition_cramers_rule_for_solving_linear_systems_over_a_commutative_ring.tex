\begin{proposition}[Cramer's rule] \label{proposition:cramers_rule_for_solving_linear_systems_over_a_commutative_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:commutative_ring}{commutative ring}, and let $A = (a_{ij}) \in M_n(R)$ be an invertible $n \times n$ matrix with inverse given by $\det(A)^{-1}\operatorname{adj}(A)$. Let $b \in R^{n \times 1}$ be a column vector, and let $x \in R^{n \times 1}$ be the unique solution to $Ax = b$.

    For each $k = 1, \ldots, n$, the $k$-th component of $x$ is
    $$x_k = \frac{\det(A_k)}{\det(A)},$$
    where $A_k$ is the matrix obtained from $A$ by replacing its $k$-th column with the vector $b$.
\end{proposition}
