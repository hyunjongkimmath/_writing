\begin{proposition} \label{proposition:fundamental_group_of_a_galois_category_is_profinite}
 Let $(\mathcal{C}, F)$ be a \CrefAndHyperrefIfExist{definition:fiber_functor_for_a_functor_from_a_galois_category_to_the_category_of_finite_sets}{Galois category}.
 The \CrefAndHyperrefIfExist{definition:fundamental_group_of_a_galois_category}{fundamental group} of $(\calC, F)$ is \CrefAndHyperrefIfExist{definition:profinite_group}{profinite}. 
\end{proposition}
