\begin{proposition} \label{proposition:equivalent_formulations_of_solvable_finite_groups}
Let $G$ be a finite \CrefAndHyperrefIfExist{definition:group}{group}. 

\begin{enumerate}
    \item $G$ is a \CrefAndHyperref{definition:solvable_group_via_derived_series}{solvable group}.

    \item There exists a finite chain of \CrefAndHyperrefIfExist{definition:normal_subgroup_of_a_group}{normal subgroups}
    $$1 = G_0 \triangleleft G_1 \triangleleft \cdots \triangleleft G_k = G$$
    such that each \CrefAndHyperrefIfExist{definition:quotient_group_of_a_group_by_a_normal_subgroup}{quotient group} $G_{i+1}/G_i$ is an \CrefAndHyperrefIfExist{definition:abelian_group}{abelian group}.
    \item There exists a finite chain of subgroups
    $$1 = H_0 \triangleleft H_1 \triangleleft \cdots \triangleleft H_m = G$$
    such that each factor $H_{j+1}/H_j$ is a \CrefAndHyperrefIfExist{definition:cyclic_group}{cyclic group}.

    \item Every \CrefAndHyperref{definition:composition_series_and_composition_factors_of_a_group}{composition factor} of $G$ is a cyclic group of prime order.
\end{enumerate}
\end{proposition}
