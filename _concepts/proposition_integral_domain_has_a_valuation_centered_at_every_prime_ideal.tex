\begin{proposition} \label{proposition:integral_domain_has_a_valuation_centered_at_every_prime_ideal}
    Let $R$ be an \CrefAndHyperrefIfExist{definition:domain_integral_domain}{integral domain}. for any \CrefAndHyperrefIfExist{definition:prime_and_maximal_ideal_of_a_ring}{prime ideal} $\mathfrak{p}$, there is at least one valuation $v$ \CrefAndHyperrefIfExist{definition:centered_at_for_a_prime_of_an_integral_domain_point_of_an_integral_scheme}{centered at} $\mathfrak{p}$.
\end{proposition}
