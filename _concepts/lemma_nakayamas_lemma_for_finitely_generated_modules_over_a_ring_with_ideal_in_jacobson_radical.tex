\begin{lemma}[Nakayama's lemma]  \label{lemma:nakayamas_lemma_for_finitely_generated_modules_over_a_ring_with_ideal_in_jacobson_radical}
	Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{ring}, $M$ a \CrefAndHyperrefIfExist{definition:finitely_generated_modules_over_rings}{finitely generated} right $R$-module, and $I$ a left ideal of $R$ with $I \subseteq J(R)$\CrefIfExists{definition:jacobson_radical_of_a_ring}.

	If $IM = M$, then $M = 0$.
\end{lemma}
\begin{proof}
	Suppose that $IM = M$. Apply \CrefAndHyperref{lemma:nakayamas_submodule_lemma_for_finitely_generated_modules_over_a_ring}{Nakayama's submodule lemma} with $N = 0$. The hypothesis gives $0 + IM = M$, i.e., $IM = M$. Since $M$ is finitely generated and $I \subseteq J(R)$, Nakayama's submodule lemma implies that $0 = M$. Therefore, $M = 0$.
\end{proof}
