\begin{lemma} \label{lemma:base_change_functor_for_cats_of_finite_etale_covers_for_morphism_of_connected_schemes_is_exact_functor_of_galois_categories}
    Let $f: X \to Y$ be a morphism of \CrefAndHyperrefIfExist{definition:disconnected_and_connected_topological_spaces}{connected} \CrefAndHyperrefIfExist{definition:scheme}{schemes}. Let $\overline{x}$ be a \CrefAndHyperrefIfExist{definition:geometric_point_of_a_scheme}{geometric point} of $X$ and let $\overline{y} = f(\overline{x})$. 

    The \hldef{base change functor on $\FEt_Y$}
    $$\FEt_Y \to \FEt_X, \quad Z \mapsto Z \times_Y X$$
    \CrefIfExists{definition:category_of_finite_etale_covers_of_a_scheme}
    \CrefIfExists{definition:cartesian_product_of_two_objects_in_a_category_over_an_object}
    is an \CrefAndHyperrefIfExist{definition:exact_functor_between_galois_categories}{exact functor of Galois categories} when $\FEt_X$ and $\FEt_Y$ are equipped with the \CrefAndHyperrefIfExist{definition:fiber_functor_of_category_of_finite_etale_covers_of_a_scheme_given_by_a_geometric_point}{fiber functors} $F_{\overline{x}}$ and $F_{\overline{y}}$ for any \CrefAndHyperrefIfExist{definition:geometric_point_of_a_scheme}{geometric point} $\overline{x}$ of $X$ and $\overline{y} = f(\overline{x})$. 
\end{lemma}
