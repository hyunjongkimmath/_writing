\begin{lemma} \label{lemma:normalizer_of_subgroup_is_subgroup_of_group}
    Let $(G,\cdot)$ be a \CrefAndHyperrefIfExist{definition:group}{group} and let $H \subseteq G$ be a \CrefAndHyperrefIfExist{definition:subgroup_of_a_group}{subgroup}. The \CrefAndHyperrefIfExist{definition:normalizer_of_a_subgroup_of_a_group}{normalizer $N_G(H)$} is a \CrefAndHyperrefIfExist{definition:subgroup_of_a_group}{subgroup} of $G$.
\end{lemma}

\begin{proof}
    We verify the subgroup axioms for $N_G(H)$ inside $G$. The identity element $e \in G$ satisfies $e H e^{-1} = e H e = H$, so $e \in N_G(H)$. For any $g_1, g_2 \in N_G(H)$, we have
    $$(g_1 g_2) H (g_1 g_2)^{-1} = g_1 (g_2 H g_2^{-1}) g_1^{-1} = g_1 H g_1^{-1} = H,$$
    so $g_1 g_2 \in N_G(H)$. For any $g \in N_G(H)$, we have
    $$g^{-1} H g = g^{-1} (g H g^{-1}) g = H,$$
    so $g^{-1} \in N_G(H)$. Hence, $N_G(H)$ is a subgroup of $G$\CrefIfExists{proposition:subgroup_criterion_for_a_subset_of_a_group}, i.e. $N_G(H) \leq G$.
\end{proof}
