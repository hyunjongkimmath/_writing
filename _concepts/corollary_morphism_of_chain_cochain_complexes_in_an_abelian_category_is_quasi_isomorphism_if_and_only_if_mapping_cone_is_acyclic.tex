\begin{corollary} \label{corollary:morphism_of_chain_cochain_complexes_in_an_abelian_category_is_quasi_isomorphism_if_and_only_if_mapping_cone_is_acyclic}
    Let $\calA$ be an \CrefAndHyperrefIfExist{definition:abelian_category}{abelian category}. Let $f : C \to D$ be a morphism of \CrefAndHyperrefIfExist{definition:chain_complex_of_objects_in_an_additive_category}{chain or cochain complexes} in $\calA$. 
    $f$ is a \CrefAndHyperrefIfExist{definition:quasi_isomorphism_of_chain_complexes_of_objects_in_an_abelian_category}{quasi-isomorphism} if and only if the \CrefAndHyperrefIfExist{definition:mapping_cone_of_a_map_of_chain_cochain_complexes_in_an_additive_category}{mapping cone} $\operatorname{cone}(f)$ is \CrefAndHyperrefIfExist{definition:acyclic_complex_of_objects_in_an_abelian_category}{acyclic}.
\end{corollary}
