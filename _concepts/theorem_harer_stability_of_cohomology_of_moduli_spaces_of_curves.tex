\begin{theorem}[Harer Stability Theorem] \label{theorem:harer_stability_of_cohomology_of_moduli_spaces_of_curves}
  Let $g \geq 2$ be an integer and let $H^k(\mathcal{M}_g; \mathbb{Q})$ denote the $k$-th rational singular cohomology group of the moduli stack of curves of genus $g$. There exist stabilization maps
  $$ H^k(\mathcal{M}_{g+1}; \mathbb{Q}) \to H^k(\mathcal{M}_g; \mathbb{Q}) $$
  induced by the inclusion of surfaces (gluing a torus with one boundary component). For integers $g$ and $k$ satisfying the stability range
  $$ k \leq \frac{2g-2}{3}, $$
  these stabilization maps are isomorphisms. Consequently, the cohomology groups $H^k(\mathcal{M}_g; \mathbb{Q})$ are independent of $g$ in this range.

\end{theorem}
