\begin{theorem} \label{theorem:cauchys_theorem_that_a_fintie_group_contains_an_element_of_prime_factor_order_of_cardinality}
Let $G$ be a \CrefAndHyperrefIfExist{definition:countable_finite_uncountable_sets}{finite} \CrefAndHyperrefIfExist{definition:group}{group} and let $p$ be a \CrefAndHyperrefIfExist{definition:prime_number}{prime number}. 
If $p$ divides the order of $G$, then there exists an element $g \in G$ of order $p$.

Equivalently, $G$ contains a \CrefAndHyperrefIfExist{definition:p_subgroup_of_a_finite_group}{$p$-subgroup} of order $p$.
\end{theorem}
