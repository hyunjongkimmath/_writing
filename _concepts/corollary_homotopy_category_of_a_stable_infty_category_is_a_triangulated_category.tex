\begin{corollary}[{\cite[Remark 1.1.2.15.]{lurie_ha}}] \label{corollary:homotopy_category_of_a_stable_infty_category_is_a_triangulated_category}
    The \CrefAndHyperrefIfExist{definition:homotopy_category_of_a_simplicial_set}{homotopy category} $h\calC$ of a \CrefAndHyperrefIfExist{definition:stable_infinity_category}{stable $\infty$-category} is \CrefAndHyperrefIfExist{definition:triangulated_category}{triangulated}.
\end{corollary}
\begin{proof}
    This follows from \Cref{theorem:homotopy_category_of_a_pointed_infinity_category_with_cofibers_whose_suspension_is_an_equivalence_is_a_triangulated_category} --- every stable $\infty$-category satisfies the hypotheses there.
\end{proof}
