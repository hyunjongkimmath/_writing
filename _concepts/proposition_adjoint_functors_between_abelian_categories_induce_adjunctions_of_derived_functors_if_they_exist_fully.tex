\begin{proposition}[{\cite[Tag 09T5]{stacks-project}}] \label{proposition:adjoint_functors_between_abelian_categories_induce_adjunctions_of_derived_functors_if_they_exist_fully}
Let $F : \mathcal{A} \to \mathcal{B}$ and $G : \mathcal{B} \to \mathcal{A}$ be \CrefAndHyperrefIfExist{definition:functor_between_categories}{functors} of \CrefAndHyperrefIfExist{definition:abelian_category}{abelian categories} such that $F$ is a \CrefAndHyperrefIfExist{definition:adjoint_functors_between_categories_unit_counit_of_adjoint_functors}{right adjoint} to $G$. If the \CrefAndHyperrefIfExist{definition:right_left_derived_multifunctor_of_an_additive_multifunctor_between_abelian_categories}{derived functors} $RF : D(\mathcal{A}) \to D(\mathcal{B})$ and $LG : D(\mathcal{B}) \to D(\mathcal{A})$ exist, then $RF$ is a right adjoint to $LG$.
\end{proposition}
