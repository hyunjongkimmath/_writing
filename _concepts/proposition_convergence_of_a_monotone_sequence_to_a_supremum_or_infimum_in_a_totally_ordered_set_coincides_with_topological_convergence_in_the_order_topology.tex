% MONOTONE SEQUENCE CONVERGENCE COINCIDES WITH TOPOLOGICAL CONVERGENCE IN ORDER TOPOLOGY
\begin{proposition}[Convergence of monotone sequences to supremum or infimum in a totally ordered set coincides with topological convergence in the order topology] \label{proposition:convergence_of_a_monotone_sequence_to_a_supremum_or_infimum_in_a_totally_ordered_set_coincides_with_topological_convergence_in_the_order_topology}
    \TODO{AI generated, verify}
Let $(L, \leq)$ be a \CrefAndHyperrefIfExist{definition:total_order_linear_order_on_a_set}{totally ordered set} equipped with the \CrefAndHyperrefIfExist{definition:order_topology_on_a_partially_ordered_set}{order topology}.

Let $(x_n)_{n=1}^\infty$ be a sequence in $L$.
\begin{enumerate}
    \item If $(x_n)$ is non-decreasing and the set $\{x_n : n \geq 1\}$ has a \CrefAndHyperrefIfExist{definition:supremum_infimum_of_a_subset_of_an_ordered_set}{supremum} $s = \sup_{n} x_n$ in $L$, then $(x_n)$ \CrefAndHyperrefIfExist{definition:convergence_and_limit_of_sequence_in_a_topological_space}{converges} to $s$ in the order topology.

    \item If $(x_n)$ is non-increasing and the set $\{x_n : n \geq 1\}$ has an \CrefAndHyperrefIfExist{definition:supremum_infimum_of_a_subset_of_an_ordered_set}{infimum} $i = \inf_{n} x_n$ in $L$, then $(x_n)$ converges to $i$ in the order topology.
\end{enumerate}
Consequently, for a monotone sequence whose supremum or infimum exists in $L$, the order-theoretic limit coincides with the topological limit\CrefIfExists{definition:convergence_of_a_monotone_sequence_in_a_partially_ordered_set_with_adjoined_infinity}.
\end{proposition}
\begin{proof}
    \TODO{AI generated, verify}
    We prove item (1); item (2) follows by a symmetric argument applied to the reverse order.

    Suppose that $(x_n)$ is non-decreasing and let $s = \sup_{n} x_n$. Let $U$ be an open neighborhood of $s$ in the order topology. By \Cref{definition:order_topology_on_a_partially_ordered_set}, there exists a basis element $I$ such that
    $$s \in I \subseteq U.$$

    We consider cases based on the form of the basis element $I$.
    \begin{enumerate}
        \item If $I = (a, b)$ for some $a, b \in L$ with $a < s < b$, then since $s$ is the supremum of $\{x_n : n \geq 1\}$ and $a < s$, the element $a$ is not an upper bound of the sequence. Hence there exists an index $N$ such that $x_N > a$. Since $(x_n)$ is non-decreasing, for all $n \geq N$ we have $x_n \geq x_N > a$. Moreover, $x_n \leq s < b$ for all $n$, so $x_n \in (a, b) = I \subseteq U$ for all $n \geq N$.

        \item If $I = (-\infty, b)$ is an open lower ray with $s < b$, then for every $n$ we have $x_n \leq s < b$, so $x_n \in (-\infty, b) = I \subseteq U$ for all $n$.

        \item If $I = (a, \infty)$ is an open upper ray with $a < s$, then as in case (1), the element $a$ is not an upper bound of the sequence. Hence there exists an index $N$ such that $x_N > a$. Since $(x_n)$ is non-decreasing, for all $n \geq N$ we have $x_n \geq x_N > a$, so $x_n \in (a, \infty) = I \subseteq U$ for all $n \geq N$.
    \end{enumerate}

    In every case, there exists an index $N$ such that for all $n \geq N$, we have $x_n \in U$. Therefore $(x_n)$ converges to $s$ in the order topology.
\end{proof}
