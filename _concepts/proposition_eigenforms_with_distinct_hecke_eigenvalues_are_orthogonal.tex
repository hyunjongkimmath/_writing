\begin{proposition} \label{proposition:eigenforms_with_distinct_hecke_eigenvalues_are_orthogonal}
\TODO{AI generated, verify}
Let $k$ be a positive even integer and let $\Gamma = \mathrm{SL}_2(\bbZ)$. Let $f, g$ in the \CrefAndHyperref{definition:vector_space_of_modular_and_cusp_forms}{space of cusp forms} $S_k(\Gamma)$ be nonzero simultaneous eigenforms for all Hecke operators with eigenvalues $\{\lambda_n(f)\}_{n\geq 1}$ and $\{\lambda_n(g)\}_{n\geq 1}$ respectively. If there exists some positive integer $m$ such that
$$\hlin{\lambda_m(f) \neq \lambda_m(g)},$$
then $f$ and $g$ are orthogonal with respect to the \CrefAndHyperref{definition:petersson_inner_product_on_cusp_forms_of_a_congruence_subgroup}{Petersson inner product}:
$$\hlin{\langle f, g \rangle = 0.}$$

In particular, if $\lambda_p(f) \neq \lambda_p(g)$ for some prime $p$, then $\langle f, g \rangle = 0$.
\end{proposition}
