\begin{proposition} \label{proposition:determinant_of_identity_matrix_equals_one_over_a_commutative_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:commutative_ring}{commutative ring}, and let $I_n$\CrefIfExists{definition:identity_matrix} be the $n \times n$ \CrefAndHyperrefIfExist{definition:identity_matrix}{identity matrix}. Then
    $$\det(I_n) = 1_R.$$
\end{proposition}
