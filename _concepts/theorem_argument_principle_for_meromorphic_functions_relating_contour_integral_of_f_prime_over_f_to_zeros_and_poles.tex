\begin{theorem}[Argument Principle] \label{theorem:argument_principle_for_meromorphic_functions_relating_contour_integral_of_f_prime_over_f_to_zeros_and_poles}

    Let $U \subset \mathbb{C}$ be a \CrefAndHyperrefIfExist{definition:disconnected_and_connected_topological_spaces}{connected} open subset and let $f : U \to \mathbb{C}$ be a \CrefAndHyperrefIfExist{definition:meromorphic_function_on_an_open_subset_of_the_complex_plane}{meromorphic function} on $U$. Let $\gamma$ be a \CrefAndHyperrefIfExist{definition:path_and_loop_in_a_topological_space}{closed} \CrefAndHyperrefIfExist{definition:piecewise_continuously_differentiable_path_from_a_closed_real_interval_to_the_complex_plane}{piecewise continuously differentiable path} in $U$ that does not pass through any zero or pole of $f$ and whose image is a simple closed curve. Let $z_1, \ldots, z_m$ be the zeros of $f$ inside $\gamma$ with multiplicities $n_1, \ldots, n_m$, and let $p_1, \ldots, p_k$ be the poles of $f$ inside $\gamma$ with orders $m_1, \ldots, m_k$. Then
    $$\hlin{\frac{1}{2\pi i} \int_{\gamma} \frac{f'(z)}{f(z)} \, dz = N - P,}$$
    \CrefIfExists{definition:contour_integral_of_a_complex_valued_function_along_a_piecewise_continuously_differentiable_path}
    where $N = \sum_{i=1}^{m} n_i$ is the total number of zeros of $f$ inside $\gamma$ (counted with multiplicity) and $P = \sum_{j=1}^{k} m_j$ is the total number of poles of $f$ inside $\gamma$ (counted with order).
\end{theorem}

\begin{proof}
    \TODO{AI generated, verify}
    We factor $f$ as a product of its zero and pole factors times a nonvanishing holomorphic function, then compute the logarithmic derivative.

    \begin{enumerate}
        \item Since $f$ is meromorphic on $U$, around each zero $z_i$ of multiplicity $n_i$ we can write
        $$f(z) = (z - z_i)^{n_i} g_i(z)$$
        where $g_i$ is holomorphic and $g_i(z_i) \neq 0$ in a neighborhood of $z_i$. Similarly, around each pole $p_j$ of order $m_j$ we can write
        $$f(z) = (z - p_j)^{-m_j} h_j(z)$$
        where $h_j$ is holomorphic and $h_j(p_j) \neq 0$ in a neighborhood of $p_j$.

        \item The \hldef{logarithmic derivative} of $f$ is $\frac{f'}{f}$. Using the factorizations above, we compute:
        \begin{itemize}
            \item At a zero $z_i$ of multiplicity $n_i$:
            $$\frac{f'(z)}{f(z)} = \frac{n_i}{z - z_i} + \frac{g_i'(z)}{g_i(z)},$$
            where $\frac{g_i'}{g_i}$ is holomorphic at $z_i$.
            \item At a pole $p_j$ of order $m_j$:
            $$\frac{f'(z)}{f(z)} = -\frac{m_j}{z - p_j} + \frac{h_j'(z)}{h_j(z)},$$
            where $\frac{h_j'}{h_j}$ is holomorphic at $p_j$.
        \end{itemize}

        \item The function $\frac{f'}{f}$ is therefore meromorphic on $U$ with simple poles at the zeros and poles of $f$, and with residues $n_i$ at each zero $z_i$ and $-m_j$ at each pole $p_j$.

        \item By \Cref{theorem:the_residue_theorem_relating_the_contour_integral_of_a_meromorphic_function_along_a_closed_piecewise_continuously_differentiable_path_to_the_sum_of_residues_at_isolated_singularities_enclosed_by_the_path}, we have
        \begin{align*}
        \frac{1}{2\pi i} \int_{\gamma} \frac{f'(z)}{f(z)} \, dz &= \sum_{\text{singularities inside } \gamma} \operatorname{Res}\left(\frac{f'}{f}, \cdot\right) \\
        &= \sum_{i=1}^{m} n_i - \sum_{j=1}^{k} m_j \\
        &= N - P.
        \end{align*}
        \CrefIfExists{definition:residue_of_a_holomorphic_function_at_an_isolated_singularity_defined_as_the_coefficient_of_z_minus_z_0_to_the_negative_1_in_its_laurent_expansion}
    \end{enumerate}

    This completes the proof.
\end{proof}
