\begin{lemma}[see {\cite[Chapter I, 5.1]{knus_qhfr}}] \label{lemma:base_change_of_a_commutative_unitary_ring_by_a_unitary_algebra_is_a_unitary_algebra}
    Let $(R, \varepsilon, \Lambda)$ be a \CrefAndHyperrefIfExist{definition:commutative_ring}{commutative} \CrefAndHyperrefIfExist{definition:unitary_ring}{unitary ring}.

    For any commutative $R$-algebra $R'$, the triple $(A \otimes_R R', \varepsilon' = \varepsilon \otimes 1, \Lambda' = \Lambda \otimes_R R')$ is a \CrefAndHyperrefIfExist{definition:unitary_algebra_over_an_involution_ring}{unitary $R'$-algebra}.
\end{lemma}
