\begin{proposition} \label{proposition:group_homology_cohomology_for_module_of_finite_cyclic_group.tex}
    Let $m \geq 1$ be an integer. Let $C_m$ be the cyclic group of $m$ elements generated by $\sigma$. Let $R$ be a (not necessarily commutative) ring. Let $A$ be a (left) $R[C_m]$-module. Let $N = \sum_{i=0}^{m-1} \sigma^i$ be the \CrefAndHyperrefIfExist{definition:norm_element_of_a_group_algebra_of_a_finite_group_over_a_ring}{norm element} and $D = \sigma - 1$ be the \CrefAndHyperrefIfExist{definition:augmentation_map_of_a_group_ring_to_its_base_ring}{augmentation ideal} generator. We have
    \TODO{cyclic group}
    \begin{align*}
        H_n(C_m, A) &\cong \begin{cases} 
            A_{C_m} = A/(\sigma-1)A & n = 0 \\
            \ker(\sigma-1)/\text{im}(N) & n \text{ odd, } n \geq 1 \\
            \ker(N)/\text{im}(\sigma-1) & n \text{ even, } n \geq 2 
        \end{cases} \\
        H^n(C_m, A) &\cong \begin{cases} 
            A^{C_m} = \ker(\sigma-1) & n = 0 \\
            \ker(N)/\text{im}(\sigma-1) & n \text{ odd, } n \geq 1 \\
            \ker(\sigma-1)/\text{im}(N) & n \text{ even, } n \geq 2 
        \end{cases}
    \end{align*}
    (\Cref{definition:group_cohomology_homology_of_a_module_over_a_group_ring_over_a_ring}) (\Cref{definition:invariant_subgroup_of_a_group_acting_on_an_abelian_group}) (\Cref{definition:coinvariant_quotient_group_of_a_group_acting_on_an_abelian_group})
\end{proposition}
\begin{proof}
    By \Cref{corollary:group_homology_cohomology_are_just_tor_and_ext_functors_by_trivial_module_over_group_ring}, 
    \begin{align*}
        H_n(C_m, A) &\cong \Tor_n^{R[C_m]}(R, A) \\
        H^n(C_m, A) &\cong \Ext^n_{R[C_m]}(R, A).
    \end{align*}
    (\Cref{definition:tor_functors_of_bimodules_of_rings}) (\Cref{definition:ext_functors_of_bimodules_of_rings}) 
    The objects $\Tor_n^{R[C_m]}(R, A)$ and $\Ext^n_{R[C_m]}(R, A)$ can be (\Cref{corollary:tor_objects_of_modules_over_rings_are_well_defined}) (\Cref{definition:ext_functors_of_bimodules_of_rings}) computed by taking a general \CrefAndHyperrefIfExist{definition:flat_resolution_of_an_object_in_an_abelian_category_with_respect_to_a_right_exact_monoidal_product_functor}{flat resolution} (with respect to $\otimes_{R[C_m]}$) of $R$ as an $R[C_m]$-module and a \CrefAndHyperrefIfExist{definition:left_right_resolution_of_an_object_by_a_class_of_objects_in_an_abelian_category}{projective resolution} of $R$ as an $R[C_m]$-module respectively.

    The resolution 
    $$\cdots \xrightarrow{N} R[C_m] \xrightarrow{\sigma-1} R[C_m] \xrightarrow{N} R[C_m] \xrightarrow{\sigma-1} R[C_m] \xrightarrow{\varepsilon} R \to 0$$
    in \Cref{lemma:group_ring_of_a_cyclic_group_has_periodic_free_resolution} is a \CrefAndHyperrefIfExist{definition:left_right_resolution_of_an_object_by_a_class_of_objects_in_an_abelian_category}{left resolution} of $R$ by \CrefAndHyperrefIfExist{definition:free_bimodule_over_rings}{free} $R[C_m]$-modules; in particular, \CrefAndHyperrefIfExist{lemma:examples_of_projective_and_injective_objects}{it is} a projective resolution and \CrefAndHyperrefIfExist{proposition:projective_modules_are_flat}{it is} a flat resolution. 
    
    The desired isomorphisms all follow from the Tor and Ext computations using this resolution.
\end{proof}
