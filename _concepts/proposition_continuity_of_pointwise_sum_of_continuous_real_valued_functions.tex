\begin{proposition} \label{proposition:continuity_of_pointwise_sum_of_continuous_real_valued_functions}
    Let $X$ be a \CrefAndHyperrefIfExist{definition:topological_space}{topological space} and let $\bbR$ be the \CrefAndHyperrefIfExist{definition:field_of_real_numbers}{field of real numbers} equipped with its \CrefAndHyperrefIfExist{definition:standard_absolute_value_on_real_numbers}{standard topology}. Let $f,g : X \to \bbR$ be \CrefAndHyperrefIfExist{definition:continuous_map_of_topological_spaces}{continuous} functions. Their pointwise sum $f+g : X \to \bbR$, defined by $(f+g)(x) = f(x) + g(x)$, is continuous.

\end{proposition}
\begin{proof}
    Let $a \in X$. Since $f$ and $g$ are continuous at $a$, we have
    $$\lim_{x \to a} f(x) = f(a) \quad\text{and}\quad \lim_{x \to a} g(x) = g(a)$$
    \CrefIfExists{definition:limit_of_a_function_between_extended_metric_spaces_valued_in_ordered_commutative_monoid_with_adjoined_infinity_at_an_accumulation_point}\CrefIfExists{proposition:continuity_at_a_point_in_extended_metric_space_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity_equivalent_to_topological_definition_and_limit_at_accumulation_point}. By \Cref{proposition:limit_algebra_for_real_valued_functions}, the limit of the sum is the sum of the limits, so
    $$\lim_{x \to a} (f+g)(x) = f(a) + g(a) = (f+g)(a).$$
    Thus $f+g$ is continuous at $a$ \CrefIfExists{proposition:continuity_at_a_point_in_extended_metric_space_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity_equivalent_to_topological_definition_and_limit_at_accumulation_point}. Since $a$ was arbitrary, $f+g$ is continuous \CrefIfExists{lemma:function_between_topological_spaces_is_continuous_iff_continuous_at_all_points}.
\end{proof}
