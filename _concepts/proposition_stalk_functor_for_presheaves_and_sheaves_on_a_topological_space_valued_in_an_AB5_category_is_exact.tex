\begin{proposition} \label{proposition:stalk_functor_for_presheaves_and_sheaves_on_a_topological_space_valued_in_an_AB5_category_is_exact}

    Let $\calA$ be an \CrefAndHyperrefIfExist{definition:grothendiecks_additional_axioms_for_abelian_categories}{AB5 category}. Let $X$ be a \CrefAndHyperrefIfExist{definition:topological_space}{topological space}, and let $x \in X$ be a point. 
    \begin{enumerate}
        \item The \CrefAndHyperrefIfExist{definition:stalk_of_a_presheaf_on_a_topological_space_at_a_point}{stalk} functor $\calF \mapsto \calF_x$ is an \CrefAndHyperrefIfExist{definition:exact_functor_between_abelian_categories}{exact functor} $\PreShv(X, \calA) \to \calA$ (\Cref{definition:presheaf_on_a_topological_space}). 
        \item Assume that a \CrefAndHyperrefIfExist{definition:sheafification_of_a_presheaf_on_a_topological_space}{sheafification} functor 
                $$a: \PreShv(X, \calA) \to \Shv(X, \calA) $$
        exists. The stalk functor $\calF \mapsto \calF_x$ is an exact functor $\Shv(X, \calA) \to \calA$. 
        \item Assume that a \CrefAndHyperrefIfExist{definition:sheafification_of_a_presheaf_on_a_topological_space}{sheafification} functor 
                $$a: \PreShv(X, \calA) \to \Shv(X, \calA) $$
        exists. A sequence 
        $$0 \to \calF \to \calG \to \calH \to 0$$
        is \CrefAndHyperrefIfExist{definition:exact_sequence_of_morphisms_in_an_abelian_category}{exact} if and only if the induced sequences 
        $$0 \to \calF_x \to \calG_x \to \calH_x \to 0$$
        on stalks are exact for all $x \in X$. 
    \end{enumerate}
\end{proposition}
