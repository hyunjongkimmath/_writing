\begin{theorem}[{\cite[3.4. Theorem, 3.8. Proposition]{spaltenstein_ruc}}] \label{theorem:AB5_AB5_star_categories_with_left_right_resolutions_for_bounded_complexes_by_a_class_has_resolutions_for_all_complexes_by_the_class}
    \begin{enumerate}
        \item Let $\calA$ be an \CrefAndHyperrefIfExist{definition:grothendiecks_additional_axioms_for_abelian_categories}{AB5} \CrefAndHyperrefIfExist{definition:abelian_category}{abelian category}. Suppose that $\mathfrak{P}$ is a class of objects in $\mathbf{Ch}(\calA)$ satisfying the following:

        \begin{center}
            Every \CrefAndHyperrefIfExist{definition:bounded_complexes_on_an_additive_category_and_homologically_bounded_objects_on_an_abelian_category}{(cohomologically) bounded above complex} $A^\bullet \in \mathbf{Ch}(\calA)$ has a left \CrefAndHyperref{definition:left_right_resolution_of_a_chain_complex_in_an_abelian_category}{$\mathfrak{P}$-resolution}.
        \end{center}

        Then every complex in $\mathbf{Ch}(\calA)$ has a left $\mathfrak{P}$-resolution. 

        \item Dually, let $\calA$ be an \CrefAndHyperrefIfExist{definition:grothendiecks_additional_axioms_for_abelian_categories}{AB$5^*$} \CrefAndHyperrefIfExist{definition:abelian_category}{abelian category}. Suppose that $\mathfrak{I}$ is a class of objects in $\mathbf{Ch}(\calA)$ satisfying the following:

        \begin{center}
            Every \CrefAndHyperrefIfExist{definition:bounded_complexes_on_an_additive_category_and_homologically_bounded_objects_on_an_abelian_category}{(cohomologically) bounded below complex} $A^\bullet \in \mathbf{Ch}(\calA)$ has a left \CrefAndHyperref{definition:left_right_resolution_of_a_chain_complex_in_an_abelian_category}{$\mathfrak{I}$-resolution}.
        \end{center}

        Then every complex in $\mathbf{Ch}(\calA)$ has a right $\mathfrak{I}$-resolution. 
    \end{enumerate}
\end{theorem}
