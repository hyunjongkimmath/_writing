\begin{theorem}[Universal coefficient theorem for cohomology, see e.g. {\cite[Theorem 3.6.5, Exercise 3.6.2]{weibel}}]\label{theorem:universal_coefficient_theorem_for_cohomology_for_modules_over_a_ring}

Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{(not necessarily commutative) ring}. Let $P$ be a \CrefAndHyperrefIfExist{definition:chain_complex_of_objects_in_a_preadditive_category}{chain complex} of \CrefAndHyperrefIfExist{definition:projective_module}{projective right $R$ modules} such that each \CrefAndHyperrefIfExist{definition:submodule_of_a_module_over_a_ring}{submodule} $d\left(P_{n}\right)$ of $P_{n-1}$ is also projective. Then for every $n$ and every \CrefAndHyperrefIfExist{definition:module_of_a_ring}{right $R$-module} $M$, there is an \CrefAndHyperrefIfExist{definition:short_exact_sequence_in_an_abelian_category}{exact sequence}

$$ 0 \rightarrow \operatorname{Ext}_{R}^{1}\left(H_{n-1}(P), M\right) \rightarrow H^{n}\left(\operatorname{Hom}_{R}(P, M)\right) \rightarrow \operatorname{Hom}_{R}\left(H_{n}(P), M\right) \rightarrow 0 $$
\CrefIfExists{definition:ext_functors_of_bimodules_of_rings} \CrefIfExists{definition:homology_and_cohomology_objects_for_a_chain_complex_in_an_abelian_category}
When $R$ is a \CrefAndHyperrefIfExist{definition:hereditary_ring}{hereditary ring} (e.g. $R$ is a \CrefAndHyperrefIfExist{definition:principal_ideal_ring}{PID}), this exact sequence \CrefAndHyperrefIfExist{definition:split_exact_sequence_in_an_additive_category}{splits} non-canonically.

\end{theorem}
