\begin{theorem}[Smooth base change for integral adic coefficients, cf. {\cite[Theorem 7.7.2]{fu_ect}}] \label{theorem:smooth_base_change_theorem_for_derived_category_of_adic_sheaves_with_integral_coefficients_on_schemes}
    Let
    \begin{center}
        \begin{tikzcd}
            X' \ar[r, "f'"] \ar[d,"g'"] & Y' \ar[d, "g"] \\
            X \ar[r, "f"] & Y
        \end{tikzcd}
    \end{center}
    be a Cartesian diagram of schemes such that $f$ is \CrefAndHyperrefIfExist{definition:smooth_morphism_of_schemes}{smooth} and $g$ is \CrefAndHyperrefIfExist{definition:quasi_compact_morphism_of_schemes}{quasi-compact} and \CrefAndHyperrefIfExist{definition:quasi_separated_morphism_of_schemes}{quasi-separated}.

    Let $\ell$ be a prime invertible on $Y$. Let $R$ be the \CrefAndHyperrefIfExist{definition:integral_element_over_a_ring}{integral closure} of $\bbZ_\ell$ in a finite extension $E$ of $\Qell$, and let $\lambda$ be a uniformizer of $R$.

    For any $L = (L_n) \in D_c^b(Y', R)$ (\Cref{definition:derived_category_of_bounded_constructible_complexes_of_adic_sheaves_with_integral_coefficients_on_a_noetherian_scheme}), there is a natural isomorphism
    $$f^* Rg_* L \cong Rg_*' {f'}^* L$$
    in $D_c^b(X, R)$ (\Cref{definition:six_functor_formalism_on_derived_category_of_bdd_constr_complexes_of_adic_sheaves_with_integral_coefficients_on_a_noetherian_scheme}).

    The isomorphism follows levelwise from \Cref{theorem:smooth_base_change_theorem_for_derived_category_of_Z_nZ_sheaves_on_schemes}, since $f^*$ and $Rg_*$ act levelwise on inverse systems.
\end{theorem}
