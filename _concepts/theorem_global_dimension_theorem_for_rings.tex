\begin{theorem} \label{theorem:global_dimension_theorem_for_rings}
(Global Dimension Theorem) For any \CrefAndHyperrefIfExist{definition:ring}{not necessarily commutative ring} $R$, the following values are equal:
\begin{enumerate}
    \item $\operatorname{lgldim}(R)$, i.e. $\sup \{ \operatorname{pd}_R(M) \mid M \in {}_R\mathsf{Mod} \}$. (\Cref{definition:left_right_global_dimension_of_a_ring}) (\Cref{definition:left_right_projective_injective_flat_dimension_of_a_module_over_a_ring})
    \item $\sup \{ \operatorname{id}_R(M) \mid M \in {}_R\mathsf{Mod} \}$.
    \item $\sup \{ \operatorname{pd}_R(R/I) \mid I \text{ is a left ideal of } R \}$.
\end{enumerate}
If $R$ is \CrefAndHyperrefIfExist{definition:locally_noetherian_and_noetherian_scheme}{Noetherian}, then $\operatorname{lgldim}(R) = \operatorname{rgldim}(R)$. In general, these may differ.
\end{theorem}
