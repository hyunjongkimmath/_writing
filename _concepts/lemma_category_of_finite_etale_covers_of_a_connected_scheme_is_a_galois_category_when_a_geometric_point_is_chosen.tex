\begin{lemma}[see {\cite[0BNB]{stacks-project}}] \label{lemma:category_of_finite_etale_covers_of_a_connected_scheme_is_a_galois_category_when_a_geometric_point_is_chosen}
    Let $X$ be a \CrefAndHyperrefIfExist{definition:disconnected_and_connected_topological_spaces}{connected} \CrefAndHyperrefIfExist{definition:scheme}{scheme} and let $\overline{x}$ be a \CrefAndHyperrefIfExist{definition:geometric_point_of_a_scheme}{geometric point} of $X$. 
    The \CrefAndHyperref{definition:fiber_functor_of_category_of_finite_etale_covers_of_a_scheme_given_by_a_geometric_point}{fiber functor} 
    $$F_{\bar{x}}: \FEt_X \to \Sets, \quad Y \mapsto |Y_{\overline{x}}|$$
    sending $Y$ to the underlying set of the geometric fiber $Y_{\overline{x}} = Y \times_X \overline{x}$ makes $\FEt_X$ a \CrefAndHyperrefIfExist{definition:fiber_functor_for_a_functor_from_a_galois_category_to_the_category_of_finite_sets}{Galois category}.
\end{lemma}
