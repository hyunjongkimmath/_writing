\begin{theorem}[See {\cite[Theorem II.3.1]{kiehl_weissauer_wcps}}] \label{theorem:core_of_a_triangulated_category_with_a_t_structure_is_abelian_category}
    Let $D$ be a \CrefAndHyperrefIfExist{definition:triangulated_category}{triangulated category} and let $(\Dleqzero, \Dgeqzero)$ be a \CrefAndHyperrefIfExist{definition:t_structure_on_a_triangulated_category}{$t$-structure} on $D$.
    The \CrefAndHyperrefIfExist{definition:t_structure_on_a_triangulated_category}{core $\Core(D)$} is an \CrefAndHyperrefIfExist{definition:abelian_category}{abelian category}. A sequence 
    $$0 \to X \xrightarrow{u} Y \xrightarrow{v} Z \to 0$$
    is \CrefAndHyperrefIfExist{definition:exact_sequence_of_morphisms_in_an_abelian_category}{exact} if and only if there exists a \CrefAndHyperrefIfExist{definition:triangulated_category}{distinguished triangle} $(X,Y,Z,u,v,w)$ in $D$. 
\end{theorem}
