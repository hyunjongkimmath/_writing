\begin{proposition} \label{proposition:integral_of_reciprocal_is_natural_logarithm}
For $0 < a \leq b$, the \CrefAndHyperrefIfExist{definition:lower_upper_riemann_sum_integral_of_a_bounded_function_on_a_closed_interval_to_R_and_riemann_integrable_function}{Riemann integral} of $\frac{1}{x}$ is given by:
$$\int_a^b \frac{1}{x} \, dx = \ln(b) - \ln(a)$$
\CrefIfExists{definition:real_natural_logarithm_function_as_inverse_of_real_exponential_function}
In particular, $\ln(x) = \int_1^x \frac{1}{t} \, dt$ for all $x > 0$.
\end{proposition}
\begin{proof}
    The function $x \mapsto 1/x$ is continuous on $(0, \infty)$ (as the reciprocal of a nonzero continuous function)\CrefIfExists{proposition_reciprocal_function_on_real_numbers_is_continuous.tex}. On any closed interval $[a, b] \subset (0, \infty)$, continuity implies Riemann integrability\CrefIfExists{proposition:continuous_functions_are_riemann_integrable_on_closed_intervals}.

    The natural logarithm $\ln : (0, \infty) \to \mathbb{R}$ is differentiable with derivative
    $$\frac{d}{dx} \ln(x) = \frac{1}{x},$$
    \CrefIfExists{proposition:derivative_of_natural_logarithm_is_reciprocal}. Since $\ln$ is an antiderivative of $1/x$, we apply the \CrefAndHyperrefIfExist{theorem:fundamental_theorem_of_calculus_part_two_evaluation_by_antiderivative}{Fundamental Theorem of Calculus, Part II}:
    $$\int_a^b \frac{1}{x}\,dx = \ln(b) - \ln(a).$$

    Setting $a = 1$ and using $\ln(1) = 0$, we obtain the particular case:
    $$\ln(x) = \int_1^x \frac{1}{t}\,dt.$$
\end{proof}
