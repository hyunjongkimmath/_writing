\begin{corollary} \label{corollary:real_exponential_function_is_infinitely_differentiable}
    The \CrefAndHyperrefIfExist{definition:real_exponential_function}{exponential function} $\exp : \mathbb{R} \to \mathbb{R}$ is \CrefAndHyperrefIfExist{definition:derivative_of_a_real_valued_function_on_a_real_interval}{smooth} at every point $x \in \mathbb{R}$.
\end{corollary}
\begin{proof}
    Recall that 
    $$\frac{d}{dx} e^x = e^x.$$
    \CrefIfExists{proposition:derivative_of_exponential_function_is_itself} Therefore, $e^x$ is infinitely differentiable at all $x \in \bbR$.
\end{proof}
