\begin{proposition} \label{proposition:projective_module_over_local_ring_or_pid_are_free}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:local_ring}{local ring} or a \CrefAndHyperrefIfExist{definition:principal_ideal_ring}{PID}. 
    All \CrefAndHyperrefIfExist{definition:projective_bimodule_over_rings}{projective} \CrefAndHyperrefIfExist{definition:module_of_a_ring}{modules} over $R$ are \CrefAndHyperrefIfExist{definition:free_bimodule_over_rings}{free}.
\end{proposition}
