\begin{lemma}[Mapping lemma, see {\cite[Mapping Lemma 5.2.4.]{weibel}}] \label{lemma:mapping_lemma_for_spectral_sequences_in_abelian_categories}
Let $\mathcal{A}$ be an abelian category and $r_0$ an integer.
\begin{enumerate}
    \item Let $f: E \to E'$ be a \CrefAndHyperrefIfExist{definition:morphism_of_spectral_sequences_in_an_abelian_category}{morphism of homological spectral sequences} in $\mathcal{A}$ starting at page $r_0$. If there exists some $r \ge r_0$ such that the morphism $f^r_{p,q}: E^r_{p,q} \to E'^r_{p,q}$ is an isomorphism for all $p, q \in \mathbb{Z}$, then $f^s_{p,q}: E^s_{p,q} \to E'^s_{p,q}$ is an isomorphism for all $s \ge r$ and all $p, q \in \mathbb{Z}$.
    \item Let $f: E \to E'$ be a morphism of \CrefAndHyperrefIfExist{definition:morphism_of_spectral_sequences_in_an_abelian_category}{cohomological spectral sequences} in $\mathcal{A}$ starting at page $r_0$. If there exists some $r \ge r_0$ such that the morphism $f_r^{p,q}: E_r^{p,q} \to E_r'^{p,q}$ is an isomorphism for all $p, q \in \mathbb{Z}$, then $f_s^{p,q}: E_s^{p,q} \to E_s'^{p,q}$ is an isomorphism for all $s \ge r$ and all $p, q \in \mathbb{Z}$.
\end{enumerate}
\textit{Remark:} In the limit, if the spectral sequences are \CrefAndHyperrefIfExist{definition:converges_to_a_graded_object_for_a_spectral_sequence_in_an_abelian_category}{convergent}, this implies an isomorphism on the abutments.
\end{lemma}
