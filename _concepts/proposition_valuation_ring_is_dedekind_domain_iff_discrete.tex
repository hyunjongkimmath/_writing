\begin{proposition} \label{proposition:valuation_ring_is_dedekind_domain_iff_discrete}
    Let $(K,v)$ be a \CrefAndHyperrefIfExist{definition:valuation_on_a_field}{valuation field} and let $R$ be its \CrefAndHyperrefIfExist{definition:valuation_ring_of_a_nonarchimedean_valued_field}{valuation ring}. 
    The \CrefAndHyperrefIfExist{definition:ring}{ring} $R$ is a \CrefAndHyperrefIfExist{definition:dedekind_domain}{Dedekind domain} if and only if if the \CrefAndHyperrefIfExist{definition:valuation_on_a_field}{valuation} $v$ is \CrefAndHyperrefIfExist{definition:discrete_valuation_on_a_field}{discrete}.
\end{proposition}
