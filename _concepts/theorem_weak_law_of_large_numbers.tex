\begin{theorem}[Weak law of large numbers (Khintchine)] \label{theorem:weak_law_of_large_numbers}
Let $(\Omega, \mathcal{F}, P)$ be a \CrefAndHyperref{definition:probability_space}{probability space}. Let $\{X_n\}_{n=1}^\infty$ be a sequence of \CrefAndHyperref{definition:independence_of_random_variables_on_a_probability_space}{independent and identically distributed} real-valued random variables on $(\Omega, \mathcal{F}, P)$, with $\mathbb{E}[X_1] = \mu$ and \CrefAndHyperref{definition:variance_and_covariance}{$\mathrm{Var}(X_1) = \sigma^2 < \infty$}.

Let $S_n = X_1 + \cdots + X_n$ and let $\bar{X}_n = S_n / n$. Then for every $\varepsilon > 0$,
$$\lim_{n \to \infty} P\left(\left|\bar{X}_n - \mu\right| \geq \varepsilon\right) = 0.$$
That is, $\bar{X}_n$ \CrefAndHyperrefIfExist{definition:convergence_in_measure_for_a_sequence_of_measurable_functions_on_a_measure_space_into_an_extended_metric_space_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity}{converges to $\mu$ in probability}. This follows from \CrefAndHyperref{proposition:chebyshev_inequality_for_random_variables_with_finite_variance}{Chebyshev's inequality} applied to $\bar{X}_n$, since $\mathrm{Var}(\bar{X}_n) = \sigma^2 / n \to 0$.
\end{theorem}
