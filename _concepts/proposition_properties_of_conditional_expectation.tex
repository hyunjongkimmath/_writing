\begin{proposition}[Properties of conditional expectation] \label{proposition:properties_of_conditional_expectation}
Let $(\Omega, \mathcal{F}, P)$ be a \CrefAndHyperrefIfExist{definition:probability_measure_space_for_a_measure_on_a_measurable_space}{probability measure space}, let $\mathcal{G} \subseteq \mathcal{F}$ be a \CrefAndHyperrefIfExist{definition:sigma_algebra_on_a_set_and_measure_space}{sub-$\sigma$-algebra}, and let $X$ be an integrable random variable on $(\Omega, \mathcal{F}, P)$. Then the \CrefAndHyperrefIfExist{definition:conditional_expectation}{conditional expectation} $\mathbb{E}[X \mid \mathcal{G}]$ satisfies the following properties:

\begin{enumerate}
    \item \textbf{(Linearity)} For any $a, b \in \mathbb{R}$ and integrable random variables $Y$,
    $$\mathbb{E}[aX + bY \mid \mathcal{G}] = a\,\mathbb{E}[X \mid \mathcal{G}] + b\,\mathbb{E}[Y \mid \mathcal{G}] \quad \text{almost surely.}$$

    \item \textbf{(Tower property)} If $\mathcal{H} \subseteq \mathcal{G}$ is a sub-$\sigma$-algebra, then
    $$\mathbb{E}[\mathbb{E}[X \mid \mathcal{G}] \mid \mathcal{H}] = \mathbb{E}[X \mid \mathcal{H}] \quad \text{almost surely.}$$

    \item \textbf{(Pulling out what is known)} If $Z$ is $\mathcal{G}$-measurable and $ZX$ is integrable, then
    $$\mathbb{E}[ZX \mid \mathcal{G}] = Z\,\mathbb{E}[X \mid \mathcal{G}] \quad \text{almost surely.}$$

    \item \textbf{(Jensen's inequality)} If $\varphi: \mathbb{R} \to \mathbb{R}$ is a convex function such that $\varphi(X)$ is integrable, then
    $$\varphi(\mathbb{E}[X \mid \mathcal{G}]) \leq \mathbb{E}[\varphi(X) \mid \mathcal{G}] \quad \text{almost surely.}$$
\end{enumerate}
\end{proposition}
