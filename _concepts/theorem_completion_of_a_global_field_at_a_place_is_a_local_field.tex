\begin{theorem}  \label{theorem:completion_of_a_global_field_at_a_place_is_a_local_field}
    Let $L$ be a \CrefAndHyperrefIfExist{definition:global_field}{global field} and let $v$ be any \CrefAndHyperrefIfExist{definition:place_of_a_global_field}{place}. The \CrefAndHyperrefIfExist{definition:completion_of_a_global_field_at_a_place}{completion $L_v$} of $L$ with respect to $v$ is a \CrefAndHyperrefIfExist{definition:local_field}{local field}.
\end{theorem}
