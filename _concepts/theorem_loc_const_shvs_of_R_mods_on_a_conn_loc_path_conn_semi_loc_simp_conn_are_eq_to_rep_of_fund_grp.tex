\begin{theorem}  \label{theorem:loc_const_shvs_of_R_mods_on_a_conn_loc_path_conn_semi_loc_simp_conn_are_eq_to_rep_of_fund_grp}
    Let $X$ Be a \CrefAndHyperrefIfExist{definition:disconnected_and_connected_topological_spaces}{connected}, \CrefAndHyperrefIfExist{definition:locally_path_connected_topological_space}{locally path connected}, and \CrefAndHyperrefIfExist{definition:semi_locally_simply_connected_space}{semi-locally simply connected}. Let $x \in X$. Let $R$ be a ring.
    There is an \CrefAndHyperrefIfExist{definition:equivalence_of_categories}{equivalence} 
    $$\mathbf{LCS}(X,R) \simeq \mathbf{Rep}_R(\pi_1(X,x))$$

    of categories where $\mathbf{LCS}(X,R)$ is the category of \CrefAndHyperrefIfExist{definition:locally_constant_sheaf_on_a_site_with_sheafification}{locally constant sheaves} of $R$-modules, and $\mathbf{Rep}_R(\pi_1(X,x))$ is the category of \CrefAndHyperrefIfExist{definition:representation_of_a_group_on_a_module_over_a_not_necessarily_commutative_ring}{$R$-linear representations} of the fundamental group $\pi_1(X,x)$.

    \CrefAndHyperrefIfExist{definition:equivalence_of_categories}{Natural inverse equivalences} are given as follows: given a locally constant sheaf $\calF$, the fiber at the basepoint $x$ is the module $M = \calF_x$\CrefIfExists{definition:stalk_of_a_presheaf_on_a_topological_space_at_a_point}, and for any \CrefAndHyperrefIfExist{definition:path_and_loop_in_a_topological_space}{loop} $\gamma$ in $X$ based at $x$, ``lifting'' sections along the loop defines an automorphism of $M$, which in turn defines a representation $\rho: \pi_1(X,x) \to \Aut_R(M)$. Given a representation  $\rho: \pi_1(X,x) \to \Aut_R(M)$, and letting $\widetilde{X}$ be the \CrefAndHyperrefIfExist{definition:universal_cover_of_a_topological_space}{universal cover} of $X$, form the associated bundle of the locally constant sheaf as the quotient
    $$\calF = (\widetilde{X} \times M) / \pi_1(X,x)$$
    where $\pi_1$ acts on $\widetilde{X}$ via deck transformations and on $M$ via the representation $\rho$.
\end{theorem}
