\begin{proposition} \label{proposition:any_fiber_bundle_of_topological_spaces_is_a_serre_fibration}
Let $\pi: E \to B$ be a \CrefAndHyperrefIfExist{definition:fiber_bundle_of_topological_spaces}{fiber bundle}. Then $\pi$ is a \CrefAndHyperrefIfExist{definition:serre_fibration_of_topological_spaces}{Serre fibration}.
\end{proposition}
