\begin{lemma}[Pascal's identity for binomial coefficients] \label{lemma:pascal_identity_for_binomial_coefficients}
    For any nonnegative integer $n \in \mathbb{Z}_{\geq 0}$ and any integer $j \in \mathbb{Z}$, we have
    $$\binom{n}{j-1} + \binom{n}{j} = \binom{n+1}{j}$$
    where $\binom{n}{k}$ is the \CrefAndHyperrefIfExist{definition:combinatorial_choose_function_binomial_coefficient}{binomial coefficient} (with the convention $\binom{n}{k} = 0$ for $k < 0$ or $k > n$).
\end{lemma}

\begin{proof}
    If $j \leq 0$, then $\binom{n}{j-1} = \binom{n}{j} = 0$ and $\binom{n+1}{j} = 0$, so the identity holds. If $j = 1$, then $\binom{n}{0} + \binom{n}{1} = 1 + n = n+1 = \binom{n+1}{1}$. If $j = n+1$, then $\binom{n}{n} + \binom{n}{n+1} = 1 + 0 = 1 = \binom{n+1}{n+1}$. If $j \geq n+2$, then all three terms are $0$. For $2 \leq j \leq n$, we compute directly using the factorial definition:
    $$\begin{aligned}
        \binom{n}{j-1} + \binom{n}{j} &= \frac{n!}{(j-1)!(n-j+1)!} + \frac{n!}{j!(n-j)!} \\
        &= \frac{n! \cdot j}{j!(n-j+1)!} + \frac{n! \cdot (n-j+1)}{j!(n-j+1)!} \\
        &= \frac{n!(j + n - j + 1)}{j!(n-j+1)!} \\
        &= \frac{n!(n+1)}{j!(n+1-j)!} \\
        &= \frac{(n+1)!}{j!(n+1-j)!} \\
        &= \binom{n+1}{j}.
    \end{aligned}$$
    \CrefIfExists{definition:factorial_of_a_nonnegative_integer}
    All cases are covered, so the identity holds for all $j \in \mathbb{Z}$.
\end{proof}
