\begin{theorem}[Chevalley's extension theorem] \label{theorem:chevalleys_extension_theorem_for_valuations_on_extension_of_fields}
    Let $L/K$ be an \CrefAndHyperrefIfExist{definition:extension_of_a_field}{extension of fields}. Let $v$ be a \CrefAndHyperrefIfExist{definition:valuation_on_a_field}{valuation} on $K$. There exists at least one valuation of $L$ that \CrefAndHyperrefIfExist{definition:extension_of_a_valuation_of_a_field_to_an_extension_field}{extends} $v$. 
\end{theorem}
