\begin{lemma} \label{lemma:relative_cellular_chain_group_of_CW_pair_with_coefficients_in_an_abelian_group_is_isomorphic_to_a_singular_homology_group}
Let $(X, A)$ be a \CrefAndHyperrefIfExist{definition:CW_pair}{CW pair} and $G$ an abelian group. There is a canonical isomorphism:
$$C_n^{CW}(X, A; G) \cong H_n(X^n \cup A, X^{n-1} \cup A; G)$$
\Cref{definition:relative_absolute_group_of_cellular_n_chains_of_a_CW_pair_complex} where $H_n$ denotes \CrefAndHyperrefIfExist{definition:singular_homology_and_relative_singular_homology_of_topological_space_with_coefficients_in_an_abelian_group}{relative singular homology}. Furthermore, $H_k(X^n \cup A, X^{n-1} \cup A; G) = 0$ for $k \neq n$.
\end{lemma}
