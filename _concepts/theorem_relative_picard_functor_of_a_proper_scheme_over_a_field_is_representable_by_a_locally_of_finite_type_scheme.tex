\begin{theorem}[\cite{murre_ocfcpfcag}, \cite{oort_ssp}, see {\cite[Section 8.2, Theorem 3]{bosch_lutkebohmert_raynaud_nm}}] \label{theorem:relative_picard_functor_of_a_proper_scheme_over_a_field_is_representable_by_a_locally_of_finite_type_scheme}
    Let $X$ be a \CrefAndHyperrefIfExist{definition:proper_morphism_of_schemes}{proper} \CrefAndHyperrefIfExist{definition:scheme_over_a_scheme}{scheme over} a \CrefAndHyperrefIfExist{definition:field}{field} $k$. 
    The \CrefAndHyperrefIfExist{definition:big_fppf_site_of_a_scheme}{fppf}-\CrefAndHyperrefIfExist{definition:sheaf_on_a_site}{sheaf} 
    $$\Pic_{X/k}: (\Sch/k)_{\mathrm{fppf}}^\circ \to \Sets$$
    (\Cref{definition:relative_picard_functor_of_a_scheme_over_a_scheme}) is \CrefAndHyperrefIfExist{definition:representable_functor_on_a_locally_small_category}{representable by} by a \CrefAndHyperrefIfExist{definition:finite_type_morphism_of_schemes}{locally of finite type} scheme over $k$.
\end{theorem}
