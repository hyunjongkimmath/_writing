\begin{proposition} \label{proposition:projective_modules_are_flat}
    Let $R,S$ be \CrefAndHyperrefIfExist{definition:ring}{not necessarily commutative rings}.
    If $P$ is a \CrefAndHyperrefIfExist{definition:projective_bimodule_over_rings}{projective} \CrefAndHyperrefIfExist{definition:module_of_a_ring}{$R$-$S$-bimodule}, then $P$ is both \CrefAndHyperrefIfExist{definition:flat_left_right_module_over_a_ring}{flat} with respect to the \CrefAndHyperrefIfExist{definition:module_of_a_ring}{left $R$-module} structure and flat with respect to the right $S$-\CrefAndHyperrefIfExist{definition:module_of_a_ring}{module} structure. In other words, the \CrefAndHyperrefIfExist{definition:functor_between_categories}{functors}
    $$-\otimes_R P : \mathrm{Mod}_R \to \mathbf{Mod}_S$$
    and 
    $$P \otimes_S - : {}_S \mathrm{Mod} \to {}_R \mathbf{Mod}$$
    \CrefIfExists{definition:tensor_product_of_bimodules_of_rings}
    are \CrefAndHyperrefIfExist{definition:acyclic_complex_of_objects_in_an_abelian_category}{exact}.
\end{proposition}
