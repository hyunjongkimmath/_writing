\begin{theorem} \label{theorem:isomorphism_theorem_between_cellular_homology_and_singular_homology_of_CW_pair_with_coefficients_in_an_abelian_group}
For any \CrefAndHyperrefIfExist{definition:CW_pair}{CW pair} $(X, A)$ and any abelian group $G$, there exists a natural isomorphism:
$$H_n^{CW}(X, A; G) \cong H_n(X, A; G)$$
(\Cref{definition:relative_absolute_cellular_homology_groups_of_CW_pair_complex}) (\Cref{definition:singular_homology_and_relative_singular_homology_of_topological_space_with_coefficients_in_an_abelian_group})
for all $n \ge 0$.
\end{theorem}
