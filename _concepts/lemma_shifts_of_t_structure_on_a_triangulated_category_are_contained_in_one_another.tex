\begin{lemma} \label{lemma:shifts_of_t_structure_on_a_triangulated_category_are_contained_in_one_another}
    Let $D$ be a \CrefAndHyperrefIfExist{definition:t_structure_on_a_triangulated_category}{triangulated category} and let $(\Dleqzero, \Dgeqzero)$ be a \CrefAndHyperrefIfExist{definition:t_structure_on_a_triangulated_category}{$t$-structure} on $D$.
    If $m,n \in \mathbb{Z}$ satisfy $m \leq n$, then 
    \begin{align*}
        \Dleqn[m] &\supset \Dleqn \\
        \Dgeqn[m] &\subset \Dgeqn
    \end{align*}
\end{lemma}
\begin{proof}
    This is clear from the definitions and axioms on $\Dleqn, \Dgeqn$. 
\end{proof}
