\begin{theorem}[Freudenthal suspension, see e.g. {\cite[Chapter 11]{may}}] \label{theorem:freudenthal_suspension_theorem}
If $X$ is \CrefAndHyperrefIfExist{definition:nondegenerately_based_well_pointed_topological_space}{nondegenerately based} and \CrefAndHyperrefIfExist{definition:n_connected_topological_space}{$(n-1)$-connected} where $n \geq 1$, then the \CrefAndHyperrefIfExist{definition:suspension_homomorphism_of_a_homotopy_group_of_a_pointed_topological_space}{suspension homomorphism} $\Sigma: \pi_q(X) \to \pi_{q+1}(\Sigma X)$ is a bijection if $q < 2n-1$ and a surjection if $q = 2n-1$. 
\end{theorem}
