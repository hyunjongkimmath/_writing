\begin{proposition} \label{proposition:connection_between_dedekind_prime_decomposition_and_valuation_extensions}

    Let $R$ be a \CrefAndHyperref{definition:dedekind_domain}{Dedekind domain} with \CrefAndHyperref{definition:field_of_fractions_of_an_integral_domain}{fraction field $K$}, and let $L$ be a \CrefAndHyperref{definition:finite_field_extension}{finite extension} of $K$. Let $\calO_L$ be the \CrefAndHyperref{definition:ring_of_integers_of_an_extension_of_fraction_field_of_a_dedekind_domain}{ring of integers of $L$ over $R$}. Let $\mathfrak{p}$ be a \CrefAndHyperref{definition:prime_of_a_dedekind_domain}{prime of $R$} with associated $\mathfrak{p}$-adic \CrefAndHyperref{definition:p_adic_valuation_of_a_prime_of_a_dedekind_domain}{valuation} $v_{\mathfrak{p}}$ on $K$.

    For each prime $\mathfrak{P}$ of $\calO_L$ appearing in the factorization $\mathfrak{p}\calO_L = \prod \mathfrak{P}^{e(\mathfrak{P}/\mathfrak{p})}$, let $v_{\mathfrak{P}}$ be the $\mathfrak{P}$-adic valuation on $L$. Then $v_{\mathfrak{P}}$ is an \CrefAndHyperref{definition:extension_of_a_valuation_of_a_field_to_an_extension_field}{extension of $v_{\mathfrak{p}}$ to $L$}, and the invariants coincide:
    $$e(\mathfrak{P}/\mathfrak{p}) = e(v_{\mathfrak{P}}/v_{\mathfrak{p}}), \qquad f(\mathfrak{P}/\mathfrak{p}) = f(v_{\mathfrak{P}}/v_{\mathfrak{p}}),$$
    where $e(v_{\mathfrak{P}}/v_{\mathfrak{p}})$ and $f(v_{\mathfrak{P}}/v_{\mathfrak{p}})$ are the \CrefAndHyperref{definition:ramification_index_of_extension_of_valuation_fields}{ramification index} and \CrefAndHyperref{definition:inertial_degree_of_extension_of_valuation_fields}{inertial degree} of the extension of discrete valuations.

    Conversely, every extension of $v_{\mathfrak{p}}$ to $L$ arises as $v_{\mathfrak{P}}$ for a unique prime $\mathfrak{P}$ of $\calO_L$ lying above $\mathfrak{p}$.
\end{proposition}
