\begin{proposition} \label{proposition:identity_function_is_continuous_between_topological_spaces}
    Let $X$ be a \CrefAndHyperrefIfExist{definition:topological_space}{topological space}. The \CrefAndHyperrefIfExist{definition:identity_function_on_a_set}{identity function} $\mathrm{id}_X : X \to X$ is \CrefAndHyperrefIfExist{definition:continuous_map_of_topological_spaces}{continuous}.
\end{proposition}
\begin{proof}
    Let $V \subseteq X$ be an open set. Then $\mathrm{id}_X^{-1}(V) = V$, which is open in $X$ by hypothesis. Hence $\mathrm{id}_X$ is continuous.
\end{proof}
