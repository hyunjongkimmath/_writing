\begin{proposition}[{\cite[Proposition 1.3.3(ii)]{beilinson_berstein_deligne_atlesI}}] \label{proposition:distinguished_triangles_for_t_structures_on_triangulated_categories_given_by_truncation_are_unique}
    Let $D$ be a \CrefAndHyperrefIfExist{definition:triangulated_category}{triangulated category} and let $(\Dleqzero, \Dgeqzero)$ be a \CrefAndHyperrefIfExist{definition:t_structure_on_a_triangulated_category}{$t$-structure} on $D$.

    For any object $E$ in $D$, there exists a unique morphism $d \in \Hom(\tau_{\geq 1} E, \tau_{\leq 0} E[1])$ such that the triangle
    $$\tau_{\leq E} \to E \to \tau_{\geq 1} E \xrightarrow{d} \tau_{\leq 0} E[1]$$
    is \CrefAndHyperrefIfExist{definition:triangulated_category}{distinguished}. Up to unique isomorphism, every distinguished triangle $(A,E,B)$ in $D$ with $A \in D^{\leq 0}$ and $B \in D^{\geq 1}$ is equivalent to such a triangle.

    % $$\tauleqzero(E) \xrightarrow{\ad_{\leq 0}} E \xrightarrow{\ad_{\geq 1}} \taugeqn[1](E) \to \tauleqzero(E)[1].$$
\end{proposition}
