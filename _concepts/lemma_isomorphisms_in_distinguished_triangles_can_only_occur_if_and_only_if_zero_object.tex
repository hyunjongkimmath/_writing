\begin{lemma}[See {\cite[Corollary II.1.6]{kiehl_weissauer_wcps}}] \label{lemma:isomorphisms_in_distinguished_triangles_can_only_occur_if_and_only_if_zero_object}
    Let $D$ be a \CrefAndHyperrefIfExist{definition:triangulated_category}{triangulated category}. Let $(X,Y,Z,u,v,w)$ be a \CrefAndHyperrefIfExist{definition:triangulated_category}{distinguished triangle} of $D$.
    The morphism $u$ is an isomorphism if and only if $Z \cong 0$.
\end{lemma}
