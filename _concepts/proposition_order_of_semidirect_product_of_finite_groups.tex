\begin{proposition} \label{proposition:order_of_semidirect_product_of_finite_groups}
    Let $N$ and $H$ be finite \CrefAndHyperrefIfExist{definition:group}{groups}, and let $\varphi: H \to \operatorname{Aut}(N)$ be a \CrefAndHyperrefIfExist{definition:group_homomorphism}{homomorphism}. The \CrefAndHyperrefIfExist{definition:semidirect_product_of_groups_external_and_internal}{external semidirect product} $N \rtimes_\varphi H$ is a finite group of order
    $$|N \rtimes_\varphi H| = |N| \cdot |H|.$$
\end{proposition}
