\begin{theorem}[Weil restriction over a finite field extension] \label{theorem:weil_restriction_over_finite_field_extension}
    \TODO{quasi-projective scheme}
    Let $L/K$ be a \CrefAndHyperrefIfExist{definition:finite_field_extension}{finite degree field extension} and let $f : \mathrm{Spec}(L) \to \mathrm{Spec}(K)$ be the induced morphism. For any quasi-projective scheme $X'$ over $\mathrm{Spec}(L)$, the \CrefAndHyperrefIfExist{definition:weil_restriction_on_schemes}{Weil restriction} $\mathrm{Res}_{X'/\mathrm{Spec}(L)}$ exists as a quasi-projective scheme over $\mathrm{Spec}(K)$. In particular:
    \begin{itemize}
        \item If $X' = \mathbb{A}^n_L$, then $\mathrm{Res}_{X'/\mathrm{Spec}(L)} \cong \mathbb{A}^{n \cdot [L:K]}_K$.
        \item If $X' = \mathbb{G}_{m,L}$, then $\mathrm{Res}_{\mathbb{G}_{m,L}/\mathrm{Spec}(L)}$ is an \CrefAndHyperrefIfExist{definition:algebraic_torus_over_a_scheme}{algebraic torus} over $K$ of dimension $[L:K]$, called the \hldef{norm torus}.
    \end{itemize}
\end{theorem}
