\begin{proposition}[see {\cite[Chapter 8, after Definition 2 and after Proposition 3]{bosch_lutkebohmert_raynaud_nm}}] \label{proposition:relative_picard_functor_of_a_scheme_over_a_scheme_is_sections_functor_of_first_derived_direct_image_of_sheaf_of_units}
    Let $f: X \to S$ be a \CrefAndHyperrefIfExist{definition:scheme_over_a_scheme}{scheme over a scheme}. 
    The \CrefAndHyperrefIfExist{definition:relative_picard_functor_of_a_scheme_over_a_scheme}{relative Picard functor} 
    $$\Pic_{X/S}: (\Sch/S)_{\mathrm{fppf}}^\circ \to \Sets$$
    \CrefIfExists{definition:big_fppf_site_of_a_scheme}\CrefIfExists{definition:opposite_category_of_a_category}\CrefIfExists{definition:category_of_sets}\CrefIfExists{definition:cartesian_product_of_two_objects_in_a_category_over_an_object} can be constructed as
    $$\Pic_{X/S}(T) = H^0(T, R^1 f_*(\mathbb{G}_m))$$
    \CrefIfExists{definition:sheaf_cohomology_group_of_a_sheaf_of_modules_over_a_sheaf_of_rings_on_a_site}\CrefIfExists{definition:left_right_derived_functors_of_a_right_left_exact_functor_between_abelian_categories_where_source_has_enough_projectives_injectives}\CrefIfExists{definition:direct_image_of_a_sheaf_on_a_site_under_a_continuous_functor_of_sites_or_a_site_morphism}\CrefIfExists{definition:multiplicative_group_sheaf_of_a_ringed_site}. 

    If $f$ is proper, then this formula remains valid when $H^0$ and $R^1 f_*(\mathbb{G}_m)$ are computed via the \CrefAndHyperrefIfExist{definition:big_etale_site_of_a_scheme}{\'etale topology}.
\end{proposition}
