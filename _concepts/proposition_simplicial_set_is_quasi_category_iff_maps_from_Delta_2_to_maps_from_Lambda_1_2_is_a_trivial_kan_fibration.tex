\begin{proposition} \label{proposition:simplicial_set_is_quasi_category_iff_maps_from_Delta_2_to_maps_from_Lambda_1_2_is_a_trivial_kan_fibration}
Let $K$ be a \CrefAndHyperrefIfExist{definition:simplicial_cosimplicial_object_in_a_category}{simplicial set}. The following are equivalent:
\begin{enumerate}
    \item $K$ is a \CrefAndHyperref{definition:infty_category_quasi_category}{quasi-category}.
    \item The map of simplicial sets induced by the inner horn inclusion $\Lambda_1^2 \hookrightarrow \Delta^2$,
    $$\Map(\Delta^2, K) \to \Map(\Lambda_1^2, K),$$
    is a \CrefAndHyperrefIfExist{definition:trivial_Kan_fibration_between_simplicial_sets}{trivial Kan fibration}.
\end{enumerate}
\end{proposition}
