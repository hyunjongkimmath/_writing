\begin{lemma} \label{lemma:hurewicz_fibrations_are_serre_fibrations}
    Any \CrefAndHyperrefIfExist{definition:hurewicz_fibration_of_topological_spaces}{Hurewicz fibration} is a \CrefAndHyperrefIfExist{definition:serre_fibration_of_topological_spaces}{Serre fibration}.
\end{lemma}
\begin{proof}
    This holds trivially from the definitions --- Hurewicz fibrations satisfy \CrefAndHyperrefIfExist{definition:homotopy_lifting_property_for_a_continuous_map_of_topological_spaces_with_respect_to_a_topological_space}{HLP} for all topological spaces whereas Serre fibrations satisfy HLP for all disks $D^n$. 
\end{proof}
