\begin{corollary} \label{corollary:limit_of_reciprocal_function_on_real_numbers_at_accumulation_point}
    Let $\bbR$ be the \CrefAndHyperrefIfExist{definition:field_of_real_numbers}{field of real numbers} equipped with its \CrefAndHyperrefIfExist{definition:standard_absolute_value_on_real_numbers_standard_topology_on_real_numbers}{standard absolute value}, let $\bbR^\times = \bbR \setminus \{0\}$ be equipped with the \CrefAndHyperrefIfExist{definition:subspace_of_a_topological_space}{subspace topology}, and let $\operatorname{rec} : \bbR^\times \to \bbR^\times$ be the reciprocal function $\operatorname{rec}(x) = 1/x$, which is continuous by \Cref{proposition:reciprocal_function_on_real_numbers_is_continuous}. For every $a \in \bbR^\times$,
    $$\lim_{x \to a} \frac{1}{x} = \frac{1}{a}.$$
    \CrefIfExists{definition:limit_of_a_function_between_extended_metric_spaces_valued_in_ordered_commutative_monoid_with_adjoined_infinity_at_an_accumulation_point}
\end{corollary}
\begin{proof}
    The reciprocal function $\operatorname{rec} : \bbR^\times \to \bbR^\times$ is continuous by \Cref{proposition:reciprocal_function_on_real_numbers_is_continuous}. Hence $\operatorname{rec}$ is continuous at every point of $\bbR^\times$ by \Cref{lemma:function_between_topological_spaces_is_continuous_iff_continuous_at_all_points}. To see that each $a \in \bbR^\times$ is an accumulation point of $\bbR^\times$, note that there exists an interval $I \subseteq \bbR^\times$ containing $a$ that is not a singleton: for instance, $I = (a/2, 3a/2)$ if $a > 0$, and $I = (3a/2, a/2)$ if $a < 0$. Since $I \subseteq \bbR^\times$, \Cref{lemma:point_in_subset_of_real_numbers_is_accumulation_point_if_and_only_if_there_exists_a_convergent_sequence_in_the_complement_whose_limit_is_the_point} part~(3) implies $a$ is an accumulation point of $\bbR^\times$. Now $\bbR^\times$ with its restriction of the standard absolute value is an \CrefAndHyperrefIfExist{definition:extended_metric_on_a_set_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity}{extended metric space}, so \Cref{proposition:continuity_at_a_point_in_extended_metric_space_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity_equivalent_to_topological_definition_and_limit_at_accumulation_point} implies
    $$\lim_{x \to a} \frac{1}{x} = \frac{1}{a}.$$
    \CrefIfExists{definition:limit_of_a_function_between_extended_metric_spaces_valued_in_ordered_commutative_monoid_with_adjoined_infinity_at_an_accumulation_point}

    \medskip
    \noindent\textit{Remark.} This is a special case of \Cref{corollary:limit_of_reciprocal_function_on_topological_field_at_accumulation_point}, because $\bbR$ equipped with its standard topology induced by the absolute value is a \CrefAndHyperrefIfExist{definition:topological_field}{topological field}.
\end{proof}
