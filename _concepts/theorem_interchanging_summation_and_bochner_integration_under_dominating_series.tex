\begin{theorem}[Interchanging summation and Bochner integration under a dominating series] \label{theorem:interchanging_summation_and_bochner_integration_under_dominating_series}

    Let $(X, \mathcal{A}, \mu)$ be a \CrefAndHyperrefIfExist{definition:measure_on_a_measurable_space_and_measure_space}{measure space} and let $V$ be a \CrefAndHyperrefIfExist{definition:extended_norm_on_a_vector_space_over_a_field_with_absolute_value}{Banach space} over $\mathbb{R}$ or $\mathbb{C}$\CrefIfExists{definition:field_of_real_numbers}\CrefIfExists{definition:field_of_complex_numbers} with norm $\|\cdot\|$. Let $(f_n)_{n=1}^\infty$ be a sequence of functions $f_n : X \to V$ satisfying:

    \begin{enumerate}
        \item For every $n$, the function $f_n$ is \CrefAndHyperrefIfExist{definition:vector_valued_lebesgue_integral_into_normed_vector_space}{Bochner integrable}.
        \item There exists an \CrefAndHyperrefIfExist{definition:lebesgue_integral_of_measurable_functions_into_the_real_line_extended_by_infinities}{integrable} function $g : X \to \mathbb{R}_{\ge 0}$ such that for $\mu$-almost every $x \in X$,
        $$\sum_{n=1}^\infty \|f_n(x)\| \le g(x).$$
    \end{enumerate}

    Then the series $\sum_{n=1}^\infty f_n(x)$ converges in norm for $\mu$-almost every $x \in X$, the function $x \mapsto \sum_{n=1}^\infty f_n(x)$ is Bochner integrable, and
    $$\int_X \left( \sum_{n=1}^\infty f_n(x) \right) d\mu(x) = \sum_{n=1}^\infty \int_X f_n(x) \, d\mu(x),$$
    where the series on the right \CrefAndHyperrefIfExist{definition:infinite_series_and_convergence_of_infinite_series_in_a_topological_abelian_semigroup}{converges} in $V$.
\end{theorem}
