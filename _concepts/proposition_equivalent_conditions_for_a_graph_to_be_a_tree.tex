Let $G$ be a \CrefAndHyperrefIfExist{definition:simple_graph_directed_graph_multigraph_weighted_graph}{simple graph} with $n = |V(G)|$ vertices. The following statements are equivalent:

\begin{enumerate}
    \item $G$ is a \CrefAndHyperrefIfExist{definition:tree_forest_in_graph_theory}{tree}.
    \item $G$ is \CrefAndHyperrefIfExist{definition:connected_graph_connectivity_of_a_graph}{connected} and has exactly $n-1$ edges.
    \item $G$ contains no \CrefAndHyperrefIfExist{definition:walk_path_trail_cycle_in_a_graph}{cycle} and has exactly $n-1$ edges.
    \item Any two vertices of $G$ are joined by a unique \CrefAndHyperrefIfExist{definition:walk_path_trail_cycle_in_a_graph}{path}.
    \item $G$ contains no cycle, and adding any edge between non-adjacent vertices creates exactly one cycle.
\end{enumerate}
\end{proposition}
