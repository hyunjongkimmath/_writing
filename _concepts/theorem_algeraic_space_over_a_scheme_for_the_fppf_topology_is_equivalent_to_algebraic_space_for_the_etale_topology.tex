\begin{theorem}{cf. {\cite[Tag 076M]{stacks-project}}} \label{theorem:algeraic_space_over_a_scheme_for_the_fppf_topology_is_equivalent_to_algebraic_space_for_the_etale_topology}
    Let $S$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme}. 
    Every \CrefAndHyperrefIfExist{definition:algebraic_space_on_a_site_on_Sch_S_that_preserves_under_base_change_and_has_tau_local_surjectivity_and_etaleness}{algebraic space} for the \CrefAndHyperrefIfExist{definition:big_etale_site_of_a_scheme}{(big) \'etale topology} on \CrefAndHyperrefIfExist{definition:scheme_over_a_scheme}{$\mathit{Sch}/S$} is an algebraic space for the \CrefAndHyperrefIfExist{definition:big_fppf_site_of_a_scheme}{(big) fppf topology} on $\mathit{Sch}/S$.
\end{theorem}
