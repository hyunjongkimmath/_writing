\begin{theorem}[Classical Riemann--Roch] \label{theorem:classical_riemann_roch_for_a_curve_over_a_field}

    Let $k$ be a \CrefAndHyperref{definition:algebraically_closed_field}{algebraically closed field}, and let $C$ be a \CrefAndHyperref{definition:nice_curve_over_a_field}{nice curve over} $k$ of \CrefAndHyperref{definition:geometric_genus_of_a_smooth_projective_variety_over_a_field}{genus} $g$.

    Let $\mathcal{L}$ be a \CrefAndHyperref{definition:free_and_locally_free_modules_on_a_ringed_site}{line bundle} on $C$, and let $\deg(\mathcal{L})$ denote the degree of $\mathcal{L}$. The Euler characteristic of $\mathcal{L}$ is
    $$\chi(\mathcal{L}) := \dim_k H^0(C, \mathcal{L}) - \dim_k H^1(C, \mathcal{L}),$$
    where $H^i(C, \mathcal{L})$ are the \CrefAndHyperref{definition:sheaf_cohomology_group_of_a_sheaf_of_modules_over_a_sheaf_of_rings_on_a_site}{sheaf cohomology groups}. Then
    $$\chi(\mathcal{L}) = \deg(\mathcal{L}) + 1 - g.$$
\end{theorem}
