\begin{lemma} \label{lemma:zeroth_left_right_derived_object_of_a_right_left_exact_functor_is_naturally_isomorphic_to_functor_object}
    Let $F: \calA \to \calB$ be an additive functor between abelian categories.
    \begin{enumerate}
        \item Suppose that the functor $F$ is \CrefAndHyperrefIfExist{definition:exact_functor_between_abelian_categories}{right exact} and suppose that $A \in \calA$ is an object for which a \CrefAndHyperrefIfExist{definition:left_right_resolution_of_an_object_by_a_class_of_objects_in_an_abelian_category}{projective resolution}
        $$\cdots \to P_2 \to P_1 \to P_0 \to A \to 0$$
        exists in $\calA$. 
        We have a \CrefAndHyperrefIfExist{definition:natural_transformation_between_functors_between_categories}{natural isomorphism}
        $$L_0 F (A) \cong F(A).$$
        (\Cref{definition:left_right_derived_functors_of_a_right_left_exact_functor_between_abelian_categories_where_source_has_enough_projectives_injectives})

        \item Suppose that the functor $F$ is \CrefAndHyperrefIfExist{definition:exact_functor_between_abelian_categories}{left exact} and suppose that $A \in \calA$ is an object for which a \CrefAndHyperrefIfExist{definition:left_right_resolution_of_an_object_by_a_class_of_objects_in_an_abelian_category}{injective resolution}
        $$0 \to A \to I^0 \to I^1 \to I^2 \to \cdots$$
        exists in $\calA$. 

        We have a \CrefAndHyperrefIfExist{definition:natural_transformation_between_functors_between_categories}{natural isomorphism}
        $$R^0 F (A) \cong F(A).$$
        (\Cref{definition:left_right_derived_functors_of_a_right_left_exact_functor_between_abelian_categories_where_source_has_enough_projectives_injectives})
    \end{enumerate}
\end{lemma}
\begin{proof}
    The isomorphism is immediate from the definition of $L_0 F$ and $R^0 F$ and the exactness assumptions on $F$. The naturality follows from \Cref{lemma:projective_injective_complex_with_map_to_from_object_with_left_right_resolution_lifts_uniquely_up_to_chain_homotopy}
\end{proof}
