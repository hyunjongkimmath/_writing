\begin{lemma} \label{lemma:extended_metric_spaces_are_hausdorff}
    Let $T$ be an \CrefAndHyperrefIfExist{definition:ordered_semiring}{ordered semiring}, let $T^\infty$ be the \CrefAndHyperrefIfExist{definition:ordered_semiring_with_adjoined_infinity}{ordered semiring with adjoined infinity} obtained from $T$, and let $(M,d)$ be an \CrefAndHyperrefIfExist{definition:extended_metric_on_a_set_valued_in_an_ordered_semiring_with_adjoined_infinity}{extended metric space valued in $T^\infty$}. The induced \CrefAndHyperrefIfExist{definition:topological_space}{topological space} on $M$ is \CrefAndHyperrefIfExist{definition:separation_axioms_of_topology}{Hausdorff}.
\end{lemma}
