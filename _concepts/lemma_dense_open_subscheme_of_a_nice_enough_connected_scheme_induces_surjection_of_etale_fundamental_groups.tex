\begin{lemma}[see {\cite[0BQG]{stacks-project}}, see also the various statements in {\cite[0BQD]{stacks-project}}] \label{lemma:dense_open_subscheme_of_a_nice_enough_connected_scheme_induces_surjection_of_etale_fundamental_groups}
    Let $X$ Be a \CrefAndHyperrefIfExist{definition:connected_object_of_a_galois_category}{connected} \CrefAndHyperrefIfExist{definition:scheme}{scheme}. Let $j: U \hookrightarrow X$ be the embedding of a dense open subscheme. Assume that 
    \begin{enumerate}
        \item every \CrefAndHyperrefIfExist{definition:quasi_compact_morphism_of_schemes}{quasi-compact} \CrefAndHyperrefIfExist{definition:topological_space}{open} of $X$ has finitely many \CrefAndHyperrefIfExist{definition:irreducible_component_of_a_topological_space}{irreducible components}
        \TODO{punctured spectrum}
        \item for every $x \in X \setminus U$ the punctured spectrum of the \CrefAndHyperrefIfExist{definition:strict_henselization_of_a_local_ring}{strict henselization} of \CrefAndHyperrefIfExist{definition:stalk_of_a_presheaf_on_a_topological_space_at_a_point}{$\calO_{X,x}$} is \CrefAndHyperrefIfExist{definition:connected_object_of_a_galois_category}{connected}.
    \end{enumerate}
    then the \CrefAndHyperrefIfExist{definition:base_change_functor_between_categories_of_finite_etale_covers_of_schemes}{base change functor} $\FEt_X \to \FEt_U$ is \CrefAndHyperrefIfExist{definition:full_and_faithful_functor_between_locally_small_categories}{fully faithful} and hence (\Cref{lemma:equivalent_conditions_for_a_map_of_fundamental_groups_of_galois_categories_to_be_surjective}) the induced homomorphism \Cref{lemma:morphism_of_connected_schemes_induces_continuous_homomoprhism_of_etale_fundamental_groups} $\pi_1(U,\overline{x}) \to \pi_1(X,\overline{x})$ is surjective.
\end{lemma}
