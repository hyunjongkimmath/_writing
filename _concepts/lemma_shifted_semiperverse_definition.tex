\begin{lemma} \label{lemma:shifted_semiperverse_definition}

    For $B \in \Dbc(X)$, 
    \begin{align*}
        B &\in \pDleqn(X) \Leftrightarrow \dim \Supp(\scrH^{-i} B) \leq i+n \quad \text{for all } i \in \bbZ \\
        B &\in \pDgeqn(X) \Leftrightarrow \dim \Supp(\scrH^{-i} D B) \leq i-n \quad \text{for all } i \in \bbZ
    \end{align*}
\end{lemma}
\begin{proof}
    By definition, $\pDleqn(X) = \pDleqzero(X)[-n]$, so $B \in \pDleqn(X)$ if and only if $B[n] \in \pDleqzero(X)$. In turn, this is equivalent to 
    $$\dim \Supp(\scrH^{-i} (B[n])) \leq i \quad \text{for all } i \in \bbZ$$
    or to 
    $$\dim \Supp(\scrH^{-i+n} B) \leq i \quad \text{for all } i \in \bbZ$$
    by Lemma \ref{lemma:indices_of_cohomology_for_t_structures}. Changing indices, this is equivalent to 
    $$\dim \Supp(\scrH^{-i} B) \leq i+n \quad \text{for all } i \in \bbZ$$
    as desired. The second equivalence can be demonstrated similarly, taking into account that $D(B[n]) \cong (DB)[-n]$. 
\end{proof}
