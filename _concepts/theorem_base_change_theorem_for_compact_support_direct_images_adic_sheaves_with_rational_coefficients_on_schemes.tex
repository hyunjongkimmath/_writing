\begin{theorem}[Base change for compact-support direct images, rational adic coefficients, cf. {\cite[Expos\'e XVII 5.2]{SGA4_III}}, {\cite[Theorem 7.4.4]{fu_ect}}] \label{theorem:base_change_theorem_for_compact_support_direct_images_adic_sheaves_with_rational_coefficients_on_schemes}
    Let $S$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme}. Let
    \begin{center}
        \begin{tikzcd}
            X' \ar[r, "f'"] \ar[d,"g'"] & Y' \ar[d, "g"] \\
            X \ar[r, "f"] & Y
        \end{tikzcd}
    \end{center}
    be a Cartesian diagram of $S$-morphisms between $S$-schemes such that $f$ is \CrefAndHyperrefIfExist{definition:compactification_of_a_morphism_of_schemes}{$S$-compactifiable} and $Y'$ is \CrefAndHyperrefIfExist{definition:quasi_compact_morphism_of_schemes}{quasi-compact} and \CrefAndHyperrefIfExist{definition:quasi_separated_morphism_of_schemes_and_separated_morphism_of_schemes}{(absolutely) quasi-separated}.

    Let $\ell$ be a prime invertible on $Y$. Let $R$ be the \CrefAndHyperrefIfExist{definition:integral_element_over_a_ring}{integral closure} of $\bbZ_\ell$ in a finite extension $E$ of $\Qell$, and let $\lambda$ be a uniformizer of $R$.

    For any $K \in Dbc(X, E)$ (\Cref{definition:derived_category_of_cohomologically_bounded_constructible_sheaves_with_rational_adic_coefficients_on_a_noetherian_scheme}), there is a canonical isomorphism
    $$g^* Rf_! K \cong Rf_!' {g'}^* K$$
    in $Dbc(Y', E)$ (\Cref{definition:six_functor_formalism_on_derived_category_of_bdd_constr_complexes_with_rational_coefficients_on_a_noetherian_scheme}).

    The isomorphism follows from \Cref{theorem:base_change_theorem_for_compact_support_direct_images_adic_sheaves_with_integral_coefficients_on_schemes} by tensoring with $E$ over $R$, since the functors on $Dbc(\cdot, E)$ descend from those on $D_c^b(\cdot, R)$ via localization by multiplication by $\lambda^m$.
\end{theorem}
