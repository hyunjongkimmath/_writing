\begin{proposition} \label{proposition:first_sheaf_cohomology_of_sheaf_of_units_on_scheme_is_same_for_fpqc_fppf_etale_zariski_tops}
    Let $X$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme}. 
    The \CrefAndHyperrefIfExist{definition:sheaf_cohomology_group_of_a_sheaf_of_modules_over_a_sheaf_of_rings_on_a_site}{sheaf cohomology groups}
    $$H^1(X,\bbG_m)$$
    (\Cref{definition:multiplicative_group_sheaf_of_a_ringed_site})
    with respect to the \CrefAndHyperrefIfExist{definition:big_fpqc_site_of_a_scheme}{(big) fpqc}, \CrefAndHyperrefIfExist{definition:big_fppf_site_of_a_scheme}{fppf}, \CrefAndHyperrefIfExist{definition:big_etale_site_of_a_scheme}{\'etale}, and \CrefAndHyperrefIfExist{definition:big_zariski_site_of_a_scheme}{Zariski} topologies are all \CrefAndHyperrefIfExist{definition:natural_transformation_between_functors_between_categories}{naturally isomorphic} to one another and are isomorphic to the \CrefAndHyperrefIfExist{definition:picard_group}{Picard group} of $X$.
\end{proposition}
