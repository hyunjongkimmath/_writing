\begin{theorem}[See {\cite[Theorem 4.4]{kiehl_weissauer_wcps}}] \label{theorem:long_exact_cohomological_sequence_in_core_for_t_structure_on_triangulated_categories_from_distinguished_triangle}
    Let $D$ be a \CrefAndHyperrefIfExist{definition:triangulated_category}{triangulated category} and let $(\Dleqzero, \Dgeqzero)$ be a \CrefAndHyperrefIfExist{definition:t_structure_on_a_triangulated_category}{$t$-structure} on $D$. Let $(X,Y,Z)$ be a distinguished triangle in $D$. 
    The \CrefAndHyperrefIfExist{definition:cohomology_functors_for_a_triangulated_category_with_a_t_structure}{cohomological functors} attached to a $t$-structure of $D$ induce a long exact cohomology sequence 
    $$\cdots \to H^{-1}(Z) \to H^0(X) \to H^0(Y) \to H^0(Z) \to H^1(X) \to \cdots$$
    in \CrefAndHyperrefIfExist{definition:t_structure_on_a_triangulated_category}{$\Core(D)$}.
\end{theorem}
