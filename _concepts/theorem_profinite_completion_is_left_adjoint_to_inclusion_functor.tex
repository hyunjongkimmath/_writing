\begin{theorem} \label{theorem:profinite_completion_is_left_adjoint_to_inclusion_functor}
Let $G$ be a \CrefAndHyperrefIfExist{definition:group}{group} and $\widehat{G}$ its \CrefAndHyperrefIfExist{definition:profinite_completion_of_a_group}{profinite completion} with the canonical map $\iota: G \to \widehat{G}$. For every \CrefAndHyperrefIfExist{definition:profinite_group}{profinite group} $H$ and every \CrefAndHyperrefIfExist{definition:continuous_map_of_topological_spaces}{continuous} \CrefAndHyperrefIfExist{definition:group_homomorphism}{homomorphism} $f: G \to H$ (where $G$ is given the \CrefAndHyperrefIfExist{definition:discrete_topological_space_of_a_set}{discrete topology}), there exists a unique continuous homomorphism $\hat{f}: \widehat{G} \to H$ such that $\hat{f} \circ \iota = f$.

\TextIfExists{definition:adjoint_functors_between_categories_unit_counit_of_adjoint_functors}{
In other words, the \CrefAndHyperrefIfExist{definition:functor_between_categories}{functor} $G \mapsto \widehat{G}$ from the \CrefAndHyperrefIfExist{definition:category_of_groups_of_abelian_groups}{category of groups} to the \CrefAndHyperrefIfExist{definition:category_of_profinite_groups}{category of profinite groups} is \CrefAndHyperref{definition:adjoint_functors_between_categories_unit_counit_of_adjoint_functors}{left adjoint} to the inclusion functor $\widehat{G} \hookrightarrow G$.
}
\end{theorem}
