\begin{proposition} \label{proposition:nice_circumstances_for_an_open_cover_to_be_acyclic_for_a_sheaf_of_abelian_groups}
     Let $X$ be a \CrefAndHyperrefIfExist{definition:topological_space}{topological space}, let $\calF$ be a \CrefAndHyperrefIfExist{definition:sheaf_on_a_topological_space_valued_in_a_category_with_a_terminal_object}{sheaf of} \CrefAndHyperrefIfExist{definition:group}{abelian groups} on $X$, and let $\calU = \{U_i\}_{i \in I}$ be an \CrefAndHyperrefIfExist{definition:open_covering_of_a_topological_space}{open cover} of $X$.

     Under the following circumstances, $\calU$ is \CrefAndHyperrefIfExist{definition:acyclic_for_a_sheaf_of_abelian_groups_for_an_open_cover_of_a_topological_space}{acyclic for $\calF$}:
     \begin{enumerate}
        \item $X$ is a \CrefAndHyperrefIfExist{definition:quasi_separated_morphism_of_schemes_and_separated_morphism_of_schemes}{separated scheme}, $\calF$ is a \CrefAndHyperrefIfExist{definition:quasi_coherent_sheaf_on_a_general_scheme}{quasi-coherent} $\calO_X$-module, and every $U_i \in \calU$ is an \CrefAndHyperrefIfExist{definition:affine_open_subscheme_of_a_scheme}{affine open subset} of $X$.

        \item $X$ is \CrefAndHyperrefIfExist{definition:locally_contractible_topological_space}{locally contractible}, $\calF$ is a \CrefAndHyperrefIfExist{definition:constant_sheaf_on_a_topological_space_valued_in_category_admitting_sheafification}{constant sheaf}, and $\calU$ is a \CrefAndHyperrefIfExist{definition:semi_good_open_cover_of_a_topological_space}{semi-good cover} of $X$.

        \item \TODO{something about complex manifolds, holomorphic sheaves, and each $U_i$ being a Stein manfiold}

        \item $\calF$ is a \CrefAndHyperrefIfExist{definition:flasque_flabby_sheaf_of_sets_on_a_topological_space}{flasque sheaf}.
     \end{enumerate}
\end{proposition}
