\begin{context}[{\cite[Section 1.]{ekedahl_af}}]\label{context:ekedahl_af_topos_morphisms_to_and_from_Pro_N_S_to_S}
    % Assume \Cref{context:ekedahl_af_ringed_topos_category_of_inverse_sequences}. 
    Let $S$ be a \CrefAndHyperrefIfExist{definition:topos}{topos}.
    Write \hl{$\pi: \mathrm{Pro}_\bbN(S) \to S$} (\Cref{definition:category_of_inverse_direct_sequences_of_objects_in_a_category}) for the \CrefAndHyperrefIfExist{definition:morphism_of_topoi}{topos morphism} given by $\pi_* M_\bullet = \varprojlim M_\bullet$ and $\pi^* M = \{M, \id\}_{n \geq 1}$, i.e. the system consisting of the objects $M$ at each level and the identity maps between them. For each $n > 0$, further write \hl{$i_n: S \to S^N$} for the topos morphism satisfying $i_n^* M_\bullet = M_n$. 
\end{context}
