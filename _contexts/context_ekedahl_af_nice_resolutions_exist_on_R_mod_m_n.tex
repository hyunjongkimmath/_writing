\begin{context}[{\cite[Section 1.]{ekedahl_af}}] \label{context:ekedahl_af_nice_resolutions_exist_on_R_mod_m_n} 
    Assume \Cref{context:ekedahl_af_ringed_topos_category_of_inverse_sequences} and \Cref{context:ekedahl_af_topos_morphisms_to_and_from_Pro_N_S_to_S}. 

    Assume for every $n$ there exists, locally on $S$, a \CrefAndHyperrefIfExist{definition:left_right_resolution_of_an_object_by_a_class_of_objects_in_an_abelian_category}{resolution} $F_n^\bullet \to R/m^n$ of $R/m^n$ by a complex of right $R$-modules which is finitely generated free in each degree and such that modulo \CrefAndHyperrefIfExist{definition:essentially_zero_inverse_sequence_essentially_bounded_complex_of_sheaves_of_abelian_groups_on_a_site}{essentially zero systems} $\pi^* F_n^\bullet \to \pi^*(R/m^n)$ is a resolution.

    For example, this is satisfied when $R$ is the \CrefAndHyperrefIfExist{definition:constant_sheaf_on_a_site_with_sheafification}{constant} sheaf of a noetherian, commutative, and local ring whose maximal ideal is $m$. Moreover, if this assumption is true of $(R,m)$, then it is also true of $(R[G], m[G])$ where $G$ is a group object in $S$.

    This assumption is assumed throughout \cite{ekedahl_af}
\end{context}
