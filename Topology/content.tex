\subsection{Constructions of topological spaces from other topological spaces}


\input{../_definitions/definition_product_of_sets.tex}

\input{../_definitions/definition_box_topology_on_a_cartesian_product_of_topological_spaces.tex}
\input{../_definitions/definition_product_topology_on_a_cartesian_product_of_topological_spaces.tex}


\begin{proposition}
Let $I$ be an index set and $\{X_i\}_{i \in I}$ be a collection of \CrefAndHyperrefIfExist{definition:topological_spaces}{topological spaces}. Let $X = \prod_{i \in I} X_i$ be the \CrefAndHyperrefIfExist{definition:product_of_sets}{set theoretic Cartesian product} of these spaces. 

If $X = \prod_{i \in I} X_i$ is endowed with the \CrefAndHyperrefIfExist{definition:product_topology_on_a_cartesian_product_of_topological_spaces}{product topology}, then a \CrefAndHyperrefIfExist{definition:function_of_sets}{function} $f: Z \to X$ from a \CrefAndHyperrefIfExist{definition:topological_space}{topological space} $Z$ is \CrefAndHyperrefIfExist{definition:continuous_map_of_topological_spaces}{continuous} if and only if each component function $f_i = p_i \circ f: Z \to X_i$ is continuous. 

This universal property generally fails for the \CrefAndHyperrefIfExist{definition:box_topology_on_a_cartesian_product_of_topological_spaces}{box topology} when $I$ is \CrefAndHyperrefIfExist{definition:countable_finite_uncountable_sets}{infinite}.
\end{proposition}

\subsection{Convergence in extended metric spaces}

\input{../_definitions/definition_convergence_and_limits_in_extended_metric_space.tex}
\input{../_definitions/definition_cauchy_sequence_in_an_extended_metric_space.tex}

\input{../_definitions/definition_completion_of_an_extended_metric_space.tex}

\begin{proposition}
Let $(X,d)$ be an \CrefAndHyperrefIfExist{definition:extended_metric_on_a_set}{extended metric space}. A \CrefAndHyperrefIfExist{definition:completion_of_an_extended_metric_space}{completion} $(\widehat{X}, \widehat{d})$ of $(X,d)$ exists and is unique up to unique isometry. Moreover, if $(X,d)$ is a metric space, then so is $(\widehat{X}, \widehat{d})$.
\end{proposition}
\begin{proof}

Let $\calC$ be the set of all Cauchy sequences in \((X,d)\):
\[
\mathcal{C} = \{(x_n)_{n \in \mathbb{N}} : (x_n) \text{ is a Cauchy sequence in } (X,d)\},
\]
and define a relation \(\sim\) on \(\mathcal{C}\) by
\[ (x_n) \sim (y_n) \iff \lim_{n \to \infty} d(x_n, y_n) = 0.  \]
This relation is well-defined and in fact an equivalence relation. \TODO{Show that the relation is well-defined and an equivalence relation} Let \(\widehat{X} = \mathcal{C} / \sim\) and let \(\widehat{d} : \widehat{X} \times \widehat{X} \to [0, \infty]\) be defined by
\[ \widehat{d}([(x_n)], [(y_n)]) := \lim_{n \to \infty} d(x_n, y_n).  \]
This limit exists as a value in $[0,\infty]$ and is independent of representative sequences chosen. \TODO{Demonstrate why the limit exists and is well defined}

It can be verified that \(\widehat{d}\) is an extended metric on \(\widehat{X}\) and that \((\widehat{X}, \widehat{d})\) is complete by construction.

Define an embedding
\[
i : X \to \widehat{X}, \quad i(x) := [(x,x,x,\ldots)],
\]
which is an isometry:
\[
\widehat{d}(i(x), i(y)) = \lim_{n \to \infty} d(x,y) = d(x,y).
\]

Moreover, \(i(X)\) is dense in \(\widehat{X}\) because any equivalence class \([(x_n)]\) of a Cauchy sequence can be approximated arbitrarily closely by points \(i(x_n)\) in the image of \(X\).

\bigskip

\textbf{Uniqueness:}

Suppose \((\widehat{X}', \widehat{d}')\) is another completion of \((X,d)\) with isometric embedding \(i' : X \to \widehat{X}'\).

By the universal property of completions, there exists a unique isometry
\[
\Phi : \widehat{X} \to \widehat{X}'
\]
such that \(\Phi \circ i = i'\).

The map \(\Phi\) is onto since \(i'(X)\) is dense in \(\widehat{X}'\) and \(\Phi\) preserves limits of Cauchy sequences.

Therefore, \(\Phi\) is a unique isometry between the completions, and the completion is unique up to unique isometry.
\end{proof}
