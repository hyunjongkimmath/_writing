\subsubsection{Cancellation theorem for a property of morphisms of schemes}

\begin{theorem}[see [{\cite[10.1.19.]{foag}}]] \label{theorem:cancellation_theorem_for_a_property_of_morphisms_of_schemes}
    \TODO{list of properties that fit in here}
    Let $P$ be a class of \CrefAndHyperrefIfExist{definition:morphism_of_schemes}{morphisms of schemes} that is preserved by \CrefAndHyperrefIfExist{definition:base_change_of_a_morphism_in_a_category_by_a_morphism}{base change} and composition. Suppose that 
    \begin{center}
        \begin{tikzcd}
            X \ar[rr, "\pi"] \ar[rd, "\tau"] & & Y \ar[dl, "\rho"] \\
            & Z & 
        \end{tikzcd}
    \end{center}
    is a commuting diagram of \CrefAndHyperrefIfExist{definition:scheme}{schemes}. Suppose that the \CrefAndHyperrefIfExist{definition:diagonal_morphism_of_a_morphism_of_schemes}{diagonal morphism} $\delta_\rho: Y \to Y \times_Z Y$ is in $P$ and that $\tau: X \to Z$ is in $P$. Then $\pi: X \to Y$ is in $P$.

    In particular, 
    \begin{enumerate}
        \item Suppose that \CrefAndHyperrefIfExist{definition:locally_closed_imersion_of_schemes}{locally closed embeddings} are in $P$. If $\tau$ is in $P$, then $\pi$ is in $P$. 
        \item Suppose that \CrefAndHyperrefIfExist{definition:open_and_closed_immersions_of_schemes}{closed embeddings} are in $P$ (e.g. $P$ could be the class of \CrefAndHyperrefIfExist{definition:finite_morphism_of_schemes}{finite morphisms}, morphisms of \CrefAndHyperrefIfExist{definition:finite_type_morphism_of_schemes}{finite type}, \CrefAndHyperrefIfExist{definition:open_and_closed_subscheme_of_a_scheme}{closed embeddings}, \CrefAndHyperrefIfExist{definition:affine_morphism_of_schemes}{affine morphisms}, etc.) If $\tau$ is in $P$ and $\rho$ is \CrefAndHyperrefIfExist{definition:quasi_separated_morphism_of_schemes_and_separated_morphism_of_schemes}{separated}, then $\pi$ is in $P$.

        \item Suppose that \CrefAndHyperrefIfExist{definition:quasi_compact_scheme}{quasi-compact} morphisms are in $P$. If $\tau$ is in $P$ and $\rho$ is \CrefAndHyperrefIfExist{definition:quasi_separated_morphism_of_schemes_and_separated_morphism_of_schemes}{quasi-separated}, then $\pi$ is in $P$.
    \end{enumerate}
\end{theorem}

\begin{proposition}[see e.g. {\cite[10.3.4. Proposition.]{foag}}] \label{proposition:proper_morphisms_of_schemes_cancel_by_sepearated_morphisms_to_proper_morphisms}
    Let 
    \begin{center}
        \begin{tikzcd}
            X \ar[rr, "\pi"] \ar[rd, "\tau"] & & Y \ar[dl, "\rho"] \\
            & Z & 
        \end{tikzcd}
    \end{center}
    be a commuting diagram of \CrefAndHyperrefIfExist{definition:scheme}{schemes}. Suppose that $\tau$ is \CrefAndHyperrefIfExist{definition:proper_morphism_of_schemes}{proper} and $\rho$ is \CrefAndHyperrefIfExist{definition:quasi_separated_morphism_of_schemes_and_separated_morphism_of_schemes}{separated}. Then $\pi$ is proper. 
\end{proposition}
\begin{proof}
    \TODO{closed immersions are proper}
    All \CrefAndHyperrefIfExist{definition:open_and_closed_subscheme_of_a_scheme}{closed embeddings} of \CrefAndHyperrefIfExist{definition:scheme}{schemes} are \CrefAndHyperrefIfExist{definition:proper_morphism_of_schemes}{proper}. Thus, \Cref{theorem:cancellation_theorem_for_a_property_of_morphisms_of_schemes} shows that $\pi$ is proper.
\end{proof}

\begin{remark}
The connection to "codimension $p$ cycles" arises from the map $A^p(X) \to A_{n-p}(X)$ defined by $c \mapsto c_{\text{id}_X}([X])$. If $X$ is smooth, this map is an isomorphism. In that case, for any cycle $\gamma \in A^p(X)$, the corresponding operation $c_{\gamma}$ is given by the \CrefAndHyperrefIfExist{definition:intersection_product}{intersection product}: $c_{\gamma, f}(\alpha) \coloneqq f^*(\gamma) \cdot \alpha$.
\end{remark}

\begin{proposition}

Let $X$ be a \CrefAndHyperrefIfExist{definition:locally_noetherian_and_noetherian_scheme}{Noetherian scheme} of dimension $n$. 
The relationship between the \CrefAndHyperrefIfExist{definition:operational_chow_ring}{operational Chow ring} $A^*(X)$ and the \CrefAndHyperrefIfExist{definition:k_dimensional_algebraic_cycles}{classical Chow groups} $A_*(X)$ is governed by the following properties:

\begin{enumerate}
    \item \textbf{The Cap Product (Action):} The total Chow group $A_*(X)$ is a \CrefAndHyperrefIfExist{definition:module}{graded module} over the ring $A^*(X)$. The action, denoted by the cap product 
    \[ \cap: A^p(X) \otimes A_k(X) \to A_{k-p}(X), \]
    is defined by $c \cap \alpha = c_{\mathrm{id}_X}(\alpha)$, where $c_{\mathrm{id}_X}$ is the homomorphism associated to the identity morphism on $X$.
    
    \item \textbf{The Poincaré Duality Map:} There exists a canonical homomorphism 
    \[ \Phi: A^p(X) \to A_{n-p}(X) \]
    defined by $\Phi(c) = c \cap [X]$, where $[X]$ is the \CrefAndHyperrefIfExist{definition:fundamental_cycle}{fundamental class} of $X$.
    
    \item \textbf{Smooth Case (Isomorphism):} If $X$ is a \CrefAndHyperrefIfExist{definition:smooth_variety}{smooth variety} (or more generally, a regular scheme), then $\Phi$ is an \CrefAndHyperrefIfExist{definition:isomorphism}{isomorphism}. In this case, every $k$-dimensional cycle $\alpha$ corresponds uniquely to an operational class $c_\alpha$ such that the ring multiplication in $A^*(X)$ coincides with the classical intersection product on $A_*(X)$.
    
    \item \textbf{Singular Case:} If $X$ is singular, $\Phi$ is generally neither injective nor surjective. The operational ring $A^*(X)$ captures only those "well-behaved" classes that can be pulled back and intersected consistently, whereas $A_*(X)$ contains all cycles regardless of their intersection properties.
\end{enumerate}
\end{proposition}

\section{Hirzebruch Surfaces}

\input{../_definitions/definition_hirzebruch_surface_over_a_field.tex}

\begin{theorem}[Intersection Pairing on $\mathbb{F}_n$]
Let $\mathbb{F}_n$ be the Hirzebruch surface of degree $n \ge 0$. The intersection form on the Picard group $\operatorname{Pic}(\mathbb{F}_n) \cong \mathbb{Z} f \oplus \mathbb{Z} s$ (where $f$ is the fiber class and $s$ is the class of the unique section with $s^2 = -n$) is given by the matrix:
$$
\begin{pmatrix}
f^2 & f \cdot s \\
s \cdot f & s^2
\end{pmatrix}
=
\begin{pmatrix}
0 & 1 \\
1 & -n
\end{pmatrix}.
$$
Consequently, the intersection pairing is unimodular, and the determinant of the intersection form is $-1$.
\end{theorem}

\begin{theorem}[Canonical Divisor of $\mathbb{F}_n$]
The canonical divisor class $K_{\mathbb{F}_n}$ of the Hirzebruch surface $\mathbb{F}_n$ is given in terms of the basis $\{f, s\}$ by the formula:
$$\hlin{K_{\mathbb{F}_n} \sim -2s - (n+2)f.}$$
\end{theorem}

\begin{theorem}[Cohomology of Line Bundles on $\mathbb{F}_n$]
Let $D \equiv a s + b f$ be a divisor on $\mathbb{F}_n$ with $a, b \in \mathbb{Z}$. The dimension of the space of global sections $H^0(\mathbb{F}_n, \mathcal{O}(D))$ is given by:
$$\hlin{h^0(\mathbb{F}_n, \mathcal{O}(as+bf)) = \sum_{i=0}^a \max(0, b - ni + 1), \quad \text{if } a \ge 0.}$$
If $a < 0$, then $h^0(\mathbb{F}_n, \mathcal{O}(D)) = 0$.
\end{theorem}

\begin{proposition}[Curve Classes]
Let $C \subset \mathbb{F}_n$ be an irreducible curve effectively equivalent to $as + bf$.
\begin{itemize}
    \item If $C = s$ (the unique negative section), then $a=1, b=0$.
    \item If $C \neq s$, then $a \ge 0$ and $b \ge na$.
\end{itemize}
This implies that the cone of effective curves (the Mori cone) $\overline{NE}(\mathbb{F}_n)$ is generated by the classes of the negative section $s$ and the fiber $f$.
\end{proposition}



\begin{theorem}[Structure of Hirzebruch Surfaces]
Let $\mathbb{F}_n = \mathbb{P}(\mathcal{O}_{\mathbb{P}^1} \oplus \mathcal{O}_{\mathbb{P}^1}(n))$ be a Hirzebruch surface over an algebraically closed field $k$, where $n \ge 0$. Let $\pi: \mathbb{F}_n \to \mathbb{P}^1$ be the ruling.
\begin{enumerate}
    \item \textbf{Picard Group:} The Picard group $\operatorname{Pic}(\mathbb{F}_n)$ is free abelian of rank 2, generated by the class of a fiber $f$ of $\pi$ and the class of a section $s$ satisfying $s^2 = -n$. Explicitly,
    $$\hlin{\operatorname{Pic}(\mathbb{F}_n) \cong \mathbb{Z} f \oplus \mathbb{Z} s,}$$
    where $f \cdot f = 0$, $f \cdot s = 1$, and $s$ corresponds to the quotient line bundle $\mathcal{O}_{\mathbb{P}^1}(n)$ of the defining vector bundle.
    \item \textbf{Unique Negative Section:} If $n > 0$, there exists a unique irreducible curve $C_0 \subset \mathbb{F}_n$ with negative self-intersection. This curve is a section of $\pi$ and satisfies $C_0^2 = -n$. It corresponds to the class $s$ in the Picard group.
    \item \textbf{Toric Description:} $\mathbb{F}_n$ is the complete toric surface associated to the fan $\Sigma_n \subset \mathbb{R}^2$ generated by the rays
    $$v_1 = (1, 0), \quad v_2 = (0, 1), \quad v_3 = (-1, n), \quad v_4 = (0, -1).$$
\end{enumerate}
\end{theorem}

\begin{proposition}[Isomorphism Classification]
Two Hirzebruch surfaces $\mathbb{F}_n$ and $\mathbb{F}_m$ are isomorphic as abstract algebraic varieties if and only if $n = m$.
\end{proposition}

\begin{theorem}[Rationality and Exceptional Cases]
\begin{itemize}
    \item $\mathbb{F}_0$ is isomorphic to $\mathbb{P}^1 \times \mathbb{P}^1$.
    \item $\mathbb{F}_1$ is isomorphic to the blow-up of $\mathbb{P}^2$ at a single point. The unique negative section $C_0$ (with $C_0^2 = -1$) is the exceptional divisor of the blow-up.
    \item All Hirzebruch surfaces $\mathbb{F}_n$ are rational surfaces.
\end{itemize}
\end{theorem}
