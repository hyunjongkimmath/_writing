\begin{definition} \label{definition:abelianization_of_a_group}
Let $G$ be a \CrefAndHyperrefIfExist{definition:group}{group}. The \hldef{abelianization of~$G$} is the quotient group
$$\hlin{G^{\operatorname{ab}} = G/[G,G]}$$
where $[G,G]$ is the \CrefAndHyperrefIfExist{definition:commutator_of_two_subsets_of_a_group}{commutator subgroup} of~$G$.
\end{definition}
