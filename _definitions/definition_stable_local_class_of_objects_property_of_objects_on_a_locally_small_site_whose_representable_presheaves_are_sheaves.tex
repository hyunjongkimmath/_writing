\begin{definition}[{\cite[Chapter I, 1.5]{knutson_as}}, See {\cite[0348]{stacks-project}}, {\cite[Definition 5.1.1]{olsson_ass}}] \label{definition:stable_local_class_of_objects_property_of_objects_on_a_locally_small_site_whose_representable_presheaves_are_sheaves}
    Let $\calC$ be a \CrefAndHyperrefIfExist{definition:grothendieck_topology_on_a_category_site_covering_sieve_topologically_generating_family}{site} on a \CrefAndHyperrefIfExist{definition:locally_small_category}{locally small category} for which \CrefAndHyperrefIfExist{definition:representable_functor_on_a_locally_small_category}{representable presheaves} are \CrefAndHyperrefIfExist{definition:sheaf_on_a_site}{sheaves} on $\calC$.
    \begin{enumerate}
        \item A class of objects $S$ of $\calC$ is said to be \hldef{stable} or \hldef{local in the site $\calC$ (or the Grothendieck topology thereof)} if for every \CrefAndHyperrefIfExist{definition:grothendieck_topology_on_a_category_site_covering_sieve_topologically_generating_family}{covering} $\{U_i \to U\}$ of an object $U$ of $\calC$, the object $U$ is in $S$ if and only if all $U_i$ are in $S$.

        \item A property $P$ of objects of $\calC$ is said to be \hldef{stable} or \hldef{local in the site $\calC$ (or the Grothendieck topology thereof)} if the class of objects satisfying $P$ is stable.
    \end{enumerate}
\end{definition}
