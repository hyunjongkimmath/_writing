\begin{definition} \label{definition:universal_homological_cohomological_delta_functor_between_abelian_cateogries}
    Let $\mathcal{A}$ and $\mathcal{B}$ be \CrefAndHyperrefIfExist{definition:abelian_category}{abelian categories}. 
A \CrefAndHyperrefIfExist{definition:homological_cohomological_delta_functor_between_abelian_categories}{cohomological (resp. homological) $\delta$-functor} $T = (T_n: \calA \to \calB, \delta_n)_{n \geq 0}$ (resp. $T = (T^n: \calA \to \calB, \delta^n)_{n \geq 0}$) is called \hldef{universal} if, for any other $\delta$-functor $S$ of the same type and any natural transformation $f: T_0 \to S_0$ (or $f: T^0 \to S^0$), there exists a unique \CrefAndHyperrefIfExist{definition:homological_cohomological_delta_functor_between_abelian_categories}{morphism of $\delta$-functors} extending $f$.
\end{definition}
