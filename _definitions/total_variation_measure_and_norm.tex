\begin{definition}[Total variation measure and norm of a signed measure]
\label{definition:total_variation_measure_and_norm}
Let $(X, \mathcal{M}, \mu)$ be a \CrefAndHyperrefIfExist{definition:signed_measure}{signed measure}. For each $E \in \mathcal{M}$, the \hldef{total variation measure of $\mu$} is the set function \hl{$|\mu|$} defined by
$$|\mu|(E) = \sup \left\{ \sum_{i=1}^{\infty} |\mu(E_i)| : \{E_i\}_{i=1}^{\infty} \text{ is a measurable partition of } E \right\}$$
and the \hldef{total variation norm of $\mu$} is the non-negative real number \hl{$\|\mu\|$} given by
$$\|\mu\| = |\mu|(X) = \sup \left\{ \sum_{i=1}^{\infty} |\mu(E_i)| : \{E_i\}_{i=1}^{\infty} \text{ is a measurable partition of } X \right\}.$$
\end{definition}
