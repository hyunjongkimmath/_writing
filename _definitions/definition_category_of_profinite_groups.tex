\begin{definition} \label{definition:category_of_profinite_groups}
    The \hldef{category of profinite groups} refers to the category whose objects are \CrefAndHyperrefIfExist{definition:profinite_group}{profinite groups} and whose morphisms are \CrefAndHyperrefIfExist{definition:continuous_map_of_topological_spaces}{continuous} \CrefAndHyperrefIfExist{definition:group_homomorphism}{homomorphisms} between them. 
\end{definition}
