\begin{definition}[Verdier dual of a constructible complex with rational coefficients] \label{definition:verdier_dual_of_an_object_in_the_bdd_constr_derived_category_of_adic_sheaves_with_rational_coefficients_on_a_finite_type_scheme_over_a_noetherian_scheme}
    Let $(R, \lambda)$ be a \CrefAndHyperrefIfExist{definition:complete_ring_with_respect_to_an_ideal}{complete} \CrefAndHyperrefIfExist{definition:discrete_valuation_ring}{DVR} such that $R/\lambda^n$ is finite for all $n \geq 1$. Let $E$ be the \CrefAndHyperrefIfExist{definition:field_of_fractions_of_an_integral_domain}{fraction field} of $R$. Let $S$ be a \CrefAndHyperrefIfExist{definition:locally_noetherian_and_noetherian_scheme}{noetherian scheme}. Let $X$ be an $S$-scheme of \CrefAndHyperrefIfExist{definition:finite_type_morphism_of_schemes}{finite type}.  Consider sheaves on the (small) \'etale sites of $X$ and $S$.

    For any object $M \in \Dbc(X, E)$, the \hldef{Verdier dual of $M$ on $X$ relative to $S$ with rational coefficients} is the object in $\Dbc(X,E)$ given by 
    $$\hlin{\mathbb{D}_X(M) := R\mathscr{H}om_E(M, \mathbb{D}_{X/S}^\bullet(E))}.$$
    \CrefIfExists{definition:derived_internal_hom_on_bdd_constructible_complexes_with_rational_coefficients_on_a_noetherian_scheme} \CrefIfExists{definition:dualizing_complex_on_a_finite_type_scheme_relative_to_a_noetherian_base_with_rational_coefficients}

    Equivalently, its underlying object in \CrefAndHyperrefIfExist{definition:derived_category_of_bounded_constructible_complexes_of_adic_sheaves_with_integral_coefficients_on_a_noetherian_scheme}{$\Dbc(X, R)$} is the \CrefAndHyperrefIfExist{definition:verdier_dual_of_an_object_in_the_bdd_constr_derived_category_of_adic_sheaves_with_integral_coefficients_on_a_finite_type_scheme_over_a_noetherian_scheme}{Verdier dual} $\bbD_X(M_0)$ where $M_0 \in \Dbc(X,R)$ is the underlying object of $M$. 
\end{definition}
