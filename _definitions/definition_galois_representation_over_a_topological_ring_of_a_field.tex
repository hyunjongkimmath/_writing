\begin{definition} \label{definition:galois_representation_over_a_topological_ring_of_a_field}
    Let $K$ be a \CrefAndHyperrefIfExist{definition:field}{field}. Let $R$ be a \CrefAndHyperrefIfExist{definition:topological_ring}{topological ring}. 

    \begin{enumerate}
        \item Let $L/K$ be a \CrefAndHyperrefIfExist{definition:galois_extension_of_fields}{Galois field extension}. A \hldef{Galois representation of $\Gal(L/K)$ over $R$} refers to a \CrefAndHyperrefIfExist{definition:continuous_representation_of_a_topological_group_on_a_topological_module_over_a_topological_ring}{continuous representation} 
        $$\rho: \operatorname{Gal}(L/K) \to \Aut_R(M)$$
        of $\operatorname{Gal}(L/K)$ as a \CrefAndHyperrefIfExist{definition:topological_group}{topological group} (with its \CrefAndHyperrefIfExist{definition:profinite_group}{profinite topology}) for some \CrefAndHyperrefIfExist{definition:topological_module_over_topological_rings}{topological $R$-module} $M$.

        The category of Galois representations of $\Gal(L/K)$ over $R$ is equivalent to the category of \CrefAndHyperrefIfExist{definition:galois_module_over_a_topological_ring_of_a_field}{Galois modules of $\Gal(L/K)$ over $R$}.

        \item A \hldef{Galois representation of $K$ over $R$} refers to a Galois representation of $\operatorname{Gal}(\overline{K}/K)$ over $R$.
    \end{enumerate}
    
    If $R$ is a ring and $M$ is a module both without explicit topologies specified, then $R$ and $M$ are implicitly equipped with their discrete topologies.
\end{definition}
