\begin{definition}[Geometric stalk of object of $D_c^b(X,K)$ for algebraic $K$] \label{definition:geometric_stalk_of_object_of_bounded_const_derived_category_of_complexes_of_adic_etale_sheaves_on_a_noetherian_scheme_with_algebraic_rational_coefficients}
    
    Let $(R, \lambda)$ be a \CrefAndHyperrefIfExist{definition:complete_ring_with_respect_to_an_ideal}{complete} \CrefAndHyperrefIfExist{definition:discrete_valuation_ring}{DVR} such that $R/\lambda^n$ is finite for all $n \geq 1$. Let $E$ be the fraction field of $R$. Let $K$ be an algebraic extension of $E$. 
    Let $X$ be a \CrefAndHyperrefIfExist{definition:locally_noetherian_and_noetherian_scheme}{noetherian scheme}, and consider \CrefAndHyperrefIfExist{definition:sheaf_on_a_site}{sheaves} on the \CrefAndHyperrefIfExist{definition:small_etale_site_of_a_scheme}{(small) \'etale site on $X$}. Let $x \in X$ be a point. Let $\barx: \Spec \kappa(x)^s \to X$ be a \CrefAndHyperrefIfExist{definition:geometric_point_of_a_scheme}{geometric point} over $X$. 

    Given an object $M \in \Dbc(X,K)$ (\Cref{definition:derived_category_of_cohomologically_bounded_constructible_sheaves_with_rational_adic_coefficients_on_a_noetherian_scheme}) which is represented by an object $L \in \Dbc(X,E')$ (\Cref{definition:derived_category_of_cohomologically_bounded_constructible_sheaves_with_rational_adic_coefficients_on_a_noetherian_scheme}), define the \hldef{stalk of $M$ at $\barx$} as 
    $$\hlin{M_{\barx} \coloneq L_{\barx} \otimes_{E'} K}$$
    (\Cref{definition:geometric_stalk_of_object_of_bounded_const_derived_category_of_complexes_of_adic_etale_sheaves_on_a_noetherian_scheme_with_rational_coefficients}) This is independent of the choice of $E'$ and the choice of $L$.

    For every such $x$ and $\overline{x}$, we have a functor

    $$\Dbc(X,K) \to \Gal(\kappa(x)^s / \kappa(x))-\VectorSpaces_K$$
\end{definition}
