\begin{definition}[Riemann surface] \label{definition:riemann_surface}
    Let $\bbC$ be the \CrefAndHyperrefIfExist{definition:field_of_complex_numbers}{field of complex numbers}. A \hldef{Riemann surface} is a one-dimensional \CrefAndHyperrefIfExist{definition:complex_analytic_manifold}{complex analytic manifold}. 

    A Riemann surface is called \hldef{compact} if the underlying topological space $X$ is compact.
\end{definition}
