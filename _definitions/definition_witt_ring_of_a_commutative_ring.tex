% WITT RING OF A COMMUTATIVE RING - Definition of the ring of Witt vectors
\begin{definition}\label{definition:witt_ring_of_a_commutative_ring}
Let $R$ be a \CrefAndHyperrefIfExist{definition:commutative_ring}{commutative ring}.

The \hldef{ring of Witt vectors} (or simply the \hldef{Witt ring}) of $R$, denoted \hl{$W(R)$}, is defined as the \CrefAndHyperrefIfExist{definition:limit_and_colimit_of_a_diagram_in_a_category}{inverse limit}
$$
    W(R) := \varprojlim_{n} W_n(R),
$$
where $W_n(R)$ are the \CrefAndHyperrefIfExist{definition:truncated_witt_ring_of_length_n_over_a_commutative_ring}{truncated Witt rings of length $n$ over $R$}, and the transition maps $W_{n+1}(R) \to W_n(R)$ are the projection homomorphisms onto the first $n$ components.

Equivalently, an element of $W(R)$ is a sequence $(a_0, a_1, a_2, \dots)$ with $a_i \in R$, and addition $\oplus$ and multiplication $\odot$ are defined componentwise via the Witt polynomials: for any two elements $\mathbf{a} = (a_0, a_1, \dots), \mathbf{b} = (b_0, b_1, \dots) \in W(R)$, we have
\begin{align*}
    \mathbf{a} \oplus \mathbf{b} &= (S_0(a_0, b_0), S_1(a_0, a_1, b_0, b_1), S_2(a_0, a_1, a_2, b_0, b_1, b_2), \dots), \\
    \mathbf{a} \odot \mathbf{b} &= (P_0(a_0, b_0), P_1(a_0, a_1, b_0, b_1), P_2(a_0, a_1, a_2, b_0, b_1, b_2), \dots),
\end{align*}
where $S_k, P_k$ are the unique polynomials determined by the \CrefAndHyperrefIfExist{definition:witt_polynomial_with_respect_to_a_prime_number}{Witt polynomials}.

With these operations, $W(R)$ is a \CrefAndHyperrefIfExist{definition:commutative_ring}{commutative ring} with zero element $(0, 0, \dots)$ and unit element $(1, 0, \dots)$.

If $k$ is a \CrefAndHyperrefIfExist{definition:perfect_field}{perfect field} of \CrefAndHyperrefIfExist{definition:characteristic_of_a_ring}{prime characteristic} $p$, then $W(k)$ has the following special properties:
\begin{itemize}
    \item $W(k)$ is a \CrefAndHyperrefIfExist{definition:discrete_valuation_ring}{discrete valuation ring} of characteristic zero,
    \item The \CrefAndHyperrefIfExist{definition:prime_and_maximal_ideal_of_a_ring}{maximal ideal} of $W(k)$ is generated by $p$,
    \item The \CrefAndHyperrefIfExist{definition:residue_field_of_a_ring_at_a_point}{residue field} $W(k)/pW(k)$ is isomorphic to $k$,
    \item $W(k)$ is \CrefAndHyperrefIfExist{definition:complete_extended_metric_space_valued_in_ordered_commutative_monoid_with_adjoined_infinity}{complete} with respect to the \CrefAndHyperrefIfExist{definition:p_adic_metric_for_a_prime_of_a_number_field}{$p$-adic metric}.

    In particular, if $k = \mathbb{F}_p$, then $W(\mathbb{F}_p) \cong \mathbb{Z}_p$, the ring of \CrefAndHyperrefIfExist{definition:p_adic_integers_for_a_nonzero_prime_of_a_number_field}{$p$-adic integers}.
\end{itemize}
\end{definition}
