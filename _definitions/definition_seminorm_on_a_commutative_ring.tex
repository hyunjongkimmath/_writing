\begin{definition} \label{definition:seminorm_on_a_commutative_ring}
Let $R$ be a \CrefAndHyperrefIfExist{definition:commutative_ring}{commutative ring}.

A \hldef{seminorm on $R$} is a function
$$
| \cdot | : R \to \mathbb{R}_{\geq 0}
$$
that satisfies the following properties for all $a, b \in R$:

\begin{enumerate}
    \item Subadditivity: $|a + b| \leq |a| + |b|$,
    \item Submultiplicativity: $|ab| \leq |a| \cdot |b|$,
    \item Normalization: $|0| = 0$.
\end{enumerate}

A ring equipped with a seminorm may be called a \hldef{seminormed ring}.
\end{definition}
