\begin{definition} \label{definition:cancellative_monoid}
    A \CrefAndHyperrefIfExist{definition:monoid}{monoid} $(M,\cdot,e)$ is \hldef{cancellative} if its \CrefAndHyperrefIfExist{definition:n_ary_operation_on_a_set}{binary operation} $\cdot$ is \CrefAndHyperrefIfExist{definition:cancellative_binary_operation_on_a_set}{cancellative}, i.e. for all $a,b,c \in M$, we have
    $$a \cdot b = a \cdot c \implies b = c.$$
    $$b \cdot a = c \cdot a \implies b = c.$$
\end{definition}
