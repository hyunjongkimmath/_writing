\begin{definition} \label{definition:symmetrically_self_dual_ring_over_a_ring}
    Let $\varphi: R \to A$ be a ring homomorphism. We say that $A$ is \hldef{symmetrically self-dual over $R$ (with respect to $\varphi$)} if there exists an \CrefAndHyperrefIfExist{definition:homomorphism_of_modules_over_a_ring}{isomorphism} $\psi: A \cong A^\vee = \Hom_{{}_R \Mod_R}(A,R)$\CrefIfExists{definition:dual_of_a_left_right_two_sided_module} as $(A,A)$-\CrefAndHyperrefIfExist{definition:module_over_a_sheaf_of_rings_on_a_site}{bimodules}. 
    We may refer to such a $\psi$ as a \hldef{symmetric self-dual map for $\varphi$}. With respect to a fixed choice of a symmetric self-dual map $\psi$, the element $\psi(1) \in A^\vee$ is called the \hldef{symmetric trace}, and satisfies the \CrefAndHyperrefIfExist{definition:trace_property_for_a_function_on_a_ring}{trace property}\CrefIfExists{lemma:symmetric_self_duality_on_a_ring_sends_center_elements_to_trace_functionals}.

    \TextIfExists{definition:left_right_frobenius_self_dual_ring_homomorphism}{
        Clearly, any symmetric self-dual ring homomorphism is a \CrefAndHyperrefIfExist{definition:left_right_frobenius_self_dual_ring_homomorphism}{left and a right Frobenius ring homomorphism}.
    }
\end{definition}
