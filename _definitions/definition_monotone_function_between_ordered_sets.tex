\begin{definition} \label{definition:monotone_function_between_ordered_sets}

    Let $(X, \leq_X)$ and $(Y, \leq_Y)$ be \CrefAndHyperrefIfExist{definition:partially_ordered_set}{ordered sets}, and let $f : X \to Y$ be a function.

    The function $f$ is called \hldef{monotonically increasing} (or simply \hldef{increasing}) if for all $x_1, x_2 \in X$,
    $$\text{if } x_1 \leq_X x_2, \text{ then } f(x_1) \leq_Y f(x_2).$$

    The function $f$ is called \hldef{monotonically decreasing} (or simply \hldef{decreasing}) if for all $x_1, x_2 \in X$,
    $$\text{if } x_1 \leq_X x_2, \text{ then } f(x_2) \leq_Y f(x_1).$$

    The function $f$ is called \hldef{strictly increasing} if for all $x_1, x_2 \in X$,
    $$\text{if } x_1 <_X x_2, \text{ then } f(x_1) <_Y f(x_2).$$

    The function $f$ is called \hldef{strictly decreasing} if for all $x_1, x_2 \in X$,
    $$\text{if } x_1 <_X x_2, \text{ then } f(x_2) <_Y f(x_1).$$

    The function $f$ is called \hldef{monotone} if it is either monotonically increasing or monotonically decreasing.
\end{definition}
