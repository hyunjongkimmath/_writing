\begin{definition}[Finite, $\sigma$-finite measure space] \label{definition:finite_sigma_finite_measure_space_for_a_measure_on_a_measurable_space}
Let $(X, \mathcal{A}, \mu)$ be a \CrefAndHyperrefIfExist{definition:measure_on_a_measurable_space_and_measure_space}{measure space}.

\begin{enumerate}
    \item The measure $\mu$ is called \hldef{finite} if
    $$\mu(X) < \infty.$$

    \item The measure $\mu$ is called \hldef{$\sigma$-finite} if there exists a sequence $(X_n)_{n\ge 1}$ of sets in $\mathcal{A}$ such that
    $$X = \bigcup_{n=1}^\infty X_n \quad \text{and} \quad \mu(X_n) < \infty \text{ for all } n \ge 1.$$
    The sequence $(X_n)_{n\ge 1}$ is called a \hldef{witnessing sequence for the $\sigma$-finiteness of $\mu$}.
\end{enumerate}

Every finite measure is $\sigma$-finite (take $X_1 = X$, $X_n = \varnothing$ for $n \ge 2$).
\end{definition}
