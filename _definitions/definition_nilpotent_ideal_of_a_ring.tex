\begin{definition} \label{definition:nilpotent_ideal_of_a_ring}
Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{not necessarily commutative ring}.
An ideal $I \subseteq R$ is \hldef{nilpotent} if there exists a positive integer $n \geq 1$ such that $I^n = \{0\}$\CrefIfExists{definition:product_of_ideals_of_a_ring}.
\end{definition}
