\begin{definition}[Topology induced by an extended metric valued in an ordered monoid with adjoined infinity] \label{definition:topology_induced_by_an_extended_metric_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity_and_open_ball_for_an_extended_metric}
    Let $T$ be an \CrefAndHyperrefIfExist{definition:ordered_commutative_monoid}{ordered commutative monoid}, let $T^\infty$ be the \CrefAndHyperrefIfExist{definition:ordered_monoid_with_adjoined_infinity}{ordered monoid with adjoined infinity} obtained from $T$, and let $(X, d)$ be an \CrefAndHyperrefIfExist{definition:extended_metric_on_a_set_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity}{extended metric space valued in $T^\infty$}.
    \begin{enumerate}
        \item Given $\varepsilon \in T_{>0}$ and $x \in X$, the set
        $$\hlin{B(x, \varepsilon) := \{ y \in X : d(x, y) < \varepsilon \}}$$
        is called the \hldef{open ball of radius $\varepsilon$ centered at $x$}.

        \item The \hldef{topology induced by $d$ on $X$} is the \CrefAndHyperrefIfExist{definition:topological_space}{topology} defined by declaring a subset $U \subseteq X$ to be open if for every $x \in U$, there exists $\varepsilon \in T_{>0}$ such that $B(x, \varepsilon) \subseteq U$\CrefIfExists{proposition:induced_topology_of_an_extended_metric_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity_is_a_topology}. In particular, $B(x, \varepsilon)$ is an open set for this topology for all $x \in X$ and $\varepsilon \in T_{>0}$.

    \end{enumerate}
\end{definition}
