\begin{definition}[{cf. \cite[Chapter I, 3.2]{knus_qhfr}}] \label{definition:sesquilinear_space_varepsilon_hermitian_space_over_a_ring_with_involution}
    Let $(A, \sigma)$ be a \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{ring with involution}. 
    \begin{enumerate}
        \item A \hldef{sesquilinear space} is a \CrefAndHyperrefIfExist{definition:sesquilinear_module_over_a_ring_with_involution}{sesquilinear module} $(M,b)$ whose underlying sesquilinear form $b$ is \CrefAndHyperref{definition:nondegenerate_nonsingular_regular_sesquilinear_form_on_a_module_over_an_involution_ring}{nonsingular}.

        \item A \hldef{$\varepsilon$-hermitian space} is a \CrefAndHyperrefIfExist{definition:varepsilon_hermitian_module_over_a_ring_with_involution}{$\varepsilon$-hermitian module} whose underlying Hermitian form is \CrefAndHyperrefIfExist{definition:nondegenerate_nonsingular_regular_sesquilinear_form_on_a_module_over_an_involution_ring}{nonsingular}.

        \item An \hldef{even $\varepsilon$-hermitian space} is a \CrefAndHyperrefIfExist{definition:varepsilon_hermitian_module_over_a_ring_with_involution}{$\varepsilon$-hermitian space} whose underlying Hermitian form is \CrefAndHyperrefIfExist{definition:even_varepsilon_hermitian_form_on_a_module_over_a_ring_with_involution}{even}.
    \end{enumerate}

    In each case, note that a ``space'' is simply a ``module'' whose underlying sesquilinear form is nonsingular.
    
\end{definition}
