\begin{definition} \label{definition:sesquilinear_form_associated_to_a_homomorphism_from_a_module_to_its_twisted_dual}
    Let $(A,\sigma)$ be a \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{ring with involution}, and let $M$ be a \CrefAndHyperrefIfExist{definition:module_of_a_ring}{left/right $A$-module}. 
    Let $h: M \to M^*$ be a left/right $A$-module homomorphism where $M^*$ is the \CrefAndHyperrefIfExist{definition:twisted_dual_module_of_a_module_over_a_ring_with_involution}{twisted dual module of $M$}. The \hldef{sesquilinear form associated to $h$} is the \CrefAndHyperrefIfExist{definition:sesquilinear_form_on_a_module_over_a_ring_with_involution}{sesquilinear form} \hl{$b_h: M \times M \to A$} given by 
    \begin{enumerate}
        \item $b_h(u,v) = h(v)(u)$ in the case that $M$ is a left $A$-module. 
        \item $b_h(u,v) = h(u)(v)$ in the case that $M$ is a right $A$-module. 
    \end{enumerate}
    It is indeed a sesquilinear form\CrefIfExists{lemma:sesquilinear_form_associated_to_a_homomorphsim_from_a_module_to_its_twisted_dual_is_indeeed_sesquilinear}.
\end{definition}
