\begin{definition}[Isometry for extended metric spaces] \label{definition:isometry_between_extended_metric_spaces_valued_in_ordered_commutative_monoid_with_adjoined_infinity}
Let $T$ be an \CrefAndHyperrefIfExist{definition:ordered_commutative_monoid}{ordered commutative monoid}, let $T^\infty$ be the \CrefAndHyperrefIfExist{definition:ordered_monoid_with_adjoined_infinity}{ordered monoid with adjoined infinity} obtained from $T$, and let $(X, d_X)$ and $(Y, d_Y)$ be \CrefAndHyperrefIfExist{definition:extended_metric_on_a_set_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity}{extended metric spaces valued in $T^\infty$}. A map $f : X \to Y$ is called an \hldef{isometry} if for all $x_1, x_2 \in X$,
\[
d_Y( f(x_1), f(x_2) ) = d_X( x_1, x_2 ).
\]
An isometry is automatically injective. If, in addition, $f$ is surjective, it is called an \hldef{isometric isomorphism}, and $X$ and $Y$ are said to be \hldef{isometric}.
\end{definition}
