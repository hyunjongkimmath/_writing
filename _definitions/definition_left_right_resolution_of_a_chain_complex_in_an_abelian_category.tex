\begin{definition} \label{definition:left_right_resolution_of_a_chain_complex_in_an_abelian_category}
    Let $\mathcal{A}$ be an \CrefAndHyperrefIfExist{definition:abelian_category}{abelian category}. 

    \begin{enumerate}
        \item ({\cite[After Theorem B]{spaltenstein_ruc}}) Let $f: A^\bullet \to B^\bullet$ be a \CrefAndHyperrefIfExist{definition:quasi_isomorphism_of_chain_complexes_of_objects_in_an_abelian_category}{quasi-isomorphism} of \CrefAndHyperrefIfExist{definition:chain_complex_of_objects_in_an_additive_category}{chain complexes} of objects in $\calA$. We say that $A^\bullet$ is a \hldef{left resolution of $B^\bullet$} and that $B^\bullet$ is a \hldef{right resolution of $A^\bullet$}.

        \item Given a \CrefAndHyperrefIfExist{definition:chain_complex_of_objects_in_an_additive_category}{chain complex} $K^\bullet$ of objects in $\calA$ and a class $\calY$ of chain complexes of objects in $\calA$, an \hldef{$\calY$-left resolution} may refer to a left resolution $Y^\bullet \to K^\bullet$ where $Y^\bullet \in \calY$ and an \hldef{$\calY$-right resolution} may refer to a right resolution $K^\bullet \to Y^\bullet$ where $Y^\bullet \in \calY$.

        % Given a chain complex $K^\bullet$ of objects in $\calA$ and a class $\calX$ of objects in $\calA$, an \hldef{$\calX$-left/right resolution} may refer to a $\calY$-left/right resolution in the above sense where $\calY$ is the class of chain complexes of objects in $\calA$.
        % $Y^\bullet \to K^\bullet$ where $Y^\bullet \in \calY$ and an \hldef{$\calY$-right resolution} may refer to a right resolution $K^\bullet \to Y^\bullet$ where $Y^\bullet \in \calY$.
    \end{enumerate}

\end{definition}
