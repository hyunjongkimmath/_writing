\begin{definition} \label{definition:semisimple_involution_ring}
    An involution ring $(R,\sigma)$ is said to be \hldef{semisimple (as an involution ring)} if the following equivalent conditions hold:
    \begin{enumerate}
        \item $(R,\sigma)$ is the direct sum of \CrefAndHyperref{definition:involution_simple_involution_ring}{$\sigma$-simple rings}.

        \item $(R,\sigma)$ is, as a left module of itself, isomorphic to a \CrefAndHyperrefIfExist{definition:coproduct_of_modules_of_rings}{direct sum} $\bigoplus I_i$ where each $I_i$ is a minimal, nonzero left \CrefAndHyperref{definition:involution_stable_ideal_of_an_involution_ring}{$\sigma$-ideal}. 

        \item $(R,\sigma)$ is, as a right module of itself, isomorphic to a direct sum $\bigoplus I_i$ where each $I_i$ is a minimal, nonzero right $\sigma$-ideal. 
    \end{enumerate}
\end{definition}
