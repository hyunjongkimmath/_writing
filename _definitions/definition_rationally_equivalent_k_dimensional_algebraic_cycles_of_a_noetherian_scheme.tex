Let $X$ be a \CrefAndHyperrefIfExist{definition:locally_noetherian_and_noetherian_scheme}{Noetherian scheme}. 
    \TODO{notion of cycle and divisor here}
Two \CrefAndHyperrefIfExist{definition:k_dimensional_algebraic_cycles_of_a_noetherian_scheme}{$k$-dimensional algebraic cycles} $\alpha, \beta \in Z_k(X)$ are said to be \hldef{rationally equivalent}, denoted \hl{$\alpha \sim \beta$}, if there exists a finite collection of $(k+1)$-dimensional \CrefAndHyperrefIfExist{definition:open_and_closed_subscheme_of_a_scheme}{closed} \CrefAndHyperrefIfExist{definition:integral_scheme}{integral} subschemes $W_j \subseteq X$ and non-zero rational functions $\phi_j \in K(W_j)^*$ such that $\alpha - \beta = \sum_j [\mathrm{div}(\phi_j)]$, where $[\mathrm{div}(\phi_j)]$ is the cycle associated to the divisor of the function $\phi_j$.
\end{definition}
