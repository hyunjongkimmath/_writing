\begin{definition} \label{definition:locally_compact_field}
    A \hldef{locally compact field} is a \CrefAndHyperrefIfExist{definition:topological_field}{topological field} whose underlying \CrefAndHyperrefIfExist{definition:topological_space}{topological space} is a \CrefAndHyperrefIfExist{definition:locally_compact_topological_space}{locally compact} \CrefAndHyperrefIfExist{definition:separation_axioms_of_topology}{Hausdorff space}. 
\end{definition}
