\begin{definition} \label{definition:paracompact_topological_space}
    A topological space is \hldef{paracompact} if it is \CrefAndHyperrefIfExist{definition:separation_axioms_of_topology}{Hausdorff} and every open cover of $X$ admits an \CrefAndHyperrefIfExist{definition:open_refinement_of_an_open_cover_of_a_topological_space}{open refinement} that is \CrefAndHyperrefIfExist{definition:locally_finite_open_cover_of_a_topological_spcae}{locally finite}.

    See \Cref{theorem:nice_enough_topological_spaces_are_paracompact}, which posits that nice enough topological spaces are paracompact.
\end{definition}
