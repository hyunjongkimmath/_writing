\begin{definition} \label{definition:associative_binary_operation_on_a_set}
    % EDITOR APPROVED
    Let $\cdot: S \times S \to S$ be a \CrefAndHyperrefIfExist{definition:n_ary_operation_on_a_set}{binary operation} on a \CrefAndHyperrefIfExist{definition:zermelo_fraenkel_set_theory}{set} $S$. It is said to be \hldef{associative} if 
    $$ (a \cdot b) \cdot c = a \cdot (b \cdot c) \quad \text{for all } a,b,c \in S.$$
\end{definition}
