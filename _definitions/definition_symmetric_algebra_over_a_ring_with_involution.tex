\begin{definition} \label{definition:symmetric_algebra_over_a_ring_with_involution}
    Let $R$ be an \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{involution ring}.
    A \CrefAndHyperrefIfExist{definition:symmetric_algebra_over_a_rnig}{symmetric $R$-algebra} $A$ is said to be a \hldef{symmetric $R$-algebra with involution} if 
    \begin{itemize}
        \item $A$ is finitely generated and projective as an $R$-module,
        \item $A$ has an \CrefAndHyperrefIfExist{definition:algebra_with_involution_over_a_ring_with_involution}{$R$-linear involution}, and 
        \item the \CrefAndHyperrefIfExist{definition:symmetric_algebra_over_a_rnig}{symmetric trace} $\varphi \in A^\vee$ (\Cref{definition:dual_of_a_left_right_two_sided_module}) is an \CrefAndHyperrefIfExist{definition:involution_trace_map_for_a_morphism_of_involution_rings}{involution trace}. 
    \end{itemize}
\end{definition}
