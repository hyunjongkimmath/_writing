\begin{definition} \label{definition:exponentiation_in_a_field}
    Let $F$ be a \CrefAndHyperrefIfExist{definition:field}{field}, and let $a \in F$. For each integer $n \in \mathbb{Z}$, the \hldef{$n$th power of $a$ in $F$}, denoted by \hl{$a^n$}, is defined as follows:
    \begin{itemize}
        \item If $a \neq 0_F$, then $a^n$ is the \CrefAndHyperrefIfExist{definition:exponentiation_in_a_ring}{$n$th power} of $a$ with respect to the underlying \CrefAndHyperrefIfExist{definition:ring}{ring structure} on $F$.
        \item If $a = 0_F$, then $0_F^n := 0_F$ for all $n > 0$, and by convention $0_F^0 := 1_F$. Negative powers of zero are undefined.
    \end{itemize}
\end{definition}
