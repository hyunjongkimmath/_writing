\begin{definition} \label{definition:principal_homogeneous_space_torsor_for_a_group}
    Let $G$ be a \CrefAndHyperrefIfExist{definition:group}{group}. 
    A \hldef{principal homogeneous space} or \hldef{torsor of $G$} is a set $X$ equipped with a
    \CrefAndHyperrefIfExit{definition:free_action_of_a_group_on_a_set}{free} and \CrefAndHyperrefIfExist{definition:transitive_group_action_of_a_group_acting_on_a_set}{transitive} group action.
\end{definition}
