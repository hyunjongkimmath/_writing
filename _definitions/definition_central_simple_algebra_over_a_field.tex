\begin{definition}[Central simple algebra over a field] \label{definition:central_simple_algebra_over_a_field}
    Let $F$ be a \CrefAndHyperrefIfExist{definition:field}{field}. A \hldef{central simple algebra over $F$} is an associative unital $F$-algebra $A$ such that $\dim_F A < \infty$, $A \neq 0$, and the following two conditions hold:
    \begin{enumerate}
        \item $A$ is a \CrefAndHyperrefIfExist{definition:simple_ring}{simple ring}, i.e., $A$ has no non-trivial two-sided ideals (the only two-sided ideals of $A$ are $\{0\}$ and $A$ itself).
        \item The center of $A$ is exactly $F$, meaning the structure map $F \to Z(A)$ induces an isomorphism between $F$ and the \CrefAndHyperrefIfExist{definition:center_of_a_ring}{center of $A$}. Equivalently, $A$ is a \CrefAndHyperrefIfExist{definition:central_algebra_over_a_commutative_ring}{central algebra} over $F$.
    \end{enumerate}

    The \hldef{degree} of a central simple algebra over $F$ is the integer $n = \sqrt{\dim_F A}$, so that $\dim_F A = n^2$.
\end{definition}
