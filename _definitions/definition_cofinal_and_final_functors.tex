\begin{definition}[cofinal and final functors]
\label{definition:cofinal_and_final_functors}
Let $\mathcal{I}$ and $\mathcal{J}$ be \CrefAndHyperrefIfExist{definition:category}{categories}, and let $u: \mathcal{I} \to \mathcal{J}$ be a functor.

\begin{enumerate}
    \item A \hldef{cofinal functor} is a functor $u: \mathcal{I} \to \mathcal{J}$ such that, for every object $j \in \mathcal{J}$, the \CrefAndHyperrefIfExist{definition:comma_category_of_two_functors_to_a_category}{comma category} $j \downarrow u$ is non-empty and a \CrefAndHyperrefIfExist{definition:connected_category}{connected category}.

    \item A \hldef{final functor} is a functor $u: \mathcal{I} \to \mathcal{J}$ such that, for every object $j \in \mathcal{J}$, the \CrefAndHyperrefIfExist{definition:comma_category_of_two_functors_to_a_category}{comma category} $u \downarrow j$ is non-empty and a \CrefAndHyperrefIfExist{definition:connected_category}{connected category}.
\end{enumerate}
