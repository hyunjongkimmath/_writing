\begin{definition} \label{definition:morphism_of_n_tuples_of_topological_spaces}
    A \hldef{morphism $f: (X_1, \dots, X_n) \to (Y_1, \dots, Y_n)$ of $n$-tuples of topological spaces} is a continuous map $f: X_1 \to Y_1$ that constitutes a map $(X_i, X_{i+1}) \to (Y_i, Y_{i+1})$ \CrefAndHyperrefIfExist{definition:map_of_pairs_of_topological_spaces}{of pairs} for each $i \in \{1,\ldots,n-1\}$. In other words, $f$ is a continuous map $X_1 \to Y_1$ such that $f(X_i) \subseteq Y_i$ for all $i = 1, \dots, n$.
\end{definition}
