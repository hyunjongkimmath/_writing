\begin{definition}  \label{definition:suspension_spectrum_of_a_pointed_topological_space}
    The \hldef{suspension spectrum of a pointed topological space $X$} is the \CrefAndHyperrefIfExist{definition:spectrum_of_topological_spaces}{spectrum} $\Sigma^\infty X = \Sigma^\infty(X)$ whose degree $n$ space is $\Sigma^n X$ (\Cref{definition:reduced_suspension_of_a_pointed_topological_space}). 
\end{definition}
