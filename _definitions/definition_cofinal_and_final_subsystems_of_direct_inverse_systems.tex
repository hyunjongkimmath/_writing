\begin{definition}[Cofinal and final subsystems of direct/inverse systems] \label{definition:cofinal_and_final_subsystems_of_direct_inverse_systems}

Let $\mathcal{C}$ be a \CrefAndHyperrefIfExist{definition:category}{category}.
\begin{enumerate}
    \item Let $(I, \leq)$ be a \CrefAndHyperrefIfExist{definition:partially_ordered_set}{directed partially ordered set}, and let $(A_i)_{i \in I}$ be a \CrefAndHyperref{definition:system_in_a_category_indexed_by_a_directed_poset}{direct system} in $\mathcal{C}$ indexed by $I$. Let $J \subseteq I$ be a sub-poset that is \CrefAndHyperref{definition:cofinal_and_final_sub_posets}{cofinal in $I$}. The restriction of the direct system to $J$, denoted $(A_j)_{j \in J}$, is called a \hldef{cofinal subsystem} of $(A_i)_{i \in I}$.

    \item Let $(K, \leq)$ be a \CrefAndHyperrefIfExist{definition:partially_ordered_set}{codirected partially ordered set}, and let $(B_k)_{k \in K}$ be an \CrefAndHyperref{definition:system_in_a_category_indexed_by_a_directed_poset}{inverse system} in $\mathcal{C}$ indexed by $K$. Let $L \subseteq K$ be a sub-poset that is \CrefAndHyperref{definition:cofinal_and_final_sub_posets}{final in $K$}. The restriction of the inverse system to $L$, denoted $(B_\ell)_{\ell \in L}$, is called a \hldef{final subsystem} of $(B_k)_{k \in K}$.
\end{enumerate}
\end{definition}
