\begin{definition} \label{definition:involution_stable_ideal_of_an_involution_ring}
    Let $(R,\sigma)$ be an \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{involution ring}. 
    A \CrefAndHyperrefIfExist{definition:ideal_of_a_ring}{left/right/two-sided ideal} $I$ of $R$ is said to be \hldef{involution stable}, \hldef{$*$-stable}, or \hldef{$\sigma$-stable} if $\sigma(I) = I$. 
\end{definition}
