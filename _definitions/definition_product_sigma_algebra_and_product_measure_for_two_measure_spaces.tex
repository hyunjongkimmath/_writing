\begin{definition}[Product $\sigma$-algebra and product measure] \label{definition:product_sigma_algebra_and_product_measure_for_two_measure_spaces}
Let $(X_1, \mathcal{A}_1)$ and $(X_2, \mathcal{A}_2)$ be two \CrefAndHyperrefIfExist{definition:sigma_algebra_on_a_set_and_measurable_space}{measurable spaces}.

\begin{enumerate}
    \item The \hldef{product $\sigma$-algebra on $X_1 \times X_2$} is the \CrefAndHyperrefIfExist{definition:sigma_algebra_on_a_set_and_measurable_space}{$\sigma$-algebra} generated by the measurable rectangles:
    $$\hlin{\mathcal{A}_1 \otimes \mathcal{A}_2 := \sigma(\{ A_1 \times A_2 : A_1 \in \mathcal{A}_1,\ A_2 \in \mathcal{A}_2\}).}$$
    Elements of $\mathcal{A}_1 \otimes \mathcal{A}_2$ are called \hldef{measurable rectangles} (when they are of the form $A_1 \times A_2$) or more generally \hldef{sets in the product $\sigma$-algebra}.

    \item Now suppose that $(X_1, \mathcal{A}_1, \mu_1)$ and $(X_2, \mathcal{A}_2, \mu_2)$ are two \CrefAndHyperrefIfExist{definition:measure_on_a_measurable_space_valued_in_the_extended_real_line_and_measure_space}{measure spaces}, both $\sigma$-finite. The \hldef{product measure on $(X_1 \times X_2, \mathcal{A}_1 \otimes \mathcal{A}_2)$} is the unique measure $\mu_1 \times \mu_2$ satisfying
    $$(\mu_1 \times \mu_2)(A_1 \times A_2) = \mu_1(A_1) \cdot \mu_2(A_2)$$
    for all $A_1 \in \mathcal{A}_1$ and $A_2 \in \mathcal{A}_2$. Existence follows from the Carathéodory extension theorem; uniqueness follows from the $\pi$--$\lambda$ theorem (or Dynkin's lemma), since the measurable rectangles form a $\pi$-system generating $\mathcal{A}_1 \otimes \mathcal{A}_2$.
\end{enumerate}

These constructions extend inductively to finite products $(X_1 \times \cdots \times X_n, \mathcal{A}_1 \otimes \cdots \otimes \mathcal{A}_n, \mu_1 \times \cdots \times \mu_n)$ and, with appropriate modifications, to countably infinite products.
\end{definition}
