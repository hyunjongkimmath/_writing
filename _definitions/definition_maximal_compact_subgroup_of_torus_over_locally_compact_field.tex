\begin{definition}[Maximal compact subgroup of a torus over a locally compact field {\cite[2.1]{ono_aat}}, cf. {\cite[2.1]{lombardo_graavCMt}}] \label{definition:maximal_compact_subgroup_of_torus_over_locally_compact_field}
    Let $k$ be a \CrefAndHyperrefIfExist{definition:locally_compact_field}{locally compact field} whose topology is defined by a non-trivial \CrefAndHyperrefIfExist{definition:absolute_value_on_a_ring}{absolute value}, and let $\calO_v$ be the \CrefAndHyperrefIfExist{definition:ring_of_integers_of_a_global_or_local_ring}{ring of integers} of $k$. Let $T/k$ be a \CrefAndHyperrefIfExist{definition:algebraic_torus_over_a_field}{torus}, and let $K/k$ be a finite Galois \CrefAndHyperrefIfExist{definition:algebraic_torus_over_a_field}{splitting field} for $T$ with Galois group $\Gamma = \Gal(K/k)$. Let $V$ be the unique extension of the absolute value to $K$, and let $\calO_V^\times$ be the unit group of the valuation ring of $(K,V)$, equipped with the natural continuous $\Gamma$-action. Write $\widehat{T}$ for the \CrefAndHyperrefIfExist{definition:character_module_of_a_torus_over_a_field}{character module of $T$} with its $\Gamma$-action.

    The \hldef{maximal compact subgroup of $T(k)$}, written \hl{$T(\calO_v)$}, is the subgroup
    $$T(\calO_v) := \varphi^{-1}\bigl(\Hom_\Gamma(\widehat{T}, \calO_V^\times)\bigr)$$
    where $\varphi: T(K) \xrightarrow{\sim} \Hom_{\mathrm{ab. gr.}}(\widehat{T}, K^\times)$ is the duality isomorphism of \CrefAndHyperrefIfExist{definition:character_module_of_a_torus_over_a_field}{the character module}. Equivalently, using that $\Gamma$ acts trivially on $\calO_v^\times \subset \calO_V^\times$, one has
    $$T(\calO_v) = \varphi^{-1}\bigl(\Hom_\Gamma(\widehat{T}, \calO_v^\times)\bigr)$$
    (see \CrefAndHyperrefIfExist{theorem:ono_2.1.3_equivalence_of_integral_torus_descriptions}{Ono's characterization}). In pointwise terms,
    $$T(\calO_v) = \{x \in T(k) : \chi(x) \in \calO_V^\times \text{ for all } \chi \in \widehat{T}\}.$$
\end{definition}
