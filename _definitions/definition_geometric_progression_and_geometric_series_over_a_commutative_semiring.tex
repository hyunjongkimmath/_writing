\begin{definition}[Geometric progression and geometric series over a commutative semiring] \label{definition:geometric_progression_and_geometric_series_over_a_commutative_semiring}
    \TODO{AI generated, verify}
    Let $S$ be a \CrefAndHyperrefIfExist{definition:semiring}{commutative semiring}. Let $a \in S$ and $r \in S$. For each integer $n \geq 1$, let
    $$a_n = ar^{n-1},$$
    \CrefIfExists{definition:exponentiation_in_a_semiring}
    The sequence $(a_n)_{n=1}^\infty$ is called a \hldef{geometric progression over $S$} with \hldef{first term} $a$ and \hldef{common ratio} $r$.

    For each integer $N \geq 1$, the finite series $\sum_{n=1}^N ar^{n-1}$ is called a \hldef{finite geometric series of length $N$ over $S$}.

\end{definition}
