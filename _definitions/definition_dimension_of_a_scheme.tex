\begin{definition}[Dimension of a Scheme] \label{definition:dimension_of_a_scheme}
Let $X$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme} with underlying topological space $|X|$.
\begin{itemize}
    \item The \hldef{dimension at a point $x \in |X|$}, denoted $\dim_x(X)$, is the \CrefAndHyperrefIfExist{definition:krull_dimension_of_a_commutative_ring}{Krull dimension} of the \CrefAndHyperrefIfExist{definition:local_ring}{local ring} \CrefAndHyperrefIfExist{definition:stalk_of_a_presheaf_on_a_topological_space_at_a_point}{$\mathcal{O}_{X,x}$}, which is local by definition\CrefIfExists{definition:locally_ringed_space_on_a_topological_space}. This is the supremum of the lengths $n$ of chains of prime ideals
    $$ \mathfrak{p}_0 \subsetneq \mathfrak{p}_1 \subsetneq \cdots \subsetneq \mathfrak{p}_n \subseteq \mathcal{O}_{X,x}.  $$

    \item The \hldef{dimension of the scheme $X$} is defined as
    $$ \hlin{\dim(X) := \sup_{x \in |X|} \dim_x(X). }$$
    Equivalently, it is the supremum of the lengths of chains of distinct irreducible closed subsets of $|X|$ ordered by inclusion. It is also equivalent to the \CrefAndHyperrefIfExist{definition:dimension_of_a_topological_space}{dimension} of the underlying topological space $|X|$.
\end{itemize}
\end{definition}
