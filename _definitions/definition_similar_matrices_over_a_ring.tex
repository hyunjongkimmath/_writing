\begin{definition} \label{definition:similar_matrices_over_a_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{(not necessarily commutative) ring}. Two matrices $A, B \in M_n(R)$\CrefIfExists{definition:matrix_over_a_ring} are called \hldef{similar over $R$} (or \hldef{conjugate over $R$}) if there exists an \CrefAndHyperrefIfExist{definition:invertible_matrix_over_a_ring}{invertible matrix} $P \in \GL_n(R)$\CrefIfExists{definition:general_linear_group_over_a_ring} such that
    $$B = P^{-1} A P.$$
    \TextIfExists{definition:homomorphism_of_modules_over_a_ring}{If $M$ is a \CrefAndHyperrefIfExist{definition:free_bimodule_over_rings}{free} $R$-module of finite rank and $T: M \to M$ is an \CrefAndHyperrefIfExist{definition:homomorphism_of_modules_over_a_ring}{$R$-linear} endomorphism, then the \CrefAndHyperrefIfExist{definition:matrix_representation_of_an_endomorphism_of_a_free_module_with_respect_to_a_basis}{matrices representing} $T$ with respect to two different ordered bases of $M$ are similar. Conversely, any two similar matrices arise as matrix representations of the same endomorphism with respect to different bases\CrefIfExists{definition:change_of_basis_matrix_for_a_free_module_over_a_ring}.}
\end{definition}
