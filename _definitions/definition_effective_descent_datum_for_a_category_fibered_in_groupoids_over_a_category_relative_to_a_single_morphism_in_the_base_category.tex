\begin{definition} \label{definition:effective_descent_datum_for_a_category_fibered_in_groupoids_over_a_category_relative_to_a_single_morphism_in_the_base_category}
    Let $\mathcal{C}$ be a category and let $p: \mathcal{F} \to \mathcal{C}$ be a \CrefAndHyperrefIfExist{definition:category_fibered_in_groupoids_over_a_category}{category fibered in groupoids}. We fix a \CrefAndHyperrefIfExist{definition:cleavage_splitting_for_a_category_fibered_in_groupoids}{cleavage} for $p$, denoting the chosen pullback functor along a morphism $f: V \to U$ by $f^*: \mathcal{F}_U \to \mathcal{F}_V$, and the structural natural isomorphisms for composition by $\alpha_{g,h}: (g \circ h)^* \xrightarrow{\sim} h^* \circ g^*$ for morphisms $W \xrightarrow{h} V \xrightarrow{g} U$.

    Let $\pi: U' \to U$ be a morphism in $
    \calC$ such that $U' \times_U U'$ \CrefIfExists{definition:cartesian_product_of_two_objects_in_a_category_over_an_object} exists in $\calC$. 
    % and let $S_\pi$ be the \CrefAndHyperrefIfExist{definition:sieve_on_an_object_in_a_category}{sieve on $U$} \CrefAndHyperrefIfExist{definition:sieve_on_an_object_of_a_category_generated_by_a_family_of_morphisms}{generated by} $\pi$. 
    Write $\pi_1,\pi_2: U' \times_U U' \rightrightarrows U'$ for the two canonical projections.

    A descent datum $(X',\phi)$ of $\calF$ relative to $\pi$ is \hldef{effective} if there exists an object $y \in \calF_U$ and an isomoprhism $\alpha: X' \xrightarrow{\sim} \pi^*(y)$ in $\calF_U$ such that the following diagram in $\calF_{U' \times_U U'}$ commutes:

    $$
    \begin{tikzcd}[row sep=large, column sep=large]
    \pi_1^* X' \arrow[r, "\phi"] \arrow[d, "\pi_1^*(\alpha)"'] & \pi_2^* X' \arrow[d, "\pi_2^*(\alpha)"] \\
    \pi_1^* \pi^* y \arrow[r, "\text{can}"] & \pi_2^* \pi^* y.
    \end{tikzcd}
    $$

    Equvialently, a descent datum relative to $\pi$ is effective if it is in the effective image of the \CrefAndHyperrefIfExist{definition:descent_datum_associated_to_an_object_in_a_category_fibered_in_groupoids_over_a_category_with_respect_to_a_single_morphism}{canonical functor}  
    $$\Psi:\calF_U \to \operatorname{Des}(\calF, \pi)$$

    \TextIfExists{definition:effective_descent_datum_for_a_category_fibered_in_groupoids_over_a_category_relative_to_a_sieve_on_an_object_of_the_base_category}{
        Moreover, this notion of effectiveness is equivalent to the notion of effectiveness in \Cref{definition:effective_descent_datum_for_a_category_fibered_in_groupoids_over_a_category_relative_to_a_sieve_on_an_object_of_the_base_category}, where the site is the site on $U$ \CrefAndHyperrefIfExist{definition:sieve_on_an_object_of_a_category_generated_by_a_family_of_morphisms}{generated by} $\pi$, see \Cref{theorem:descent_datum_for_a_category_fibered_in_groupoids_over_a_category_with_respect_to_a_single_morphism_is_eq_cat_of_sieve_gen_by_morphism}.
    }

\end{definition}
