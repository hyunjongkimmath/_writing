\begin{definition}[Relative ampleness for a line bundle on a scheme] \label{definition:relative_ampleness_of_a_line_bundle_on_a_scheme}
    Let $f\colon X \to S$ be a \CrefAndHyperrefIfExist{definition:proper_morphism_of_schemes}{proper morphism} of \CrefAndHyperrefIfExist{definition:scheme}{schemes} and let $\mathcal{L}$ be a \CrefAndHyperrefIfExist{definition:free_and_locally_free_modules_on_a_ringed_site}{line bundle} (i.e., an \CrefAndHyperrefIfExist{definition:free_and_locally_free_modules_on_a_ringed_site}{invertible sheaf}) on $X$.
    The line bundle $\mathcal{L}$ is called \hldef{$f$-ample} (or \hldef{relatively ample with respect to $f$}) if for every \CrefAndHyperrefIfExist{definition:quasi_coherent_sheaf_on_a_general_scheme}{quasi-coherent sheaf} $\mathcal{F}$ on $S$, the natural map
    $$\operatorname{Sym}(\mathbf{R}^0 f_* (\mathcal{L}^{\otimes n})) \to \mathbf{R}^0 f_* (\mathcal{L}^{\otimes n} \otimes \mathcal{G})$$
    \TODO{usual sense of ampleness}
    is an isomorphism for all quasi-coherent sheaves $\mathcal{G}$ and all $n \gg 0$. Equivalently, $\mathcal{L}$ is $f$-ample if its restriction to every fiber $X_s = X \times_S \operatorname{Spec}(\kappa(s))$ is ample in the usual sense for every point $s \in S$.
\end{definition}
