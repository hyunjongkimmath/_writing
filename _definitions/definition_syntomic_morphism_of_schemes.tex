\begin{definition}[Syntomic morphism of schemes] \label{definition:syntomic_morphism_of_schemes}

    Let $f: X \to Y$ be a \CrefAndHyperrefIfExist{definition:morphism_of_schemes}{morphism of schemes}.

    The morphism $f$ is called \hldef{syntomic} if it satisfies the following conditions:
    \begin{itemize}
        \item $f$ is \CrefAndHyperrefIfExist{definition:flat_morphism_of_schemes}{flat},
        \item $f$ is \CrefAndHyperrefIfExist{definition:locally_of_finite_presentation_finite_presentation_morphism_of_schemes}{locally of finite presentation},
        \item $f$ is \CrefAndHyperrefIfExist{definition:unramified_morphism_of_schemes}{unramified}.
    \end{itemize}

    Equivalently, $f$ is syntomic if and only if it is flat, locally of finite presentation, and the sheaf of relative Kähler differentials $\Omega_{X/Y}$ vanishes.
\end{definition}
