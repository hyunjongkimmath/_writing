\begin{definition}[Random variable on a probability measure space] \label{definition:random_variable_on_a_probability_measure_space}
Let $(\Omega, \mathcal{F}, P)$ be a \CrefAndHyperref{definition:probability_measure_space_for_a_measure_on_a_measurable_space}{probability measure space}.

A function $X : \Omega \to S$ into a measurable space $(S, \mathcal{S})$ is called a \hldef{$(S, \mathcal{S})$-valued random variable on $(\Omega, \mathcal{F}, P)$} if it is a \CrefAndHyperref{definition:measurable_function_from_a_measure_space_into_the_extended_real_line_or_complex_plane}{measurable function} from $(\Omega, \mathcal{F})$ into $(S, \mathcal{S})$. That is, $X^{-1}(A) \in \mathcal{F}$ for every $A \in \mathcal{S}$.

Special cases:
\begin{enumerate}
    \item A function $X : \Omega \to \mathbb{R}$ is called a \hldef{real-valued random variable on $(\Omega, \mathcal{F}, P)$} if it is $(\mathbb{R}, \mathcal{B}(\mathbb{R}))$-valued, where $\mathcal{B}(\mathbb{R})$ is the Borel $\sigma$-algebra on $\mathbb{R}$ generated by the open intervals. That is, $X^{-1}(B) \in \mathcal{F}$ for every Borel set $B \subseteq \mathbb{R}$.

    \item A function $X : \Omega \to \mathbb{C}$ is called a \hldef{complex-valued random variable on $(\Omega, \mathcal{F}, P)$} if it is $(\mathbb{C}, \mathcal{B}(\mathbb{C}))$-valued, where $\mathcal{B}(\mathbb{C})$ is the Borel $\sigma$-algebra on $\mathbb{C}$ generated by the open disks. That is, $X^{-1}(B) \in \mathcal{F}$ for every Borel set $B \subseteq \mathbb{C}$.

    \item A function $X : \Omega \to \overline{\mathbb{R}} = \mathbb{R} \cup \{-\infty, \infty\}$ is called a \hldef{random variable taking values in the extended real line on $(\Omega, \mathcal{F}, P)$} if it is $(\overline{\mathbb{R}}, \mathcal{B}(\overline{\mathbb{R}}))$-valued, where $\mathcal{B}(\overline{\mathbb{R}})$ is the Borel $\sigma$-algebra on $\overline{\mathbb{R}}$ generated by the intervals $(a, \infty]$. That is, $X^{-1}(B) \in \mathcal{F}$ for every Borel set $B \subseteq \overline{\mathbb{R}}$.
\end{enumerate}

A function $X$ as above ($(S, \mathcal{S})$-valued, real-valued, complex-valued, or extended real-valued) is called simply a \hldef{random variable on $(\Omega, \mathcal{F}, P)$} when the codomain and $\sigma$-algebra are clear from context.
\end{definition}
