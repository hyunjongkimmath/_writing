% P-ADIC INTEGERS FOR A NONZERO PRIME OF A NUMBER FIELD - Definition of the ring of p-adic integers for a nonzero prime of a number field
\begin{definition}\label{definition:p_adic_integers_for_a_nonzero_prime_of_a_number_field}
    Let $K$ be a \CrefAndHyperrefIfExist{definition:number_field}{number field} with \CrefAndHyperrefIfExist{definition:ring_of_integers_of_a_global_or_local_ring}{ring of integers} $\calO_K$. Let $\mathfrak{p}$ be a nonzero \CrefAndHyperrefIfExist{definition:prime_and_maximal_ideal_of_a_ring}{prime ideal} of $\calO_K$.

    The \hldef{$\mathfrak{p}$-adic integers of $K$}, denoted \hl{$\calO_{K,\mathfrak{p}}$}, is the \CrefAndHyperrefIfExist{definition:completion_of_a_metric_space}{completion} of $\calO_K$ with respect to the \CrefAndHyperrefIfExist{definition:metric_induced_by_an_absolute_value_on_a_field}{metric induced by} the \CrefAndHyperrefIfExist{definition:p_adic_valuation_and_absolute_value_on_a_number_field}{$\mathfrak{p}$-adic absolute value}. Equivalently, it is the set
    $$
        \calO_{K,\mathfrak{p}} := \left\{ (x_n)_{n=1}^\infty \in \prod_{n=1}^\infty \calO_K/\mathfrak{p}^n : x_{n+1} \equiv x_n \pmod{\mathfrak{p}^n} \text{ for all } n \geq 1 \right\},
    $$
    where the operations are defined componentwise. This is a complete \CrefAndHyperrefIfExist{definition:discrete_valuation_ring}{discrete valuation ring}.

    In the special case where $K = \mathbb{Q}$ and $\mathfrak{p} = (p)$ for a rational prime $p$, we write $\calO_{\mathbb{Q},(p)} = \mathbb{Z}_p$ and call it the ring of \hl{$p$-adic integers}.
\end{definition}
