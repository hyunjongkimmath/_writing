\begin{definition}[{\cite[Definition 1.2.12.1]{lurie_htt}}] \label{definition:final_object_of_a_topological_or_simplicial_category_or_simplicial_set}
Let $\mathcal{C}$ be a \CrefAndHyperref{definition:topological_category}{topological category} (or a \CrefAndHyperref{definition:simplicial_category}{simplicial category}, or a \CrefAndHyperrefIfExist{definition:simplicial_set}{simplicial set}). An object $X \in \mathcal{C}$ is called \hldef{final} if it is \CrefAndHyperrefIfExist{definition:initial_and_final_object_of_a_V_enriched_category_in_a_monoidal_category}{final} in the \CrefAndHyperref{definition:homotopy_category_of_a_topological_category}{homotopy category} $\mathrm{h}\mathcal{C}$ of a category, regarded as a category \CrefAndHyperrefIfExist{definition:category_enriched_in_a_monoidal_category}{enriched over} the \CrefAndHyperref{definition:homotopy_category_of_spaces}{homotopy category of spaces} $\mathcal{H}$. In other words, $X$ is final if and only if for each object $Y \in \mathcal{C}$, the mapping space $\mathrm{Map}_{\mathrm{h}\mathcal{C}}(Y,X)$ is \CrefAndHyperref{definition:weakly_contractible_simplicial_set}{weakly contractible} (that is, a final object of $\mathcal{H}$).

\end{definition}
