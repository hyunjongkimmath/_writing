For $x \in \mathbb{R}$, the \hldef{sine function} and \hldef{cosine function}, denoted by \hl{$\sin : \mathbb{R} \to \mathbb{R}$} and \hl{$\cos : \mathbb{R} \to \mathbb{R}$}, are defined by the power series
$$\sin(x) = \sum_{n=0}^\infty (-1)^n \frac{x^{2n+1}}{(2n+1)!}$$
$$\cos(x) = \sum_{n=0}^\infty (-1)^n \frac{x^{2n}}{(2n)!}$$
Both series \CrefAndHyperrefIfExist{definition:absolute_convergence_of_a_series_in_module_over_a_commutative_ring_with_an_absolute_value_valued_in_an_ordered_semiring}{converge absolutely} for every $x \in \mathbb{R}$. As with other trigonometric functions, exponents of $\sin$ and $\cos$ are written in the following manner:
\begin{align*}
    \sin^n(x) &\coloneq (\sin(x))^n
    \cos^n(x) &\coloneq (\cos(x))^n
\end{align*}

Key properties:
\begin{enumerate}
    \item $\sin^2(x) + \cos^2(x) = 1$ for all $x \in \mathbb{R}$ (Pythagorean identity).
    \item $\cos(0) = 1$, $\sin(0) = 0$. The set $\{x > 0 : \cos(x) = 0\}$ has a unique minimum; we denote it by $\pi/2$ (see \Cref{definition:archimedes_constant_pi_via_cosine_root}).
    \item Addition formulas: $\sin(x+y) = \sin x \cos y + \cos x \sin y$ and $\cos(x+y) = \cos x \cos y - \sin x \sin y$.
    \item Both functions are $2\pi$-periodic: $\sin(x + 2\pi) = \sin(x)$ and $\cos(x + 2\pi) = \cos(x)$.
\end{enumerate}
\end{definition}
