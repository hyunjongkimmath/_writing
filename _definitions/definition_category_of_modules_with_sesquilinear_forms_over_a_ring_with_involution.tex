\begin{definition} \label{definition:category_of_modules_with_sesquilinear_forms_over_a_ring_with_involution}
    Let $(A, \sigma)$ be a \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{ring with involution}.

    The \hldef{category of modules with sesquilinear forms over $(A,\sigma)$} is the \CrefAndHyperrefIfExist{definition:category}{category} whose objects are pairs $(M,b)$ where $M$ is a \CrefAndHyperrefIfExist{definition:module_of_a_ring}{(right) $A$-module} and $b: M \times M \to A$ is a \CrefAndHyperrefIfExist{definition:sesquilinear_form_on_a_module_over_a_ring_with_involution}{sesquilinear form} and whose morphisms are \CrefAndHyperrefIfExist{definition:isometry_between_modules_with_sesquilinear_forms_over_ring_with_involution}{isometries}.

    In particular, such a category is a \CrefAndHyperrefIfExist{definition:groupoid}{groupoid}.
\end{definition}
