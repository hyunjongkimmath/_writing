\begin{definition} \label{definition:character_lattice_of_a_torus_over_a_field}
Let $k$ be a \CrefAndHyperrefIfExist{definition:field}{field}. Fix a \CrefAndHyperrefIfExist{definition:separable_closure_of_a_field}{separable closure} $\overline{k}$ of $k$. Let $T/k$ be an \CrefAndHyperrefIfExist{definition:algebraic_torus_over_a_scheme}{algebraic torus}. 
\begin{enumerate}
  \item The \hldef{character lattice of $T$} is the \CrefAndHyperrefIfExist{definition:lattice_over_a_commutative_ring}{lattice} \hl{$X^\ast(T) = \mathrm{Hom}_{\overline{k}\text{-groups}}(T_{\bark},\mathbb{G}_{m,\bark})$}.
  \item The \hldef{cocharacter lattice of $T$} is the lattice \hl{$X_\ast(T) = \mathrm{Hom}_{\overline{k}\text{-groups}}(\mathbb{G}_{m,\bark}, T_{\bark})$}.
\end{enumerate}
\end{definition}
