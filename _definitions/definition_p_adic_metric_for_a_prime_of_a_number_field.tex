% P-ADIC METRIC - Definition of the p-adic metric on a number field
\begin{definition}\label{definition:p_adic_metric_for_a_prime_of_a_number_field}
    Let $K$ be a \CrefAndHyperrefIfExist{definition:number_field}{number field} with \CrefAndHyperrefIfExist{definition:ring_of_integers_of_a_global_or_local_ring}{ring of integers} $\calO_K$. Let $\mathfrak{p}$ be a nonzero \CrefAndHyperrefIfExist{definition:prime_and_maximal_ideal_of_a_ring}{prime ideal} of $\calO_K$, and let $|\cdot|_\mathfrak{p}$ denote the \CrefAndHyperrefIfExist{definition:p_adic_valuation_and_absolute_value_on_a_number_field}{$\mathfrak{p}$-adic absolute value on $K$}.

    The \hldef{$\mathfrak{p}$-adic metric} (or \hldef{$\mathfrak{p}$-adic distance}) on $K$, denoted \hl{$d_\mathfrak{p}$}, refers to the \CrefAndHyperrefIfExist{definition:metric_induced_by_an_absolute_value_on_a_field}{metric induced by} $|\cdot|_{\mathfrak{p}}$, i.e.
    $$
        d_\mathfrak{p}(x, y) := |x - y|_\mathfrak{p}
    $$
    for all $x, y \in K$. This is a \CrefAndHyperrefIfExist{definition:archimedean_and_nonarchimedean_metrics_on_sets}{non-archimedean metric}.

\end{definition}
