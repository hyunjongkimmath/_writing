\begin{definition}[Verdier dual of a constructible complex with algebraic coefficients] \label{definition:verdier_dual_of_an_object_in_the_bdd_constr_derived_category_of_adic_sheaves_with_algebraic_coefficients_on_a_finite_type_scheme_over_a_noetherian_scheme}
    Let $(R, \lambda)$ be a \CrefAndHyperrefIfExist{definition:complete_ring_with_respect_to_an_ideal}{complete} \CrefAndHyperrefIfExist{definition:discrete_valuation_ring}{DVR} such that $R/\lambda^n$ is finite for all $n \geq 1$. Let $E$ be the \CrefAndHyperrefIfExist{definition:field_of_fractions_of_an_integral_domain}{fraction field} of $R$, and let $K/E$ be a \CrefAndHyperrefIfExist{definition:algebraic_element_minimal_polynomial_of_an_algebraic_element_algebraic_field_extension_transcendental_element_field_extension}{algebraic extension}. Let $S$ be a \CrefAndHyperrefIfExist{definition:locally_noetherian_and_noetherian_scheme}{noetherian scheme}. Let $X$ be an $S$-scheme of \CrefAndHyperrefIfExist{definition:finite_type_morphism_of_schemes}{finite type}.  Consider sheaves on the (small) \'etale sites of $X$ and $S$.

    For any object $M \in \Dbc(X, K)$, the \hldef{Verdier dual of $M$ on $X$ relative to $S$ with algebraic coefficients} is
    $$\mathbb{D}_X(M) := R\mathscr{H}om_K(M, \mathbb{D}_{X/S}^\bullet(K)).$$
    \CrefIfExists{definition:derived_internal_hom_on_bdd_constructible_complexes_with_algebraic_coefficients_on_a_noetherian_scheme} \CrefIfExists{definition:dualizing_complex_for_verdier_duality_in_the_bdd_constr_derived_category_of_adic_sheaves_with_algebraic_coefficients_on_a_finite_type_scheme_over_a_noetherian_scheme}
\end{definition}
