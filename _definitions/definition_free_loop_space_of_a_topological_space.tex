\begin{definition}[Free loop space of a topological space] \label{definition:free_loop_space_of_a_topological_space}
    Let $Y$ be a \CrefAndHyperrefIfExist{definition:topological_space}{topological space}. 

    The \hldef{free loop space}, denoted \hl{$\mathcal{L}Y$}, is the \CrefAndHyperrefIfExist{definition:compact_open_topology_on_the_set_of_continuous_maps_between_topological_spaces_and_function_space}{mapping space} $\operatorname{Map}(S^1, Y)$ of maps from the circle to $Y$ with no basepoint restriction.
\end{definition}
