\begin{definition} \label{definition:strict_morphism_of_filtered_objects_in_an_abelian_category}
Let $\mathcal{A}$ be an \CrefAndHyperrefIfExist{definition:abelian_category}{abelian category}. Let $f: (M, F) \to (N, G)$ be a \CrefAndHyperrefIfExist{definition:morphism_of_filtered_objects_in_a_category}{morphism of filtered objects}.
\begin{enumerate}
    \item In the case that $f$ is a morphism of filtered objects with \CrefAndHyperrefIfExist{definition:filtration_on_an_object_of_a_category}{increasing filtrations}, it is said to be a \hldef{strict morphism} if for every $p \in \mathbb{Z}$, the following equality of subobjects of $N$ holds:
    $$f(M) \cap G_p N = f(F_p M)$$
    (\Cref{definition:image_coimage_of_a_morphism_in_a_category}) (\Cref{definition:intersection_of_subobjects_of_an_object_in_a_category})

    \item In the case that $f$ is a morphism of filtered objects with decreasing filtrations, it is said to be a \hldef{strict morphism} if for every $p \in \mathbb{Z}$, the following equality of subobjects of $N$ holds:
    $$f(M) \cap G^p N = f(F^p M)$$
\end{enumerate}
\end{definition}
