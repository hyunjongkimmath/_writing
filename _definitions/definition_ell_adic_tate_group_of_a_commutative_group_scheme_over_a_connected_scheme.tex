\begin{definition}[$\ell$-adic Tate group] \label{definition:ell_adic_tate_group_of_a_commutative_group_scheme_over_a_connected_scheme}
Let $S$ be a \CrefAndHyperrefIfExist{definition:disconnected_and_connected_topological_spaces}{connected} \CrefAndHyperrefIfExist{definition:scheme}{scheme} with a \CrefAndHyperrefIfExist{definition:geometric_point_of_a_scheme}{geometric point} $\bar{s}: \operatorname{Spec}(k(\bar{s})) \to S$. Let $f: \mathcal{A} \to S$ be a \CrefAndHyperrefIfExist{definition:algebraic_group_scheme_over_a_scheme}{commutative group scheme} of \CrefAndHyperrefIfExist{definition:finite_type_morphism_of_schemes}{finite type}, and let $\ell$ be a prime number. For each integer $\nu \geq 1$, let $\mathcal{A}_{\bar{s}}[\ell^\nu](k(\bar{s}))$ denote the group of \CrefAndHyperrefIfExist{definition:rational_point_on_a_scheme_over_a_field}{$k(\bar{s})$-rational points} of the \CrefAndHyperrefIfExist{definition:kernel_of_a_homomorphism_of_group_schemes_over_a_scheme}{kernel} of $[\ell^\nu]: \mathcal{A}_{\bar{s}} \to \mathcal{A}_{\bar{s}}$, where $\mathcal{A}_{\bar{s}} = \mathcal{A} \times_S \operatorname{Spec}(k(\bar{s}))$ is the \CrefAndHyperrefIfExist{definition:geometric_fiber_of_a_morphism_of_schemes_at_a_geometric_point}{geometric fiber}.

The \hldef{$\ell$-adic Tate group of $\mathcal{A}$ over $S$}, denoted by \hl{$T_\ell(\mathcal{A}_{\bar{s}})$}, is the inverse limit
$$\hlin{T_\ell(\mathcal{A}_{\bar{s}}) = \varprojlim_{\nu} \mathcal{A}_{\bar{s}}[\ell^\nu](k(\bar{s}))}$$
where the inverse limit is taken with respect to the multiplication-by-$\ell$ transition maps. The Tate group $T_\ell(\mathcal{A}_{\bar{s}})$ is a module over $\mathbb{Z}_\ell$ on which the \CrefAndHyperrefIfExist{definition:etale_fundamental_group_of_a_connected_scheme}{étale fundamental group} $\pi_1^{\text{ét}}(S, \bar{s})$ \CrefAndHyperrefIfExist{definition:continuous_action_on_a_topological_space_by_a_topological_group}{acts continuously}. The \hldef{rational $\ell$-adic Tate group}, denoted by \hl{$V_\ell(\mathcal{A}_{\bar{s}})$}, is
$$\hlin{V_\ell(\mathcal{A}_{\bar{s}}) = T_\ell(\mathcal{A}_{\bar{s}}) \otimes_{\mathbb{Z}_\ell} \mathbb{Q}_\ell.}$$
When $f$ is an \CrefAndHyperrefIfExist{definition:abelian_scheme_over_a_scheme}{abelian scheme} of \CrefAndHyperrefIfExist{definition:relative_dimension_for_a_morphism_of_schemes}{relative dimension} $g$ and $\ell$ is invertible on $S$, one has $T_\ell(\mathcal{A}_{\bar{s}}) \cong \mathbb{Z}_\ell^{2g}$ and $V_\ell(\mathcal{A}_{\bar{s}}) \cong \mathbb{Q}_\ell^{2g}$.
\end{definition}
