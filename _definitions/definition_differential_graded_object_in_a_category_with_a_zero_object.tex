\begin{definition} \label{definition:differential_graded_object_in_a_category_with_a_zero_object}
    Let $\calC$ be a \CrefAndHyperrefIfExist{definition:category}{category} with a \CrefAndHyperrefIfExist{definition:initial_final_zero_objects_of_a_category}{zero object}. Let $r$ be an integer.
    A \hldef{differential graded object of degree $r$ in $\calC$} is a \CrefAndHyperrefIfExist{definition:graded_object_in_a_category}{graded object} $M = \{M_n\}_{n \in \bbZ}$ equipped with morphisms $d_n: M_n \to M_{n+r}$ such that $d_{n+r} \circ d_n = 0$.
\end{definition}
