\begin{definition}[Closed upper half-plane] \label{definition:closed_upper_half_plane_in_the_complex_plane}
    Let $\bbC$ denote the \CrefAndHyperrefIfExist{definition:field_of_complex_numbers}{field of complex numbers}. The \hldef{closed upper half-plane in the complex plane} is the subset
    $$\hlin{\overline{\mathbb{H}} = \{z \in \bbC : \operatorname{Im}(z) \geq 0\}}$$
    where $\operatorname{Im}(z)$ denotes the \CrefAndHyperrefIfExist{definition:real_and_imaginary_parts_of_a_complex_number}{imaginary part of $z$}.

    The closed upper half-plane is the \CrefAndHyperrefIfExist{definition:closure_of_a_subspace_of_a_topological_space}{closure} of the \CrefAndHyperrefIfExist{definition:upper_half_plane_in_the_complex_plane}{upper half-plane in the complex plane} in $\bbC$, and its boundary $\overline{\mathbb{H}} \setminus \mathbb{H} = \{z \in \bbC : \operatorname{Im}(z) = 0\}$ is canonically identified with the \CrefAndHyperrefIfExist{definition:field_of_real_numbers}{field of real numbers}.
\end{definition}
