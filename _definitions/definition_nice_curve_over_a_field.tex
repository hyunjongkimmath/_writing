\begin{definition} \label{definition:nice_curve_over_a_field}
    Let $k$ be a \CrefAndHyperrefIfExist{definition:field}{field}. 

    To many writers, a \hldef{nice curve over $k$} refers to a \CrefAndHyperrefIfExist{definition:smooth_morphism_of_schemes}{smooth}, \CrefAndHyperrefIfExist{definition:geometric_property_of_a_scheme_over_a_field}{geometrically} \CrefAndHyperrefIfExist{definition:integral_scheme}{integral}, and \CrefAndHyperrefIfExist{definition:projective_variety_over_a_field}{projective} \CrefAndHyperrefIfExist{definition:algebraic_curve_over_a_field}{curve} over $k$.
\end{definition}
