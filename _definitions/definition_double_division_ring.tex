\begin{definition}[Double Division Ring] \label{definition:double_division_ring}
    A \hldef{double division ring} is an \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{involution ring} isomorphic to a \CrefAndHyperrefIfExist{definition:hyperbolic_ring_of_a_ring}{hyperbolic ring} $H(A)$ equipped with the \CrefAndHyperrefIfExist{definition:standard_involution_on_the_hyperbolic_ring_of_a_ring}{standard involution structure} where $A$ is a \CrefAndHyperrefIfExist{definition:division_ring}{division ring}. 

    % The \hldef{double division ring} associated to $A$ is the hyperbolic ring
    % $$ H(A) := A \times A^{\mathrm{op}}, $$
    % where $A^{\mathrm{op}}$ is the \CrefAndHyperrefIfExist{definition:opposite_ring_of_a_ring}{opposite ring} of $A$.

    % Explicitly, as a set $H(A) = A \times A$, with componentwise addition and multiplication given by
    % $$ (a,b) \cdot (c,d) := (a \cdot c,\; d \cdot b), $$
    % for all $(a,b), (c,d) \in H(A)$, where the first component uses the multiplication of $A$ and the second component uses the reversed multiplication of $A^{\mathrm{op}}$.

    % The standard involution on $H(A)$ is the map
    % $$ \tau: H(A) \to H(A), \qquad \tau(a,b) := (b,a). $$
\end{definition}
