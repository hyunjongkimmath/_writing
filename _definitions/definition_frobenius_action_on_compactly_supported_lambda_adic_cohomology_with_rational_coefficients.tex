    Let $(R, \lambda)$ be a \CrefAndHyperrefIfExist{definition:complete_ring_with_respect_to_an_ideal}{complete} \CrefAndHyperrefIfExist{definition:discrete_valuation_ring}{DVR} such that $R/\lambda^n$ is finite for all $n \geq 1$. Let $E = \Frac(R)$ be its \CrefAndHyperrefIfExist{definition:field_of_fractions_of_an_integral_domain}{fraction field}. Let $\mathbb{F}_q$ be a finite field with $q = p^f$ elements. Let $X$ be a scheme of finite type and separated over $\mathbb{F}_q$, with structure morphism $f_X \colon X \to \Spec(\mathbb{F}_q)$. Consider sheaves on the \CrefAndHyperrefIfExist{definition:small_etale_site_of_a_scheme}{(small) \'etale site $X_{\et}$}.

    Let $\Frob_X \colon X \to X$ be the \CrefAndHyperrefIfExist{definition:frobenius_endomorphism_on_a_scheme_over_a_scheme_with_invertible_finite_order_absolute_frobenius_morphism}{Frobenius endomorphism} of $X$ over $\mathbb{F}_q$.

    Let $N \in \Dbc(X, E)$ be an object represented by some $M \in D_c^b(X, R)$. The \CrefAndHyperrefIfExist{definition:frobenius_action_on_compactly_supported_lambda_adic_cohomology_with_integral_coefficients}{Frobenius action on integral compactly supported cohomology}
    $$\Frob_* \colon H_c^q(X, M) \to H_c^q(X, M)$$
    is $R$-linear. The \hldef{Frobenius action on the compactly supported $\lambda$-adic cohomology of $N$ with rational coefficients} is the endomorphism of $E$-vector spaces
    $$\hlin{\Frob_* \colon H_c^q(X, N) \to H_c^q(X, N)}$$
    defined by tensoring the integral Frobenius action with $E$:
    $$\Frob_* \otimes \id_E \colon H_c^q(X, M) \otimes_R E \to H_c^q(X, M) \otimes_R E.$$
    This is independent of the choice of representative $M$ for $N$.
\end{definition}
