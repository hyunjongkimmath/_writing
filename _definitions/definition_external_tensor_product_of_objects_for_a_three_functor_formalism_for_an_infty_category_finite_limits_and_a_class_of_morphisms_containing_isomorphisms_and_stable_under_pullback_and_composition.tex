\begin{definition} \label{definition:external_tensor_product_of_objects_for_a_three_functor_formalism_for_an_infty_category_finite_limits_and_a_class_of_morphisms_containing_isomorphisms_and_stable_under_pullback_and_composition}
    Let $(C,E)$ be as in \Cref{context:infty_category_with_finite_limits_and_a_class_of_morphisms_stable_under_pullback_and_composition_for_correspondences}. 

    Let 
    $$D \colon \operatorname{Corr}(C,E) \to \operatorname{Cat}_\infty$$
    (\Cref{definition:correspondence_symmetric_monoidal_infty_category_of_an_infty_category_with_finite_limtis_and_class_of_morphisms_containing_isomorphisms_and_stable_under_pulback_and_composition})\CrefIfExists{definition:infty_category_of_infty_categories}
    be a \CrefAndHyperrefIfExist{definition:three_functor_formalism_for_an_infty_category_with_finite_limtis_and_class_of_morphisms_containing_isomorphisms_and_stable_under_pulback_and_composition}{$3$-functor formalism}.

    For objects $X$ and $Y$ of $C$, the \hldef{external tensor product functor} is the functor
    $$\hlin{\boxtimes: D(X) \times D(Y) \to D(X \times Y), \quad (M,N) \mapsto \pr_X^* M \otimes \pr_Y^*}$$
    where $\pr_X$ and $\pr_Y$ are the projections from $X \times Y$ to $X$ and $Y$ respectively.
\end{definition}
