\begin{definition}[Middle extension functor for perverse sheaves with algebraic coefficients] \label{definition:middle_extension_functor_for_perverse_sheaves_with_algebraic_coefficients_on_open_immersion}

Let $(R, \lambda)$ be a \CrefAndHyperrefIfExist{definition:complete_ring_with_respect_to_an_ideal}{complete} \CrefAndHyperrefIfExist{definition:discrete_valuation_ring}{DVR} such that $R/\lambda^n$ is finite for all $n \geq 1$. Let $E$ be the \CrefAndHyperrefIfExist{definition:field_of_fractions_of_an_integral_domain}{fraction field} of $R$, and let $K/E$ be a \CrefAndHyperrefIfExist{definition:algebraic_element_minimal_polynomial_of_an_algebraic_element_algebraic_field_extension_transcendental_element_field_extension}{algebraic extension}. Let $X$ be \CrefAndHyperrefIfExist{definition:finite_type_morphism_of_schemes}{finite type} \CrefAndHyperrefIfExist{definition:scheme_over_a_scheme}{scheme over} a \CrefAndHyperrefIfExist{definition:field}{field} that is \CrefAndHyperrefIfExist{definition:algebraically_closed_field_or_separably_closed_field}{separably closed} or \CrefAndHyperrefIfExist{definition:finite_field}{finite}, and consider sheaves on the \CrefAndHyperrefIfExist{definition:small_etale_site_of_a_scheme}{(small) \'etale site} of $X$. Let $j: U \hookrightarrow X$ be an \CrefAndHyperrefIfExist{definition:open_and_closed_immersions_of_schemes}{open immersion}.

% Let $\Perv(U, K)$ and $\Perv(X, K)$ denote the hearts of the perverse $t$-structures on $\Dbc(U, K)$ and $\Dbc(X, K)$ respectively. Let $\pscrH^0_K(-)$ denote the zeroth perverse cohomology functor with respect to these perverse $t$-structures.

For $\mathcal{P} \in \Perv(U, K)$\CrefIfExists{definition:self_dual_middle_perverse_t_structure_on_bdd_constructible_derived_category_with_algebraic_coefficients}, there is a \CrefAndHyperrefIfExist{definition:canonical_forgets_support_map_from_derived_direct_image_with_compact_support_to_derived_direct_image_for_bdd_constructible_complexes_with_algebraic_coefficients_on_compactifiable_morphisms}{canonical forgets support morphism}
$$\alpha_j(\mathcal{P}): Rj_! \mathcal{P} \to Rj_* \mathcal{P}$$
in $\Dbc(X, K)$.
\CrefIfExists{definition:derived_direct_image_functor_with_compact_support_on_bounded_constructible_complexes_with_algebraic_coefficients_on_a_noetherian_scheme}\CrefIfExists{definition:derived_direct_image_functor_on_bounded_constructible_complexes_with_algebraic_coefficients_on_a_noetherian_scheme}

The \hldef{middle/intermediate extension of $\mathcal{P}$ along $j$} is the object in $\Perv(X, K)$
$$\hlin{j_{!*} \mathcal{P} := \Im\bigl(\pscrH^0_K(Rj_! \mathcal{P}) \xrightarrow{\pscrH^0_K(\alpha_j(\mathcal{P}))} \pscrH^0_K(Rj_* \mathcal{P})\bigr)}$$
where $\Im(-)$ denotes the \CrefAndHyperrefIfExist{definition:image_coimage_of_a_morphism_in_a_category}{image} in the \CrefAndHyperrefIfExist{definition:abelian_category}{abelian category} $\Perv(X, K)$.

The assignment $\mathcal{P} \mapsto j_{!*}\mathcal{P}$ extends to a functor
$$\hlin{j_{!*}: \Perv(U, K) \to \Perv(X, K)}$$
called the \hldef{middle extension functor with algebraic coefficients along $j$}. It is defined as the colimit of the middle extension functors for rational coefficients over finite extensions $E'$ of $E$ inside $K$, since the base change functors $\Dbc(\cdot, E') \to \Dbc(\cdot, K)$ commute with both $Rj_!$ and $Rj_*$.
\end{definition}
