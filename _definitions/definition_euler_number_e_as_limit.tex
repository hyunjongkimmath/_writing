\begin{definition} \label{definition:euler_number_e_as_limit}
The \hldef{Euler's number $e$} is the \CrefAndHyperrefIfExist{definition:field_of_real_numbers}{real number} defined as 
$$\hlin{e = \lim_{n \to \infty} \left(1 + \frac{1}{n}\right)^n}.$$
\CrefIfExists{definition:convergence_and_limit_of_sequence_in_extended_metric_space}\CrefIfExists{definition:exponentiation_in_a_field}
This limit indeed converges in the \CrefAndHyperrefIfExist{definition:field_of_real_numbers}{field of real numbers}\CrefIfExists{theorem:limit_of_one_plus_one_over_n_to_the_n_exists}.

\TextIfExists{corollary:eulers_number_equals_series_of_reciprocals_of_integer_factorials}{
A posteriori, 
$$e = \sum_{n=0}^\infty \frac{1}{n!}$$
\CrefIfExists{definition:factorial_of_a_nonnegative_integer}
\CrefIfExists{lemma:series_of_1_over_n_factorial_converges}\CrefIfExists{corollary:eulers_number_equals_series_of_reciprocals_of_integer_factorials}
}
\end{definition}
