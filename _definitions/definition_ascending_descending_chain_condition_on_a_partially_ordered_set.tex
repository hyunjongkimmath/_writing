\begin{definition} \label{definition:ascending_descending_chain_condition_on_a_partially_ordered_set}
Let $(S, \leq)$ be a \CrefAndHyperrefIfExist{definition:partially_ordered_set}{partially ordered set}. 
\begin{enumerate}
    \item $S$ satisfies the \hldef{Ascending Chain Condition} \hldef{(ACC)} if every sequence $s_1 \leq s_2 \leq s_3 \leq \dots$ of elements in $S$ eventually stabilizes, i.e., there exists $n \in \mathbb{N}$ such that $s_n = s_{n+1} = s_{n+2} = \dots$.
    \item $S$ satisfies the \hldef{Descending Chain Condition} \hldef{(DCC)} if every sequence $s_1 \geq s_2 \geq s_3 \geq \dots$ of elements in $S$ eventually stabilizes, i.e., there exists $n \in \mathbb{N}$ such that $s_n = s_{n+1} = s_{n+2} = \dots$.
\end{enumerate}
\end{definition}
