\begin{definition}[{\cite[Chapter I, 7.1]{knus_qhfr}}] \label{definition:base_change_of_a_sesquilinear_form_on_a_module_over_a_ring_with_involution}
    Let $\eta \to A \to A'$ be a \CrefAndHyperrefIfExist{definition:homomorphism_of_involution_rings}{homomorphism of rings with involution}. Let $M$ be a \CrefAndHyperrefIfExist{definition:module_of_a_ring}{right $A$-module} and let $b: M \times M \to A$ be a \CrefAndHyperrefIfExist{definition:sesquilinear_form_on_a_module_over_a_ring_with_involution}{sesquilinear form} on $M$.


    By the \hldef{base change of $b$} (which may also be called by names such as \hldef{extension of scalars of $b$} or the \hldef{induction of $b$ along $\eta$} as per \Cref{definition:extension_of_scalars_base_change_induction_for_modules_along_ring_homomorphisms}) as the sesquilinear form \hl{$b \otimes_A A': (M \otimes_A A') \times (M \otimes_A A') \to A'$} (denoted by \hl{$b'$} or \hl{$b \otimes 1_{A'}$} in \cite{knus_qhfr}) on the right $A'$-module \CrefAndHyperref{definition:extension_of_scalars_base_change_induction_for_modules_along_ring_homomorphisms}{$M \otimes_A A'$} given by 
    $$(b \otimes_A A')(m \otimes a', n \otimes a'') = \overline{a} ' \eta(b(m,n)) a'' \quad \text{for all } m,n \in M, a',a'' \in A'.$$
    and extended $A'$-sesquilinearly.

    The pair $(M \otimes_A A', b \otimes_A A')$ may also be written as \hl{$(M, b) \otimes_A A'$}.

\end{definition}
