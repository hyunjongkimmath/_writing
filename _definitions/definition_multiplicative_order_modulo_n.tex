    Let $n \in \bbZ_{>0}$ and let $a \in \bbZ$ with $\gcd(a, n) = 1$. The \hldef{multiplicative order of $a$ modulo $n$}, denoted by \hl{$\operatorname{ord}_n(a)$}, is the smallest positive integer $d$ such that
    $$a^d \equiv 1 \pmod{n}.$$
\end{definition}
