\begin{definition} \label{definition:left_right_inverse_of_an_element_for_a_binary_operation_on_a_set}
    % EDITOR APPROVED
    Let $\cdot: S \times S \to S$ be a \CrefAndHyperrefIfExist{definition:n_ary_operation_on_a_set}{binary operation} on a \CrefAndHyperrefIfExist{definition:zermelo_fraenkel_set_theory}{set} $S$ such that there exists an \CrefAndHyperrefIfExist{definition:left_right_identity_element_of_a_binary_operation}{identity element} $e \in S$. Let $s \in S$ be an element. 
    \begin{enumerate}
        \item A \hldef{left inverse of $s$} is an element $a \in S$ such that $a \cdot s = e$.
        \item A \hldef{right inverse of $s$} is an element $a \in S$ such that $s \cdot a = e$.
        \item An \hldef{inverse of $s$} is an element $a \in S$ that is both a left and a right inverse of $a$. 

        In the case that $\cdot$ is \CrefAndHyperrefIfExist{definition:associative_binary_operation_on_a_set}{associative}, i.e. $(S,\cdot)$ is a \CrefAndHyperrefIfExist{definition:monoid}{monoid}, inverses of $s$ are unique\CrefIfExists{lemma:inverse_of_an_element_for_a_binary_operation_on_a_set_is_unique_if_exists} if they exist, so one may refer to an inverse as \emph{the} inverse of $s$. In this case, the inverse of $s$ is denoted by \hl{$s^{-1}$}.
    \end{enumerate}
\end{definition}
