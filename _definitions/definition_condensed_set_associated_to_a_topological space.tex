\begin{definition} \label{definition:condensed_set_associated_to_a_topological space}
    Given a \CrefAndHyperrefIfExist{definition:category_of_topological_spaces}{topological space} $T$, define its \hldef{associated condensed set} \hl{$\underline{T}$} as the \CrefAndHyperrefIfExist{definition:functor_between_categories}{functor}

    $$\operatorname{Prof}^{\op} \to \Sets, \quad S \mapsto C(S,T),$$
    \CrefIfExists{definition:opposite_category_of_a_category}
    where $C(S,T)$ is the set of \CrefAndHyperrefIfExist{definition:continuous_map_of_topological_spaces}{continuous maps} from $S$ to $T$.

    Note that if $T$ has the structure of a \CrefAndHyperrefIfExist{definition:topological_group}{topological group}, \CrefAndHyperrefIfExist{definition:topological_ring}{topological ring}, etc., then $\underline{T}$ is \CrefAndHyperrefIfExist{definition:condensed_object_of_a_category}{condensed group}, condensed ring, etc.
\end{definition}
