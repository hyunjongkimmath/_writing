\begin{definition}[Algebraically Ordered Semiring] \label{definition:algebraically_ordered_naturally_ordered_commutative_monoid}
    An \CrefAndHyperrefIfExist{definition:ordered_semiring}{ordered semiring} $M$ is said to be \hldef{algebraically ordered} (or \hldef{naturally ordered}) if for all $a, b \in M$, $b \geq a$ implies there exists some $x \in M_{\geq 0}$ such that $b = a + x$.
\end{definition}
