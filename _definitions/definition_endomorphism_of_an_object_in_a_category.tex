\begin{definition} \label{definition:endomorphism_of_an_object_in_a_category}
    Let $\mathcal{C}$ be a \CrefAndHyperrefIfExist{definition:category}{category} and let $X$ be an object of $\mathcal{C}$.

    An \hldef{endomorphism of $X$} is a morphism $f: X \to X$ in $\mathcal{C}$.
\end{definition}
