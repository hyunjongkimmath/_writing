\begin{definition}[{\cite[Definition 2.1]{ekedahl_af}}] \label{definition:ekedahl_af_R_comp_neg_norm_objs_of_the_der_cat_of_modules_over_ring_of_cat_of_inv_seqs_essentially_an_isomorphism}
    Assume \Cref{context:ekedahl_af_topos_morphisms_to_and_from_Pro_N_S_to_S}

    Let $M,N$ be objects \CrefAndHyperrefIfExist{notation:derived_category_of_category_of_modules_of_a_ringed_site}{$D(\mathrm{Pro}_\bbN(S), R_\bullet)$}.
    \begin{enumerate}
        \item We may say that $M$ is an \hldef{$R$-complex} if $\pi^*(R_1) \otimes_{R_\bullet}^L M$ (\Cref{context:ekedahl_af_topos_morphisms_to_and_from_Pro_N_S_to_S}) (\Cref{definition:derived_tensor_product_of_complexes_of_modules_on_an_essentially_small_ringed_site_with_a_pretopology}) is \CrefAndHyperrefIfExist{definition:essentially_constant_complex_of_abelian_group_objects_in_the_category_of_inverse_direct_sequences_of_objects_in_a_topos}{essentially constant}

        \item $M$ is \hldef{negligible} if $\pi^*(R_1) \otimes_{R_\bullet}^L M$ is \CrefAndHyperrefIfExist{definition:essentially_zero_inverse_sequence_essentially_bounded_complex_of_sheaves_of_abelian_groups_on_a_site}{essentially zero}.


        \item $M$ is \hldef{normalised} if the morphism $\tau(\widehat{M}) \to \tau(M)$ (\Cref{notation:ekedahl_af_tau_M_either_pro_object_of_truncations_or_M_depending_on_whether_ekedahls_condition_A_holds}) (\Cref{definition:ekedahls_normalisation_functor_on_essentially_bounded_complexes_of_inverse_systems_on_ringed_topoi}) is a \CrefAndHyperrefIfExist{definition:quasi_isomorphism_of_chain_complexes_of_objects_in_an_abelian_category}{quasi-isomorphism}. 

        \item A morphism $M \to N$ is \hldef{essentially an isomorphism} if it has a negligible \CrefAndHyperrefIfExist{definition:mapping_cone_of_a_map_of_chain_cochain_complexes_in_an_additive_category}{mapping cone}.
    \end{enumerate}
\end{definition}
