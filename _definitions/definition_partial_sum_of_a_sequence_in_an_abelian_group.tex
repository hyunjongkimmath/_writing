\begin{definition}[Partial sum] \label{definition:partial_sum_of_a_sequence_in_an_abelian_group}
    Let $A$ be an abelian group written additively. Let $(a_n)_{n=1}^\infty$ be a sequence in $A$. For each integer $N \geq 1$, the \hldef{$N$-th partial sum of the sequence $(a_n)$} is the element
    \hl{$S_N = \sum_{n=1}^N a_n$}
    of $A$. The sequence $(S_N)_{N=1}^\infty$ in $A$ is called the \hldef{sequence of partial sums of $(a_n)$}.
\end{definition}
