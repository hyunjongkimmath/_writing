\begin{definition} \label{definition:partition_of_a_closed_interval_in_the_real_line}
    Let $[a,b] \subset \mathbb{R}$ be a \CrefAndHyperrefIfExist{definition:interval_in_the_real_line}{closed}. 
    A \hldef{partition of $[a,b]$} is a finite ordered set $P = \{x_0,\dots,x_n\}$ such that $a = x_0 < x_1 < \dots < x_n = b$.
\end{definition}
