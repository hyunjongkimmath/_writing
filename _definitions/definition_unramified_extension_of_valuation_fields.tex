\begin{definition} \label{definition:unramified_extension_of_valuation_fields}
    Assume \Cref{context:extension_of_valuation_fields}. 
    \begin{enumerate}
        \item Further assume that $L/K$ is \CrefAndHyperrefIfExist{definition:degree_of_a_field_extension}{finite}. The extension $(L,w)/(K,v)$ of valuation fields is \hldef{unramified} if 
        \begin{enumerate}
            \item $e(w/v) = 1$ (\Cref{definition:ramification_index_of_extension_of_valuation_fields})
            \item $f(w/v) = [L:K]$ (\Cref{definition:inertial_degree_of_extension_of_valuation_fields})
            \item The \CrefAndHyperrefIfExist{definition:residue_field_of_a_ring_at_a_point}{residue field} \CrefAndHyperrefIfExist{definition:extension_of_a_field}{extension} $k_w/k_v$ is \CrefAndHyperrefIfExist{definition:separable_algebraic_field_extension}{separable}.
        \end{enumerate}

        \item Removing the assumption that $L/K$ is finite, the extension $(L,w)/(K,v)$ of valuation fields is \hldef{unramified} if for every finite subextension $K \subseteq E \subseteq L$, $(E, w|_E) /(K,v)$ is unramified in the above sense.
    \end{enumerate}
\end{definition}
