\begin{definition} \label{definition:isotypic_component_of_a_representation_of_a_group_on_a_left_module_over_a_ring_for_an_irreducible_representation}

    Let $G$ be a \CrefAndHyperrefIfExist{definition:group}{group}, $R$ a \CrefAndHyperrefIfExist{definition:ring}{ring with unity}, and $(\rho, M)$ a \CrefAndHyperrefIfExist{definition:representation_of_a_group_on_a_module_over_a_not_necessarily_commutative_ring}{representation of $G$ on a left $R$-module}. Let $(\sigma, S)$ be an \CrefAndHyperrefIfExist{definition:irreducible_simple_representation_of_a_group_on_a_left_module_over_a_ring}{irreducible representation} of $G$.

    The \hldef{$S$-isotypic component of $M$} is the submodule
    $$\hlin{M_{(S)} := \sum \{N \subseteq M \mid N \text{ is a subrepresentation and } N \cong S\}.}$$
    It is the largest subrepresentation of $M$ that is a direct sum of copies of $S$.

    If $(\rho, M)$ is \CrefAndHyperrefIfExist{definition:semisimple_representation_of_a_group_on_a_left_module_over_a_ring}{semisimple}, then for each isomorphism class $[S]$ of irreducible representations appearing in $M$, the isotypic component $M_{(S)}$ is a direct summand, and
    $$ M = \bigoplus_{[S]} M_{(S)},$$
    where the sum runs over a set of representatives of isomorphism classes of irreducible constituents. Each $M_{(S)}$ is isomorphic to $S^{\oplus m_S}$ for some multiplicity $m_S \in \mathbb{N} \cup \{\infty\}$, called the \hldef{multiplicity of $S$ in $M$}.
\end{definition}
