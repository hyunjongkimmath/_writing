\begin{definition} \label{definition:morphism_of_pointed_n_tuples_of_topological_spaces}
    A \hldef{morphism $f: (X_1, \dots, X_n, x) \to (Y_1, \dots, Y_n, y)$ of pointed $n$-tuples of topological spaces} is a continuous map $f: X_1 \to Y_1$ that constitutes
    \begin{enumerate}
        \item a \CrefAndHyperrefIfExist{definition:morphism_of_n_tuples_of_topological_spaces}{morphism of $n$-tuples} $(X_1, \dots, X_n) \to (Y_1, \dots, Y_n)$, and
        \item a \CrefAndHyperrefIfExist{definition:pointed_topological_space}{pointed map} $(X_i, x) \to (Y_i, y)$ for each $i \in \{1, \dots, n\}$.
    \end{enumerate}
    In other words, $f$ is a continuous map $X_1 \to Y_1$ such that $f(X_i) \subseteq Y_i$ for all $i$ and $f(x) = y$.
\end{definition}
