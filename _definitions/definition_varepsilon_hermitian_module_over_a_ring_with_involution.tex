\begin{definition}[{\cite[Chapter I, 3.1]{knus_qhfr}}] \label{definition:varepsilon_hermitian_module_over_a_ring_with_involution}
    Let $A$ be a \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{ring with involution}, and let $\varepsilon \in Z(A)$ satisfy $\varepsilon \cdot \overline{\varepsilon} = 1$.

    \begin{enumerate}
        \item A \hldef{$\varepsilon$-Hermitian module over $A$} is a \CrefAndHyperrefIfExist{definition:sesquilinear_module_over_a_ring_with_involution}{sesquilinear module} $(M,b)$ over $A$ such that the \CrefAndHyperrefIfExist{definition:sesquilinear_form_on_a_module_over_a_ring_with_involution}{sesquilinear form} $b$ is in fact a \CrefAndHyperrefIfExist{definition:varepsilon_hermitian_form_on_a_module_over_a_ring_with_involution}{$\varepsilon$-Hermitian form} on $M$. 

        In other words, a $\varepsilon$-Hermitian module over $A$ is a pair $(M,b)$ with $M$ a \CrefAndHyperrefIfExist{definition:finitely_generated_modules_over_rings}{finitely generated} \CrefAndHyperrefIfExist{definition:projective_bimodule_over_rings}{projective} \CrefAndHyperrefIfExist{definition:module_of_a_ring}{(right) $A$-module} and $b$ a \CrefAndHyperrefIfExist{definition:varepsilon_hermitian_form_on_a_module_over_a_ring_with_involution}{$\varepsilon$-Hermitian form} on $M$. 

        As with sesquilinear modules, we may identify $b$ with its \CrefAndHyperrefIfExist{definition:adjoint_map_of_a_sesquilinear_form_on_a_module_over_a_ring_with_involution}{adjoint} and write \hl{$(M, h)$} to denote an $\varepsilon$-Hermitian module with $h \in \Hom_A(M, M^*)$\CrefIfExists{definition:twisted_dual_module_of_a_module_over_a_ring_with_involution}. 

        \item An \hldef{even $\varepsilon$-hermitian module over $A$} is a $\varepsilon$-hermitian module whose underlying $\varepsilon$-hermitian form is \CrefAndHyperrefIfExist{definition:even_varepsilon_hermitian_form_on_a_module_over_a_ring_with_involution}{even}.

        \item A \hldef{symmetric bilinear form module over $A$} is an $1$-Hermitian module over $A$ where $A$ has the trivial involution.
        \item A \hldef{skew-symmetric bilinear form module over $A$} is a $(-1)$-Hermitian module over $A$ where $A$ has the trivial involution.
    \end{enumerate}
\end{definition}
