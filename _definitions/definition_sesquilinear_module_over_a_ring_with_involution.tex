\begin{definition} \label{definition:sesquilinear_module_over_a_ring_with_involution}
    Let $A$ be a \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{ring with involution}.

    A \hldef{sesquilinear module over $A$} is a pair $(M,b)$ with $M$ a \CrefAndHyperrefIfExist{definition:finitely_generated_modules_over_rings}{finitely generated} \CrefAndHyperrefIfExist{definition:projective_bimodule_over_rings}{projective} \CrefAndHyperrefIfExist{definition:module_of_a_ring}{(right) $A$-module} and $b$ a \CrefAndHyperrefIfExist{definition:sesquilinear_form_on_a_module_over_a_ring_with_involution}{sesquilinear form} on $M$. 

    We may identify $b$ with its \CrefAndHyperrefIfExist{definition:adjoint_map_of_a_sesquilinear_form_on_a_module_over_a_ring_with_involution}{adjoint} and write \hl{$(M, h)$} to denote an $\varepsilon$-Hermitian module with $h \in \Hom_A(M, M^*)$\CrefIfExists{definition:twisted_dual_module_of_a_module_over_a_ring_with_involution}. 
\end{definition}

\begin{remark}
    This terminology is not strictly present in \cite{knus_qhfr}.
\end{remark}
