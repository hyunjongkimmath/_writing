\begin{definition} \label{definition:category_of_relative_chow_motives_over_a_noetherian_base_and_coefficients_in_a_commutative_ring}
Let $S$ be a \CrefAndHyperrefIfExist{definition:noetherian_scheme}{Noetherian base scheme} and $R$ a \CrefAndHyperrefIfExist{definition:commutative_ring}{commutative ring}. 
The \hldef{category of relative Chow motives}, denoted by \hl{$\mathfrak{Chow}(S, R)$}, is the additive category defined by the following data:

\begin{enumerate}
    \item \textbf{Objects:} The objects are \CrefAndHyperref{definition:relative_chow_motive_over_a_noetherian_base_and_commutative_ring_of_coefficients}{relative Chow motives} $(X, p, n)$

    \item \textbf{Morphisms:} Given two relative Chow motives $M = (X, p, m)$ and $N = (Y, q, n)$, the morphisms are the \CrefAndHyperref{definition:morphism_of_relative_chow_motives_over_a_noetherian_base_and_coefficients_in_a_commutative_ring}{morphisms of relative Chow motives}
\end{enumerate}

\end{definition}
