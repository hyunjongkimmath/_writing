\begin{definition}[Cumulative distribution function] \label{definition:cumulative_distribution_function_of_a_real_valued_random_variable_on_a_probability_space}
Let $(\Omega, \mathcal{F}, P)$ be a \CrefAndHyperref{definition:probability_space}{probability space}. Let $X: \Omega \to \mathbb{R}$ be a function such that for every $t \in \mathbb{R}$, the set $\{\omega \in \Omega : X(\omega) \leq t\}$ belongs to $\mathcal{F}$. The \hldef{cumulative distribution function of $X$}, denoted by \hl{$F_X$}, is the function \hl{$F_X: \mathbb{R} \to [0, 1]$} defined by
$$F_X(t) = P(\{\omega \in \Omega : X(\omega) \leq t\}).$$
\end{definition}
