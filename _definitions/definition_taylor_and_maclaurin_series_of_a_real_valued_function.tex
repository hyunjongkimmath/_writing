\begin{definition} \label{definition:taylor_and_maclaurin_series_of_a_real_valued_function}

    Let $I \subseteq \mathbb{R}$ be an open interval, let $c \in I$, and let $f : I \to \mathbb{R}$ be a function that is \CrefAndHyperrefIfExist{definition:derivative_of_a_real_valued_function_on_a_real_interval}{differentiable} at $c$. 

    For each nonnegative integer $n$, the $n$th derivative of $f$ at $c$, denoted by \hl{$f^{(n)}(c)$}, is defined recursively as follows:
    \begin{align*}
        f^{(0)}(c) &:= f(c), \\
        f^{(n)}(c) &:= \left(f^{(n-1)}\right)'(c) \quad\text{for } n \geq 1,
    \end{align*}
    where the derivative exists whenever it is invoked.

    The \hldef{Taylor series of $f$ centered at $c$}, denoted by \hl{$T_c f(x)$}, is the \CrefAndHyperrefIfExist{definition:real_power_series_and_radius_of_convergence}{real power series}
    $$\hlin{T_c f(x) := \sum_{n=0}^\infty \frac{f^{(n)}(c)}{n!} (x - c)^n}$$
    where $n!$ is the \CrefAndHyperrefIfExist{definition:factorial_of_a_nonnegative_integer}{factorial of $n$}.

    When $c = 0$, the Taylor series of $f$ centered at $0$ is called the \hldef{Maclaurin series of $f$}, denoted by \hl{$T_0 f(x)$} or simply \hl{$M_f(x)$}:
    $$M_f(x) := \sum_{n=0}^\infty \frac{f^{(n)}(0)}{n!} x^n.$$
\end{definition}
