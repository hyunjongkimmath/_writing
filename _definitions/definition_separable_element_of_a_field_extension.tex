\begin{definition} \label{definition:separable_algebraic_field_extension}
   Let $L/K$ be a \CrefAndHyperrefIfExist{definition:extension_of_a_field}{field extension}. 
   An element $\alpha$ of $K$ is said to be \hldef{separable} if it is \CrefAndHyperrefIfExist{definition:algebraic_element_minimal_polynomial_of_an_algebraic_element_algebraic_field_extension_transcendental_element_field_extension}{algebraic over $K$} and the \CrefAndHyperrefIfExist{definition:algebraic_element_minimal_polynomial_of_an_algebraic_element_algebraic_field_extension_transcendental_element_field_extension}{minimal polynomial of $\alpha$ over $K$} has no repeated roots in any (eqiuvalently all) \CrefAndHyperrefIfExist{definition:algebraic_element_minimal_polynomial_of_an_algebraic_element_algebraic_field_extension_transcendental_element_field_extension}{algebraic closure}/\CrefAndHyperrefIfExist{definition:splitting_field_of_polynomials_over_a_field_normal_field_extension}{splitting field}.
\end{definition}
