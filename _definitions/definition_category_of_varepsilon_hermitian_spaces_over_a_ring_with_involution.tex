\begin{definition}[cf. {\cite[Chapter I, 3.2]{knus_qhfr}}]\label{definition:category_of_varepsilon_hermitian_spaces_over_a_ring_with_involution}
    Let $(A, \sigma)$ be a \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{ring with involution}, and let $\varepsilon \in Z(A)$ be an element such that $\varepsilon \cdot \overline{\varepsilon} = 1$.

    The \hldef{category of $\varepsilon$-Hermitian spaces over $(A,\sigma)$} is the full subcategory \hl{$\mathfrak{H}^\varepsilon(A)$} of the \CrefAndHyperrefIfExist{definition:category_of_modules_with_varepsilon_hermitian_forms_over_a_ring_with_involution}{category of $\varepsilon$-hermitian spaces over $(A,\sigma)$} whose objects are the \CrefAndHyperrefIfExist{definition:varepsilon_hermitian_module_over_a_ring_with_involution}{$\varepsilon$-Hermitian spaces}, i.e. pairs $(M,h)$ where $M$ is a \CrefAndHyperrefIfExist{definition:finitely_generated_modules_over_rings}{finitely generated} \CrefAndHyperrefIfExist{definition:projective_bimodule_over_rings}{projective} (one-sided) \CrefAndHyperrefIfExist{definition:module_of_a_ring}{$A$-module} and $h$ is \CrefAndHyperrefIfExist{definition:nondegenerate_nonsingular_regular_sesquilinear_form_on_a_module_over_an_involution_ring}{nonsingular}. 

    The \hldef{category of even $\varepsilon$-Hermitian modules over $(A,\sigma)$} is the full subcategory \hl{$\mathfrak{H}_+^\varepsilon(A)$} of $\mathfrak{H}^\varepsilon(A)$ whose objects are the \CrefAndHyperrefIfExist{definition:even_varepsilon_hermitian_form_on_a_module_over_a_ring_with_involution}{even} $\varepsilon$-hermitian spaces over $A$.
    
    In particular, such categories are \CrefAndHyperrefIfExist{definition:groupoid}{groupoids}.
\end{definition}
