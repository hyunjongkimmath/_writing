\begin{definition} \label{definition:projective_cover_of_a_module_over_a_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{ring}, not assumed commutative. Let $M$ be an (left or right) $R$-module. 
    A \hldef{projective cover of $M$} is a pair $(P, \pi)$ where $P$ is a \CrefAndHyperrefIfExist{definition:projective_bimodule_over_rings}{projective $R$-module} and $\pi: P \to M$ is a superfluous \CrefAndHyperrefIfExist{definition:monomorphism_and_epimorphism_in_categories}{epimorphism}, meaning that for any \CrefAndHyperrefIfExist{definition:submodule_of_a_module_over_a_ring}{submodule} $K \subseteq P$, if $\pi(K) = M$, then $K = P$.

\end{definition}
