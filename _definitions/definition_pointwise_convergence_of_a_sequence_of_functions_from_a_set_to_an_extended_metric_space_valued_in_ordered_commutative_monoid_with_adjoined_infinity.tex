\begin{definition} \label{definition:pointwise_convergence_of_a_sequence_of_functions_from_a_set_to_an_extended_metric_space_valued_in_ordered_commutative_monoid_with_adjoined_infinity}
    Let $T$ be an \CrefAndHyperrefIfExist{definition:ordered_commutative_monoid}{ordered commutative monoid}, let $T^\infty$ be the \CrefAndHyperrefIfExist{definition:ordered_monoid_with_adjoined_infinity}{ordered monoid with adjoined infinity} obtained from $T$, and let $X$ be a set and let $(Y, d)$ be an \CrefAndHyperrefIfExist{definition:extended_metric_on_a_set_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity}{extended metric space valued in $T^\infty$}.
    A sequence of functions $f_n : X \to Y$ is said to \hldef{converge pointwise to a function $f : X \to Y$} if for every $x \in X$, the sequence $(f_n(x))_{n=1}^\infty$ \CrefAndHyperrefIfExist{definition:convergence_and_limit_of_sequence_in_extended_metric_space_valued_in_ordered_commutative_monoid_with_adjoined_infinity}{converges} to $f(x)$ in $(Y, d)$. That is, for every $\varepsilon \in T_{>0}$ and every $x \in X$, there exists $N \in \mathbb{N}$ such that for all $n \geq N$, one has $d(f_n(x), f(x)) < \varepsilon$. One calls $f$ the \hldef{(pointwise) limit of the sequence $(f_n)_{n=1}^\infty$}.
\end{definition}
