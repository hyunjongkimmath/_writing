\begin{definition} \label{definition:special_linear_group_over_a_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{ring}. 
    The \hldef{special linear group of $n \times n$ matrices over $R$} is the the \CrefAndHyperrefIfExist{definition:subgroup_of_a_group}{subgroup} \hl{$\SL_n(R)$}  of $\GL_n(R)$ consisting of the \CrefAndHyperrefIfExist{definition:matrix_over_a_ring}{matrices} of \CrefAndHyperrefIfExist{definition:determinant_of_a_matrix_over_a_ring}{determinant} $1$. 
\end{definition}
