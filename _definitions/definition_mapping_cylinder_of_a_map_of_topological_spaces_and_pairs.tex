\begin{definition} \label{definition:mapping_cylinder_of_a_map_of_topological_spaces_and_pairs}

\begin{enumerate}
    \item Let $f: X \to Y$ be a \CrefAndHyperrefIfExist{definition:continuous_map_of_topological_spaces}{continuous map} of \CrefAndHyperrefIfExist{definition:topological_space}{topological spaces}.

    The \hldef{mapping cylinder of $f$}, denoted by \hl{$M_f$}, is the \CrefAndHyperrefIfExist{definition:quotient_of_a_topological_space_with_respect_to_an_equivalence_relations}{quotient space} of the \CrefAndHyperrefIfExist{definition:disjoint_union_of_topological_spaces}{disjoint union} $(X \times I) \sqcup Y$ (where $I = [0, 1]$ is the unit interval) by the \CrefAndHyperrefIfExist{definition:equivalence_relation}{equivalence relation} generated by:
    $$ (x, 1) \sim f(x) \quad \text{for all } x \in X. $$
    In other words, $M_f$ is obtained by attaching the cylinder $X \times I$ to $Y$ by identifying the "top" end $X \times \{1\}$ with the image of $X$ in $Y$ under $f$.

    \item Let $f: (X, A) \to (Y, B)$ be a \CrefAndHyperrefIfExist{definition:map_of_pairs}{map of pairs}.
    The \hldef{relative mapping cylinder of $f$} (or \hldef{mapping cylinder of the map of pairs $f$}) is the pair of topological spaces \hl{$(M_f, M_{f|_A})$}, where $M_f$ is the mapping cylinder of the total map and $M_{f|_A} \subseteq M_f$ is the mapping cylinder of the restricted map $f|_A: A \to B$.
\end{enumerate}
