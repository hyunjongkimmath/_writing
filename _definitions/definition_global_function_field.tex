\begin{definition} \label{definition:global_function_field}
    A \hldef{global function field} is a \CrefAndHyperrefIfExist{definition:finite_field_extension}{finite extension} of the \CrefAndHyperref{definition:function_field_over_an_integral_domain_in_finitely_many_variables}{field of rational functions} $\mathbb{F}_q(t)$ in one variable over a \CrefAndHyperrefIfExist{definition:finite_field}{finite field} $\mathbb{F}_q$.
\end{definition}
