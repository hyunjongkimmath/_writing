\begin{definition}[Dualizing complex for Verdier duality on a noetherian scheme with torsion coefficients] \label{definition:dualizing_complex_for_verdier_duality_in_the_bounded_below_derived_category_of_etale_sheaves_with_torsion_coefficients_on_a_finite_type_scheme_over_a_noetherian_scheme}
    Let $S$ be a \CrefAndHyperrefIfExist{definition:locally_noetherian_and_noetherian_scheme}{noetherian scheme}. Let $A$ be a \CrefAndHyperrefIfExist{definition:noetherian_ring}{noetherian} \CrefAndHyperrefIfExist{definition:torsion_ring}{torsion ring}. Consider \CrefAndHyperrefIfExist{definition:sheaf_on_a_site}{sheaves} of \CrefAndHyperrefIfExist{definition:module_of_a_ring}{$A$-modules} on the \CrefAndHyperrefIfExist{definition:small_etale_site_of_a_scheme}{(small) \'etale site} of $S$.

    \begin{enumerate}
        \item The \hldef{dualizing complex for Verdier duality on $S$ with coefficients in $A$}, denoted by \hl{$\mathbb{D}_S^\bullet$}, is the constant sheaf $A_S$ concentrated in degree $0$.

        \item Let $X$ be a finite type scheme over $S$. The \hldef{(relative) dualizing complex for Verdier duality on $X/S$ with coefficients in $A$}, denoted by \hl{$\mathbb{D}_{X/S}^\bullet$}, is defined as the object in $D^+(X_{\et}, A)$\CrefIfExists{notation:derived_category_of_category_of_modules_of_a_ringed_site} defined as 
        $$\hlin{\mathbb{D}_{X/S}^\bullet = Rf^! \, \mathbb{D}_S^\bullet}$$
        \CrefIfExists{definition:exceptional_inverse_image_functor_on_the_bounded_below_derived_categories_of_etale_sheaves_with_torsion_coefficients_on_schemes_of_finite_type_over_a_noetherian_scheme_via_a_compactifiable_morphism}
    \end{enumerate}
\end{definition}
