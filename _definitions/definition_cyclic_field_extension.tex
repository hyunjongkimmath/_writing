\begin{definition} \label{definition:cyclic_field_extension}
    A \hldef{cyclic field extension} $K/F$ is a \CrefAndHyperrefIfExist{definition:extension_of_a_field}{field extension} that is \CrefAndHyperrefIfExist{definition:galois_extension_of_fields}{Galois} and whose \CrefAndHyperrefIfExist{definition:galois_extension_of_fields}{Galois group} is a \CrefAndHyperrefIfExist{definition:cyclic_group}{cyclic group}, almost always \CrefAndHyperrefIfExist{definition:cardinal_number_and_cardinality_of_a_set}{finite}. 
\end{definition}
