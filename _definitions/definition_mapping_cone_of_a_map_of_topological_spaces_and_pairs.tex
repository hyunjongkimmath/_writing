\begin{definition} \label{definition:mapping_cone_of_a_map_of_topological_spaces_and_pairs}

\begin{enumerate}
    \item 
    Let $f: X \to Y$ be a \CrefAndHyperrefIfExist{definition:continuous_map_of_topological_spaces}{continuous map} of \CrefAndHyperrefIfExist{definition:topological_space}{topological spaces}. 

    The \hldef{mapping cone of $f$}, denoted by \hl{$C_f$}, is the topological space obtained by taking the \CrefAndHyperrefIfExist{definition:disjoint_union_of_topological_spaces}{disjoint union} of the \CrefAndHyperrefIfExist{definition:cone_of_a_topological_space}{cone} on $X$ and the space $Y$, and identifying the base of the cone with its image in $Y$ under $f$. Formally:
    $$ C_f = \left((X \times I) \coprod Y \right) / \sim $$
    where the \CrefAndHyperrefIfExist{definition:equivalence_relation_on_a_set}{equivalence relation} $\sim$ is generated by:
    \TODO{quotient of a topological space by an equivalence relation}
    \begin{itemize}
        \item $(x, 0) \sim (x', 0)$ for all $x, x' \in X$ (collapsing one end of the cylinder to a vertex).
        \item $(x, 1) \sim f(x)$ for all $x \in X$ (attaching the other end to $Y$ via $f$).
    \end{itemize}
    The mapping cone fits into a sequence $X \xrightarrow{f} Y \xrightarrow{i} C_f$, where $i$ is the natural inclusion.

    \TextIfExists{definition:mapping_cylinder_of_a_map_of_topological_spaces}{Note that the mapping cone is a quotient of the mapping cylinder.}
    
    % The \hldef{mapping cone of $f$}, denoted \hl{$C_f$}, is the \CrefAndHyperrefIfExist{definition:quotient_space}{quotient space} of the \CrefAndHyperrefIfExist{definition:mapping_cylinder_of_a_map_of_topological_spaces_and_pairs}{mapping cylinder} $M_f$ by the equivalence relation $(x, 0) \sim (x', 0)$ for all $x, x' \in X$. Equivalently:
    % $$ C_f = \left( (X \times I) \sqcup Y \right) / \{ (x, 0) \sim (x', 0), (x, 1) \sim f(x) \}. $$

    \item Let $f: (X, A) \to (Y, B)$ be a \CrefAndHyperrefIfExist{definition:map_of_pairs}{map of pairs}. 


    The \hldef{relative mapping cone of $f$} (or \hldef{mapping cone of the map of pairs $f$}) is the pair of topological spaces \hl{$(C_f, C_{f|_A})$}, where $C_f$ is the mapping cone of the total map and $C_{f|_A} \subseteq C_f$ is the mapping cone of the restricted map $f|_A: A \to B$ sharing the same cone vertex.
\end{enumerate}
