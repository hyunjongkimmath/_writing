\begin{definition}[Cyclic subgroup scheme of order $N$ of a group scheme over a scheme] \label{definition:cyclic_subgroup_scheme_of_order_N_of_a_group_scheme_over_a_scheme}
    Let $S$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme} and let $N$ be a positive integer. Let $E/S$ be an \CrefAndHyperrefIfExist{definition:elliptic_curve_over_a_scheme}{elliptic curve over $S$}. 
    
    % The \hldef{$N$-torsion subgroup scheme of $E$}, denoted by \hl{$E[N]$}, is the kernel of the multiplication-by-$N$ morphism $[N] : E \to E$, as in the \CrefAndHyperrefIfExist{definition:torsion_subgroup_scheme_of_a_group_scheme}{torsion subgroup scheme of a group scheme}.

    A \hldef{cyclic subgroup scheme of order $N$ on $E$} is a \CrefAndHyperref{definition:finite_flat_subgroup_scheme_of_rank_N}{finite flat subgroup scheme} $C \subset E$ of rank $N$ that is locally on $S$ in the fppf topology isomorphic to $\bbZ/N\bbZ$ as a group scheme. More precisely, there exists an \CrefAndHyperrefIfExist{definition:big_fppf_site_of_a_scheme}{fppf cover} $\{U_i \to S\}$ such that for each $i$,
    $$\hlin{C|_{U_i} \cong (\bbZ/N\bbZ)_{U_i}}$$
    as group schemes over $U_i$.

    \TODO{valued point of a scheme}
    Equivalently, a finite flat subgroup scheme $C \subset E$ of rank $N$ is cyclic if and only if there exists an fppf cover $\{U_i \to S\}$ and a section $P \in C(U_i)$ such that the map
    $$\bbZ/N\bbZ \longrightarrow C|_{U_i}, \quad 1 \longmapsto P$$
    is an isomorphism of group schemes over $U_i$. In this case, $P$ is said to be a \hldef{generating section of $C$}.

    When $\operatorname{char}(\bar{k}(s))$ does not divide $N$ for all geometric points $s \in S$, the group scheme $\bbZ/N\bbZ_S$ is etale and every finite flat subgroup scheme of rank $N$ that is locally constant (etale) is cyclic. In this case, a cyclic subgroup scheme of order $N$ corresponds to a subsheaf of $E[N]$ (\Cref{definition:torsion_subgroup_scheme_of_a_group_scheme}) generated by a section of exact order $N$.
\end{definition}
