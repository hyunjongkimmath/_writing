\begin{definition}[Convergence in distribution] \label{definition:convergence_in_distribution_of_real_valued_random_variables}
    \TODO{cumulative distribution}
Let $(\Omega, \mathcal{F}, P)$ be a \CrefAndHyperrefIfExist{definition:probability_measure_space_for_a_measure_on_a_measurable_space}{probability measure space}. Let $\{X_n\}_{n=1}^\infty$ and $X$ be real-valued \CrefAndHyperrefIfExist{definition:random_variable_on_a_probability_measure_space}{random variables} on $(\Omega, \mathcal{F}, P)$, with cumulative distribution functions $F_{X_n}$ and $F_X$, respectively. We say that $X_n$ \hldef{converges in distribution} to $X$ (denoted \hl{$X_n \xrightarrow{d} X$}) if for every $t \in \mathbb{R}$ at which $F_X$ is \CrefAndHyperrefIfExist{definition:continuous_map_between_open_subsets_of_euclidean_spaces}{continuous},
$$\lim_{n \to \infty} F_{X_n}(t) = F_X(t).$$

\end{definition}
