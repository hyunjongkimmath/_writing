\begin{definition}[Extended metric induced by an extended seminorm on an ordered abelian group] \label{definition:extended_metric_induced_by_an_extended_seminorm_on_an_ordered_abelian_group}

    Let $(G, +, \leq)$ be an \CrefAndHyperrefIfExist{definition:ordered_abelian_group}{ordered abelian group} (written additively), and let $T$ be an \CrefAndHyperrefIfExist{definition:ordered_commutative_monoid}{ordered commutative monoid} (written additively with identity $0_T$). Let $T^\infty$ be the \CrefAndHyperrefIfExist{definition:ordered_monoid_with_adjoined_infinity}{ordered commutative monoid with adjoined infinity} obtained from $T$.

    Given an \hldef{extended seminorm on $G$ valued in $T^\infty$}\CrefAndHyperrefIfExist{definition:extended_seminorm_on_an_ordered_abelian_group_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity}{extended seminorm}
    $$v : G \to T^\infty,$$
    the function $d : G \times G \to T^\infty$ defined by
    $$\hlin{d(x, y) := v(x - y)}$$
    (where $x - y$ denotes $x + (-y)$ in the group $G$) is an \CrefAndHyperrefIfExist{definition:extended_metric_on_a_set_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity}{extended metric on $G$ valued in $T^\infty$}. This is called the \hldef{extended metric induced by $v$}.
\end{definition}
