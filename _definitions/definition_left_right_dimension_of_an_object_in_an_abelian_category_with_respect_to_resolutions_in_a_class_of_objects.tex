\begin{definition} \label{definition:left_right_dimension_of_an_object_in_an_abelian_category_with_respect_to_resolutions_in_a_class_of_objects}
Let $\mathcal{A}$ be an \CrefAndHyperrefIfExist{definition:abelian_category}{abelian category} and $\mathcal{X} \subseteq \mathcal{A}$ a class of objects. For an object $M \in \mathcal{A}$:
\begin{enumerate}
    \item The \hldef{left $\mathcal{X}$-dimension of $M$}, denoted \hl{$\operatorname{res}_{\mathcal{X}}(M)$}, is the infimum of the set of integers $n \geq 0$ such that there exists an \CrefAndHyperrefIfExist{definition:acyclic_complex_of_objects_in_an_abelian_category}{exact sequence}:
    $$0 \to X_n \to X_{n-1} \to \dots \to X_0 \to M \to 0$$
    with $X_i \in \mathcal{X}$ for all $0 \leq i \leq n$. If no such finite sequence exists, the dimension is $\infty$.
    \item The \hldef{right $\mathcal{X}$-dimension of $M$}, denoted \hl{$\operatorname{cores}_{\mathcal{X}}(M)$}, is the infimum of the set of integers $n \geq 0$ such that there exists an exact sequence:
    $$0 \to M \to X^0 \to X^1 \to \dots \to X^n \to 0$$
    with $X^i \in \mathcal{X}$ for all $0 \leq i \leq n$. If no such finite sequence exists, the dimension is $\infty$.
\end{enumerate}
\end{definition}
