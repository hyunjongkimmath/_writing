\begin{definition} \label{definition:totally_ramified_extension_of_valuation_fields}
    Assume \Cref{context:extension_of_valuation_fields}. 
    \begin{enumerate}
        \item Further assume that $L/K$ is \CrefAndHyperrefIfExist{definition:degree_of_a_field_extension}{finite}. The extension $(L,w)/(K,v)$ of valuation fields is \hldef{totally ramified} if 
        \begin{enumerate}
            \item $e(w/v) = [L:K]$ (\Cref{definition:ramification_index_of_extension_of_valuation_fields})
            \item $f(w/v) = 1$ (\Cref{definition:inertial_degree_of_extension_of_valuation_fields})
        \end{enumerate}

        \item Removing the assumption that $L/K$ is finite, the extension $(L,w)/(K,v)$ of valuation fields is \hldef{totally ramified} if for every finite subextension $K \subseteq E \subseteq L$, $(E, w|_E) /(K,v)$ is totally ramified in the above sense.
    \end{enumerate}
\end{definition}
