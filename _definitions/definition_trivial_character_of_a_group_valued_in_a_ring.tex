\begin{definition} \label{definition:trivial_character_of_a_group_valued_in_a_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:commutative_ring}{commutative ring}, and $G$ a \CrefAndHyperrefIfExist{definition:group}{group}.

    \TODO{define and reference trivial group homomorphism}
    A \CrefAndHyperrefIfExist{definition:character_of_a_group_valued_in_a_ring}{character} $\chi: G \to R^\times$\CrefIfExists{definition:unit_of_a_ring_invertible_element_of_a_ring_unit_group_of_a_group} is said to be \hldef{trivial} if it is the trivial group homomorphism, i.e. $\chi(g) = 1_R$ for all $g \in G$.
\end{definition}
