\begin{definition}
\label{definition:fppf_descent_for_a_category_fibered_in_groupoids_over_a_scheme}
Let $X$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme}. Let $\mathcal{F}$ be a \CrefAndHyperrefIfExist{definition:category_fibered_in_groupoids_over_a_category}{category fibered in groupoids} over the \CrefAndHyperrefIfExist{definition:small_fppf_site_of_a_scheme}{small fppf site $X_{\mathrm{fppf}}$}.

We say that $\mathcal{F}$ has \hldef{fppf descent on $X$} if $\mathcal{F}$ \CrefAndHyperrefIfExist{definition:descent_with_respect_to_a_grothendieck_topology_on_a_site_for_a_category_fibered_in_groupoids}{satisfies descent with respect to the fppf topology} on $X_{\mathrm{fppf}}$. That is, for every $U \in X_{\mathrm{fppf}}$ and every covering sieve $S$ in the fppf topology on $U$, every \CrefAndHyperrefIfExist{definition:descent_datum_for_a_category_fibered_in_groupoids_over_a_category_relative_to_a_sieve_on_an_object_of_the_base_category}{descent datum for $\mathcal{F}$ relative to $S$} is \CrefAndHyperrefIfExist{definition:effective_descent_datum_for_a_category_fibered_in_groupoids_over_a_category_relative_to_a_sieve_on_an_object_of_the_base_category}{effective}.
\end{definition}
