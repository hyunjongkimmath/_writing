
\begin{definition}[Linear algebraic group over a scheme] \label{definition:linear_algebraic_group_over_a_scheme}
Let \(S\) be a base scheme.  
A \hldef{linear algebraic group over $S$} is an \CrefAndHyperrefIfExist{definition:algebraic_group_scheme_over_a_scheme}{affine group scheme} \(G\) over \(S\) that is \CrefAndHyperrefIfExist{definition:locally_of_finite_presentation_finite_presentation_morphism_of_schemes}{finitely presented} and \CrefAndHyperrefIfExist{definition:smooth_morphism_of_schemes}{smooth} over \(S\),  
and such that for some integer \(n \geq 1\), there exists a \CrefAndHyperrefIfExist{definition:open_and_closed_immersions_of_schemes}{closed immersion} of \(S\)-group schemes
$$ G \hookrightarrow \operatorname{GL}_{n, S}.$$
\CrefIfExists{definition:general_linear_group_over_a_scheme}
\end{definition}
