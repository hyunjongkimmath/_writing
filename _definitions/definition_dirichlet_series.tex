\begin{definition}[Dirichlet series]
\label{definition:dirichlet_series}
\TODO{AI generated, verify}
Let $(a_n)_{n=1}^\infty$ be a sequence of complex numbers. The \hldef{Dirichlet series with coefficients $(a_n)$} is the formal expression
$$\hlin{\sum_{n=1}^\infty \frac{a_n}{n^s}}$$
where $s \in \mathbb{C}$ is a complex variable. For each fixed $s$, the series is interpreted as the \CrefAndHyperrefIfExist{definition:infinite_series_and_convergence_of_infinite_series_in_a_topological_abelian_semigroup}{infinite series} $\sum_{n=1}^\infty a_n n^{-s}$, where $n^{-s}$ is the \CrefAndHyperrefIfExist{definition:real_exponentiation_via_supremum}{real exponentiation} when $s \in \mathbb{R}$ and the \CrefAndHyperrefIfExist{definition:complex_exponential_function}{complex exponentiation} when $s \in \mathbb{C}$ (i.e., $n^{-s} = \exp(-s \ln n)$).
\end{definition}
