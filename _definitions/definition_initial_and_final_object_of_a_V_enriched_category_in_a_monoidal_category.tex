Let $(\mathcal{V}, \otimes, \mathbf{1})$ be a \CrefAndHyperrefIfExist{definition:monoidal_category}{monoidal category}. Let $\mathcal{C}$ be a \CrefAndHyperrefIfExist{definition:category_enriched_in_a_monoidal_category}{$\mathcal{V}$-enriched category}.

\begin{enumerate}
    \item Suppose that $\mathcal{V}$ has an \CrefAndHyperrefIfExist{definition:initial_final_zero_objects_of_a_category}{initial object}, denoted by \hl{$\bot_{\mathcal{V}}$}. An object $I \in \operatorname{Ob}(\mathcal{C})$ is called an \hldef{initial object of $\mathcal{C}$} if for every pair of objects $X, Y \in \operatorname{Ob}(\mathcal{C})$ there exist isomorphisms in $\mathcal{V}$
    $$\underline{\operatorname{Hom}}_{\mathcal{C}}(X, I) \cong \bot_{\mathcal{V}} \quad \text{and} \quad \underline{\operatorname{Hom}}_{\mathcal{C}}(I, Y) \cong \bot_{\mathcal{V}}.$$

    \item Suppose that $\mathcal{V}$ has a \CrefAndHyperrefIfExist{definition:initial_final_zero_objects_of_a_category}{final object}, denoted by \hl{$*_{\mathcal{V}}$}. An object $T \in \operatorname{Ob}(\mathcal{C})$ is called a \hldef{final object of $\mathcal{C}$} (or a \hldef{terminal object of $\mathcal{C}$}) if for every pair of objects $X, Y \in \operatorname{Ob}(\mathcal{C})$ there exist isomorphisms in $\mathcal{V}$
    $$\underline{\operatorname{Hom}}_{\mathcal{C}}(X, T) \cong *_{\mathcal{V}} \quad \text{and} \quad \underline{\operatorname{Hom}}_{\mathcal{C}}(T, Y) \cong *_{\mathcal{V}}.$$
\end{enumerate}
\end{definition}
