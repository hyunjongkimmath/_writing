\begin{definition} \label{definition:flat_functor_between_categories}
   A \CrefAndHyperrefIfExist{definition:functor_between_categories}{functor} $u: \calC \to \calD$ of \CrefAndHyperrefIfExist{definition:category}{categories} is said to be \hldef{flat} if the \CrefAndHyperrefIfExist{definition:left_kan_extension_of_presheaves_along_a_functor_of_sheaves_along_cocontinuous_functor_of_sites}{Left Kan extension} functor 
   $$\operatorname{Lan}_u: \PreShv(\calC, \Sets) \to \PreShv(\calD, \Sets)$$
   is \CrefAndHyperrefIfExist{definition:left_right_exact_functor_between_categories}{left exact}.
\end{definition}
