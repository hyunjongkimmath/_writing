\begin{definition} \label{definition:commutator_of_two_subsets_of_a_group}
   Let $G$ be a \CrefAndHyperrefIfExist{definition:group}{group}. 

   For nonempty subsets $S, T \subseteq G$, the \hldef{commutator subgroup of $S$ and $T$}, denoted by \hl{$[S,T]$}, is the subgroup of $G$ \CrefAndHyperrefIfExist{definition:subgroup_of_a_group_generated_by_elements}{generated by} all commutators of elements from $S$ and $T$:
   $$[S,T] = \langle [s,t] = sts^{-1}t^{-1} \mid s \in S, t \in T \rangle \leq G.$$

   When $S = T$, we write $[S,S]$ instead of $[S,T]$. In particular, the \hldef{commutator subgroup (or derived subgroup) of $G$}, denoted by \hl{$[G,G]$}, is
   $$[G,G] = \langle [g,h] \mid g,h \in G \rangle \leq G.$$

   The group $G$ is \CrefAndHyperrefIfExist{definition:group}{abelian} if and only if $[G,G]$ is the trivial subgroup.
\end{definition}
