\begin{definition} \label{definition:measurable_function_from_a_measure_space_into_the_extended_real_line_or_complex_plane}
Let $(X, \mathcal{A}, \mu)$ be a \CrefAndHyperrefIfExist{definition:sigma_algebra_on_a_set_and_measure_space}{measure space}.

\TODO{borel set}
\begin{enumerate}
  \item A function $f : X \to \overline{\mathbb{R}} = \mathbb{R} \cup \{-\infty, \infty\}$\CrefIfExists{definition:extended_real_line} is called a \hldef{measurable function into the extended real line} if $f^{-1}((a, \infty]) \in \mathcal{A}$ for every $a \in \mathbb{R}$. Equivalently, $f^{-1}(B) \in \mathcal{A}$ for every Borel set $B \subseteq \overline{\mathbb{R}}$.

  \item A function $f : X \to \mathbb{C}$ is called a \hldef{measurable function into the complex plane} if both $\operatorname{Re}(f)$ and $\operatorname{Im}(f)$ are measurable real-valued functions. Equivalently, $f^{-1}(B) \in \mathcal{A}$ for every Borel set $B \subseteq \mathbb{C}$.

  % \item A function $f : X \to \overline{\mathbb{R}}$ or $f : X \to \mathbb{C}$ is called simply a \hldef{measurable function} (when the codomain is clear from context) if it satisfies the appropriate condition above.
\end{enumerate}
\end{definition}
