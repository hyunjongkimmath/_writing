\begin{definition} \label{definition:p_adic_valuation_of_a_point_of_a_dedekind_scheme}
    Let $S$ be a \CrefAndHyperrefIfExist{definition:dedekind_scheme}{Dedekind scheme}. For each closed point $s \in S$, the local ring $\mathcal{O}_{S,s}$ is a DVR. The \hldef{$s$-adic valuation on the function field $K(S)$} is the discrete valuation $v_s$ defined by $v_s(f) = \operatorname{ord}_s(f)$, i.e., for any nonzero rational function $f \in K(S)^\times$, one writes $f = u \cdot \pi^n$ in $\mathcal{O}_{S,s}$ with $u \in \mathcal{O}_{S,s}^\times$ and a \CrefAndHyperrefIfExist{definition:discrete_valuation_ring}{uniformizer} $\pi$, setting $v_s(f) = n$.
\end{definition}
