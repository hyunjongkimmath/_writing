\begin{definition}[Finite flat subgroup scheme of rank $N$] \label{definition:finite_flat_subgroup_scheme_of_rank_N}
    Let $S$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme}, let $G$ be an \CrefAndHyperrefIfExist{definition:algebraic_group_scheme_over_a_scheme}{algebraic group scheme over $S$}, and let $N$ be a positive integer. A \hldef{finite flat subgroup scheme of rank $N$} of $G$ is a closed subscheme $H \hookrightarrow G$ such that:
    \begin{enumerate}
        \item $H$ is a \CrefAndHyperrefIfExist{definition:subgroup_scheme_of_an_algebraic_group_scheme_over_a_scheme}{subgroup scheme} of $G$,
        \item the structural morphism $H \to S$ is a \CrefAndHyperrefIfExist{definition:finite_morphism_of_schemes}{finite morphism} and a \CrefAndHyperrefIfExist{definition:flat_morphism_of_schemes}{flat morphism}, and
        \item for every geometric point $\bar{s} : \operatorname{Spec}(\bar{k}(s)) \to S$, the fiber $H_{\bar{s}}$ is a finite group scheme over $\bar{k}(s)$ of order $N$.
    \end{enumerate}
    The integer $N$ is called the \hldef{rank} (or \hldef{degree}) of $H$ over $S$, denoted
    $$\hlin{\operatorname{rank}_S(H) = N}.$$
\end{definition}
