\begin{definition} \label{definition:closed_geometric_point_of_a_scheme}
    A \CrefAndHyperrefIfExist{definition:geometric_point_of_a_scheme}{geometric point} $\overline{s}: \Spec \Omega \to S$ of a scheme $S$ is said to be \hldef{closed} if its image is a \CrefAndHyperrefIfExist{definition:closed_point_of_a_topological_space}{closed point} of $S$.
\end{definition}
