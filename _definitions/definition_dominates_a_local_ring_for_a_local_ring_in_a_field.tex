\begin{definition} \label{definition:dominates_a_local_ring_for_a_local_ring_in_a_field}
    Let $K$ be a \CrefAndHyperrefIfExist{definition:field}{field}. Let $(A,\mathfrak{m}_A)$ and $(B,\mathfrak{m}_B)$ be \CrefAndHyperrefIfExist{definition:local_ring}{local rings} contained in $K$. 
    We say that $B$ \hldef{dominates $A$} if 
    \begin{itemize}
        \item $A \subseteq B$ and 
        \item $\mathfrak{m}_B \cap A = \mathfrak{m}_A$.
    \end{itemize}
\end{definition}
