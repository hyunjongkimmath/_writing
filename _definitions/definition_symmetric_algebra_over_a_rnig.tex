\begin{definition}[cf. {\cite[Chapter I, 7.4]{knus_qhfr}} in the case that the base ring is commutative.] \label{definition:symmetric_algebra_over_a_rnig}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{not necessarily commutative ring}. A \hldef{symmetric $R$-algebra} is an \CrefAndHyperrefIfExist{definition:algebra_of_a_ring}{$R$-algebra $A$} such that:
    \begin{enumerate}
        \item $A$ is \CrefAndHyperrefIfExist{definition:finitely_generated_algebra_over_a_not_necessarily_commutative_ring}{finitely generated} and \CrefAndHyperrefIfExist{definition:projective_bimodule_over_rings}{projective} as a \CrefAndHyperrefIfExist{definition:module_of_a_ring}{one-sided $R$-module}.

        \item There exists an isomorphism of $(A, A)$-bimodules $\psi: A \xrightarrow{\cong} A^\vee$.
        % , where the \CrefAndHyperrefIfExist{definition:dual_of_a_left_right_two_sided_module}{dual} $A^\vee = \operatorname{Hom}_{R}(A, R) = \Hom_{R-\bbZ}(A,R)$ is the set of \CrefAndHyperrefIfExist{definition:homomorphism_of_modules_over_a_ring}{left $R$-linear maps}.
    \end{enumerate}
    The $(A, A)$-bimodule structure on $A^*$ is defined for $f \in A^*$ and $u, v, a \in A$ by:
    \begin{align*}
        (u \cdot f)(a) &= f(au) \\
        (f \cdot v)(a) &= f(va)
    \end{align*}
    Under this isomorphism, the element $\varphi = \psi(1)$ is called the \hldef{symmetric trace}, and necessarily satisfies $\varphi(ab) = \varphi(ba)$ for all $a, b \in A$.

    A posteriori, the fact that $A$ is a symmetric $R$-algebra implies that $A$ is in fact projective as an $R$-$R$-bimodule. 
\end{definition}
