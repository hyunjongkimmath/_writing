\begin{definition} \label{definition:milnor_K_theory_of_a_field}
    Let $F$ be a \CrefAndHyperrefIfExist{definition:field}{field}. The \hldef{Milnor $K$-theory of $F$} is the graded associated ring \hl{$\MKtheory(F)$} generated by the degree $1$ symbols \hl{$[u]$} for each $u \in F^\times$ subject to the Steinberg relations: $[a] \cdot [1-a] = 0$ for each $a \in F^\times - \{1\}$.
\end{definition}
