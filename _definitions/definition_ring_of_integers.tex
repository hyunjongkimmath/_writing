\begin{definition} \label{definition:ring_of_integers}
    The \hldef{ring of integers}, universally denoted by \hl{$\bbZ$}, is the unique (up to \CrefAndHyperref{definition:ring_homomorphism}{isomorphism}) \CrefAndHyperref{definition:ordered_ring}{ordered} \CrefAndHyperrefIfExist{definition:domain_integral_domain}{integral domain} such that there exists a subset $P \subset \bbZ$ such that 
    \begin{enumerate}
        \item $P$ is nonempty, and
        \item Every non-empty subset of $P$ contains a least element.
    \end{enumerate}
    \CrefIfExists{corollary:integers_existence_uniqueness}
\end{definition}
