\begin{definition}[see {\cite[Chapter I, 7.2]{knus_qhfr}}] \label{definition:scharlau_transfer_for_sesquilinear_forms_on_modules_over_unitary_rings}
   Let $\eta: (A,\Lambda, \varepsilon) \to (A', \Lambda', \varepsilon')$ be a \CrefAndHyperrefIfExist{definition:unitary_ring}{homomorphism of unitary rings}. Write $\overline{\cdot}$ for both of the involutions on $A$ and $A'$. Let $\tau: A' \to A$ be a \CrefAndHyperrefIfExist{definition:homomorphism_of_modules_over_a_ring}{morphism of $A$-bimodules} and assume that 
    $$\tau(\overline{a}) = \overline{\tau(a)}$$
    for all $a \in A'$.

    The \CrefAndHyperrefIfExist{definition:scharlau_transfer_of_sesquilinear_forms_on_modules_under_involution_ring_homomorphism}{Scharlau transfer}
    $$\tau: \Sesq_{A'}(M') \to \Sesq_A(M'_{(A)})$$
    (\Cref{definition:sesquilinear_form_on_a_module_over_a_ring_with_involution}) restricts to a map 
    $$\hlin{\tau: \Sesq_{\Lambda'}^{\varepsilon'}(M') \to \Sesq_\Lambda^{\varepsilon}(M'_{(A)})}$$

\end{definition}
