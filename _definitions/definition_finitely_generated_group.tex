\begin{definition} \label{definition:finitely_generated_group}
    Let $G$ be a \CrefAndHyperrefIfExist{definition:group}{group}.
    It is said to be \hldef{finitely generated} if there exists some \CrefAndHyperrefIfExist{definition:countable_finite_uncountable_sets}{finite} \CrefAndHyperrefIfExist{definition:subset_of_a_set}{subset} $S \subseteq G$ such that $\langle S \rangle = G$ (\Cref{definition:subgroup_of_a_group_generated_by_elements}).
\end{definition}
