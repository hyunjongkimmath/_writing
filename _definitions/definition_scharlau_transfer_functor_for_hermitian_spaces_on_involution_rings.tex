\begin{definition}[see {\cite[Chapter I, 7.2]{knus_qhfr}}] \label{definition:scharlau_transfer_functor_for_hermitian_spaces_on_involution_rings}

    Let $\eta: A \to A'$ be a \CrefAndHyperrefIfExist{definition:homomorphism_of_involution_rings}{homomorphism of rings with involution}, and assume that $A'$ is \CrefAndHyperrefIfExist{definition:finitely_generated_modules_over_rings}{finitely generated} and \CrefAndHyperrefIfExist{definition:projective_bimodule_over_rings}{projective} as a \CrefAndHyperrefIfExist{definition:bimodule_over_rings_module_over_a_ring}{right $A$-module}. Let $\varepsilon \in A$ be an element satisfying $\varepsilon \cdot \overline{\varepsilon} = 1$ and let $\varepsilon' = \eta(\varepsilon)$.
    Given a right $A'$-module $M'$, write $M'_{(A)}$ for the \CrefAndHyperrefIfExist{definition:restriction_of_scalars_of_a_left_or_right_module_along_a_ring_homomorphism_from_the_base_ring}{restriction} of $M'$ as a right $A$-module (via $\eta$). Let $\tau: A' \to A$ be a \CrefAndHyperrefIfExist{definition:homomorphism_of_modules_over_a_ring}{morphism of $A$-bimodules} and assume that 
    $$\tau(\overline{a}) = \overline{\tau(a)}$$
    for all $a \in A'$. 


    The \CrefAndHyperrefIfExist{definition:scharlau_transfer_for_sesquilinear_forms_on_modules_over_unitary_rings}{Scharlau transfer} map
    $$\tau: \Sesq_{A'}(M') \to \Sesq_A(M'_{(A)})$$
    (\Cref{definition:scharlau_transfer_of_sesquilinear_forms_on_modules_under_involution_ring_homomorphism}) induces functors

    \hlalign{    \begin{align*}
            T: \mathfrak{Herm}^{\varepsilon'}(A') \to \mathfrak{Herm}^\varepsilon(A) \\ 
            T: \mathfrak{Herm}_+^{\varepsilon'}(A') \to \mathfrak{Herm}_+^\varepsilon(A)
    \end{align*}}

    (\Cref{definition:category_of_varepsilon_hermitian_modules_over_a_ring_with_involution})

    It is appropriate to call these induced functors the \hldef{Scharlau transfer functor for Hermitian spaces over involution rings}.


\end{definition}
