\begin{definition}[Normalization of a reduced scheme] \label{definition:normalization_of_a_reduced_scheme}
    Let $X$ be a \CrefAndHyperrefIfExist{definition:reduced_scheme}{reduced scheme}.
    \TODO{dominant morphism}
    The \hldef{normalization of $X$} is a morphism $\nu\colon \widetilde{X} \to X$ from a \CrefAndHyperrefIfExist{definition:normal_and_factorial_scheme}{normal scheme} $\widetilde{X}$ such that for any normal scheme $Y$ and any dominant morphism $f\colon Y \to X$, there exists a unique factorization through $\nu$. Concretely, if $X = \operatorname{Spec}(A)$ is an \CrefAndHyperrefIfExist{definition:affine_open_subscheme_of_a_scheme}{affine open subset} with $A$ reduced, then $\widetilde{X} = \operatorname{Spec}(\widetilde{A})$ where $\widetilde{A}$ is the integral closure of $A$ in the \CrefAndHyperrefIfExist{definition:field_of_fractions_of_an_integral_domain}{total ring of fractions} $\operatorname{Frac}(A)$.

    \TODO{separate out the following}
    More generally, let $X$ be an \CrefAndHyperrefIfExist{definition:integral_scheme}{integral scheme} with \CrefAndHyperrefIfExist{definition:function_field_over_a_field}{function field} $K$, and let $L/K$ be a \CrefAndHyperrefIfExist{definition:finite_field_extension}{finite field extension}. The \hldef{normalization of $X$ in $L$} is a morphism $\nu\colon X' \to X$ from an integral scheme $X'$ such that the function field of $X'$ is $L$ and for any integral scheme $Y$ with function field contained in $L$ and any dominant morphism $f\colon Y \to X$, there exists a unique factorization through $\nu$. Concretely, if $X = \operatorname{Spec}(A)$ is affine, then $X' = \operatorname{Spec}(\widetilde{A})$ where $\widetilde{A}$ is the integral closure of $A$ in $L$.
\end{definition}
