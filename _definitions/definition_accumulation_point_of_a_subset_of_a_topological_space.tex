\begin{definition} \label{definition:accumulation_point_of_a_subset_of_a_topological_space}
    Let $X$ be a \CrefAndHyperrefIfExist{definition:topological_space}{topological space}. Let $A \subseteq X$.
    A point $p \in X$ is an \hldef{accumulation point of $A$} if the following equivalent conditions hold:
    \begin{enumerate}
        \item every \CrefAndHyperrefIfExist{definition:neighborhood_and_neighborhood_basis_of_a_point_in_a_topological_space}{open neighborhood} $U$ of $p$ contains a point $y \in A$ such that $y \neq p$.
        \item every \CrefAndHyperrefIfExist{definition:neighborhood_and_neighborhood_basis_of_a_point_in_a_topological_space}{neighborhood} $U$ of $p$ contains a point $y \in A$ such that $y \neq p$.
    \end{enumerate}
\end{definition}
