\begin{definition} \label{definition:equicontinuous_family_of_functions_between_metric_spaces_valued_in_ordered_commutative_monoid_with_adjoined_infinity}

    Let $T$ be an \CrefAndHyperrefIfExist{definition:ordered_commutative_monoid}{ordered commutative monoid}, let $T^\infty$ be the \CrefAndHyperrefIfExist{definition:ordered_monoid_with_adjoined_infinity}{ordered monoid with adjoined infinity} obtained from $T$, and let $(X, d_X)$ and $(Y, d_Y)$ be \CrefAndHyperrefIfExist{definition:extended_metric_on_a_set_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity}{extended metric spaces valued in $T^\infty$}. Let $\mathcal{F}$ be a set of \CrefAndHyperrefIfExist{definition:function_of_sets}{functions} $f : X \to Y$.

    \begin{itemize}
        \item The family $\mathcal{F}$ is called \hldef{equicontinuous at a point $x_0 \in X$} if for every $\varepsilon \in T_{>0}$ there exists $\delta \in T_{>0}$ such that for all $f \in \mathcal{F}$ and all $x \in X$,
        $$d_X(x, x_0) < \delta \implies d_Y(f(x), f(x_0)) < \varepsilon.$$

        \item The family $\mathcal{F}$ is called \hldef{equicontinuous} (or \hldef{uniformly equicontinuous}) if for every $\varepsilon \in T_{>0}$ there exists $\delta \in T_{>0}$ such that for all $f \in \mathcal{F}$ and all $x, y \in X$,
        $$d_X(x, y) < \delta \implies d_Y(f(x), f(y)) < \varepsilon.$$

        \item The family $\mathcal{F}$ is called \hldef{pointwise equicontinuous} if it is equicontinuous at every point $x \in X$.
    \end{itemize}

    If $(Y, d_Y)$ is \CrefAndHyperrefIfExist{definition:complete_extended_metric_space_valued_in_ordered_commutative_monoid_with_adjoined_infinity}{complete}, then a pointwise equicontinuous family of functions on a \CrefAndHyperrefIfExist{definition:compact_topological_space}{compact} metric space is uniformly equicontinuous.
\end{definition}
