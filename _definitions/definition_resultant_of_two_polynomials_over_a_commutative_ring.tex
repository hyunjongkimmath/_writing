\begin{definition} \label{definition:resultant_of_two_polynomials_over_a_commutative_ring}

    Let $R$ be a \CrefAndHyperrefIfExist{definition:commutative_ring}{commutative ring}. Let $f, g \in R[x]$ be two non-zero polynomials of degrees $m = \deg(f)$ and $n = \deg(g)$, with leading coefficients $a_m$ and $b_n$, respectively. Write
    $$ f = a_m x^m + a_{m-1} x^{m-1} + \cdots + a_0, \qquad g = b_n x^n + b_{n-1} x^{n-1} + \cdots + b_0. $$

    The \hldef{Sylvester matrix of $f$ and $g$}, denoted by \hl{$\operatorname{Syl}(f,g)$}, is the $(m+n) \times (m+n)$ matrix over $R$ whose first $n$ rows consist of the coefficients of $f$ shifted right by one entry at each successive row, and whose last $m$ rows consist of the coefficients of $g$ similarly shifted:
    $$
    \operatorname{Syl}(f,g) =
    \begin{pmatrix}
        a_m & a_{m-1} & \cdots & a_0 & 0 & \cdots & 0 \\
        0 & a_m & \cdots & a_1 & a_0 & \cdots & 0 \\
        \vdots & \vdots & \ddots & \vdots & \vdots & \ddots & \vdots \\
        0 & 0 & \cdots & a_m & a_{m-1} & \cdots & a_0 \\
        b_n & b_{n-1} & \cdots & b_0 & 0 & \cdots & 0 \\
        0 & b_n & \cdots & b_1 & b_0 & \cdots & 0 \\
        \vdots & \vdots & \ddots & \vdots & \vdots & \ddots & \vdots \\
        0 & 0 & \cdots & b_n & b_{n-1} & \cdots & b_0
    \end{pmatrix}.
    $$

    The \hldef{resultant of $f$ and $g$ over $R$}, denoted by \hl{$\operatorname{Res}(f,g)$}, is the determinant of the Sylvester matrix:
    $$ \hlin{\operatorname{Res}(f,g) = \det \operatorname{Syl}(f,g).} $$

    If either $f = 0$ or $g = 0$, we set $\operatorname{Res}(f,g) = 0$.
\end{definition}
