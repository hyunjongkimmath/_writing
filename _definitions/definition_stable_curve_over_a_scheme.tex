\begin{definition} \label{definition:stable_curve_over_a_scheme}
    Let $S$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme}.
    A \hldef{stable curve over $S$} is a \CrefAndHyperrefIfExist{definition:proper_morphism_of_schemes}{proper} \CrefAndHyperrefIfExist{definition:curve_over_a_scheme}{curve} $(C, \pi)$ over $S$ such that:
    \begin{enumerate}

        \item The \CrefAndHyperrefIfExist{definition:geometric_fiber_of_a_morphism_of_schemes_at_a_geometric_point}{geometric fibers} of $\pi$ have at most \CrefAndHyperrefIfExist{definition:ordinary_double_point_of_a_finite_type_scheme_over_a_field}{ordinary double points} as \CrefAndHyperrefIfExist{definition:singularity_of_a_scheme}{singularities} (i.e., each fiber is a connected, reduced scheme whose only singularities are nodal), and
        \item the \CrefAndHyperrefIfExist{definition:dualizing_complex_of_proper_finite_type_scheme_over_locally_noetherian_scheme}{dualizing sheaf} $\omega_{C/S}$ of the morphism is \CrefAndHyperrefIfExist{definition:relative_ampleness_of_a_line_bundle_on_a_scheme}{relatively ample} over $S$.
    \end{enumerate}

\end{definition}
