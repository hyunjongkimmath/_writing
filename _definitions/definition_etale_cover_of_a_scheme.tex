\begin{definition} \label{definition:etale_cover_of_a_scheme}
    Let $X$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme}. 
    A \hldef{\'etale cover of $X$} is a scheme $Y$ equipped with a \CrefAndHyperrefIfExist{definition:injective_surjective_bijective_map_of_sets}{surjective} \CrefAndHyperrefIfExist{definition:morphism_of_schemes}{morphism} $Y \to X$.
\end{definition}
