\begin{definition} \label{definition:morphism_of_pointed_pair_of_topological_spaces}
    A \hldef{morphism $f: (X,A,x) \to (Y,B,y)$ of pointed pairs of topological spaces} is a continuous map $f: X \to Y$ that constitutes 
    \begin{enumerate}
        \item a map $(X,A) \to (Y,B)$ \CrefAndHyperrefIfExist{definition:map_of_pairs_of_topological_spaces}{of pairs},
        \item a map $(x,x) \to (Y,y)$ \CrefAndHyperrefIfExist{definition:pointed_topological_space}{of pointed spaces}, and
        \item a map $(A,x) \to (B,y)$ of pointed spaces, and
    \end{enumerate}
    In other words, $f$ is a continuous map $X \to Y$ such that $f(A) \subseteq B$ and $f(x) = y$. 
\end{definition}
