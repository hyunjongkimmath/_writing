\begin{definition} \label{definition:holomorph_of_a_group}
    Let $G$ be a \CrefAndHyperrefIfExist{definition:group}{group}. The \hldef{holomorph of $G$}, denoted \hl{$\operatorname{Hol}(G)$}, is the \CrefAndHyperrefIfExist{definition:semidirect_product_of_groups_external_and_internal}{external semidirect product}
    $$\operatorname{Hol}(G) = G \rtimes \operatorname{Aut}(G)$$
    where $\operatorname{Aut}(G)$ is the \CrefAndHyperrefIfExist{definition:automorphism_of_a_group_automorphism_group_of_a_group}{automorphism group} of $G$, and the action $\operatorname{Aut}(G) \to \operatorname{Aut}(G)$ is the natural evaluation action: for $\alpha \in \operatorname{Aut}(G)$ and $g \in G$,
    $$\alpha \cdot g = \alpha(g).$$
\end{definition}
