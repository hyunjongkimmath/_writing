\begin{definition} \label{definition:trivial_torsor_principal_homogeneous_space_of_a_group}
    Let $G$ be a \CrefAndHyperrefIfExist{definition:group}{group}. 
    The \hldef{trivial principal homogeneous space} or \hldef{trivial torsor} \hl{$G$} is the \CrefAndHyperrefIfExist{definition:principal_homogeneous_space_torsor_for_a_group}{principal homogeneous space} whose underlying set is $G$ and whose group action by $G$ is multiplication (either on the left or right).
\end{definition}
