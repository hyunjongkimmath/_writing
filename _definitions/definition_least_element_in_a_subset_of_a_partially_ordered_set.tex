\begin{definition} \label{definition:least_element_in_a_subset_of_a_partially_ordered_set}
Let $(X, \leq)$ be a \CrefAndHyperrefIfExist{definition:partially_ordered_set}{partially ordered set} and let $A \subseteq X$ be a subset. 
An element $m \in A$ is a \hldef{least element of $A$} (or \hldef{minimum of $A$}) if for all $x \in A$, $m \leq x$. If a least element exists, it is unique\CrefIfExists{lemma:maximum_minimum_element_of_a_subset_of_a_poset_is_unique_if_exists}.
\end{definition}
