\begin{definition}[cf. {\cite[Definition 5.2.10]{weibel}}, {\cite[Tag 08DU]{stacks-project}}] \label{definition:regular_spectral_sequence_in_an_abelian_category}
    Let $\calA$ be an abelian category. 
    \begin{enumerate}
        \item Let $E$ be a \CrefAndHyperrefIfExist{definition:spectral_sequence_in_an_abelian_category}{homological spectral sequence} in $\calA$. We say that $E$ is \hldef{regular} if for each $p$ and $q$, the differentials $d_{pq}^r: E_{p,q}^r \to E_{p-r, q+r-1}^r$ are $0$ for all large $r$.  
 
        \item Let $E$ be a \CrefAndHyperrefIfExist{definition:spectral_sequence_in_an_abelian_category}{cohomological spectral sequence} in $\calA$. We say that $E$ is \hldef{regular} if for each $p$ and $q$, the differentials $d^{pq}_r: E^{p,q}_r \to E^{p-r, q+r-1}_r$ are $0$ for all large $r$.  
    \end{enumerate}
\end{definition}
