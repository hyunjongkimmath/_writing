\begin{definition} \label{definition:n_ary_operation_on_a_set}
    % EDITOR APPROVED
    Let $S$ be a \CrefAndHyperrefIfExist{definition:zermelo_fraenkel_set_theory}{set}. Let $n \geq 1$ be an \CrefAndHyperrefIfExist{definition:ring_of_integers}{integer}.
    An \hldef{$n$-ary operation/operator on $S$} is a \CrefAndHyperrefIfExist{definition:function_of_sets}{function} $S^n \to S$\CrefIfExists{definition:product_of_sets}. 
    
    For $n=2$ and $3$, an $n$-ary operator on $S$ is called a \hldef{binary operator} and \hldef{ternary operator} respectively.

    One common way to denote a generic binary operator is as ``multiplication'' --- given elements $a,b \in S$, one often write \hl{$a \cdot b$}, or \hl{$ab$} for short, as the \CrefAndHyperrefIfExist{definition:function_of_sets}{image} of the pair $(a,b)$ via the binary operator $S \times S \to S$.
\end{definition}
