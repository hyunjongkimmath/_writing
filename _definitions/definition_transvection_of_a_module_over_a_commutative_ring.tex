\begin{definition} \label{definition:transvection_of_a_module_over_a_commutative_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:commutative_ring}{commutative ring}. Let $M$ be a \CrefAndHyperrefIfExist{definition:bimodule_over_rings_module_over_a_ring}{module} over $R$.
    A \hldef{transvection} is an automorphism $T \in \operatorname{Aut}_R(M)$ of the form
    $T(x) = x + \phi(x)v$
    where $\phi \in M^*$ \CrefIfExists{definition:dual_of_a_left_right_two_sided_module} and $v \in M$ such that 
    \begin{enumerate}
        \item $\phi(v) = 0$, and
        \item $v$ is a unimodular element of $M$ or $\phi$ is a unimodular element of $M^*$.
    \end{enumerate}
\end{definition}
