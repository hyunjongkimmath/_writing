\begin{definition} \label{definition:unit_motive_over_a_scheme}
Let $S$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme}.
The \hldef{unit (Chow) motive} is the \CrefAndHyperrefIfExist{definition:relative_chow_motive_over_a_noetherian_base_and_commutative_ring_of_coefficients}{relative Chow motive} \hl{$\mathds{1}_S = (S, \mathrm{id}_S, 0)$}.
\end{definition}
