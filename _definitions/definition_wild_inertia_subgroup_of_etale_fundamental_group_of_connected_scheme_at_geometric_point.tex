\begin{definition} \label{definition:wild_inertia_subgroup_of_etale_fundamental_group_of_connected_scheme_at_geometric_point}
    Let $X$ be a \CrefAndHyperrefIfExist{definition:disconnected_and_connected_topological_spaces}{connected} \CrefAndHyperrefIfExist{definition:scheme}{scheme}, let $x \in X$ be a point, and let $\overline{x}$ be a \CrefAndHyperrefIfExist{definition:geometric_point_of_a_scheme}{geometric point} of $X$ whose image is $x$. Let $p$ be the characteristic of the residue field $k(x)$.

    Let $I_{\overline{x}}$ be the \CrefAndHyperrefIfExist{definition:inertia_group_of_the_etale_fundamental_group_of_a_connected_scheme_at_a_geometric_point}{inertia subgroup} of the \CrefAndHyperrefIfExist{definition:etale_fundamental_group_of_a_connected_scheme}{\'etale fundamental group} $\pioneet(X, \overline{x})$.
    For each finite \'etale \CrefAndHyperrefIfExist{definition:galois_morphism_of_schemes}{Galois cover} $Y$ of $X$ with geometric point $\overline{y}$ mapping to $\overline{x}$, let $I_{\overline{y}} \subseteq \operatorname{Aut}(Y/X)$ be the \CrefAndHyperrefIfExist{definition:inertia_group_of_a_morphism_of_schemes_at_a_geometric_point}{inertia subgroup} and let $P_{\overline{y}}$ be its unique smallest normal subgroup such that $I_{\overline{y}}/P_{\overline{y}}$ has order prime to $p$.
    The \hldef{wild inertia subgroup} \hl{$P_{\overline{x}}$} is the inverse limit
    $$P_{\overline{x}} = \varprojlim_{(Y, \overline{y})} P_{\overline{y}}$$

    Equivalently, $P_{\overline{x}}$ is the unique smallest closed normal subgroup of $I_{\overline{x}}$ such that $I_{\overline{x}}/P_{\overline{x}}$ is a procyclic group of order prime to $p$. It is a closed normal subgroup of $\pioneet(X, \overline{x})$, and the quotient $I_{\overline{x}}/P_{\overline{x}}$ is the \CrefAndHyperrefIfExist{definition:tame_inertia_subgroup_of_etale_fundamental_group_of_connected_scheme_at_geometric_point}{tame inertia subgroup}.
\end{definition}
