\begin{definition} \label{definition:unramified_extension_of_absolute_valued_fields_general}
    \begin{enumerate}
        \item An \CrefAndHyperrefIfExist{definition:extension_of_absolute_valued_fields}{extension of absolute valued fields} $(L, |\cdot|_L)/(K, |\cdot|_K)$ is \hldef{unramified} (or that $(L, |\cdot|_L)$ is \hldef{unramified over $(K, |\cdot|_K)$}) if:
        \begin{enumerate}
            \item \textbf{Non-Archimedean case:} If $|\cdot|_K$ is \CrefAndHyperrefIfExist{definition:archimedean_absolute_value_on_a_field}{non-archimedean}, let $\mathcal{O}_K$ and $\mathcal{O}_L$ be the respective rings of integers $\{x \mid |x| \leq 1\}$. The extension is unramified if:
            \begin{enumerate}
                \item The value groups satisfy $|L^\times|_L = |K^\times|_K$.
                \item The maximal ideal $\mathfrak{m}_L$ is generated by $\mathfrak{m}_K$.
                \item The \CrefAndHyperrefIfExist{definition:valuation_ring_of_a_nonarchimedean_valued_field}{residue field} extension $l/k$ is \CrefAndHyperrefIfExist{definition:separable_algebraic_field_extension}{separable algebraic}.
            \end{enumerate}
            \item \textbf{Archimedean case:} If $|\cdot|_K$ is \CrefAndHyperrefIfExist{definition:archimedean_absolute_value_on_a_field}{archimedean}, the extension is unramified if for every (equivalently, for any) embedding of $L$ into its \CrefAndHyperrefIfExist{definition:completion_of_a_valuation_field}{completion} $\widehat{L}$, the local degree $[\widehat{L} : \widehat{K}]$ is $1$.
        \end{enumerate}

        \item Let $L/K$ be an \CrefAndHyperrefIfExist{definition:algebraic_element_minimal_polynomial_of_an_algebraic_element_algebraic_field_extension_transcendental_element_field_extension}{algebraic field extension}. 
        \begin{enumerate}
            \item Let $S$ be a set of (\CrefAndHyperrefIfExist{definition:equivalent_absolute_values_on_a_field}{equivalence classes} of) \CrefAndHyperrefIfExist{definition:absolute_value_on_a_ring_valued_in_an_ordered_semiring}{absolute values} on $K$. 
            \begin{enumerate}
                \item We say that $L/K$ is \hldef{unramified over $S$} if for all $|\cdot|_K \in S$, the \CrefAndHyperref{theorem:every_field_extension_of_an_absolute_valued_field_has_an_extension_of_the_absolute_value}{unique} absolute value on $L$ extending $|\cdot|_K$ is unramified over $|\cdot|_K$. 

                \item We say that $L/K$ is \hldef{unramified outside of $S$} if for all absolute value $|\cdot|_K$ on $K$ not in $S$, the \CrefAndHyperref{theorem:every_field_extension_of_an_absolute_valued_field_has_an_extension_of_the_absolute_value}{unique} absolute value on $L$ extending $|\cdot|_K$ is unramified over $|\cdot|_K$. 
            \end{enumerate}

        \item We may say that $L/K$ is \hldef{(everywhere) unramified} if it is unramified over the set of all \CrefAndHyperrefIfExist{definition:archimedean_and_nonarchimedean_extended_metrics_on_sets_valued_in_ordered_commutative_monoid_with_adjoined_infinity}{nonarchimedean} \CrefAndHyperrefIfExist{definition:absolute_value_on_a_field}{absolute values} of $K$. 
        
        % However, in the case that $K$ is a number field, to say that $L/K$ is \hldef{(everywhere) unramified} almost always means that $L/K$ is unramified over the set of all \CrefAndHyperrefIfExist{definition:archimedean_and_nonarchimedean_extended_metrics_on_sets_valued_in_ordered_commutative_monoid_with_adjoined_infinity}{nonarchimedean} \CrefAndHyperrefIfExist{definition:absolute_value_on_a_ring_valued_in_an_ordered_semiring}{absolute values} (i.e. the \CrefAndHyperrefIfExist{definition:place_of_a_global_field}{finite places}) of $K$ 

        \end{enumerate}
    \end{enumerate}


    
\end{definition}
