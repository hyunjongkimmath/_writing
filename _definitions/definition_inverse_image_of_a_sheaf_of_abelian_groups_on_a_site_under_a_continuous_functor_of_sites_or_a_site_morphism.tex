\begin{definition} \label{definition:inverse_image_of_a_sheaf_of_abelian_groups_on_a_site_under_a_continuous_functor_of_sites_or_a_site_morphism}
    % \label{definition:inverse_image_of_a_module_on_a_ringed_site_under_a_continuous_functor_of_sites_or_a_site_morphism}
    Let $(\calC,J)$ and $(\calD,K)$ be \CrefAndHyperrefIfExist{definition:grothendieck_topology_on_a_category_site_covering_sieve_topologically_generating_family}{sites}  with small \CrefAndHyperrefIfExist{definition:grothendieck_topology_on_a_category_site_covering_sieve_topologically_generating_family}{topological generating families} (or $U$-small topologically generating families if a \CrefAndHyperrefIfExist{definition:grothendieck_universe}{universe} $U$ is available), and let $u : \calC \to \calD$ be a \CrefAndHyperrefIfExist{definition:continuous_functor_of_sites}{continuous functor of sites}. 

    As per \Cref{definition:inverse_image_of_a_sheaf_under_a_continuous_functor_of_sites_or_a_site_morphism} and by \Cref{theorem:sheafification_of_a_presheaf_of_sets_on_a_small_enough_site}, there is a pullback/inverse image functor 
    $$u_s: \Sh(\calC, J; \Ab) \to \Sh(\calD, K; \Ab)$$
    % If $\calR$ is a \CrefAndHyperrefIfExist{definition:sheaf_on_a_site}{sheaf} of rings on $(\calC, J)$, then $u_s \calR$ is a sheaf of rings on $(\calD, K)$ and $u_s$ restricts to a functor 
    % $$\hlin{u_s: \Mod(\calR) \to \Mod(u_* \calR).}$$
    % \CrefIfExists{definition:module_over_a_sheaf_of_rings_on_a_site}

    As discussed in \Cref{definition:inverse_image_of_a_sheaf_under_a_continuous_functor_of_sites_or_a_site_morphism}, if $u$ is the functor underlying a site morphism $f: (\calD, K) \to (\calC, J)$, then we may alternatively denote $u_s$ by 
    $$\hlin{f^*: \Sh(\calC, J; \Ab) \to \Sh(\calD, K; \Ab)}$$
    and call this a \hldef{inverse image/pullback functor via $f$}.
\end{definition}
