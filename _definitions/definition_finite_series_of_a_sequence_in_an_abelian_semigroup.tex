\begin{definition}[Finite series] \label{definition:finite_series_of_a_sequence_in_an_abelian_semigroup}
    Let $A$ be an \CrefAndHyperrefIfExist{definition:semigroup}{abelian semigroup} written additively. Let $(a_n)_{n=1}^N$ be a finite sequence in $A$ of length $N \geq 1$. The \hldef{finite series} (or simply the \hldef{sum}) of the sequence is the element
    \hl{$\sum_{n=1}^N a_n \in A$},
    i.e., the $N$-th partial sum $\CrefAndHyperrefIfExist{definition:partial_sum_of_a_sequence_in_an_abelian_group}{S_N}$ of the sequence $(a_n)$.
\end{definition}
