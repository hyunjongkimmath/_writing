\begin{definition}[Probability space] \label{definition:probability_space}
% NOTE TO THINGS THAT REFER TO THIS: NOTiCE HOW THIS DISCUSSION AVOIDS MEASURE THEORY. IF MEASURE THEORY MUST BE USED, REFER TO definition:probability_measure_space_for_a_measure_on_a_measurable_space INSTEAD.
A \hldef{probability space} is a triple $(\Omega, \mathcal{F}, P)$ consisting of:

\begin{enumerate}
    \item A nonempty set $\Omega$, called the \hldef{sample space}, whose elements are the possible outcomes of a random experiment.

    \item A collection $\mathcal{F}$ of subsets of $\Omega$, called the \hldef{events}, such that:
    \begin{enumerate}
        \item $\Omega \in \mathcal{F}$,
        \item if $E \in \mathcal{F}$ then its complement $E^c = \Omega \setminus E$ is in $\mathcal{F}$,
        \item if $(E_n)_{n=1}^\infty$ is a sequence with each $E_n \in \mathcal{F}$, then $\bigcup_{n=1}^\infty E_n \in \mathcal{F}$.
    \end{enumerate}

    \item A map $P: \mathcal{F} \to [0, 1]$, called the \hldef{probability measure}, satisfying:
    \begin{enumerate}
        \item $P(\Omega) = 1$,
        \item for every sequence $(E_n)_{n=1}^\infty$ of pairwise disjoint events (i.e., $E_i \cap E_j = \varnothing$ for $i \neq j$),
        $$P\left(\bigcup_{n=1}^\infty E_n\right) = \sum_{n=1}^\infty P(E_n).$$
    \end{enumerate}
\end{enumerate}

The value $P(E)$ is called the \hldef{probability of the event $E$}.
\end{definition}
