\begin{definition} \label{definition:tor_dimension_of_a_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{not necessarily commutative ring}.
The \hldef{Tor-dimension} (or \hldef{weak global dimension}) of a ring $R$, denoted \hl{$\operatorname{Tor-dim}(R)$} or \hl{$\operatorname{wdim}(R)$}, is defined as:
$$\operatorname{wdim}(R) = \sup \{ \operatorname{fd}_R(M) \mid M \in {}_R\mathsf{Mod} \}$$
\end{definition}
