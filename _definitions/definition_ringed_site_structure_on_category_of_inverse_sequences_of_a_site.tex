\begin{definition}[{\cite[Section 1.]{ekedahl_af}}]\label{definition:ringed_site_structure_on_category_of_inverse_sequences_of_a_site}
    Assume \Cref{context:ekedahl_af_topos_morphisms_to_and_from_Pro_N_S_to_S}.
    There is a ring object, which we might sometimes denote by \hl{$R_\bullet$}, given by the projective system
    $$\cdots \to R/m^{n+1} \to R/m^n \to  \cdots.$$
    When \CrefAndHyperrefIfExist{definition:category_of_inverse_direct_sequences_of_objects_in_a_category}{$\mathrm{Pro}_\bbN(S)$} is equipped with $R_\bullet$, $\pi$ is a \CrefAndHyperrefIfExist{definition:morphism_of_ringed_topoi}{morphism of ringed topoi}. In particular, for an object $M$ of the ringed \CrefAndHyperrefIfExist{definition:topos}{topos} $(S, R)$, its pullback $\pi^* M$ is the inverse system $\{M \otimes_R R_n\}_{n \geq 1}$. 
\end{definition}
