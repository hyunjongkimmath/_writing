\begin{definition} \label{definition:prime_subfield_of_a_field}
Let $K$ be a \CrefAndHyperrefIfExist{definition:field}{field}. The \hldef{prime subfield of $K$} is the intersection of all \CrefAndHyperrefIfExist{definition:extension_of_a_field}{subfields} of $K$.
\TODO{it should be stated and referenced here that the intersection of subfields of a field are themselves subfields}
\begin{itemize}
    \item If $\operatorname{char}(K) = p > 0$, the prime subfield is isomorphic to the \CrefAndHyperrefIfExist{definition:finite_field}{finite field} $\mathbb{F}_p \cong \mathbb{Z}/p\mathbb{Z}$.
    \item If $\operatorname{char}(K) = 0$, the prime subfield is isomorphic to the field of \CrefAndHyperrefIfExist{definition:field_of_rational_numbers}{rational numbers} $\mathbb{Q}$.
\end{itemize}
\end{definition}
