\begin{definition} \label{definition:strongly_complete_profinite_group}
    A \CrefAndHyperrefIfExist{definition:profinite_group}{profinite group} $G$ is said to be \hldef{strongly complete} if the following equivalent conditions hold:
    \begin{enumerate}
        \item Every \CrefAndHyperrefIfExist{definition:subgroup_of_a_group}{subgroup} of \CrefAndHyperrefIfExist{definition:index_of_a_subgroup_in_a_group}{finite index} in $G$ is \CrefAndHyperrefIfExist{definition:topological_space}{open}.
        \item $G$ equals its own \CrefAndHyperrefIfExist{definition:profinite_completion_of_a_group}{profinite completion}.
        \item Every \CrefAndHyperrefIfExist{definition:group_homomorphism}{group homomorphism} from $G$ to any profinite group is \CrefAndHyperrefIfExist{definition:continuous_map_of_topological_spaces}{continuous}.
    \end{enumerate}
\end{definition}
