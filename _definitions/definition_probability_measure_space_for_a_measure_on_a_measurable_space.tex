\begin{definition}[Probability measure space] \label{definition:probability_measure_space_for_a_measure_on_a_measurable_space}
    A \CrefAndHyperrefIfExist{definition:measure_on_a_measurable_space_and_measure_space}{measure space} $(\Omega, \calF, P)$ is called a \hldef{probability measure space}, and $P$ is called a \hldef{probability measure} if $P(\Omega) = 1$. 

    In this case, $\Omega$ is called the \hldef{space of outcomes}, elements of $\calF$ are called \hldef{events} and $P$ is called the \hldef{probability measure}.
\end{definition}
