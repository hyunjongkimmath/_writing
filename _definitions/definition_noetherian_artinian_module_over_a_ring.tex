\begin{definition} \label{definition:noetherian_artinian_module_over_a_ring}
Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{ring} and $M$ be a left or \CrefAndHyperrefIfExist{definition:bimodule_over_rings_module_over_a_ring}{right $R$-module}.
\begin{enumerate}
    \item $M$ is a \hldef{Noetherian module} if the set of its \CrefAndHyperrefIfExist{definition:submodule_of_a_module_over_a_ring}{submodules}, ordered by inclusion, satisfies the \CrefAndHyperrefIfExist{definition:ascending_descending_chain_condition_on_a_partially_ordered_set}{ACC}.
    \item $M$ is an \hldef{Artinian module} if the set of its submodules, ordered by inclusion, satisfies the \CrefAndHyperrefIfExist{definition:ascending_descending_chain_condition_on_a_partially_ordered_set}{DCC}.
\end{enumerate}
\end{definition}
