\begin{definition}[Meromorphic function] \label{definition:meromorphic_function_on_an_open_subset_of_the_complex_plane}
Let $U \subseteq \mathbb{C}$ be an open set. A function $f: U \to \widehat{\mathbb{C}} = \mathbb{C} \cup \{\infty\}$ is called \hldef{meromorphic on $U$} if there exists a subset $S \subseteq U$ such that:
\begin{enumerate}
  \item $S$ has no \CrefAndHyperrefIfExist{definition:accumulation_point_in_an_extended_metric_space}{accumulation point} in $U$ (i.e., every point of $U$ has a neighborhood intersecting $S$ in at most finitely many points),
  \item $f$ is \CrefAndHyperrefIfExist{definition:holomorphic_function_from_a_subset_of_Cn_to_Cm}{holomorphic} on $U \setminus S$,
  \item For each $z_0 \in S$, $z_0$ is a \CrefAndHyperrefIfExist{definition:isolated_singularity_of_a_holomorphic_function_from_an_open_subset_of_the_complex_plane_to_the_complex_plane_classified_as_removable_pole_or_essential}{pole} of $f|_{U \setminus S}$.
\end{enumerate}
The set $S$ is called the \hldef{pole set of $f$}.
\end{definition}
