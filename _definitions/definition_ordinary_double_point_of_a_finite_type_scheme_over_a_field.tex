\begin{definition}[Ordinary double point] \label{definition:ordinary_double_point_of_a_finite_type_scheme_over_a_field}
    Let $k$ be a \CrefAndHyperrefIfExist{definition:field}{field}, let $X$ be a scheme of \CrefAndHyperrefIfExist{definition:finite_type_morphism_of_schemes}{finite type} over $k$, and let $x \in X$ be a point. 

    \TODO{completion}
    The point $x$ is called an \hldef{ordinary double point} (or a \hldef{node}) if the completion $\widehat{\mathcal{O}}_{X,x}$ of the local ring at $x$ is isomorphic to
    $$\hlin{k[[u, v]]/(uv)}$$
    as a $k$-algebra. In other words, in an affine neighbourhood of $x$, $X$ can be locally analytically described by the equation $xy = 0$, which geometrically represents two smooth branches crossing transversely at $x$.

    A $k$-scheme $X$ is said to have \hldef{at most ordinary double points} (or is \hldef{nodal}) if every singular point of finite type over $k$ is an ordinary double point.
\end{definition}
