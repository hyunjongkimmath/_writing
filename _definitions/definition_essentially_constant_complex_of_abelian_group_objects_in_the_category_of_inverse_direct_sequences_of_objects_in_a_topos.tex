\begin{definition} \label{definition:essentially_constant_complex_of_abelian_group_objects_in_the_category_of_inverse_direct_sequences_of_objects_in_a_topos}
    Assume \Cref{context:ekedahl_af_ringed_topos_category_of_inverse_sequences}.

    Let $M$ be a \CrefAndHyperrefIfExist{definition:cochain_complex_of_objects_in_an_additive_category}{cochain complex} of \CrefAndHyperrefIfExist{definition:group_object_in_a_category_with_a_final_object}{abelian group objects} of \CrefAndHyperrefIfExist{definition:category_of_inverse_direct_sequences_of_objects_in_a_category}{$\mathrm{Pro}_{\bbN}(S)$}. 
    We say that $M$ is \hldef{essentially constant} if there exists a complex $N$ of abelian group objects in $\mathrm{Pro}_{\bbN}(S)$, a complex $P$ of abelian group objects in $S$, and \CrefAndHyperrefIfExist{definition:chain_complex_of_objects_in_an_additive_category}{morphism of complexes} $M \stackrel{f}{\leftarrow} N \xrightarrow{g} \pi^{*} P$  such that the \CrefAndHyperrefIfExist{definition:mapping_cone_of_a_map_of_chain_cochain_complexes_in_an_additive_category}{mapping cones} of $f$ and $g$ are \CrefAndHyperrefIfExist{definition:essentially_zero_inverse_sequence_essentially_bounded_complex_of_sheaves_of_abelian_groups_on_a_site}{essentially zero}. 
\end{definition}
