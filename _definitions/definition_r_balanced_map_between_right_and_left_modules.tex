\begin{definition}[\CrefAndHyperrefIfExist{definition:ring}{$R$}-balanced map] \label{definition:r_balanced_map_between_right_and_left_modules}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{ring}. Let $M$ be a \CrefAndHyperrefIfExist{definition:right_module_over_a_ring}{right $R$-module}, let $N$ be a \CrefAndHyperrefIfExist{definition:left_module_over_a_ring}{left $R$-module}, and let $X$ be a set.

    A map $b: M \times N \to X$ is called \hldef{$R$-balanced} (or simply \hldef{balanced} when $R$ is clear from context) if
    $$b(mr, n) = b(m, rn)$$
    for all $m \in M$, $n \in N$, and $r \in R$.
\end{definition}
