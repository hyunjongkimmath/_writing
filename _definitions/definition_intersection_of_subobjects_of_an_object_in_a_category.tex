\begin{definition} \label{definition:intersection_of_subobjects_of_an_object_in_a_category}
    Let $\calC$ be a \CrefAndHyperrefIfExist{definition:category}{category}, let $X$ be an object in $\calC$, and let $A$ and $B$ be \CrefAndHyperrefIfExist{definition:subobject_of_an_object_of_an_additive_category}{subobjects} of $X$. 
    The \hldef{intersection of $A$ and $B$}, denoted \hl{$A \cap B$}, is the \CrefAndHyperrefIfExist{definition:cartesian_product_of_two_objects_in_a_category_over_an_object}{fiber product} $A \times_C B$, assuming that it exists.
\end{definition}
