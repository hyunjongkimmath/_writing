\begin{definition} \label{definition:irreducible_element_of_a_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{ring}. 
    A element $x$ of $R$ is said to be \hldef{irreducible} if it is nonzero, it is not a \CrefAndHyperrefIfExist{definition:unit_of_a_ring}{unit}, and if $x = ab$ implies that exactly one of $a$ and $b$ is a unit for any elements $a,b \in R$.
\end{definition}
