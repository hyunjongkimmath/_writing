\begin{definition} \label{definition:quasi_finite_etale_sheaf_of_sets_modules_on_a_scheme}
    Let $\calF$ be an \CrefAndHyperrefIfExist{definition:small_etale_site_of_a_scheme}{\'etale} \CrefAndHyperrefIfExist{definition:sheaf_on_a_site}{sheaf on} a \CrefAndHyperrefIfExist{definition:scheme}{scheme} $X$. 
    \begin{enumerate}
        \item If $\calF$ is a sheaf of sets, then we say that it is \hldef{quasi-finite} if for all \CrefAndHyperrefIfExist{definition:geometric_point_of_a_scheme}{geometric points} $\barx$ of $X$, the \CrefAndHyperrefIfExist{definition:stalk_of_a_presheaf_on_the_small_etale_site_of_a_schem_at_a_geometric_point}{stalk} $\calF_{\barx}$ is a finite set.
        
        \item If $\calF$ is a \CrefAndHyperrefIfExist{definition:module_over_a_sheaf_of_rings_on_a_site}{sheaf of modules} over a ring $R$, then we say that it is \hldef{quasi-finite} if for all \CrefAndHyperrefIfExist{definition:geometric_point_of_a_scheme}{geometric points} $\barx$ of $X$, the \CrefAndHyperrefIfExist{definition:stalk_of_a_presheaf_on_the_small_etale_site_of_a_schem_at_a_geometric_point}{stalk} $\calF_{\barx}$ is a finite rank $R$-module.
    \end{enumerate}
\end{definition}
