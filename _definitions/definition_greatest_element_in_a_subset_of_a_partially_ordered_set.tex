\begin{definition} \label{definition:greatest_element_in_a_subset_of_a_partially_ordered_set}
Let $(X, \leq)$ be a \CrefAndHyperrefIfExist{definition:partially_ordered_set}{partially ordered set} and let $A \subseteq X$ be a subset. An element $M \in A$ is a \hldef{greatest element of $A$} (or \hldef{maximum of $A$}) if for all $x \in A$, $x \leq M$. If a greatest element exists, it is unique\CrefIfExists{lemma:maximum_minimum_element_of_a_subset_of_a_poset_is_unique_if_exists}.
\end{definition}
