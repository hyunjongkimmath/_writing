\begin{definition} \label{definition:general_linear_group_over_a_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{ring}. 
    The \hldef{general linear group of $n \times n$ matrices over $R$} is the the \CrefAndHyperrefIfExist{definition:group}{group} \hl{$\GL_n(R)$} consisting of $n \times n$ matrices $A$ over $R$ such that there exists a \CrefAndHyperrefIfExist{definition:matrix_over_a_ring}{matrix} $A$ over $R$, denoted \hl{$A^{-1}$}, such that 
    $$A \cdot A^{-1}, A^{-1} \cdot A = I_n$$
    (\Cref{definition:identity_matrix})
\end{definition}
