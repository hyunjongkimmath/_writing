\begin{definition}  \label{definition:n_connected_topological_space}
    A \CrefAndHyperrefIfExist{definition:topological_space}{topological space} $X$ is said to be \hldef{$n$-connected} if $\pi_q(X,x) = 0$ (\Cref{definition:homotopy_group_of_a_spectrum}) for $0 \leq q \leq n$ for all $x \in X$. 
\end{definition}
