\begin{definition} \label{definition:unimodular_element_of_a_module_over_a_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{not necessarily commutative ring}. Let $M$ be a \CrefAndHyperrefIfExist{definition:module_of_a_ring}{left/right $R$-module}.
    An element $v \in M$ is \hldef{unimodular} if there exists a linear functional $f \in M^*$ \CrefIfExists{definition:dual_of_a_left_right_two_sided_module} such that $f(v) = 1$.
\end{definition}
