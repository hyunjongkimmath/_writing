\begin{definition} \label{definition:flasque_flabby_sheaf_of_sets_on_a_topological_space}
    Let $\calF$ be a \CrefAndHyperrefIfExist{definition:sheaf_on_a_topological_space_valued_in_a_category_with_a_terminal_object}{sheaf of sets} on a \CrefAndHyperrefIfExist{definition:topological_space}{topological space} $X$. 
    The sheaf $\calF$ is called \hldef{flasque} or \hldef{flabby} if for every open $U \subseteq X$, the \CrefAndHyperrefIfExist{definition:restriction_of_a_presheaf_sheaf_on_a_topological_space}{restriction map}
    $$\rho_{X,U} : \calF(X) \to \calF(U)$$
    is \CrefAndHyperrefIfExist{definition:injective_surjective_bijective_map_of_sets}{surjective}.
\end{definition}
