\begin{definition} \label{definition:exponentiation_in_a_semiring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:semiring}{semiring}, and let $a \in R$. For each nonnegative integer $n \in \mathbb{Z}_{\geq 0}$, the \hldef{$n$th power of $a$ in $R$}, denoted by \hl{$a^n$}, is defined as the \CrefAndHyperrefIfExist{definition:power_of_a_monoid_element_by_a_nonnegative_integer}{$n$th power} of $a$ with respect to the \CrefAndHyperrefIfExist{definition:monoid}{multiplicative monoid} $(R, \cdot, 1_R)$. That is,
    \begin{align*}
        a^0 &:= 1_R, \\
        a^{n+1} &:= a \cdot a^n.
    \end{align*}
\end{definition}
