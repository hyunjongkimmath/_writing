\begin{definition} \label{definition:matrix_group_over_a_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{ring}. 
    A \hldef{matrix group over $R$} is any subgroup of the \CrefAndHyperrefIfExist{definition:general_linear_group_over_a_ring}{general linear group} $\operatorname{GL}_n(R)$. 
\end{definition}
