\begin{definition} \label{definition:relative_picard_functor_of_a_scheme_over_a_scheme}
    Let $X$ be a \CrefAndHyperrefIfExist{definition:scheme_over_a_scheme}{scheme over} a \CrefAndHyperrefIfExist{definition:scheme}{scheme} $S$. 
    The \CrefAndHyperrefIfExist{definition:big_fppf_site_of_a_scheme}{fppf} \CrefAndHyperrefIfExist{definition:sheaf_on_a_site}{sheaf} \CrefAndHyperrefIfExist{definition:sheafification_functor_on_a_site}{associated to} the \CrefAndHyperrefIfExist{definition:functor_between_categories}{functor} 
    $$P_{X/S}: (\Sch/S)^\circ \to \Sets, \quad T \mapsto \Pic(X \times_S T)$$
    \CrefIfExists{definition:scheme_over_a_scheme}\CrefIfExists{definition:opposite_category_of_a_category}\CrefIfExists{definition:category_of_sets}\CrefIfExists{definition:cartesian_product_of_two_objects_in_a_category_over_an_object} is denoted by \hl{$\Pic_{X/S}$} and is called the \hldef{relative Picard functor of $X$ over $S$}.

    For an $S$-scheme $T$, the group $\Pic_{X/S}(T)$ is called the \hldef{relative Picard group of $X \times_S T$ over $T$}.
\end{definition}
