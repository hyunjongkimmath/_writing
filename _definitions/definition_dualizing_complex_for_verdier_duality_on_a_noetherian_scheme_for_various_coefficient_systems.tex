\begin{definition}[Dualizing complex for Verdier duality on a noetherian scheme for various coefficient systems] \label{definition:dualizing_complex_for_verdier_duality_on_a_noetherian_scheme_for_various_coefficient_systems}
\TODO{AI generated, verify}
    Let $S$ be a \CrefAndHyperrefIfExist{definition:locally_noetherian_and_noetherian_scheme}{noetherian scheme}. Consider sheaves on the \CrefAndHyperrefIfExist{definition:small_etale_site_of_a_scheme}{(small) \'etale site} of $S$.

    Let the coefficient system be one of the following:
    \begin{enumerate}
        \item $A$ a \CrefAndHyperrefIfExist{definition:noetherian_ring}{noetherian} \CrefAndHyperrefIfExist{definition:torsion_ring}{torsion ring} (torsion coefficients)
        \item Assuming that $S$ is \CrefAndHyperrefIfExist{definition:locally_noetherian_and_noetherian_scheme}{noetherian} and with $R,E,K$ as in \Cref{context:setup_for_coefficients_for_adic_sheaves_on_etale_sites_of_schemes}
    \end{enumerate}

    In each case, the \hldef{dualizing complex for Verdier duality on $S$}, denoted by \hl{$\mathbb{D}_S^\bullet$} (notation suppresses coefficient dependence), is the object in the corresponding category characterized by the following property: for every $S$-scheme $X$ of \CrefAndHyperrefIfExist{definition:finite_type_morphism_of_schemes}{finite type} with structure morphism $f: X \to S$, the relative dualizing complex satisfies
    $$\hlin{\mathbb{D}_{X/S}^\bullet = Rf^! \, \mathbb{D}_S^\bullet}$$
    where $Rf^!$ is the exceptional inverse image functor appropriate to the coefficient system.

    See \Cref{definition:dualizing_complex_for_verdier_duality_in_the_bounded_below_derived_category_of_etale_sheaves_with_torsion_coefficients_on_a_finite_type_scheme_over_a_noetherian_scheme} for torsion coefficients, \Cref{definition:dualizing_complex_for_verdier_duality_in_the_bdd_constr_derived_category_of_adic_sheaves_with_integral_coefficients_on_a_finite_type_scheme_over_a_noetherian_scheme} for integral adic coefficients, \Cref{definition:dualizing_complex_for_verdier_duality_in_the_bdd_constr_derived_category_of_adic_sheaves_with_rational_coefficients_on_a_finite_type_scheme_over_a_noetherian_scheme} for rational coefficients, and \Cref{definition:dualizing_complex_for_verdier_duality_in_the_bdd_constr_derived_category_of_adic_sheaves_with_algebraic_coefficients_on_a_finite_type_scheme_over_a_noetherian_scheme} for algebraic coefficients.
\end{definition}
