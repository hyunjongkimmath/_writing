\begin{definition} \label{definition:complex_torus}
    \TODO{rank}
    Let $V$ be a \CrefAndHyperrefIfExist{definition:basis_and_dimension_of_a_vector_space_over_a_field}{finite-dimensional} \CrefAndHyperrefIfExist{definition:field_of_complex_numbers}{complex} \CrefAndHyperrefIfExist{definition:vector_space_over_a_field}{vector space} and $\Lambda \subset V$ a \CrefAndHyperrefIfExist{definition:lattice_over_a_commutative_ring}{lattice over $\bbZ$} of rank $2n = 2\dim_{\mathbb{C}} V$, i.e., a \CrefAndHyperrefIfExist{definition:discrete_group}{discrete} subgroup such that $V/\Lambda$ is compact. 
    The quotient $X = V/\Lambda$ is called a \hldef{complex torus of dimension $n$}. Equivalently, as a \CrefAndHyperrefIfExist{definition:complex_analytic_manifold}{complex manifold},
    $$X = V/\Lambda \cong (S^1)^{\times n}$$
    where the group structure on $X$ is induced by vector addition on $V$.
\end{definition}
