\begin{definition}\label{definition:commutative_binary_operation_on_a_set}
    % EDITOR APPROVED
    Let $\cdot: S \times S \to S$ be a \CrefAndHyperrefIfExist{definition:n_ary_operation_on_a_set}{binary operation} on a \CrefAndHyperrefIfExist{definition:zermelo_fraenkel_set_theory}{set}. It is said to be \hldef{commutative} if 
    $$ a \cdot b = b \cdot a \quad \text{for all } a,b \in S.$$
\end{definition}
