\begin{definition} \label{definition:subgroup_of_a_group}
Let $(G,\cdot)$ be a \CrefAndHyperrefIfExist{definition:group}{group}. A \CrefAndHyperrefIfExist{definition:subset_of_a_set}{subset} $H \subseteq G$ is called a \hldef{subgroup of $G$} if 
\begin{itemize}
    \item the \CrefAndHyperrefIfExist{definition:n_ary_operation_binary_operation_ternary_operation_on_a_set}{operations} 
    $$\cdot|_H: H \times H \to G, \quad (h_1,h_2) \mapsto h_1 \cdot h_2$$
    $${}^{-1}|_H: H H \to G, \quad h \mapsto h^{-1}$$
    have \CrefAndHyperrefIfExist{definition:image_of_a_function_between_sets}{images} contained in $H$, and 

    \item these operations along with the identity element of $G$ make $H$ into a group. 
\end{itemize}
Equivalently\CrefIfExists{proposition:subgroup_criterion_for_a_subset_of_a_group}, $H$ is a subgroup if:
\begin{itemize}
    \item $H \neq \emptyset$,
    \item For all $h_1,h_2 \in H$, the product $h_1 \cdot h_2$ is an element of $H$,
    \item For all $h \in H$, the inverse $h^{-1}$ is an element of $H$.
\end{itemize}
We denote this relation by \hl{$H \leq G$}, meaning ``$H$ is a subgroup of $G$''.
\end{definition}
