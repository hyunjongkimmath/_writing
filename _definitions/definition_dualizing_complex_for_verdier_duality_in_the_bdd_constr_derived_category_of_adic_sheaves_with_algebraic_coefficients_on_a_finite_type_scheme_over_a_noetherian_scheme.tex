\begin{definition}[Dualizing complex for Verdier duality on a noetherian scheme with algebraic coefficients] \label{definition:dualizing_complex_for_verdier_duality_in_the_bdd_constr_derived_category_of_adic_sheaves_with_algebraic_coefficients_on_a_finite_type_scheme_over_a_noetherian_scheme}
    Let $(R, \lambda)$ be a \CrefAndHyperrefIfExist{definition:complete_ring_with_respect_to_an_ideal}{complete} \CrefAndHyperrefIfExist{definition:discrete_valuation_ring}{DVR} such that $R/\lambda^n$ is finite for all $n \geq 1$. Let $E$ be the \CrefAndHyperrefIfExist{definition:field_of_fractions_of_an_integral_domain}{fraction field} of $R$, and let $K/E$ be a \CrefAndHyperrefIfExist{definition:algebraic_element_minimal_polynomial_of_an_algebraic_element_algebraic_field_extension_transcendental_element_field_extension}{algebraic extension}. Let $S$ be a \CrefAndHyperrefIfExist{definition:locally_noetherian_and_noetherian_scheme}{noetherian scheme}. Consider sheaves on the (small) \'etale site of $S$.

    \begin{enumerate}
        \item The \hldef{dualizing complex for Verdier duality on $S$ with algebraic coefficients}, denoted by \hl{$\mathbb{D}_S^\bullet(K)$}, is the image of $\mathbb{D}_S^\bullet(E)$ in $\Dbc(S, K)$ under the transition functor $\Dbc(S, E) \to \Dbc(S, K)$.

        \item Let $X$ be a finite type scheme over $S$. The \hldef{(relative) dualizing complex for Verdier duality on $X/S$ with algebraic coefficients}, denoted by \hl{$\mathbb{D}_{X/S}^\bullet(K)$}, is defined by
        $$\mathbb{D}_{X/S}^\bullet(K) = Rf^! \, \mathbb{D}_S^\bullet(K).$$
        \CrefIfExists{definition:exceptional_inverse_image_functor_on_bdd_const_derived_category_of_adic_sheaves_with_algebraic_coefficients_on_schemes_of_finite_type_over_a_a_noetherian_scheme_via_a_compactifiable_morphism}
        Equivalently, it is the image of the \CrefAndHyperrefIfExist{definition:dualizing_complex_for_verdier_duality_in_the_bdd_constr_derived_category_of_adic_sheaves_with_rational_coefficients_on_a_finite_type_scheme_over_a_noetherian_scheme}{dualizing complex} $\mathbb{D}_{X/S}^\bullet(E)$ under the transition functor $\Dbc(S, E) \to \Dbc(S, K)$.
    \end{enumerate}
\end{definition}
