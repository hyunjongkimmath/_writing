\begin{definition} \label{definition:opposite_module_of_a_module_over_a_ring_with_involution}
Let $R$ be a \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{ring with involution} $r \mapsto \bar{r}$.
    
    Let $M$ be a \CrefAndHyperrefIfExist{definition:bimodule_over_rings_module_over_a_ring}{left/right $R$-module}. The \hldef{opposite module} (not to be confused with the notion of \CrefAndHyperref{definition:opposite_module_of_a_general_bimodule_over_rings}{opposite modules} for modules over general rings), also called the \hldef{conjugate module}, is the right/left $R$-module, denoted by notations such as \hl{$M^{op}$} or \hl{$\overline{M}$}, with the same underlying abelian group as $M$, where the right $R$-action is defined as follows:
    \begin{itemize}
        \item $$m \cdot r = \bar{r} m$$ for all $m \in M$ and $r \in R$ if $M$ is a left $R$-module.

        $$r \cdot m = m \bar{r}$$
        for all $m \in M$ and $r \in R$ if $M$ is a right $R$-module.
    \end{itemize}

The opposite/conjugate module construction yields functors
\begin{align*}
\overline{(-)}: _{R} \Mod \to \Mod_R \\
\overline{(-)}: \Mod_{R}  \to _{R} \Mod.
\end{align*}
\CrefIfExists{definition:category_of_modules_and_bimodules_over_rings}
\end{definition}
