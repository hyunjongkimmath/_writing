\begin{definition} \label{definition:simple_irreducible_semisimple_module_over_a_ring}
Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{not necessarily commutative ring}.

A left/\CrefAndHyperrefIfExist{definition:bimodule_over_rings_module_over_a_ring}{right $R$-module} $M$ is \hldef{simple} (or \hldef{irreducible)} if $M \neq \{0\}$ and its only submodules are $\{0\}$ and $M$. 

A left/right $R$-module is \hldef{semisimple} if it is \CrefAndHyperrefIfExist{definition:ring_homomorphism}{isomorphic} to a \CrefAndHyperrefIfExist{definition:coproduct_of_modules_of_rings}{direct sum} of simple left/right \CrefAndHyperrefIfExist{definition:bimodule_over_rings_module_over_a_ring}{$R$-modules}.

\TextIfExists{definition:semisimple_object_of_an_additive_category}{
    Equivalently, an $R$-module is semisimple if it is a \CrefAndHyperref{definition:semisimple_object_of_an_additive_category}{semisimple object} in the abelian \CrefAndHyperrefIfExist{definition:category_of_modules_and_bimodules_over_rings}{category of $R$-modules}.
}
\end{definition}
