\begin{definition}[see {\cite[Chapter I, 7.2]{knus_qhfr}}] \label{definition:scharlau_transfer_of_sesquilinear_forms_on_modules_under_involution_ring_homomorphism}
    Let $\eta: A \to A'$ be a \CrefAndHyperrefIfExist{definition:homomorphism_of_involution_rings}{homomorphism of rings with involution}. 
    Given a right $A'$-module $M'$, write $M'_{(A)}$ for the \CrefAndHyperrefIfExist{definition:restriction_of_scalars_of_a_left_or_right_module_along_a_ring_homomorphism_from_the_base_ring}{restriction} of $M'$ as a right $A$-module (via $\eta$). Let $\tau: A' \to A$ be a \CrefAndHyperrefIfExist{definition:homomorphism_of_modules_over_a_ring}{morphism of $A$-bimodules}.

    $\tau$ induces an additive map
    $$\hlin{\tau: \Sesq_{A'}(M') \to \Sesq_A(M'_{(A)})}$$
    (\Cref{definition:sesquilinear_form_on_a_module_over_a_ring_with_involution}) that sends a sesquilinear form $b' \in \Sesq_{A'}(M')$ on an $A'$-module $M'$ to the sesquilinear form $\tau(b') \in \Sesq_A(M'_{(A)})$ on $M'_{(A)}$ constructed by
    $$\tau(b')(x,y) = \tau(b'(x,y)).$$
    It may be appropriate to call the induced map $\tau$ the \hldef{transfer} or \hldef{Scharlau transfer on sesquilinear forms}. 
\end{definition}
