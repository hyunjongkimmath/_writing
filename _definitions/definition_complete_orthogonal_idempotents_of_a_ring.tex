\begin{definition} \label{definition:complete_orthogonal_idempotents_of_a_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{ring}. 
    A set $\{e_1,\ldots,e_n\}$ in $R$ is said to consist of \hldef{complete orthogonal idempotents} if the set is \CrefAndHyperrefIfExist{definition:ortohgonal_idempotents_of_a_ring}{orthogonal idempotent} and $\sum_i e_i = 1$. 
\end{definition}
