\begin{definition} \label{definition:pointed_pair_of_topological_spaces}
    A \hldef{pointed pair of topological spaces} is a triple $(X,A,x)$ where $(X,A)$ is a \CrefAndHyperrefIfExist{definition:pair_of_topological_spaces}{pair of topological spaces} and $x \in A$, i.e. $(X,x)$ and $(A,x)$ are both \CrefAndHyperrefIfExist{definition:pointed_topological_space}{pointed spaces} and the inclusion $A \hookrightarrow X$ yields a \CrefAndHyperrefIfExist{definition:pointed_topological_space}{pointed map} $(A,x) \to (X,x)$.
\end{definition}
