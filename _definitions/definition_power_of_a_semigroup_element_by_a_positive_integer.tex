\begin{definition} \label{definition:power_of_a_semigroup_element_by_a_positive_integer}
Let $(M,\cdot)$ be a \CrefAndHyperrefIfExist{definition:semigroup}{semigroup} and let $a \in M$. For each positive integer $n \in \bbZ_{>0}$, the \hldef{$n$th power of $a$}, denoted by \hl{$a^n$}, is the element of $M$ defined recursively by
\begin{align*}
    a^1 &:= a, \\
    a^{n+1} &:= a \cdot a^n.
\end{align*}
\CrefIfExists{lemma:power_of_semigroup_element_is_in_semigroup}
If the operation of the semigroup is written additively instead, then we denote the $n$th power of $a$ by \hl{$n \cdot a$} instead.
\end{definition}
