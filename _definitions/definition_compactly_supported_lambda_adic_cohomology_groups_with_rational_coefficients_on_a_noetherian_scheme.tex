\begin{definition}[Compactly supported $\lambda$-adic cohomology with rational coefficients] \label{definition:compactly_supported_lambda_adic_cohomology_groups_with_rational_coefficients_on_a_noetherian_scheme}
    Let $(R, \lambda)$ be a \CrefAndHyperrefIfExist{definition:complete_ring_with_respect_to_an_ideal}{complete} \CrefAndHyperrefIfExist{definition:discrete_valuation_ring}{DVR} such that $R/\lambda^n$ is finite for all $n \geq 1$. Let $E = \Frac(R)$ be its \CrefAndHyperrefIfExist{definition:field_of_fractions_of_an_integral_domain}{fraction field}. Let $k$ be a field. Let $X$ be a \CrefAndHyperrefIfExist{definition:locally_noetherian_and_noetherian_scheme}{noetherian} scheme over $k$, with structure morphism $f \coloneq f_X \colon X \to \Spec(k)$, such that $f$ is \CrefAndHyperrefIfExist{definition:compactifiable_over_S}{$\Spec(k)$-compactifiable}. Consider sheaves on the \CrefAndHyperrefIfExist{definition:small_etale_site_of_a_scheme}{(small) \'etale site $X_{\et}$}.

    Let $N \in \Dbc(X, E)$ be an object of the \CrefAndHyperrefIfExist{definition:derived_category_of_cohomologically_bounded_constructible_sheaves_with_rational_adic_coefficients_on_a_noetherian_scheme}{bounded derived category with rational adic coefficients}. Since $N$ is represented by some $M \in D_c^b(X, R)$ (\CrefAndHyperrefIfExist{definition:derived_category_of_cohomologically_bounded_constructible_sheaves_with_rational_adic_coefficients_on_a_noetherian_scheme}{rational adic derived category}), the \hldef{compactly supported (rational) $\lambda$-adic cohomology groups of $X$ with coefficients in $N$} are the $E$-vector spaces:
    $$\hlin{H_c^q(X, N) \coloneq H_c^q(X, M) \otimes_R E}$$
    \CrefIfExists{definition:compactly_supported_lambda_adic_cohomology_groups_with_integral_coefficients_on_a_noetherian_scheme}
    for each $q \in \mathbb{Z}$. This is independent of the choice of representative $M$ for $N$.
\end{definition}
