\begin{definition} \label{definition:pair_of_topological_spaces}
A \hldef{pair of topological spaces} (or simply a \hldef{pair}) $(X, A)$ consists of a \CrefAndHyperrefIfExist{definition:topological_space}{topological space} $X$ and a \CrefAndHyperrefIfExist{definition:subspace_of_a_topological_space}{subspace} $A \subseteq X$. 
If $A = \emptyset$, the pair $(X, \emptyset)$ is often identified with the space $X$ itself. 


\TextIfExists{definition:category_of_pairs_of_topological_spaces}{
These form a category, see \Cref{definition:category_of_pairs_of_topological_spaces}.
}

\TextIfExists{definition:pointed_pair_of_topological_spaces}{
See also \Cref{definition:pointed_pair_of_topological_spaces}.
}

\end{definition}
