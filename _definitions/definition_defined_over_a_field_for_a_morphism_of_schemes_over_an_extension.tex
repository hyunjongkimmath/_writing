\begin{definition} \label{definition:defined_over_a_field_for_a_morphism_of_schemes_over_an_extension}
    Let $K/k$ be a \CrefAndHyperrefIfExist{definition:extension_of_a_field}{field extension}. Let $X$ and $Y$ be \CrefAndHyperrefIfExist{definition:scheme_over_a_scheme}{schemes over} $K$. Let $f: X \to Y$ be a \CrefAndHyperrefIfExist{definition:scheme_over_a_scheme}{morphism of schemes over $K$}.
    Given $k$-models $X_0$ and $Y_0$ of $X$ and $Y$ respectively, we say that \hldef{$f$ is defined over $k$ (with respect to the choice of models)} if there exists a $k$-morphism $f_0: X_0 \to Y_0$ such that $f$ is the base change of $f_0$ to $K$. 
\end{definition}
