\begin{definition} \label{definition:totally_ordered_monoid}
    A \hldef{totally ordered monoid} is an \CrefAndHyperrefIfExist{definition:ordered_monoid}{ordered monoid} whose partial order $\leq$ is a \CrefAndHyperrefIfExist{definition:total_order_linear_order_on_a_set}{total order}.
\end{definition}
