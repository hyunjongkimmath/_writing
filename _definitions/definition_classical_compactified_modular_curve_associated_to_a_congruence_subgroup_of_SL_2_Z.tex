\begin{definition}[Classical compactified modular curve associated to a congruence subgroup of $\mathrm{SL}_2(\mathbb{Z})$] \label{definition:classical_compactified_modular_curve_associated_to_a_congruence_subgroup_of_SL_2_Z}
Let $N$ be a positive integer and let $H$ be a \CrefAndHyperref{definition:congruence_subgroup_of_SL_2_Z}{congruence subgroup} of $\mathrm{SL}_2(\mathbb{Z})$. The \hldef{classical compactified modular curve associated to $H$} is the \CrefAndHyperrefIfExist{definition:riemann_surface}{Riemann surface}
\[
    X_H = H \backslash (\bbH \cup \mathbb{P}^1(\mathbb{Q})),
\]
obtained by adjoining to $Y_H = H \backslash \bbH$ the set of \CrefAndHyperrefIfExist{definition:cusps_of_a_congruence_subgroup_of_SL_2_Z}{cusps}. Each $X_H$ is a \CrefAndHyperrefIfExist{definition:compact_topological_space}{compact} Riemann surface, and the affine curve $Y_H$ is the complement of finitely many points (the cusps) in $X_H$.

% For congruence subgroups $\Gamma(N) \subseteq \Gamma_1(N) \subseteq \Gamma_0(N) \subseteq \mathrm{SL}_2(\mathbb{Z})$, the natural inclusions induce finite holomorphic maps
% \[
%     X(N) \longrightarrow X_1(N) \longrightarrow X_0(N) \longrightarrow X(1),
% \]
% which are \CrefAndHyperrefIfExist{definition:galois_cover_of_riemann_surfaces}{Galois covers} with Galois groups $(\mathrm{SL}_2(\mathbb{Z}/N\mathbb{Z}))^\times / \{\pm 1\}$ and $\mathbb{Z}/N^\times$, respectively.

These compactified modular curves carry the structure of algebraic curves over $\mathbb{C}$ (and, by uniformization theory, over number fields), and the complex analytic isomorphisms above identify them with their associated Riemann surfaces.
\end{definition}
