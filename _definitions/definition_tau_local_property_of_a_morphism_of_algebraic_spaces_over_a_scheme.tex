\begin{definition}[see {\cite[Tag 03YH]{stacks-project}}] \label{definition:tau_local_property_of_a_morphism_of_algebraic_spaces_over_a_scheme}
    \TODO{fpqc, fppf, syntomic, smooth, etale topologies on categories of algebraic spaces}
    Let $S$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme}. Let $\mathcal{P}$ be a property of morphisms of schemes over $S$. For $\tau$ some \CrefAndHyperrefIfExist{definition:grothendieck_topology_on_a_category_site_covering_sieve_topologically_generating_family}{Grothendieck topology} on the category of \CrefAndHyperrefIfExist{definition:algebraic_space_on_a_site_on_Sch_S_that_preserves_under_base_change_and_has_tau_local_surjectivity_and_etaleness}{algebraic spaces (for the big fppf topology)} over $S$\footnote{for example, $$\tau \in \{\text{fpqc}, \text{fppf}, \text{syntomic}, \text{smooth}, \text{\'etale}, \text{Zariski}\}.$$},  we say that $\mathcal{P}$ is \hldef{$\tau$-local on the base} or \hldef{$\tau$-local on the target}, or \hldef{local on the base for the $\tau$-topology} if for any $\tau$-covering $\{Y_i \to Y\}_{i \in I}$, and any \CrefAndHyperrefIfExist{definition:algebraic_space_on_a_site_on_Sch_S_that_preserves_under_base_change_and_has_tau_local_surjectivity_and_etaleness}{morphism of algebraic spaces} $f: X \to Y$ over $S$, the morphism $f$ has $\mathcal{P}$ if and only if $Y_i \times_Y X \to Y_i$\CrefIfExists{definition:cartesian_product_of_two_objects_in_a_category_over_an_object} has $\mathcal{P}$ for every $i$. 
\end{definition}
