\begin{definition} \label{definition:left_right_exact_sequences_in_an_additive_category}
Let $\mathcal{A}$ be an \CrefAndHyperrefIfExist{definition:additive_category_preadditive_category}{additive category}. 

\begin{enumerate}
    \item A sequence $0 \to A \xrightarrow{f} B \xrightarrow{g} C$ in $\mathcal{A}$ is called \hldef{left exact} if $f$ is a \CrefAndHyperrefIfExist{definition:kernel_and_cokernel_of_a_morphism_in_a_pointed_category}{kernel} of $g$. 
    
    \item A sequence $A \xrightarrow{f} B \xrightarrow{g} C \to 0$ in $\mathcal{A}$ is called \hldef{right exact} if $g$ is a \CrefAndHyperrefIfExist{definition:kernel_and_cokernel_of_a_morphism_in_a_pointed_category}{cokernel} of $f$.
\end{enumerate}

\end{definition}
