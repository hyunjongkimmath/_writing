\begin{definition} \label{definition:transpose_of_a_matrix_over_a_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{ring}.

    For any $m \times n$ \CrefAndHyperrefIfExist{definition:matrix_over_a_ring}{matrix} $A$ over $R$, the \hldef{transpose of $A$} is the $n \times m$ matrix \hl{$A^{\mathsf{T}}$} defined by $A^{\mathsf{T}}(i,j) = A(j,i)$ for all $1 \leq i \leq n$, $1 \leq j \leq m$.
\end{definition}
