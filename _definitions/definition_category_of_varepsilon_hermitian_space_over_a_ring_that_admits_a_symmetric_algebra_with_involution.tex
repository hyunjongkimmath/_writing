\begin{definition}[{cf. {\cite[Chapter I, 7.4]{knus_qhfr}} for a discussion in the case where the base ring is commutative}] \label{definition:category_of_varepsilon_hermitian_space_over_a_ring_that_admits_a_symmetric_algebra_with_involution}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{not necessarily commutative ring}, let $\varepsilon \in R^\times$ satisfy $\varepsilon^2 = 1$, and $A$ a \CrefAndHyperrefIfExist{definition:symmetric_algebra_over_a_ring}{symmetric $R$-algebra} with an \CrefAndHyperrefIfExist{definition:algebra_with_involution_over_a_ring_with_involution}{$R$-linear involution}.

    The \hldef{category of $\varepsilon$-Hermitian spaces over $R$ which admit $A$} is the \CrefAndHyperrefIfExist{definition:full_subcategory_of_a_category}{full subcategory} \hl{$\mathfrak{H}^\varepsilon(A)$} of the \CrefAndHyperrefIfExist{definition:category_of_modules_with_varepsilon_hermitian_forms_over_a_ring_with_involution}{category of modules with $\varepsilon$-Hermitian forms over $R$} whose objects are \CrefAndHyperrefIfExist{definition:sesquilinear_space_varepsilon_hermitian_space_over_a_ring_with_involution}{$\varepsilon$-Hermitian spaces} over $R$ whose underlying Hermitian forms \CrefAndHyperref{definition:admits_a_symmetric_algebra_with_involution_over_a_ring_for_a_bilinear_form_on_a_finitely_generated_projective_module}{admit $A$}. In particular, such a category is a \CrefAndHyperrefIfExist{definition:groupoid}{groupoid}.

    The \hldef{category of even $\varepsilon$-Hermitian spaces over $R$ which admit $A$} is the full subcategory \hl{$\mathfrak{H}^\varepsilon_+(A)$} whose objects are those of $\mathfrak{H}^\varepsilon(A)$ whose underlying Hermitian forms are \CrefAndHyperrefIfExist{definition:even_varepsilon_hermitian_form_on_a_module_over_a_ring_with_involution}{even}.

\end{definition}
