\begin{definition} \label{definition:subgroup_of_sections_with_identity_reduction_at_a_point_for_a_group_scheme_over_a_scheme}
    Let $G$ be a \CrefAndHyperrefIfExist{definition:algebraic_group_scheme_over_a_scheme}{group scheme} over a \CrefAndHyperrefIfExist{definition:scheme}{scheme} $S$. Let $s \in S$ be a \CrefAndHyperrefIfExist{definition:point_of_a_topos}{point}. Let $G_s$ be the \CrefAndHyperrefIfExist{definition:fiber_of_a_morphism_of_schemes_at_a_point}{fiber} of $G$ at $s$. It is a group scheme over the \CrefAndHyperrefIfExist{definition:residue_field_of_a_scheme_at_a_point}{residue field} $\kappa(s)$.

    The subgroup of $G(S)$ of \hldef{sections with identity reduction at $s$} is the subgroup denoted by \hl{$G(S)_0$} or \hl{$G_0(S)$} constructed as the preimage of $G_s^0(\kappa(s))$ (\Cref{definition:identity_component_of_a_group_scheme_over_a_scheme}) under the \CrefAndHyperref{definition:reduction_map_for_group_scheme_over_a_scheme_at_a_ponit}{reduction map} $G(S) \to G_s(\kappa(s))$.
\end{definition}
