\begin{definition} \label{definition:fiber_functor_of_category_of_finite_etale_covers_of_a_scheme_given_by_a_geometric_point}
    Let $X$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme} and let $\overline{x}$ be a \CrefAndHyperrefIfExist{definition:geometric_point_of_a_scheme}{geometric point} of $X$. 
    The \CrefAndHyperrefIfExist{definition:functor_between_categories}{functor} 
    $$F_{\bar{x}}: \FEt_X \to \Sets$$
    sending $Y \in \FEt_x$ to the underlying set of the \CrefAndHyperrefIfExist{definition:geometric_fiber_of_a_morphism_of_schemes_at_a_geometric_point}{geometric fiber} $Y \times_X \bar{x}$ is called the \hldef{fiber functor or fundamental functor of $\FEt_X$ at $\overline{x}$}.
\end{definition}
