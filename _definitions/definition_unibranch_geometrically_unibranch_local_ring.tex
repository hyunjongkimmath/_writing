\begin{definition} \label{definition:unibranch_geometrically_unibranch_local_ring}
    \begin{enumerate}
        \item A \hldef{unibranch ring} is a \CrefAndHyperrefIfExist{definition:local_ring}{local ring} $A$ 
        \begin{itemize}
            \item whose reduction $A_{\text{red}}$ is a \CrefAndHyperrefIfExist{definition:domain_integral_domain}{domain}, and 
            \item such that the \CrefAndHyperrefIfExist{definition:integral_element_over_a_ring}{integral closure} $A'$ of $A_{\text{red}}$ in its \CrefAndHyperrefIfExist{definition:field}{field} of fraction is \CrefAndHyperrefIfExist{definition:local_ring}{local}.
        \end{itemize}
        \item A \CrefAndHyperrefIfExist{definition:unibranch_scheme}{unibranch} \CrefAndHyperrefIfExist{definition:ring}{ring} $A$ is said to be \hldef{geometrically unibranch} if the \CrefAndHyperrefIfExist{definition:residue_field_of_a_scheme_at_a_point}{residue field} of the \CrefAndHyperrefIfExist{definition:integral_element_over_a_ring}{integral closure} $A'$ above is purely inseparable over the residue field of $A$.
        \TODO{purely insep}
    \end{enumerate}


\end{definition}
