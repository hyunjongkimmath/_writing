\begin{definition}[Lebesgue integral of non-negative module-valued functions] \label{definition:lebesgue_integral_of_nonnegative_measurable_function_on_a_measurable_space_valued_in_a_sigma_complete_lattice_ordered_semimodule_over_an_ordered_semiring}

Let $K$ be an \CrefAndHyperrefIfExist{definition:ordered_semiring}{ordered semiring}, let $(X,\mathcal{A},\mu)$ be a \CrefAndHyperrefIfExist{definition:measure_on_a_measurable_space_valued_in_an_ordered_semiring_and_measure_space_with_such_a_measure}{$K$-valued measure space} with $\mu : \mathcal{A} \to K$, and let $M$ be a \CrefAndHyperrefIfExist{definition:sigma_complete_lattice_ordered_semimodule_over_an_ordered_semiring}{$\sigma$-complete lattice-ordered semimodule} over $K$.

Let $f : X \to M_{\geq 0}$ be a \CrefAndHyperrefIfExist{definition:measurable_function_on_a_measurable_space_on_a_sigma_complete_lattice_ordered_semimodule_over_an_ordered_semiring}{measurable function}. The \hldef{Lebesgue integral of $f$ with respect to $\mu$} is defined by
$$\hlin{\int_X f \, d\mu := \sup \left\{ \int_X s \, d\mu : s \text{ is an } M\text{-valued simple function on } X \text{ with } 0_M \leq s(x) \leq f(x) \text{ for all } x \in X \right\}},$$
where the supremum is taken in $M$. The set of integrals of such simple functions is non-empty (containing $0_M$ from the zero function) and bounded above by any upper bound of $f$, so the supremum exists because $M$ is $\sigma$-complete.

For a measurable set $E \in \mathcal{A}$, the \hldef{Lebesgue integral of $f$ over $E$} is defined by
$$\hlin{\int_E f \, d\mu := \int_X (x \mapsto \mathbf{1}_E(x) \cdot f(x)) \, d\mu},$$
where $\mathbf{1}_E : X \to \{0_K, 1_K\} \subseteq K$ and the product is the pointwise $K$-semimodule action on $M$.

\CrefIfExists{definition:lebesgue_integral_of_measurable_functions_into_the_real_line_extended_by_infinities}{
    When $K = [0,\infty]$ and $M = [0,\infty]$, this recovers the classical \CrefAndHyperrefIfExist{definition:lebesgue_integral_of_measurable_functions_into_the_real_line_extended_by_infinities}{Lebesgue integral of non-negative measurable functions}.
}
