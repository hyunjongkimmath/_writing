\begin{definition} \label{definition:orthogonal_complement_of_a_submodule_of_a_module_with_a_sesquilinear_form_over_a_ring_with_involution}
    Let \CrefAndHyperrefIfExist{definition:ring}{$R$} be a \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{ring with involution}. Let $(M,b)$ be a \CrefAndHyperrefIfExist{definition:module_over_a_ring}{(left/right) module over $R$} with a \CrefAndHyperrefIfExist{definition:sesquilinear_form_on_a_module_over_a_ring_with_involution}{sesquilinear form} $b: M \times M \to R$.

    Let $U \subset M$ be a \CrefAndHyperrefIfExist{definition:submodule_of_a_module_over_a_ring}{submodule}. 
    \begin{enumerate}
        \item The \hldef{left orthogonal complement of $U$ in $(M,b)$} is 
        $$U^{\perp_L} = \{x \in M : b(x,u) = 0 \text{ for all } u \in U \}.$$
        \item The \hldef{right orthogonal complement of $U$ in $(M,b)$} is 
        $$U^{\perp_R} = \{y \in M : b(u,y) = 0 \text{ for all } u \in U\}.$$
        \item Let $\varepsilon \in Z(R)$\CrefIfExists{definition:center_of_a_ring} be an element such that $\varepsilon \cdot \overline{\varepsilon} = 1$ and Assume that $b$ is a \textsl{$\varepsilon$-Hermitian form}. The left and right orthogonal complements of $U$ in $(M,b)$ coincide\CrefIfExists{lemma:orthogonal_complement_of_a_submodule_of_a_module_with_a_varepsilon_hermitian_form_over_a_ring_with_involution_can_be_defined_either_using_the_left_or_right_side_of_the_form}; it is called the \hldef{orthogonal complement of $U$ in $(M,b)$} and may be denoted by \hl{$U^\perp$}.
    \end{enumerate}
\end{definition}
