\begin{definition} \label{definition:cohomology_functors_for_a_triangulated_category_with_a_t_structure}
    Let $D$ be a \CrefAndHyperrefIfExist{definition:triangulated_category}{triangulated category} and let $(\Dleqzero, \Dgeqzero)$ be a \CrefAndHyperrefIfExist{definition:t_structure_on_a_triangulated_category}{$t$-structure} on $D$.
    Define the \hldef{cohomology functors} \hl{$H^n: D \to \Core(D)$} as follows:
    \begin{align*}
        H^0(X) &= \tauleqzero \taugeqzero X \\
        H^n(X) &= H^0(X[n]).
    \end{align*}
    Note that $H^0(X) \in \Core(D)$ by \Cref{lemma:compatibility_of_t_structure_on_triangulated_category_with_truncation_functor}.
\end{definition}
