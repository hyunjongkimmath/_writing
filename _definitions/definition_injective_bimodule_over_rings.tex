\begin{definition} \label{definition:injective_bimodule_over_rings}
Let $R$ and $S$ be \CrefAndHyperrefIfExist{definition:ring}{rings}.
An \hldef{injective $R$-$S$-bimodule} is an \CrefAndHyperrefIfExist{definition:bimodule_over_rings_module_over_a_ring}{$(R,S)$-bimodule} $E$ that satisfies any of the following equivalent conditions:
\begin{enumerate}
    \item The \CrefAndHyperrefIfExist{definition:functor_between_categories}{functor}
    $$\operatorname{Hom}_{{}_R\mathsf{Mod}_S}(-, E): {}_R\mathsf{Mod}_S \to \mathsf{Ab}$$
    \CrefIfExists{definition:hom_of_left_right_bi_modules_of_rings}\CrefIfExists{definition:category_of_modules_and_bimodules_over_rings}\CrefIfExists{definition:category_of_groups_of_abelian_groups}
    is an \CrefAndHyperrefIfExist{definition:exact_functor_between_abelian_categories}{exact functor} from the \CrefAndHyperrefIfExist{definition:abelian_category}{abelian category} of \CrefAndHyperrefIfExist{definition:bimodule_over_rings_module_over_a_ring}{$(R,S)$-bimodules} to the \CrefAndHyperrefIfExist{definition:category_of_groups_of_abelian_groups}{category of abelian groups}.
    
    \item $E$ is an \CrefAndHyperrefIfExist{definition:injective_and_projective_objects_in_a_category}{injective object} in the \CrefAndHyperrefIfExist{definition:category}{category} \CrefAndHyperrefIfExist{definition:category_of_modules_and_bimodules_over_rings}{${}_R\mathsf{Mod}_S$}. That is, for every monomorphism of $(R,S)$-bimodules $f: M \to N$ and every homomorphism $g: M \to E$, there exists a homomorphism $h: N \to E$ such that $h \circ f = g$.
\end{enumerate}
\end{definition}
