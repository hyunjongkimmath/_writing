\begin{definition}[Simple abelian variety] \label{definition:simple_abelian_variety_over_a_field}
    Let $k$ be a \CrefAndHyperrefIfExist{definition:field}{field} and let $A$ be an \CrefAndHyperrefIfExist{definition:abelian_variety_over_a_field}{abelian variety over $k$}. We say that $A$ is a \hldef{simple abelian variety over $k$} if $\dim A > 0$ and the only closed subgroup schemes of $A$ which are themselves \CrefAndHyperrefIfExist{definition:abelian_variety_over_a_field}{abelian varieties over $k$} are the trivial subgroup scheme $\{0_A\}$ and $A$ itself. Equivalently, every abelian subvariety of $A$ is either $0$ or $A$.
\end{definition}
