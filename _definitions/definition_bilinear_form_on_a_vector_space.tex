\begin{definition} \label{definition:bilinear_form_on_a_vector_space}
    Let $F$ be a \CrefAndHyperrefIfExist{definition:field}{field}, and let $V$ be an \CrefAndHyperrefIfExist{definition:vector_space_over_a_field}{$F$-vector space}.

    A map $B: V \times V \to F$ is called a \hldef{bilinear form on $V$} if it is linear in each argument separately; that is, for all $u, v, w \in V$ and $\lambda \in F$,
    \begin{align*}
        B(u + v, w) &= B(u, w) + B(v, w), & B(\lambda u, w) &= \lambda B(u, w), \\
        B(u, v + w) &= B(u, v) + B(u, w), & B(u, \lambda v) &= \lambda B(u, v).
    \end{align*}

    This is a special case of a \CrefAndHyperrefIfExist{definition:multilinear_map_of_modules_over_rings}{multilinear map} when $V$ is viewed as an \CrefAndHyperrefIfExist{definition:module_of_a_ring}{$F$-module}.

\TextIfExists{definition:homomorphism_of_modules_over_a_ring}{A bilinear form on $V$ corresponds to an $F$-linear map $V \to V^*$ via $v \mapsto B(v, -)$, which is a \Cref{definition:homomorphism_of_modules_over_a_ring} from $V$ to its dual module.}

\end{definition}
