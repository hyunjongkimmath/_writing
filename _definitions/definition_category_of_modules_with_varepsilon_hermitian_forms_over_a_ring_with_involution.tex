\begin{definition} \label{definition:category_of_modules_with_varepsilon_hermitian_forms_over_a_ring_with_involution}
    Let $(A, \sigma)$ be a \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{ring with involution}, and let $\varepsilon \in Z(A)$ be an element such that $\varepsilon \cdot \overline{\varepsilon} = 1$.

    The \hldef{category of modules with $\varepsilon$-Hermitian forms over $(A,\sigma)$} is the \CrefAndHyperrefIfExist{definition:category}{category} whose objects are pairs $(M,h)$ where $M$ is a \CrefAndHyperrefIfExist{definition:bimodule_over_rings_module_over_a_ring}{(right) $A$-module} and $h: M \times M \to A$ is a \CrefAndHyperrefIfExist{definition:varepsilon_hermitian_form_on_a_module_over_a_ring_with_involution}{$\varepsilon$-Hermitian form} and whose morphisms are \CrefAndHyperrefIfExist{definition:isometry_between_modules_with_sesquilinear_forms_over_ring_with_involution}{isometries}.

    In particular, such a category is a \CrefAndHyperrefIfExist{definition:groupoid}{groupoid} and is a \CrefAndHyperrefIfExist{definition:full_subcategory_of_a_category}{full subcategory} of the \CrefAndHyperrefIfExist{definition:category_of_modules_with_sesquilinear_forms_over_a_ring_with_involution}{category of modules with sesquilinear forms over $(A,\sigma)$}.
\end{definition}
