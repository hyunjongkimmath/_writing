\begin{definition}[{\cite[1.4.9.]{beilinson_berstein_deligne_atlesI}}] \label{definition:t_structure_on_general_triangulated_category_by_gluing}
Assume \Cref{context:general_gluing_framework_for_t_structures_on_triangulated_categories}. Let $(D_U^{\leq 0}, D_U^{\geq 0})$ and $(D_F^{\leq 0}, D_F^{\geq 0})$ be \CrefAndHyperrefIfExist{definition:t_structure_on_a_triangulated_category}{$t$-structures} on $D_U$ and $D_F$ respectively. Define

\hlalign{
\begin{align*}
    D^{\leq 0} &\coloneq \{K \in D \mid j^* K \in D_U^{\leq 0} \text{ and } i^* K \in D_F^{\leq 0} \} \\
    D^{\geq 0} &\coloneq \{K \in D \mid j^* K \in D_U^{\geq 0} \text{ and } i^! K \in D_F^{\geq 0} \}
\end{align*}
}

It is a $t$-structure on $D$ (\Cref{theorem:gluing_t_structure_on_general_triangulated_category_is_indeed_a_t_structure}). We might call this $t$-structure the one \hldef{induced from those of $D_U$ and $D_F$ by gluing or recollement}. 
\end{definition}
