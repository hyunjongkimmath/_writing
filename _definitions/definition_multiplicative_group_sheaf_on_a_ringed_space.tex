\begin{definition} \label{definition:multiplicative_group_sheaf_on_a_ringed_space}
    Let $(X, \calO_X)$ be a \CrefAndHyperrefIfExist{definition:ringed_space}{ringed space}.

    The \hldef{multiplicative group} \hl{$\bbG_m = \bbG_{m,(X, \calO_X)}$}, also written \hl{$\calO_X^\times$}, is the sheaf $\mathbf{Open}(X)^{\op} \to \mathbf{Grp}$ given as follows:

    $$\calO_X^\times(U) = \{s \in \calO_X(U): \text{ there exists some } t \in \calO_X(U) \text{ such that } st = 1 \}.$$

    \TextIfExists{definition:multiplicative_group_sheaf_of_a_ringed_site}{
        This is a special case of \Cref{definition:multiplicative_group_sheaf_of_a_ringed_site}
    }
\end{definition}
