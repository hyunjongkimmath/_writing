\begin{definition} \label{definition:index_of_a_subgroup_in_a_group}
    Let $G$ be a \CrefAndHyperrefIfExist{definition:group}{group} and let $H \leq G$ be a \CrefAndHyperrefIfExist{definition:subgroup_of_a_group}{subgroup}.
    The \hldef{index of $H$ in $G$} is the \CrefAndHyperrefIfExist{definition:cardinal_number_and_cardinality_of_a_set}{cardinality} of the set 
    $$\{gH: g \in G\}$$
    of \CrefAndHyperrefIfExist{definition:coset_of_a_subgroup_of_a_group_by_an_element}{left cosets} of $H$ in $G$. This is equivalent to the cardinality of the set 
    $$\{Hg: g \in G\}$$
    of \CrefAndHyperrefIfExist{definition:coset_of_a_subgroup_of_a_group_by_an_element}{right cosets} of $H$ in $G$.
    \CrefIfExists{proposition:left_and_right_cosets_of_a_subgroup_of_a_group_have_equal_cardinality}
\end{definition}
