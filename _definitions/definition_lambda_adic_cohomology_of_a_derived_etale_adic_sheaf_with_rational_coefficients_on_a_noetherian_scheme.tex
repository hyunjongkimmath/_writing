\begin{definition} \label{definition:lambda_adic_cohomology_of_a_derived_etale_adic_sheaf_with_rational_coefficients_on_a_noetherian_scheme}
    Let $(R, \lambda)$ be a \CrefAndHyperrefIfExist{definition:complete_ring_with_respect_to_an_ideal}{complete} \CrefAndHyperrefIfExist{definition:discrete_valuation_ring}{DVR} such that $R/\lambda^n$ is finite for all $n \geq 1$. Let $X$ be a \CrefAndHyperrefIfExist{definition:locally_noetherian_and_noetherian_scheme}{noetherian scheme}, and consider \CrefAndHyperrefIfExist{definition:sheaf_on_a_site}{sheaves} on the \CrefAndHyperrefIfExist{definition:small_etale_site_of_a_scheme}{(small) \'etale site on $X$}. 
    \begin{enumerate}
        \item Let $E$ be the \CrefAndHyperrefIfExist{definition:field_of_fractions_of_an_integral_domain}{fraction field} of $R$. 

        \TODO{try to see if $D^-(X,E)$ is definable and if this notion of cohomology is definable there.}
        Given an object $N$ of $D_c^b(X,E)$ (\Cref{definition:derived_category_of_cohomologically_bounded_constructible_sheaves_with_rational_adic_coefficients_on_a_noetherian_scheme}) represented by an object $M$ of $D_c^b(X,R)$ (\Cref{definition:etale_cohomology_group_of_a_scheme_with_coefficients_in_an_abelian_group}), the \hldef{(rational) $\lambda$-adic cohomology groups} \hl{$H^q(X, N)$} are defined as 
        $$H^q(X,N) \coloneq H^q(X,M) \otimes_R E$$
        (\Cref{definition:lambda_adic_cohomology_of_a_derived_etale_adic_sheaf_with_integral_coefficients_on_a_noetherian_scheme})\CrefIfExists{definition:limit_and_colimit_of_a_diagram_in_a_category}\CrefIfExists{definition:extension_of_scalars_base_change_induction_for_modules_along_ring_homomorphisms}

        \item 
        Let $K/E$ be some \CrefAndHyperrefIfExist{definition:algebraic_element_minimal_polynomial_of_an_algebraic_element_algebraic_field_extension_transcendental_element_field_extension}{algebraic extension}. 

        Given an object $P$ of $D_c^b(X,K)$(\Cref{definition:derived_category_of_cohomologically_bounded_constructible_sheaves_with_rational_adic_coefficients_on_a_noetherian_scheme}) represented by an object $N$ of $D_c^b(X,L)$ for a finite extension $L/K$, define the \hldef{(rational) $\lambda$-adic cohomology group $H^q(X,P)$} as 
        $$H^q(X,P) := H^q(X,N) \otimes_L K.$$
    \end{enumerate}
\end{definition}
