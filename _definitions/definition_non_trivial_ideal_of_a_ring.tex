\begin{definition} \label{definition:non_trivial_ideal_of_a_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{ring}. A \hldef{non-trivial ideal of $R$} refers to an (left/right/two-sided) $R$-ideal that is neither the \CrefAndHyperrefIfExist{definition:unit_ideal_of_a_ring}{unit ideal} nor the \CrefAndHyperrefIfExist{definition:zero_ideal_of_a_ring}{zero ideal}.
\end{definition}
