\begin{definition}[Strictly ordered monoid] \label{definition:strictly_ordered_monoid}
    A \CrefAndHyperrefIfExist{definition:ordered_monoid}{commutative ordered monoid} $(M, \cdot, e, \le)$ is called \hldef{strictly ordered} if for all $a, b, c \in M$ with $a < b$ and $c > e$, we have
    $$a \cdot c < b \cdot c.$$
    Equivalently, multiplication by any strictly positive element ($c > e$) strictly preserves the order.
\end{definition>
