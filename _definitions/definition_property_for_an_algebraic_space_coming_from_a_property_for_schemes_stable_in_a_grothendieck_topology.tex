\begin{definition}[See {\cite[Tags 03E6, 03E8]{stacks-project}}, {\cite[Definition 5.4.1]{olsson_ass}}] \label{definition:property_for_an_algebraic_space_coming_from_a_property_for_schemes_stable_in_a_grothendieck_topology}
    Let $S$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme}. 
    % Let $\tau$ be a Grothendieck topology on \CrefAndHyperrefIfExist{definition:scheme_over_a_scheme}{$\mathit{Sch}/S$}.
    Let $\calP$ be a property of schemes which is \CrefAndHyperrefIfExist{definition:stable_local_class_of_objects_property_of_objects_on_a_locally_small_site_whose_representable_presheaves_are_sheaves}{local} in the \CrefAndHyperrefIfExist{definition:big_etale_site_of_a_scheme}{big \'etale topology} on \CrefAndHyperrefIfExist{definition:scheme_over_a_scheme}{$\mathit{Sch}/S$}. 

    We say that an \CrefAndHyperrefIfExist{definition:algebraic_space_on_a_site_on_Sch_S_that_preserves_under_base_change_and_has_tau_local_surjectivity_and_etaleness}{algebraic space} $X$ over $S$ \hldef{has property $\calP$} if the following equivalent conditions hold:
    \begin{enumerate}
        \item There exists a scheme $U$ and \CrefAndHyperrefIfExist{definition:property_for_a_representable_morphism_of_sheaves_coming_from_a_property_of_morphisms_of_schemes_preserved_by_base_change_and_base_local}{surjective} \CrefAndHyperrefIfExist{definition:property_for_a_representable_morphism_of_sheaves_coming_from_a_property_of_morphisms_of_schemes_preserved_by_base_change_and_base_local}{\'etale} morphism $U \to X$ such that the scheme $U$ has property $\calP$
        \item There exists a scheme $U$ and \'etale morphism $U \to X$ the scheme $U$ has property $\calP$.
    \end{enumerate}
    In fact, if $X$ is \CrefAndHyperrefIfExist{definition:representable_functor_on_a_locally_small_category}{representable} by a scheme $X$, then $\calP$ holds for the algebraic space if and only if it holds for the scheme.
\end{definition}
