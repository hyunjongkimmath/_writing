\begin{definition} \label{definition:descent_datum_for_a_category_fibered_in_groupoids_over_a_category_relative_to_a_single_morphism}
    Let $\mathcal{C}$ be a category and let $p: \mathcal{F} \to \mathcal{C}$ be a \CrefAndHyperrefIfExist{definition:category_fibered_in_groupoids_over_a_category}{category fibered in groupoids}. We fix a \CrefAndHyperrefIfExist{definition:cleavage_splitting_for_a_category_fibered_in_groupoids}{cleavage} for $p$, denoting the chosen pullback functor along a morphism $f: V \to U$ by $f^*: \mathcal{F}_U \to \mathcal{F}_V$, and the structural natural isomorphisms for composition by $\alpha_{g,h}: (g \circ h)^* \xrightarrow{\sim} h^* \circ g^*$ for morphisms $W \xrightarrow{h} V \xrightarrow{g} U$.

    Let $\pi: U' \to U$ be a morphism in $
    \calC$ such that $U' \times_U U'$ \CrefIfExists{definition:cartesian_product_of_two_objects_in_a_category_over_an_object} exists in $\calC$. 
    Write $\pi_1,\pi_2: U' \times_U U' \rightrightarrows U'$ for the two canonical projections, and write $\pi_{ij}: U' \times_U U' \times_U U' \to U' \times_U U'$ for the projections to the $(i,j)$th factors. 
    \begin{enumerate}
        \item A \hldef{descent datum for $\calF$ relative to $\pi$} is a pair $(X', \phi)$ consisting of: 
        \begin{enumerate}
            \item An object $X' \in \calF_{U'}$
            \item An isomorphism $\phi: \pi_1^*(X') \xrightarrow{\sim} \pi_2^*(X')$ in $\calF_{U' \times_U U'}$.
        \end{enumerate}
        satisfying the \hldef{cocycle condition} $\phi_{13} \circ \phi_{23} \circ \phi_{12} = \id$:
        $$
        \begin{tikzcd}[row sep=large, column sep=large]
        p_{12}^* X' \arrow[rr, "\phi_{12}"] & & p_{23}^* X' \arrow[dl, "\phi_{23}"] \\
        & p_{13}^* X' \arrow[ul, "{\phi_{13}}"'] & 
        \end{tikzcd}
        $$
        where $\phi_{ij}$ denotes the pullback of $\phi$ along the projection 

        \item A \hldef{morphism $(X'_1,\phi_1) \to (X'_2, \phi_2)$ of descent data for $\calF$ relative to $\pi$} is a morphism $X'_1 \to X_2'$ in the category $\calF_{U'}$ such that 
        $$
        \begin{tikzcd}[row sep=large, column sep=large]
        p_1^*(X'_1) \arrow[r, "\phi_1"] \arrow[d, "p_1^*(\psi)"'] & p_2^*(X'_1) \arrow[d, "p_2^*(\psi)"] \\
        p_1^*(X'_2) \arrow[r, "\phi_2"] & p_2^*(X'_2)
        \end{tikzcd}
        $$
        commutes.

    \item In particular, one may speak of the \hldef{category of descent data for $\calF$ relative to $\pi$}. 
    \TextIfExists{definition:descent_datum_for_a_category_fibered_in_groupoids_over_a_category_relative_to_a_sieve_on_an_object_of_the_base_category}{
    This category of descent data is equivalent to the category of descent data as described in \Cref{definition:descent_datum_for_a_category_fibered_in_groupoids_over_a_category_relative_to_a_sieve_on_an_object_of_the_base_category} for $\calF$ with respect to the \CrefAndHyperrefIfExist{definition:sieve_on_an_object_in_a_category}{sieve} $S_\pi$ on $U$ \CrefAndHyperrefIfExist{definition:sieve_on_an_object_of_a_category_generated_by_a_family_of_morphisms}{generated by} $\pi$. 
    }
    \end{enumerate}

\end{definition}
