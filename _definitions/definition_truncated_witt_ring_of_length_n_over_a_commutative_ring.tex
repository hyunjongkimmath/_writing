% TRUNCATED WITT RING OF LENGTH N OVER A COMMUTATIVE RING - Definition of Witt rings of finite length

\begin{definition}\label{definition:truncated_witt_ring_of_length_n_over_a_commutative_ring}
Let $R$ be a \CrefAndHyperrefIfExist{definition:commutative_ring}{commutative ring}, let $p$ be a \CrefAndHyperrefIfExist{definition:prime_number}{prime number}, and let $n \geq 1$ be an integer.

The \hldef{truncated Witt ring of length $n$ over $R$} (also called the \hldef{Witt ring of length $n$ over $R$}), denoted \hl{$W_n(R)$}, is defined as follows.

Let $W_0, W_1, \dots, W_{n-1}$ be the \CrefAndHyperrefIfExist{definition:witt_polynomial_with_respect_to_a_prime_number}{Witt polynomials} with respect to $p$. There exist unique polynomials $S_k, P_k \in \mathbb{Z}[x_0, \dots, x_k, y_0, \dots, y_k]$ such that for all $0 \leq k < n$:
\begin{align*}
    W_k(x_0, \dots, x_k) + W_k(y_0, \dots, y_k) &= W_k(S_0(\mathbf{x}, \mathbf{y}), \dots, S_k(\mathbf{x}, \mathbf{y})), \\
    W_k(x_0, \dots, x_k) \cdot W_k(y_0, \dots, y_k) &= W_k(P_0(\mathbf{x}, \mathbf{y}), \dots, P_k(\mathbf{x}, \mathbf{y})),
\end{align*}
where $\mathbf{x} = (x_0, \dots, x_k)$ and $\mathbf{y} = (y_0, \dots, y_k)$.

The truncated Witt ring $W_n(R)$ is the set $R^n$ equipped with addition \hl{$\oplus$} and multiplication \hl{$\odot$} defined by:
\begin{align*}
    (x_0, \dots, x_{n-1}) \oplus (y_0, \dots, y_{n-1}) &:= (S_0(x_0, y_0), \dots, S_{n-1}(x_0, \dots, x_{n-1}, y_0, \dots, y_{n-1})), \\
    (x_0, \dots, x_{n-1}) \odot (y_0, \dots, y_{n-1}) &:= (P_0(x_0, y_0), \dots, P_{n-1}(x_0, \dots, x_{n-1}, y_0, \dots, y_{n-1})).
\end{align*}

With these operations, $W_n(R)$ is a \CrefAndHyperrefIfExist{definition:commutative_ring}{commutative ring} with zero element $(0, \dots, 0)$ and unit element $(1, 0, \dots, 0)$.

For any integer $m$ with $1 \leq m < n$, the projection map $\pi_{n,m}: W_n(R) \to W_m(R)$ given by
$$
    (x_0, \dots, x_{n-1}) \mapsto (x_0, \dots, x_{m-1})
$$
is a surjective ring homomorphism.

The truncated Witt rings form an \CrefAndHyperrefIfExist{definition:system_in_a_category_indexed_by_a_directed_poset}{inverse system} under these projection maps, and the \CrefAndHyperrefIfExist{definition:witt_ring_of_a_commutative_ring}{ring of Witt vectors} $W(R)$ is defined as the \CrefAndHyperrefIfExist{definition:limit_and_colimit_of_a_diagram_in_a_category}{inverse limit} $\varprojlim_n W_n(R)$.
\end{definition}
