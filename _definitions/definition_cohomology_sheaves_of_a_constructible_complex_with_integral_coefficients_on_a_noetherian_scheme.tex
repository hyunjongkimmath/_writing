    Let $(R, \lambda)$ be a \CrefAndHyperrefIfExist{definition:complete_ring_with_respect_to_an_ideal}{complete} \CrefAndHyperrefIfExist{definition:discrete_valuation_ring}{DVR} such that $R/\lambda^n$ is finite for all $n \geq 1$. Let $X$ be a \CrefAndHyperrefIfExist{definition:locally_noetherian_and_noetherian_scheme}{noetherian scheme}, and consider sheaves on the \CrefAndHyperrefIfExist{definition:small_etale_site_of_a_scheme}{(small) \'etale site} of $X$.

    Let $K \in \Dbc(X, R)$ be a \CrefAndHyperrefIfExist{definition:derived_category_of_bounded_constructible_complexes_of_adic_sheaves_with_integral_coefficients_on_a_noetherian_scheme}{bounded constructible complex with integral coefficients}, represented by an inverse system $K = (K_n, u_n)_{n \geq 0}$ with $K_n \in D^-(X, R/\lambda^{n+1})$. The \hldef{$i$-th cohomology sheaf of $K$} is the \CrefAndHyperrefIfExist{definition:lambda_adic_sheaf_on_the_small_etale_site_of_a_scheme_for_an_ideal_of_a_commutative_ring}{$\lambda$-adic sheaf}
    $$\hlin{\mathscr{H}^i(K) := \bigl(\mathscr{H}^i(K_n)\bigr)_{n \geq 0}}$$
    where $\mathscr{H}^i(K_n)$ is the $i$-th \CrefAndHyperrefIfExist{definition:homology_and_cohomology_objects_for_a_chain_complex_in_an_abelian_category}{cohomology sheaf} of $K_n$ in $D(X, R/\lambda^{n+1})$, and the transition maps are induced by the isomorphisms $u_n$. Each $\mathscr{H}^i(K)$ is a constructible $\lambda$-adic sheaf, and vanishes for all but finitely many $i$.
\end{definition}
