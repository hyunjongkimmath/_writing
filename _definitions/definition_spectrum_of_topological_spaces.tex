\begin{definition} \label{definition:spectrum_of_topological_spaces}
    A \hldef{spectrum} $E$ is a \CrefAndHyperrefIfExist{definition:prespectrum_of_pointed_topological_spaces}{prespectrum} such that the adjoints $\tilde{\sigma}: E_n \to \Omega E_{n+1}$ to the structure maps $\sigma: \Sigma T_n \to T_{n+1}$ are \CrefAndHyperrefIfExist{definition:homeomorphism_of_topological_spaces}{homeomorphisms}. 
    
    A \hldef{morphism $f: E \to E'$ of spectra} is a morphism of between spectra as prespectra.
\end{definition}
