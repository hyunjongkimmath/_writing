\begin{definition} \label{definition:absolute_p_nth_power_frobenius_morphism_on_a_scheme}
    Let $p$ be a \CrefAndHyperrefIfExist{definition:prime_number}{prime number}. Let $X$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme} of \CrefAndHyperrefIfExist{definition:characteristic_a_prime_for_a_scheme}{characteristic} $p$. Let $n \geq 0$. 
    \begin{enumerate}
        \item The \hldef{$p^n$th power absolute Frobenius morphism on $X$} is the scheme endomorphism $X \to X$, denoted commonly by notations such as \hl{$F_{X,p^n}$} and \hl{$\Frob_{X,p^n}$} unless otherwise specified, given by the identity on the underlying topological space and affine locally by morphisms $\Spec A \to \Spec A$ whose corresponding ring morphism $A \to A$ is the \CrefAndHyperrefIfExist{definition:absolute_p_nth_power_frobenius_endomorphism_of_a_ring_of_prime_characteristic}{$p^n$th power absolute Frobenius endomorphism}. 

        In the case that $n = 1$, this is referred to as the \hldef{absolute Frobenius morphism on $X$} and notated as \hl{$F_{X}$}.

        \item Assuming that $F_{X,p^n}$ is invertible, we may write \hl{$F_{X, 1/p^n}$} or \hl{$\Frob_{X,1/p^n}$} to denote its inverse. We might also sometimes refer to this as the \hldef{$1/p^n$th power absolute Frobenius morphism on $X$}.
    \end{enumerate}    
\end{definition}
