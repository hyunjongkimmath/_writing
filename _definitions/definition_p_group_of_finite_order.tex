\begin{definition} \label{definition:p_group_of_finite_order}
Let $G$ be a \CrefAndHyperrefIfExist{definition:countable_finite_uncountable_sets}{finite} \CrefAndHyperrefIfExist{definition:group}{group}. Let $p$ be a \CrefAndHyperrefIfExist{definition:prime_number}{prime number}.
A \CrefAndHyperrefIfExist{definition:group}{group} $G$ is called a \hldef{$p$-group} if the order of $G$ is a power of $p$, i.e., there exists an integer $n \geq 0$ such that
$$|G| = p^n.$$
\CrefIfExists{definition:cardinal_number_and_cardinality_of_a_set}
\end{definition}
