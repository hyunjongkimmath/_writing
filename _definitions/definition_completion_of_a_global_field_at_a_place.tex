
\begin{definition} \label{definition:completion_of_a_global_field_at_a_place}
    Let $K$ be a \CrefAndHyperrefIfExist{definition:global_field}{global field} and let $v$ be a \CrefAndHyperrefIfExist{definition:place_of_a_global_field}{place} of $K$. Write $|\cdot|_v$ for an absolute value representing $v$. 
    The \hldef{completion of $K$ at $v$}, often denoted \hl{$K_v$}, is the \CrefAndHyperrefIfExist{definition:completion_of_an_extended_metric_space}{completion of $K$ with respect to} the \CrefAndHyperrefIfExist{definition:metric_induced_by_an_absolute_value_on_a_field}{metric induced by $|\cdot|_v$}.

    % \CrefAndHyperrefIfExist{definition:extended_metric_induced_by_an_extended_norm_on_a_vector_space_over_a_field_with_absolute_value}{metric induced by $|\cdot|_v$} as a \CrefAndHyperrefIfExist{definition:extended_norm_on_a_vector_space_over_a_field_with_absolute_value}{norm} on the $1$-dimensional $K$-vector space $K$ with absolute value $|\cdot|_v$. 
\end{definition}
