\begin{definition}[Verdier dual of a constructible complex] \label{definition:verdier_dual_of_an_object_in_the_bdd_constr_derived_category_of_adic_sheaves_with_integral_coefficients_on_a_finite_type_scheme_over_a_noetherian_scheme}
    Let $S$ be a \CrefAndHyperrefIfExist{definition:locally_noetherian_and_noetherian_scheme}{noetherian scheme}. Let $X$ be a finite type $S$-scheme. Let $A$ be a \CrefAndHyperrefIfExist{definition:noetherian_ring}{noetherian} \CrefAndHyperrefIfExist{definition:torsion_ring}{torsion ring}. Consider sheaves of $A$-modules on the \CrefAndHyperrefIfExist{definition:small_etale_site_of_a_scheme}{(small) \'etale sites} of $X$ and $S$.

    For any object $K \in D^+(X_{\et}, A)$, the \hldef{Verdier dual of $K$ on $X$ relative to $S$} is
    $$\hlin{\mathbb{D}_X(K) := R\mathscr{H}om_A(K, \mathbb{D}_{X/S}^\bullet).}$$
    \CrefIfExists{definition:dualizing_complex_on_a_finite_type_scheme_relative_to_a_noetherian_base_with_torsion_coefficients}\CrefIfExists{definition:derived_sheaf_internal_hom_functor_of_complexes_of_modules_on_an_essentially_small_ringed_site_with_a_pretopology}
\end{definition}
