\begin{definition}[Free path space of a topological space] \label{definition:free_path_space_of_a_topological_space}
    Let $Y$ be a \CrefAndHyperrefIfExist{definition:topological_space}{topological space}. 
    The \hldef{free path space of $Y$}, is the \CrefAndHyperrefIfExist{definition:compact_open_topology_on_the_set_of_continuous_maps_between_topological_spaces_and_function_space}{function space} $\operatorname{Map}(I, Y)$ of all \CrefAndHyperrefIfExist{definition:path_and_loop_in_a_topological_space}{paths} $\gamma : [0,1] \to Y$ equipped with the \CrefAndHyperrefIfExist{definition:compact_open_topology_on_the_set_of_continuous_maps_between_topological_spaces_and_function_space}{compact-open topology}. As such, it is also appropriate to denote the free path space of $Y$ as \hl{$Y^I$}.
\end{definition}
