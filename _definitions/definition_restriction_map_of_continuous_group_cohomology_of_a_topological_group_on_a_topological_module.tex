\begin{definition} \label{definition:restriction_map_of_continuous_group_cohomology_of_a_topological_group_on_a_topological_module}
    Let $G$ be a topological group. Let $R$ be a topological ring. 
    Let $R$ be a \CrefAndHyperrefIfExist{definition:topological_ring}{topological ring}, $G$ a \CrefAndHyperrefIfExist{definition:topological_group}{topological group}, and let $H$ be a topological subgroup of $G$. Let $M$ be a \Cref{definition:topological_module_over_a_group_ring_of_a_topological_group_over_a_topological_ring} of $G$ over $R$.

    There are restriction maps
    $$\hlin{\operatorname{res}: C^n(G,M) \to C^n(H,M) \quad f \mapsto f|_{H^n}}$$
    on the \CrefAndHyperref{definition:continuous_cochain_complex_of_a_topological_module_over_a_group_ring_of_a_topological_group_over_a_topological_ring}{continuous cochain complex groups of $M$ as $G$ and $H$ modules}. These are compatible with the coboundary maps, i.e. $d^{n+1} \circ \operatorname{res} = \operatorname{res} \circ d^n$ for all $n$ and induce \hldef{restriction maps}
    $$\hlin{\operatorname{res}: H^n(G,M) \to H^n(H,M) \quad f \mapsto f|_{H^n}}$$

    % Let $R\text{-}\mathbf{TopMod}_G$ be the \CrefAndHyperrefIfExist{definition:category_of_topological_modules_over_a_group_ring_of_a_topological_group_over_a_topological_ring}{category of (left) topological $G$-modules over $R$}.
\end{definition}
