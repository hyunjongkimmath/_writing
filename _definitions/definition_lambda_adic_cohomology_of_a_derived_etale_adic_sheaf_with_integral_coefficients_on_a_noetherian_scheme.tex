\begin{definition} \label{definition:lambda_adic_cohomology_of_a_derived_etale_adic_sheaf_with_integral_coefficients_on_a_noetherian_scheme}
    Let $(R, \lambda)$ be a \CrefAndHyperrefIfExist{definition:complete_ring_with_respect_to_an_ideal}{complete} \CrefAndHyperrefIfExist{definition:discrete_valuation_ring}{DVR} such that $R/\lambda^n$ is finite for all $n \geq 1$. Let $X$ be a \CrefAndHyperrefIfExist{definition:locally_noetherian_and_noetherian_scheme}{noetherian scheme}, and consider \CrefAndHyperrefIfExist{definition:sheaf_on_a_site}{sheaves} on the \CrefAndHyperrefIfExist{definition:small_etale_site_of_a_scheme}{(small) \'etale site on $X$}. 

    Given an object $M$ of $D^-(X,R)$ (\Cref{definition:derived_category_of_bounded_above_complexes_of_adic_sheaves_on_a_scheme}), the \hldef{$\lambda$-adic cohomology groups with coefficients in $M$} are the groups \hl{$H^q(X, M)$} defined as 
    $$H^q(X,M) \coloneq \varprojlim_n H^q(X, M_n)$$
    (\Cref{definition:etale_cohomology_group_of_a_scheme_with_coefficients_in_an_abelian_group})\CrefIfExists{definition:limit_and_colimit_of_a_diagram_in_a_category}

    In particular, this notion is defined on the subcategory $D_c^b(X,R)$ (\Cref{definition:etale_cohomology_group_of_a_scheme_with_coefficients_in_an_abelian_group}) and on the subcategory of \CrefAndHyperrefIfExist{definition:lambda_adic_sheaf_on_the_small_etale_site_of_a_scheme_for_an_ideal_of_a_commutative_ring}{$\lambda$-adic sheaves} on $X$.
\end{definition}
