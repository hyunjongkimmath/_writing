\begin{definition} \label{definition:local_and_compact_uniform_convergence_of_sequences_of_functions_from_a_topological_space_to_an_extended_metric_space_valued_in_ordered_commutative_monoid_with_adjoined_infinity}

    Let $T$ be an \CrefAndHyperrefIfExist{definition:ordered_commutative_monoid}{ordered commutative monoid}, let $T^\infty$ be the \CrefAndHyperrefIfExist{definition:ordered_monoid_with_adjoined_infinity}{ordered monoid with adjoined infinity} obtained from $T$, let $X$ be a \CrefAndHyperrefIfExist{definition:topological_space}{topological space}, let $(Y, d)$ be an \CrefAndHyperrefIfExist{definition:extended_metric_on_a_set_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity}{extended metric space valued in $T^\infty$}, and let $f_n : X \to Y$ be a sequence of \CrefAndHyperrefIfExist{definition:function_of_sets}{functions}.

    \begin{itemize}
        \item The sequence $(f_n)$ is said to \hldef{converge to $f : X \to Y$ uniformly on compact subsets} (or \hldef{converge compactly}) if for every \CrefAndHyperrefIfExist{definition:compact_topological_space}{compact subset} $K \subseteq X$,
        $$\lim_{n \to \infty} \sup_{x \in K} d(f_n(x), f(x)) = 0_T.$$

        \item The sequence $(f_n)$ is said to \hldef{converge to $f$ locally uniformly} if for every $x \in X$ there exists a \CrefAndHyperrefIfExist{definition:neighborhood_and_neighborhood_basis_of_a_point_in_a_topological_space}{neighborhood} $U$ of $x$ such that $f_n \to f$ uniformly on $U$.
    \end{itemize}

    If $X$ is locally compact, then compact convergence and local uniform convergence coincide. Compact convergence is strictly weaker than uniform convergence but strictly stronger than pointwise convergence.
\end{definition}
