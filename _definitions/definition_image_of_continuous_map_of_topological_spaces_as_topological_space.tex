\begin{definition} \label{definition:image_of_continuous_map_of_topological_spaces_as_topological_space}
    Let $f: X \to Y$ be a \CrefAndHyperrefIfExist{definition:continuous_map_of_topological_spaces}{continuous map} of \CrefAndHyperrefIfExist{definition:topological_space}{topological spaces}.    

    The \CrefAndHyperrefIfExist{definition:function_of_sets}{image} $f(X)$ of $f$ is, by default, regarded as a topological space by equipping $f(X)$ with its \CrefAndHyperrefIfExist{definition:subspace_of_a_topological_space}{subspace topology} in $Y$.
\end{definition}
