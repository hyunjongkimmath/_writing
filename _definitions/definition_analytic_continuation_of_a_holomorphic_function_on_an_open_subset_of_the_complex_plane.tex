\begin{definition}[Analytic continuation] \label{definition:analytic_continuation_of_a_holomorphic_function_on_an_open_subset_of_the_complex_plane}
Let $U$ and $V$ be non-empty open subsets of $\mathbb{C}$, and let $f: U \to \mathbb{C}$ and $g: V \to \mathbb{C}$ be \CrefAndHyperref{definition:holomorphic_function_from_a_subset_of_Cn_to_Cm}{holomorphic functions}. Suppose $U \cap V \neq \varnothing$ and there exists a non-empty open subset $W \subseteq U \cap V$ such that $f|_W = g|_W$. Then $g$ is called an \hldef{analytic continuation of $f$ to $V$}, and $f$ is called an \hldef{analytic continuation of $g$ to $U$}.
\end{definition}
