\begin{definition} \label{definition:direct_summand_of_an_object_in_an_additive_category}
    Let $\calA$ be an \CrefAndHyperrefIfExist{definition:additive_category_preadditive_category}{additive category}.
    Suppose that $A,B,C \in \calA$ are objects such that $A \cong B \oplus C$(\Cref{definition:additive_category_preadditive_category}). We say that $B$ and $C$ are \hldef{direct summands of $A$}.
\end{definition}
