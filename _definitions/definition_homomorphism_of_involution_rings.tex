\begin{definition} \label{definition:homomorphism_of_involution_rings}
    A \hldef{homomorphism/morphism $\varphi: ( A, \sigma_A) \to (B,\sigma_B)$ of involution rings} is a \CrefAndHyperrefIfExist{definition:ring_homomorphism}{ring homomorphism} $\varphi: A \to B$ such that $\sigma_B(\varphi(a)) = \varphi(\sigma_A(a))$ for all $a \in A$.
\end{definition}
