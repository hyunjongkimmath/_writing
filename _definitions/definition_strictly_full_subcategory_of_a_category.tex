\begin{definition} \label{definition:strictly_full_subcategory_of_a_category}
    A \hldef{strictly full subcategory of a category} is a \CrefAndHyperrefIfExist{definition:subcategory_of_a_category}{subcategory} that is \CrefAndHyperrefIfExist{definition:full_and_faithful_functor_between_locally_small_categories}{full} and also closed under isomorphism.
\end{definition}
