% ARCHIMEDEAN AND NONARCHIMEDEAN METRICS - Definitions of archimedean and non-archimedean metrics
\begin{definition}\label{definition:archimedean_and_nonarchimedean_extended_metrics_on_sets_valued_in_ordered_commutative_monoid_with_adjoined_infinity}
    Let $T$ be an \CrefAndHyperrefIfExist{definition:ordered_commutative_monoid}{ordered commutative monoid}, let $T^\infty$ be the \CrefAndHyperrefIfExist{definition:ordered_monoid_with_adjoined_infinity}{ordered monoid with adjoined infinity} obtained from $T$, and let $(X, d)$ be an \CrefAndHyperrefIfExist{definition:extended_metric_on_a_set_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity}{extended metric space valued in $T^\infty$}.

    \begin{enumerate}
        \item The extended metric $d$ is called a \hldef{non-Archimedean metric} (or \hldef{ultrametric}) if it satisfies the \hldef{ultrametric inequality}: for all $x, y, z \in X$,
        $$
            d(x, z) \leq \max\{d(x, y), d(y, z)\}.
        $$
        An extended metric space $(X, d)$ equipped with a non-Archimedean metric is called a \hldef{non-Archimedean metric space} (or \hldef{ultrametric space}).
    \end{enumerate}

\end{definition}
