\begin{definition}[$G$-unramified morphism of schemes] \label{definition:G_unramified_morphism_of_schemes}
    \TODO{AI generated, verify}
    Let $S$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme}, and let $G$ be a finite flat group scheme over $S$.

    A \CrefAndHyperrefIfExist{definition:morphism_of_schemes}{morphism of schemes} $f: X \to Y$ over $S$ is called \hldef{$G$-unramified} if the image of every diagonal morphism
    $$\Delta_G: G \to G \times_S G$$
    composed with any $S$-map $U \to G \times_S G$ (for affine $U$) does not factor through the base change $X \times_S Y \to X \times_S X$. Equivalently, no section of a base change of $f$ can have difference lying in the image of $G$.
\end{definition}
