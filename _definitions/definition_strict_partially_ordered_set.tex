\begin{definition}[Strict partially ordered set, strict partial order] \label{definition:strict_partially_ordered_set}
    A \hldef{strict partially ordered set} (or \hldef{strict poset}) is a pair $(P, <)$ where $P$ is a set and
    \[
    < : P \times P \to \{\text{true}, \text{false}\}
    \]
    is a \CrefAndHyperrefIfExist{definition:n_ary_relation_on_sets}{binary relation} on $P$ satisfying the following axioms for all $a, b, c \in P$:
    \begin{itemize}
        \item \CrefAndHyperrefIfExist{definition:irreflexive_binary_relation_on_a_set}{Irreflexivity}: not $a < a$,
        \item \CrefAndHyperrefIfExist{definition:reflexive_symmetric_antisymmetric_transitive_total_binary_relations_on_a_set}{Transitivity}: if $a < b$ and $b < c$, then $a < c$.
    \end{itemize}
    Consequently, $<$ is \CrefAndHyperrefIfExist{definition:asymmetric_binary_relation_on_a_set}{asymmetric}, i.e. if $a < b$, then $\lnot (b < a)$\CrefIfExists{lemma:strict_partial_order_is_asymmetric}

    The relation $<$ is called a \hldef{strict partial order}.
    \end{enumerate}

    One writes \hl{$a > b$} synonymously for $b < a$. One might also synonymously write \hl{$a \leq b$} and \hl{$a \geq b$} respectively for 
    \begin{itemize}
        \item $a < b$ or $a = b$,
        \item $a > b$ or $a = b$.
    \end{itemize}
    This binary relation $\leq$ is then a \CrefAndHyperrefIfExist{definition:partially_ordered_set}{partial order} on $P$\CrefIfExists{proposition:correspondence_between_partial_orders_and_strict_partial_orders_on_a_set}.
\end{definition}
