\begin{definition} \label{definition:trace_ideal_for_a_module_over_a_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{ring}. Let $M$ be an \CrefAndHyperrefIfExist{definition:module_of_a_ring}{(left) $R$-module}. 
    The \hldef{trace ideal of $M$}, denoted by notations such as \hl{$\tau_R(M)$} or \hl{$\Tr_R(M)$}, is the \CrefAndHyperrefIfExist{definition:ideal_of_a_ring}{$R$-ideal} of $R$ \CrefAndHyperrefIfExist{definition:ideal_generated_by_a_subset_of_a_ring}{generated by} the \CrefAndHyperrefIfExist{definition:image_coimage_of_a_morphism_in_a_category}{image} of the \CrefAndHyperrefIfExist{definition:evaluation_map_for_a_module_over_a_ring}{evaluation homomorphism} 
    $$\ev: M^\vee \otimes_R M \to R.$$
\end{definition}
