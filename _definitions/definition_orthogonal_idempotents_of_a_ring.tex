\begin{definition} \label{definition:orthogonal_idempotents_of_a_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{ring}. 
    A set $\{e_1,\ldots,e_n\}$ in $R$ is said to consist of \hldef{orthogonal idempotents} if the set is \CrefAndHyperrefIfExist{definition:orthogonal_elements_of_a_ring}{orthogonal} and each $e_i$ is \CrefAndHyperrefIfExist{definition:idempotent_element_of_a_ring}{idempotent}.
\end{definition}
