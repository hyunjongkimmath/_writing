\begin{definition} \label{definition:isolated_singularity_of_a_holomorphic_function_from_an_open_subset_of_the_complex_plane_to_the_complex_plane_classified_as_removable_pole_or_essential}
Let $U \subset \mathbb{C}$ be an \CrefAndHyperrefIfExist{definition:topological_space}{open set} and let $f : U \setminus \{z_0\} \to \mathbb{C}$ be a \CrefAndHyperrefIfExist{definition:holomorphic_function_from_a_subset_of_Cn_to_Cm}{holomorphic function} defined on $U$ with a single point $z_0 \in U$ removed. The point $z_0$ is called an \hldef{isolated singularity of $f$}. The isolated singularity $z_0$ is classified as follows:
\begin{enumerate}
  \item $z_0$ is a \hldef{removable singularity} if $f$ can be extended to a function holomorphic on all of $U$. By \Cref{theorem:riemanns_theorem_on_removable_singularities}, this is equivalent to each of the following conditions:
  \begin{enumerate}
      \item $\lim_{z \to z_0} (z - z_0)f(z) = 0$\CrefIfExists{definition:convergence_and_limit_of_sequence_in_extended_metric_space_valued_in_ordered_commutative_monoid_with_adjoined_infinity};
      \item $f$ is bounded in a \CrefAndHyperrefIfExist{definition:punctured_disk_in_the_complex_plane}{punctured neighborhood} of $z_0$;
      \item the \CrefAndHyperrefIfExist{definition:laurent_series_expansion_of_a_holomorphic_function_on_an_annulus_centered_at_an_isolated_singularity}{Laurent expansion} of $f$ at $z_0$ has no negative-degree terms, i.e.\ it reduces to a \CrefAndHyperrefIfExist{definition:taylor_and_maclaurin_series_of_a_real_valued_function}{Taylor series}.
  \end{enumerate}

  \item $z_0$ is a \hldef{pole of order $m \in \mathbb{N}$} if there exists a \CrefAndHyperrefIfExist{definition:holomorphic_function_from_a_subset_of_Cn_to_Cm}{holomorphic function} $g$ on $U$ with $g(z_0) \neq 0$ such that $f(z) = \frac{g(z)}{(z - z_0)^m}$ for all $z \in U \setminus \{z_0\}$. Equivalently, $\lim_{z \to z_0} |f(z)| = \infty$ and the \CrefAndHyperrefIfExist{definition:laurent_series_expansion_of_a_holomorphic_function_on_an_annulus_centered_at_an_isolated_singularity}{Laurent expansion} of $f$ at $z_0$ has finitely many negative-degree terms.
  \item $z_0$ is an \hldef{essential singularity} if it is neither a removable singularity nor a pole. Equivalently, the Laurent expansion of $f$ at $z_0$ has infinitely many negative-degree terms, and by \CrefAndHyperrefIfExist{theorem:weierstrass_theorem_on_essential_singularities}{Weierstrass's theorem on essential singularities}, $f$ takes every complex value (with at most one exception) infinitely often in any neighborhood of $z_0$.
\end{enumerate}
\end{definition}
