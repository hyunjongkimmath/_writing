\begin{definition}[{\cite[Expos\'e III, D\'efinition 2.1.]{SGA4_I}}] \label{definition:cocontinuous_functor_of_sites}
    Let $(\calC,J)$ and $(\calD,K)$ be \CrefAndHyperrefIfExist{definition:grothendieck_topology_on_a_category_site_covering_sieve_topologically_generating_family}{sites} (or \CrefAndHyperrefIfExist{definition:small_site_locally_small_site_U_site_U_small_site_U_topology}{$\mathscr{U}$-sites} if a \CrefAndHyperrefIfExist{definition:grothendieck_universe}{Grothendieck universe} $\mathscr{U}$ is available). 

    A functor $u: \mathcal{C} \to \mathcal{D}$ is said to be a \hldef{cocontinuous functor} if for every object $X \in \operatorname{Ob}(\mathcal{C})$ and every \CrefAndHyperrefIfExist{definition:grothendieck_topology_on_a_category_site_covering_sieve_topologically_generating_family}{covering sieve} $R \in K(u(X))$, the sieve $S$ on $X$ \CrefAndHyperrefIfExist{definition:sieve_on_an_object_of_a_category_generated_by_a_family_of_morphisms}{generated by} the set of morphisms $\{f: Y \to X \mid u(f) \in R\}$ is a covering sieve in $J(X)$.
\end{definition}
