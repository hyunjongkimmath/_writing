\begin{definition} \label{definition:fiber_functor_for_a_functor_from_a_galois_category_to_the_category_of_finite_sets}
    \TODO{category of finite sets}
Let $\mathcal{C}$ be a \CrefAndHyperrefIfExist{definition:pre_galois_category}{pre-Galois category}. 
A functor $F: \mathcal{C} \to \textbf{FinSet}$ from $\mathcal{C}$ to the category of finite sets is called a \hldef{fiber functor} if:
\begin{enumerate}
    \item $F$ is \CrefAndHyperrefIfExist{definition:left_right_exact_functor_between_categories}{exact} (preserves \CrefAndHyperrefIfExist{definition:limit_and_colimit_of_a_diagram_in_a_category}{finite limits} and \CrefAndHyperrefIfExist{definition:limit_and_colimit_of_a_diagram_in_a_category}{finite colimits}).
    \item $F$ is \CrefAndHyperrefIfExist{definition:full_and_faithful_functor_between_locally_small_categories}{faithful} and \CrefAndHyperrefIfExist{definition:reflects_a_type_of_morphism_for_a_functor_between_categories}{reflects} \CrefAndHyperrefIfExist{definition:isomorphism_in_a_category}{isomorphisms}.
\end{enumerate}
A \hldef{Galois category} is a pair $(\calC, F)$ consistsing of a pre-Galois category and a fiber functor.
\end{definition}
