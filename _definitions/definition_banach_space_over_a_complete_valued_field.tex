% A \hldef{Banach space} is a vector space \(X\) over the field \(\mathbb{R}\) or \(\mathbb{C}\) equipped with a norm \(\|\cdot\|\) such that \(X\) is complete with respect to the metric induced by the norm.
% \end{definition}

\begin{definition}[Banach space over a complete valued field] \label{definition:banach_space_over_a_complete_valued_field}
Let $k$ be a \CrefAndHyperrefIfExist{definition:field}{field} equipped with a \CrefAndHyperrefIfExist{definition:absolute_value_on_a_field}{non-trivial absolute value} $|\cdot| : k \to \mathbb{R}_{\geq 0}$ such that $k$ is \CrefAndHyperrefIfExist{definition:complete_extended_metric_space}{complete} with respect to the \CrefAndHyperrefIfExist{definition:metric_induced_by_an_absolute_value_on_a_field}{metric induced by} $|\cdot|$.  
A \hldef{Banach space over $k$} is a \CrefAndHyperrefIfExist{definition:vector_space_over_a_field}{vector space} $X$ over $k$ equipped with a \CrefAndHyperrefIfExist{definition:extended_norm_on_a_vector_space_over_a_field_with_absolute_value}{norm} $\|\cdot\| : X \to \mathbb{R}_{\geq 0}$ 
% satisfying:
% \begin{align*}
% \|x\| = 0 &\iff x = 0, \\
% \|x+y\| &\leq \|x\| + \|y\| \quad \text{for all } x,y \in X, \\
% \|\lambda x\| &= |\lambda|\cdot\|x\| \quad \text{for all } \lambda \in k, x \in X,
% \end{align*}
such that the metric $d(x,y) := \|x-y\|$ \CrefAndHyperrefIfExist{definition:extended_metric_induced_by_an_extended_norm_on_a_vector_space_over_a_field_with_absolute_value}{induced by the norm} makes $(X,d)$ into a \CrefAndHyperrefIfExist{definition:complete_extended_metric_space}{complete metric space}.

If the underlying valued field $(k, |\cdot|)$ is \CrefAndHyperrefIfExist{definition:archimedean_absolute_value_on_a_field}{non-archimedean}, then the literature often additionally assumes that $(X,d)$ is also \CrefAndHyperrefIfExist{definition:archimedean_and_nonarchimedean_metrics_on_sets}{nonarchimedean}. It would be good practice to explicitly refer to a "nonarchimedean Banach space" in this case. 
\end{definition}
