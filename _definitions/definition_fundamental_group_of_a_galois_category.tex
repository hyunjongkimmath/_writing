\begin{definition} \label{definition:fundamental_group_of_a_galois_category}
    Let $(\calC, F)$ be a \CrefAndHyperrefIfExist{definition:fiber_functor_for_a_functor_from_a_galois_category_to_the_category_of_finite_sets}{Galois category}. 
    The \hldef{fundamental group of $(\calC, F)$} is the group of \CrefAndHyperrefIfExist{definition:natural_automorphism_of_a_functor_between_categories}{natural automorphisms} of $F: \calC \to \FinSets$. It may be denoted by notations such as \hl{$\pi(\calC, F)$}, \hl{$\Pi(\calC, F)$}, or \hl{$\Aut(\calC, F)$}. If the fiber functor $F$ is clear from context, then the $F$ may be suppressed from the notation.

    It is \CrefAndHyperrefIfExist{definition:profinite_group}{profinite} (\Cref{proposition:fundamental_group_of_a_galois_category_is_profinite}).
\end{definition}
