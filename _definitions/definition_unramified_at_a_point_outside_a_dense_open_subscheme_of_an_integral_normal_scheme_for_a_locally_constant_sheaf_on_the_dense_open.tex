\begin{definition} \label{definition:unramified_at_a_point_outside_a_dense_open_subscheme_of_an_integral_normal_scheme_for_a_locally_constant_sheaf_on_the_dense_open}
    Let $X$ be an \CrefAndHyperrefIfExist{definition:integral_scheme}{integral} \CrefAndHyperrefIfExist{definition:normal_and_factorial_scheme}{normal scheme} with \CrefAndHyperrefIfExist{definition:generic_point_of_a_topological_space}{generic point} $\eta$ and \CrefAndHyperrefIfExist{definition:function_field_over_a_field}{function field} $K$. 
    % Let $\overline{\eta}$ be a geometric point of $X$ whose image is $\eta$. Let $U \subseteq X$ be a non-empty \CrefAndHyperrefIfExist{definition:open_and_closed_subscheme_of_a_scheme}{open subscheme}. 
    Let $x \in X \setminus U$ be a point. 
    % and let $\overline{x}$ be a geometric point whose image is $x$. 
    Let $\calF$ be a locally constant sheaf on $U$. In particular, there is a natural action of $\pioneet(U, \overline{x})$ on the \CrefAndHyperrefIfExist{definition:stalk_of_a_presheaf_on_the_small_etale_site_of_a_schem_at_a_geometric_point}{stalk} $\calF_{\overline{x}}$.

    We say that $\calF$ is \hldef{unramified at $x$} if the action of the \CrefAndHyperref{definition:decomposition_and_inertia_subgroups_of_etale_fundamental_group_of_dense_open_of_integral_normal_scheme_at_point_in_complement}{inertia subgroup} $I_x$ on 
    $\calF_{\overline{x}}$ is trivial. Otherwise, we say that $\calF$ is \hldef{ramified at $x$}. Note that these notions neither depend on any choice of valuation \CrefAndHyperrefIfExist{definition:centered_at_for_a_prime_of_an_integral_domain_point_of_an_integral_scheme}{centered at} $x$ nor on the choice of geometric point $\overline{x}$ lifting $x$ as these notions are insensitive to conjugation.
\end{definition}
