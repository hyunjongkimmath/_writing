\begin{definition} \label{definition:perfect_field}
A \hldef{field} $k$ is said to be a \hldef{perfect field} if it satisfies one of the following equivalent conditions:
    \TODO{irred polynomial, root}
\begin{enumerate}
    \item Every irreducible polynomial over $k$ has distinct roots in an \CrefAndHyperrefIfExist{definition:algebraic_element_minimal_polynomial_of_an_algebraic_element_algebraic_field_extension_transcendental_element_field_extension}{algebraic closure} of $k$ (i.e., every \CrefAndHyperrefIfExist{definition:algebraic_element_minimal_polynomial_of_an_algebraic_element_algebraic_field_extension_transcendental_element_field_extension}{algebraic extension} of $k$ is \CrefAndHyperrefIfExist{definition:separable_algebraic_field_extension}{separable}).
    \item The characteristic of $k$ is $0$, or the characteristic of $k$ is $p > 0$ and the Frobenius endomorphism $\Phi: k \to k$, defined by $\Phi(x) = x^p$, is an automorphism.
\end{enumerate}
\end{definition}
