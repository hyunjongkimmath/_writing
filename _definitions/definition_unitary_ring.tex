\begin{definition}[see {\cite[Chapter I, 5.1]{knus_qhfr}}] \label{definition:unitary_ring}
    Let $A$ be an \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{involution ring}. Let $\varepsilon \in Z(A)$\CrefIfExists{definition:center_of_a_ring} such that $\varepsilon \cdot \overline{\varepsilon} = 1$. Let $\Lambda$ be an additive subgroup of $A$ such that 

        \TODO{$S_{-\varepsilon}(A)$, $S^{-\varepsilon}(A)$}
    \begin{enumerate}
        \item $S_{-\varepsilon}(A) \subseteq \Lambda \subseteq S^{-\varepsilon}(A)$

        \item $a \cdot x \cdot \overline{a} \in \Lambda$ for all $x \in \Lambda$ and $a \in A$. 
    \end{enumerate}
    The triple $(A, \varepsilon, \Lambda)$ is called a \hldef{unitary ring} and the pair $(\varepsilon, \Lambda)$ is called a form parameter for $A$. 

    A \hldef{morphism of unitary rings} $f: (A,\varepsilon, \Lambda) \to (A' ,\varepsilon', \Lambda)$ is a ring homomorphism $f: A \to A'$ such that $f(\varepsilon) = \varepsilon'$ and $f(\Lambda) \subset \Lambda'$. 
\end{definition}
