\begin{definition}[Almost sure convergence] \label{definition:almost_sure_convergence_of_a_sequence_of_measurable_functions_from_a_measure_space_to_a_metric_space}
Let $(X, \mathcal{A}, \mu)$ be a \CrefAndHyperrefIfExist{definition:finite_sigma_finite_measure_space_for_a_measure_on_a_measurable_space}{measure space}. Let $(Y, d_Y)$ be a \CrefAndHyperrefIfExist{definition:extended_metric_on_a_set}{metric space} (typically $Y = \mathbb{R}$, $\overline{\mathbb{R}}$, or $\mathbb{C}$).

Let $(f_n)_{n \geq 1}$ be a sequence of \CrefAndHyperrefIfExist{definition:measurable_function_from_a_measure_space_into_the_extended_real_line_or_complex_plane}{measurable functions} $f_n : X \to Y$ and let $f : X \to Y$ be a function. The sequence $(f_n)$ is said to \hldef{converge to $f$ almost surely} (or \hldef{$\mu$-a.e.) if
$$\mu\Bigl(\bigl\{x \in X : d_Y(f_n(x), f(x)) \not\to 0 \text{ as } n \to \infty\bigr\}\Bigr) = 0.$$
Equivalently, $f_n \to f$ $\mu$-a.e. if there exists a set $N \in \mathcal{A}$ with $\mu(N) = 0$ such that for all $x \in X \setminus N$, $f_n(x) \to f(x)$ in the metric $d_Y$.
\end{definition}
