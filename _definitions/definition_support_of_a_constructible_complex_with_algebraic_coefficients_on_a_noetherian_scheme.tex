\begin{definition}[Support of a constructible complex with algebraic coefficients on a noetherian scheme] \label{definition:support_of_a_constructible_complex_with_algebraic_coefficients_on_a_noetherian_scheme}

    Let $(R, \lambda)$ be a \CrefAndHyperrefIfExist{definition:complete_ring_with_respect_to_an_ideal}{complete} \CrefAndHyperrefIfExist{definition:discrete_valuation_ring}{DVR} such that $R/\lambda^n$ is finite for all $n \geq 1$. Let $E$ be the \CrefAndHyperrefIfExist{definition:field_of_fractions_of_an_integral_domain}{fraction field} of $R$, and let $K/E$ be an \CrefAndHyperrefIfExist{definition:algebraic_element_minimal_polynomial_of_an_algebraic_element_algebraic_field_extension_transcendental_element_field_extension}{algebraic extension}. Let $X$ be a \CrefAndHyperrefIfExist{definition:locally_noetherian_and_noetherian_scheme}{noetherian scheme}, and consider sheaves on the \CrefAndHyperrefIfExist{definition:small_etale_site_of_a_scheme}{(small) \'etale site} of $X$. Let $\mathcal{K} \in \Dbc(X, K)$ be a bounded constructible complex with algebraic coefficients.

    By definition, $\mathcal{K}$ is representable by a complex $K_{E'} \in \Dbc(X, E')$ for some finite extension $E'$ of $E$ inside $K$. The \hldef{support of $\mathcal{K}$} is
    $$\hlin{\Supp(\mathcal{K}) := \Supp(K_{E'})}$$
    (\Cref{definition:support_of_a_constructible_complex_with_rational_coefficients_on_a_noetherian_scheme}) This is independent of the choice of representative $K_{E'}$, and is a closed subset of $X$.
\end{definition}
