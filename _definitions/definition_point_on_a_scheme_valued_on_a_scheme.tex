\begin{definition} \label{definition:point_on_a_scheme_valued_on_a_scheme}
    Let $X$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme}.  

    \begin{enumerate}
        \item A \hldef{$T$-point of $X$} or a \hldef{$T$-point of $X$} is a \CrefAndHyperrefIfExist{definition:morphism_of_schemes}{morphism} $T \to X$. Equivalently, a $T$-point is an element of $h_X(T)$ where $h_X$ is the \CrefAndHyperrefIfExist{definition:representable_functor_on_a_locally_small_category}{representable functor} $\Sch \to \Sets$. One writes the set of $T$-points of $X$ as \hl{$X(T)$}.

        \item For a \CrefAndHyperrefIfExist{definition:ring}{ring} $R$, an \hldef{$R$-point of $X$} is an $\operatorname{Spec} R$-point\CrefIfExists{definition:affine_scheme} of $X$ in the above sense.
    \end{enumerate}
\end{definition}
