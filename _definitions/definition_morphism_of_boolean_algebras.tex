\begin{definition} \label{definition:morphism_of_boolean_algebras}
Let $A$ and $B$ be \CrefAndHyperrefIfExist{definition:boolean_algebra}{Boolean algebras}. A \hldef{morphism of Boolean algebras} is a \CrefAndHyperrefIfExist{definition:function_of_sets}{function} $f: A \to B$ such that for all $x, y \in A$:
\begin{itemize}
    \item $f(x \lor y) = f(x) \lor f(y)$,
    \item $f(x \land y) = f(x) \land f(y)$,
    \item $f(\neg x) = \neg f(x)$,
    \item $f(0_A) = 0_B$ and $f(1_A) = 1_B$.
\end{itemize}
\end{definition}
