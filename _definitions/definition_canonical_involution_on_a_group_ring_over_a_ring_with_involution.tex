\begin{definition}[see {\cite[7.3.7]{knus_qhfr}}] \label{definition:canonical_involution_on_a_group_ring_over_a_ring_with_involution}
    Let $(R, \sigma_R)$ be a \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{ring with involution}. Let $G$ be a \CrefAndHyperrefIfExist{definition:group}{group}.

    The \hldef{canonical involution on the group ring $R[G]$} \CrefIfExists{definition:group_ring_of_a_group_over_a_ring} is the \CrefAndHyperrefIfExist{definition:algebra_with_involution_over_a_ring_with_involution}{$R$-linear involution} given by
    $$\overline{\sum r_g g} = \sum {\sigma_R(r_g) g^{-1}}.$$

    If $R$ is not specified with an involution structure, then usually, the canonical involution on $R[G]$ refers to the above in the case that $R$ is equipped with the \CrefAndHyperrefIfExist{definition:trivial_involution_on_a_ring}{trivial involution}.
\end{definition}
