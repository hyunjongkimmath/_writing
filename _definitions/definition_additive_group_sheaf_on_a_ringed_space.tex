\begin{definition} \label{definition:additive_group_sheaf_on_a_ringed_space}
    Let $(X, \calO_X)$ be a \CrefAndHyperrefIfExist{definition:ringed_space}{ringed space}.

    The \hldef{additive group} \hl{$\bbG_a = \bbG_{a,(X, \calO_X)}$} is simply the sheaf $\calO_X$ regarded as a \CrefAndHyperrefIfExist{definition:sheaf_on_a_topological_space_valued_in_a_category_with_a_terminal_object}{sheaf} of abelian groups on $X$.
\end{definition}
