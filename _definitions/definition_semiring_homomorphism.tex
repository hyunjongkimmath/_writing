    Let $R$ and $S$ be \CrefAndHyperrefIfExist{definition:semiring}{semirings}. A function $f \colon R \to S$ is called a \hldef{semiring homomorphism} if for all $r_1, r_2 \in R$ the following properties hold:
    \begin{enumerate}
        \item $f(r_1 + r_2) = f(r_1) + f(r_2)$;
        \item $f(r_1 r_2) = f(r_1) f(r_2)$;
        \item $f(0_R) = 0_S$ and $f(1_R) = 1_S$.
    \end{enumerate}

    A semiring homomorphism $f \colon R \to S$ is called a \hldef{semiring isomorphism} if it is bijective.
\end{definition}
