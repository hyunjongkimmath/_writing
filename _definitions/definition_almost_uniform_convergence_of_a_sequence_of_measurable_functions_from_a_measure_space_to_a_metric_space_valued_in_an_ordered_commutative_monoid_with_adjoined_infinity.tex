\begin{definition}[Almost uniform convergence] \label{definition:almost_uniform_convergence_of_a_sequence_of_measurable_functions_from_a_measure_space_to_a_metric_space_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity}
Let $(X, \mathcal{A}, \mu)$ be a \CrefAndHyperrefIfExist{definition:finite_sigma_finite_measure_space_for_a_measure_on_a_measurable_space}{measure space}. Let $T$ be an \CrefAndHyperrefIfExist{definition:ordered_commutative_monoid}{ordered commutative monoid}, let $T^\infty$ be the \CrefAndHyperrefIfExist{definition:ordered_monoid_with_adjoined_infinity}{ordered monoid with adjoined infinity} obtained from $T$, and let $(Y, d_Y)$ be a \CrefAndHyperrefIfExist{definition:extended_metric_on_a_set_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity}{metric space valued in $T^\infty$} (typically $Y = \mathbb{R}$, $\overline{\mathbb{R}}$, or $\mathbb{C}$).

Let $(f_n)_{n \geq 1}$ be a sequence of \CrefAndHyperrefIfExist{definition:measurable_function_between_measure_spaces}{measurable functions} $f_n : X \to Y$ and let $f : X \to Y$ be a function. The sequence $(f_n)$ is said to \hldef{converge to $f$ almost uniformly} if for every $\delta > 0$ in $\bbR$, there exists a set $E \in \mathcal{A}$ such that
$$\mu(E^c) < \delta \quad \text{and} \quad f_n \to f \text{ uniformly on } E.$$
(\Cref{definition:uniform_convergence_of_a_sequence_of_functions_from_a_set_to_an_extended_metric_space_valued_in_ordered_commutative_monoid_with_adjoined_infinity})
That is, for every $\varepsilon \in T_{>0}$, there exists $N \in \mathbb{N}$ such that for all $n \geq N$ and all $x \in E$, $d_Y(f_n(x), f(x)) < \varepsilon$.
\end{definition}
