\begin{definition}[Quaternion algebra over a field] \label{definition:quaternion_algebra_over_a_field}
    \TODO{csa}
    Let $F$ be a \CrefAndHyperrefIfExist{definition:field}{field} with $\operatorname{char}(F) \neq 2$. A \hldef{quaternion algebra over $F$} is a central simple algebra of dimension $4$ over $F$. Equivalently, a quaternion algebra over $F$ is an associative unital $F$-algebra $D$ with $\dim_F D = 4$ and a basis $\{1, i, j, k\}$ satisfying the multiplication relations
    $$i^2 = a,\quad j^2 = b,\quad ij = -ji = k$$
    for some $a, b \in F^\times$. The algebra is uniquely determined by the pair $(a, b)_F$ and is denoted by \hl{$(a, b)_F$}.

    \TODO{matrix algebra over a ring}
    A quaternion algebra over $F$ is either a \CrefAndHyperrefIfExist{definition:division_ring}{division} \CrefAndHyperrefIfExist{definition:algebra_of_a_ring}{algebra} or isomorphic to the matrix algebra $\M_2(F)$.
\end{definition}
