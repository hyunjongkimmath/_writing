\begin{definition}[Cofinal and final sub-posets] \label{definition:cofinal_and_final_sub_posets}
Let $(I, \leq)$ be a \CrefAndHyperrefIfExist{definition:partially_ordered_set}{partially ordered set}, and let $J \subseteq I$ be a sub-poset. Let $(K, \leq)$ be a \CrefAndHyperrefIfExist{definition:partially_ordered_set}{partially ordered set}, and let $L \subseteq K$ be a sub-poset.
\begin{enumerate}
    \item $J$ is \hldef{cofinal in $I$} if for every $i \in I$, there exists $j \in J$ such that $i \leq j$.

    \item $L$ is \hldef{final in $K$} if for every $k \in K$, there exists $\ell \in L$ such that $\ell \leq k$.
\end{enumerate}
\end{definition}
