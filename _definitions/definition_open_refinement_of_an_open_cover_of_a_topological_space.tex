\begin{definition} \label{definition:open_refinement_of_an_open_cover_of_a_topological_space}
    Let $X$ be a \CrefAndHyperrefIfExist{definition:topological_space}{topological space} and let $\{U_i\}_{i \in I}$ be an \CrefAndHyperrefIfExist{definition:open_covering_of_a_topological_space}{open cover} of $X$. 
    An \hldef{open refinement of $\{U_i\}_{i \in I}$} is an open cover $\{V_j\}_{j \in J}$ of $X$ such that for every $j \in J$, there exists an $i \in I$ with $V_j \subseteq U_i$.
\end{definition}
