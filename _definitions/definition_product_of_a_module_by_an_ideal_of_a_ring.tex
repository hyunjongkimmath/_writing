\begin{definition} \label{definition:product_of_a_module_by_an_ideal_of_a_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{(not necessarily commutative) ring}.

    \begin{enumerate}
        \item Let $I$ be a \CrefAndHyperrefIfExist{definition:ideal_of_a_ring}{left ideal of $R$} and let $M$ be a \CrefAndHyperrefIfExist{definition:module_of_a_ring}{left $R$-module}. The \hldef{product of $I$ and $M$}, denoted by \hl{$IM$}, is the \CrefAndHyperrefIfExist{definition:submodule_of_a_module_over_a_ring}{submodule} of $M$ \CrefAndHyperrefIfExist{definition:submodule_of_a_module_generated_by_elements}{generated by} all elements of the form $im$ with $i \in I$ and $m \in M$. Equivalently,
        $$IM = \left\{ \sum_{k=1}^n i_k m_k : n \ge 0, \; i_k \in I, \; m_k \in M \right\}.$$

        \item Let $I$ be a \CrefAndHyperrefIfExist{definition:ideal_of_a_ring}{right ideal of $R$} and let $M$ be a \CrefAndHyperrefIfExist{definition:module_of_a_ring}{right $R$-module}. The \hldef{product of $M$ and $I$}, denoted by \hl{$MI$}, is the submodule of $M$ generated by all elements of the form $mi$ with $m \in M$ and $i \in I$. Equivalently,
        $$MI = \left\{ \sum_{k=1}^n m_k i_k : n \ge 0, \; m_k \in M, \; i_k \in I \right\}.$$

        \item If $I$ is a \CrefAndHyperrefIfExist{definition:ideal_of_a_ring}{two-sided ideal of $R$}, then the product $IM$ (resp. $MI$) is defined for any left/right/two-sided $R$-module $M$ as above, and it is a submodule carrying the same structure as $M$.
    \end{enumerate}
\end{definition}
