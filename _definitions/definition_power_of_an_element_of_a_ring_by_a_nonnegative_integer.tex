\begin{definition} \label{definition:power_of_an_element_of_a_ring_by_a_nonnegative_integer}
    Let $R$ be a ring. Let $r \in R$, and let $n \in \bbZ_{\geq 0}$. The \hldef{$n$th power of $r$}, written \hl{$r^n$}, is the element of $R$ defined inducedtively as follows:
    \begin{align*}
        r^0 &= 1 \\
        r^{n+1} &= r \cdot r^n.
    \end{align*}
    \TextIfExists{definition:power_of_a_monoid_element_by_a_nonnegative_integer}{
        In other words, $r^n$ is the \CrefAndHyperrefIfExist{definition:power_of_a_monoid_element_by_a_nonnegative_integer}{$n$th power of $r$} as an element of the monoid of $R$ under multiplication.
    }
\end{definition}
