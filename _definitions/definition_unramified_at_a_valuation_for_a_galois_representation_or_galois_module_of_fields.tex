\begin{definition} \label{definition:unramified_at_a_valuation_for_a_galois_representation_or_galois_module_of_fields}
    Let $L/K$ be a \CrefAndHyperrefIfExist{definition:galois_extension_of_fields}{Galois extension of fields}. 
    Let 
    $$\rho: \operatorname{Gal}(L/K) \to \operatorname{Aut}(S)$$
    be a \CrefAndHyperrefIfExist{definition:galois_action_of_a_galois_group_of_a_galois_field_extension_on_a_set}{Galois action} on some set $S$ (e.g. $\rho$ could be a \CrefAndHyperrefIfExist{definition:galois_module_over_a_topological_ring_of_a_field}{Galois module}/\CrefAndHyperrefIfExist{definition:galois_representation_over_a_topological_ring_of_a_field}{Galois representation} over some ring.

    \begin{enumerate}
        \item In the case that $L/K$ is also an \CrefAndHyperrefIfExist{definition:extension_of_a_valuation_of_a_field_to_an_extension_field}{extension of valued fields}, we say that $\rho$ is \hldef{unramified} if the image of the \CrefAndHyperrefIfExist{definition:inertia_group_of_a_general_galois_extension_of_valuation_fields}{inertia subgroup} of $\operatorname{Gal}(L/K)$ is trivial. Otherwise, the representation/module is said to be \hldef{ramified}.

        \item For a valuation $v$ of $K$, we say that $\rho$ is \hldef{unramified at $v$} if any (equivalently every) choice of valuation $w$ extending $v$ makes $\rho$ unramified. Otherwise, we say that $\rho$ is \hldef{ramified at $v$}.
    \end{enumerate}
\end{definition}
