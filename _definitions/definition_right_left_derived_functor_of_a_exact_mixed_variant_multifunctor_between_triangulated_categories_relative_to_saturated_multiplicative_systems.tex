\begin{definition}[Mixed-Variance Derived Multifunctors, cf. {\cite[Expos\'e XVII, D\'efinition 1.2.1., 1.2.5.]{SGA4_III}}] \label{definition:right_left_derived_functor_of_a_exact_mixed_variant_multifunctor_between_triangulated_categories_relative_to_saturated_multiplicative_systems}
Let $(\mathscr{A}_i)_{1 \le i \le n}$ and $(\mathscr{B}_j)_{1 \le j \le m}$ be families of \CrefAndHyperrefIfExist{definition:triangulated_category}{triangulated categories}, and let $\mathscr{C}$ be a triangulated category. Let 
$$F: \prod_{i=1}^n \mathscr{A}_i \times \prod_{j=1}^m \mathscr{B}_j \to \mathscr{C}$$ 
be an \CrefAndHyperrefIfExist{definition:exact_mixed_variance_multifunctor_from_triangulated_categories_to_a_triangulated_category}{exact multifunctor}, covariant in $\mathscr{A}_i$ and contravariant in $\mathscr{B}_j$. Let $S_i, T_j, U$ be \CrefAndHyperrefIfExist{definition:saturated_multiplicative_system_in_a_category}{saturated multiplicative systems} in $\mathscr{A}_i, \mathscr{B}_j, \mathscr{C}$ respectively assumed \CrefAndHyperrefIfExist{definition:compatible_with_triangulation_for_a_multiplicative_system_of_a_triangulated_category}{compatible with the triangulated structure} of its respective category. Write 
\begin{align*}
Q_{A_i}&: \mathscr{A}_i \to \mathscr{A}_i(S_i^{-1}) \\ 
Q_{B_j}&: \mathscr{B}_j \to \mathscr{B}_j(T_j^{-1}) \\
Q_{C}&: \mathscr{C} \to \mathscr{C}(U^{-1})
\end{align*}
for the \CrefAndHyperrefIfExist{definition:localization_of_a_category_by_a_multiplicative_system}{localization functors}. 


\begin{enumerate}
    \item The \hldef{right derived functor $RF$ of $F$ relative to $(S_i, T_j)$ and $U$} is the functor 
    $$\hlin{RF: \prod_{i=1}^n \mathscr{A}_i(S_i^{-1}) \times \prod_{j=1}^m \mathscr{B}_j(T_j^{-1}) \to \text{Ind}(\mathscr{C}(U^{-1}))}$$
    which makes the following diagram commutative
    $$
    \begin{array}{ccc}
    \prod_{i=1}^n \mathscr{A}_i \times \prod_{j=1}^m \mathscr{B}_j & \xrightarrow{F} & \mathscr{C} \\
    \downarrow^{\prod Q_{A_i} \times \prod Q_{B_j}} & & \downarrow^{\overline{Q}_C} \\
    \prod_{i=1}^n \mathscr{A}_i(S_i^{-1}) \times \prod_{j=1}^m \mathscr{B}_j(T_j^{-1}) & \xrightarrow{RF} & \text{Ind}(\mathscr{C}(U^{-1}))
    \end{array}
    $$
    where $\overline{Q}_C$ is the composition 
    $$\mathscr{C} \xrightarrow{Q_C} \mathscr{C}(U^{-1}) \xrightarrow{\iota_{\text{Ind}}} \Ind(\mathscr{C}(U^{-1}))$$
    of $Q_C$ and the \CrefAndHyperref{definition:ind_pro_category_of_a_locally_small_category}{canonical embedding} $\iota_{\text{Ind}}$.


    \item The \hldef{left derived functor $LF$ of $F$ relative to $(S_i, T_j)$ and $U$} is the functor
    $$\hlin{LF: \prod_{i=1}^n \mathscr{A}_i(S_i^{-1}) \times \prod_{j=1}^m \mathscr{B}_j(T_j^{-1}) \to \text{Pro}(\mathscr{C}(U^{-1}))}$$
    which makes the following diagram commutative:
    $$
    \begin{array}{ccc}
    \prod_{i=1}^n \mathscr{A}_i \times \prod_{j=1}^m \mathscr{B}_j & \xrightarrow{F} & \mathscr{C} \\
    \downarrow^{\prod Q_{A_i} \times \prod Q_{B_j}} & & \downarrow^{\overline{Q}_C} \\
    \prod_{i=1}^n \mathscr{A}_i(S_i^{-1}) \times \prod_{j=1}^m \mathscr{B}_j(T_j^{-1}) & \xrightarrow{LF} & \text{Pro}(\mathscr{C}(U^{-1}))
    \end{array}
    $$
    where $\overline{Q}_C$ is the composition 
    $$\mathscr{C} \xrightarrow{Q_C} \mathscr{C}(U^{-1}) \xrightarrow{\iota_{\text{Pro}}} \text{Pro}(\mathscr{C}(U^{-1}))$$
    of $Q_C$ and the \CrefAndHyperref{definition:ind_pro_category_of_a_locally_small_category}{canonical embedding} $\iota_{\text{Pro}}$.
\end{enumerate}
\end{definition}
