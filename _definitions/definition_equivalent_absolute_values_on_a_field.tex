\begin{definition} \label{definition:equivalent_absolute_values_on_a_field}
Two \CrefAndHyperrefIfExist{definition:absolute_value_on_a_ring_valued_in_an_ordered_semiring}{absolute values} $|\cdot|_1$ and $|\cdot|_2$ on a field $F$ are \hldef{equivalent} if there exists a positive real number $c > 0$ such that
\[
|\cdot|_1 = |\cdot|_2^c.
\]
\end{definition}
