\begin{definition}[{\cite[Definition 1.1.1.9.]{lurie_ha}}] \label{definition:stable_infinity_category}
    An \CrefAndHyperrefIfExist{definition:infty_category_quasi_category}{$\infty$-category} $\calC$ is called \hldef{stable} if it satisfies the following conditions:
    \begin{enumerate}
        \item There exists a zero object of $\calC$.
        \item Every morphism in $\calC$ admits a fiber and a cofiber.
        \item A triangle in $\calC$ is a fiber sequence if and only if it a cofiber sequence.

    \end{enumerate}
\end{definition}
