\begin{definition} \label{definition:perfect_ring}
    A \CrefAndHyperrefIfExist{definition:ring}{ring} $R$ is
    \begin{enumerate}
        \item \hldef{left perfect} if every \CrefAndHyperrefIfExist{definition:flat_left_right_module_over_a_ring}{flat} \CrefAndHyperrefIfExist{definition:bimodule_over_rings_module_over_a_ring}{left $R$-module} is \CrefAndHyperrefIfExist{definition:projective_bimodule_over_rings}{projective}.
        \item \hldef{right perfect} if every flat \CrefAndHyperrefIfExist{definition:bimodule_over_rings_module_over_a_ring}{right $R$-module} is projective.
    \end{enumerate}
\end{definition}
