\begin{definition} \label{definition:almost_complex_structure_on_a_smooth_manifold_of_even_real_dimension_induces_complex_vector_space_structure_on_each_tangent_space}
    Given an \CrefAndHyperrefIfExist{definition:almost_complex_structure_on_a_smooth_manifold_of_even_real_dimension}{almost complex structure} $J$ on a \CrefAndHyperrefIfExist{definition:C_k_manifold}{smooth manifold} $M$ of real dimension $2n$, each tangent space $T_p M$ for $p \in M$ carries the structure of a complex vector space by defining scalar multiplication by 
    $$\hlin{(a+ib) \cdot v := av + bJv \quad \text{for } a,b \in \bbR, v \in T_p M.}$$
\end{definition}
