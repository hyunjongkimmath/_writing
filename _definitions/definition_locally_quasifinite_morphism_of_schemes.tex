\begin{definition}[Locally quasi-finite morphism of schemes] \label{definition:locally_quasifinite_morphism_of_schemes}
    \TODO{AI generated, verify}
    Let $f: X \to Y$ be a \CrefAndHyperrefIfExist{definition:morphism_of_schemes}{morphism of schemes}.

    The morphism $f$ is called \hldef{locally quasi-finite} if it is locally of finite type and has finite fibers, i.e., for every point $y \in Y$, the fiber $X_y = X \times_Y \Spec(\kappa(y))$ consists of a finite set of points.

    The morphism $f$ is called \hldef{quasi-finite} if it is locally quasi-finite and \CrefAndHyperrefIfExist{definition:quasi_compact_morphism_of_schemes}{quasi-compact}.
\end{definition}
