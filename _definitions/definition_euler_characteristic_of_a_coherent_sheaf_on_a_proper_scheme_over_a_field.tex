\begin{definition}[Euler characteristic of a coherent sheaf on a proper scheme over a field] \label{definition:euler_characteristic_of_a_coherent_sheaf_on_a_proper_scheme_over_a_field}
    \TODO{AI generated, verify}
    Let $k$ be a \CrefAndHyperrefIfExist{definition:field}{field}, and let $X$ be a \CrefAndHyperrefIfExist{definition:scheme_over_a_scheme}{scheme over} $k$ that is \CrefAndHyperrefIfExist{definition:proper_morphism_of_schemes}{proper}.

    Let $\mathcal{F}$ be a \CrefAndHyperrefIfExist{definition:coherent_sheaf_on_a_scheme}{coherent sheaf} on $X$. The \CrefAndHyperrefIfExist{definition:sheaf_cohomology_group_of_a_sheaf_of_modules_over_a_sheaf_of_rings_on_a_site}{sheaf cohomology groups} $H^i(X, \mathcal{F})$ are \CrefAndHyperrefIfExist{definition:basis_and_dimension_of_a_vector_space_over_a_field}{finite-dimensional} \CrefAndHyperrefIfExist{definition:vector_space_over_a_field}{$k$-vector spaces} and vanish for $i > \dim X$. The \hldef{Euler characteristic of $\mathcal{F}$}, denoted by $\hlin{\chi(\mathcal{F})}$ or $\hlin{\chi(X, \mathcal{F})}$, is defined as
    $$\chi(\mathcal{F}) := \sum_{i=0}^{\dim X} (-1)^i \dim_k H^i(X, \mathcal{F}).$$
\end{definition}
