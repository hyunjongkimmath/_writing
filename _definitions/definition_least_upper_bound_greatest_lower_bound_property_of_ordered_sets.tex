\begin{definition} \label{definition:least_upper_bound_greatest_lower_bound_property_of_ordered_sets}
    Let $(P, \leq )$ be an \CrefAndHyperrefIfExist{definition:partially_ordered_set}{ordered set}.
    \begin{enumerate}
        \item $P$ is said to have the \hldef{least upper bound property} or \hldef{supremum property} if every nonempty subset $S$ of $P$ with an \CrefAndHyperrefIfExist{definition:upper_and_lower_bounds_of_subsets_of_an_ordered_set}{upper bound} has a \CrefAndHyperrefIfExist{definition:supremum_infimum_of_a_subset_of_an_ordered_set}{supremum}.
        \item $P$ is said to have the \hldef{greatest lower bound property} or \hldef{infimum property} if every nonempty subset $S$ of $P$ with an \CrefAndHyperrefIfExist{definition:upper_and_lower_bounds_of_subsets_of_an_ordered_set}{lower bound} has an \CrefAndHyperrefIfExist{definition:supremum_infimum_of_a_subset_of_an_ordered_set}{infimum}.
    \end{enumerate}
\end{definition}
