\begin{definition} \label{definition:completion_of_an_extended_metric_space_valued_in_ordered_commutative_monoid_with_adjoined_infinity}
Let $T$ be an \CrefAndHyperrefIfExist{definition:ordered_commutative_monoid}{ordered commutative monoid}, let $T^\infty$ be the \CrefAndHyperrefIfExist{definition:ordered_monoid_with_adjoined_infinity}{ordered monoid with adjoined infinity} obtained from $T$, and let $(X, d)$ be an \CrefAndHyperrefIfExist{definition:extended_metric_on_a_set_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity}{extended metric space valued in $T^\infty$}. A \hldef{completion of $(X, d)$} is a pair $(\widehat{X}, \widehat{d})$ together with an \CrefAndHyperrefIfExist{definition:isometry_between_extended_metric_spaces_valued_in_ordered_commutative_monoid_with_adjoined_infinity}{isometric} embedding $i: X \to \widehat{X}$, where:
\begin{itemize}
    \item $(\widehat{X}, \widehat{d})$ is a \CrefAndHyperrefIfExist{definition:complete_extended_metric_space_valued_in_ordered_commutative_monoid_with_adjoined_infinity}{complete extended metric space valued in $T^\infty$},
    \item $i(X)$ is \CrefAndHyperrefIfExist{definition:dense_subset_of_a_topological_space}{dense in $\widehat{X}$}\CrefIfExists{definition:dense_subset_of_a_topological_space} (with respect to the \CrefAndHyperrefIfExist{definition:topology_induced_by_an_extended_metric_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity_and_open_ball_for_an_extended_metric}{topologies induced by} the extended metrics),
    \item $d(x,y) = \widehat{d}(i(x), i(y))$ for all $x, y \in X$.
\end{itemize}

\end{definition}
