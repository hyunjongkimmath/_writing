\begin{definition} \label{definition:sheaf_of_roots_of_unity_on_a_ringed_space}
    Let $(X, \calO_X)$ be a \CrefAndHyperrefIfExist{definition:ringed_space}{ringed space}.

    Given an integer $n \geq 1$, the \hldef{sheaf of $n$th roots of unity} is the sheaf \hl{$\mu_n = \mu_{n,(X,\calO_X)} : \mathbf{Open}(X)^{\op} \to \mathbf{Grp}$} given as follows:
    $$\mu_{n}(U) = \{s \in \calO_X^\times(U): s^n = 1\}.$$
\end{definition}
