\begin{definition}[Quotient topology] \label{definition:quotient_of_a_topological_space_with_respect_to_an_equivalence_relations}
    Let $X$ be a \CrefAndHyperrefIfExist{definition:topological_space}{topological space}, $Y$ be a set, and $q : X \to Y$ be a \CrefAndHyperrefIfExist{definition:surjective_function}{surjective function}.
    The \hldef{quotient topology on $Y$ induced by $q$} is the \CrefAndHyperrefIfExist{definition:coarser_smaller_finer_larger_for_topologies_on_a_set}{finest topology} on $Y$ such that $q$ is \CrefAndHyperrefIfExist{definition:continuous_map_of_topological_spaces}{continuous}. 
    
    Equivalently, a subset $V \subseteq Y$ is \CrefAndHyperrefIfExist{definition:open_set}{open} in the quotient topology if and only if its \CrefAndHyperrefIfExist{definition:preimage_of_a_subset_of_the_codomain_of_a_set_map}{preimage} $q^{-1}(V)$ is open in $X$. The pair $(Y, q)$ is then called a \hldef{quotient space} of $X$.
    
    Equivalently, if $\sim$ is an \CrefAndHyperrefIfExist{definition:equivalence_relation_on_a_set}{equivalence relation} on $X$, the quotient space \hl{$X/{\sim}$} is the set of \CrefAndHyperrefIfExist{definition:equivalence_class_of_an_element_of_a_set_under_an_equivalence_relation_and_quotient_of_set_by_equivalence_relation}{equivalence classes} endowed with the quotient topology induced by the canonical projection $\pi : X \to X/{\sim}$.
\end{definition}
