\begin{definition} \label{definition:automorphism_of_a_module_over_a_ring}
    Let $M$ be a \CrefAndHyperrefIfExist{definition:module_of_a_ring}{module} over a \CrefAndHyperrefIfExist{definition:ring}{ring} $R$. 
    The \hldef{automorphism group of $M$ over $R$} is the group \hl{$\operatorname{Aut}_R(M)$} of \hldef{automorphisms on $M$}, i.e. \CrefAndHyperrefIfExist{definition:homomorphism_of_modules_over_a_ring}{$R$-linear homomorphisms} $\varphi: M \to M$ that are also \CrefAndHyperrefIfExist{definition:injective_surjective_bijective_map_of_sets}{bijective functions}. Equivalently, an $R$-linear homomorphism $\varphi: M \to M$ is an automorphism on $M$ if there exists a (necessarily unique) $R$-linear homomorphism $\varphi^{-1}$ such that $\varphi \circ \varphi^{-1}, \varphi^{-1} \circ \varphi = \id_M$. The group operation is \CrefAndHyperrefIfExist{definition:composition_of_functions_between_sets}{composition of functions}, the identity element is the \CrefAndHyperrefIfExist{definition:identity_function_on_a_set}{identity function} on $M$, and the \CrefAndHyperrefIfExist{definition:group}{inverse operation on the group} is \CrefAndHyperrefIfExist{definition:injective_surjective_bijective_map_of_sets}{function inverse}.

    \TextIfExists{definition:automorphism_of_an_object_of_a_category_automorphism_group_of_an_object_of_a_locally_small_category}{
        In other words, $\operatorname{Aut}_R(M)$ is the \CrefAndHyperrefIfExist{definition:automorphism_of_an_object_of_a_category_automorphism_group_of_an_object_of_a_locally_small_category}{automorphism group} of the object $M$ in the \CrefAndHyperrefIfExist{definition:category_of_modules_and_bimodules_over_rings}{category of $R$-modules}.
    }
\end{definition}
