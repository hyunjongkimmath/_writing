\begin{definition} \label{definition:compatible_with_a_morphism_between_spectral_sequences_in_an_abelian_category_for_filtered_objects_that_the_spectral_sequences_weakly_converge_to}
   Let $\calA$ be an \CrefAndHyperrefIfExist{definition:abelian_category}{abelian category}. 
   \begin{enumerate}
    \item Let $\{E_{pq}^r\}$ and $\{E_{pq}^{'r}\}$ be \CrefAndHyperrefIfExist{definition:spectral_sequence_in_an_abelian_category}{homological spectral sequences} \CrefAndHyperrefIfExist{definition:weak_convergence_to_a_graded_object_for_a_spectral_sequence_in_an_abelian_category}{weakly converging to} $(H_*, F)$ and $(H_*', F')$, respectively. Let $f: E \to E'$ be a \CrefAndHyperrefIfExist{definition:morphism_of_spectral_sequences_in_an_abelian_category}{morphism of spectral sequences}. Let $h: H_* \to H_*'$ be a \CrefAndHyperrefIfExist{definition:morphism_of_filtered_objects_in_a_category}{morphism of filtered objects}.
    
    We say that a map $h: H_* \to H_*'$ of filtrations is \hldef{compatible with a morphism $f: E \to E'$} if $h$ maps $F_p H_n$ to $F_p H_n'$ and the associated maps $F_p H_n / F_{p-1} H_n \to F_p H_n' / F_{p-1} H_n'$ correspond under the isomorphisms $E_{pq}^\infty \cong F_p H_{p+q} / F_{p-1} H_{p+q}$  and $E_{pq}^{' \infty} \cong F_p H_{p+q}' / F_{p-1} H_{p+q}'$ to $f_{pq}^\infty: E_{pq}^\infty \to E_{pq}^{'\infty}$.


    \item Let $\{E_r^{pq}\}$ and $\{E_r'^{pq}\}$ be \CrefAndHyperrefIfExist{definition:spectral_sequence_in_an_abelian_category}{cohomological spectral sequences} \CrefAndHyperrefIfExist{definition:weak_convergence_to_a_graded_object_for_a_spectral_sequence_in_an_abelian_category}{weakly converging to} $(H^*, F)$ and $(H'^*, F')$, respectively. Let $f: E \to E'$ be a \CrefAndHyperrefIfExist{definition:morphism_of_spectral_sequences_in_an_abelian_category}{morphism of spectral sequences}. Let $h: H^* \to H'^*$ be a \CrefAndHyperrefIfExist{definition:morphism_of_filtered_objects_in_a_category}{morphism of filtered objects}.

    We say that a map $h: H^* \to H'^*$ of filtrations is \hldef{compatible with a morphism $f: E \to E'$} if $h$ maps $F^p H^n$ to $F^p H'^n$ and the associated maps $F^p H^n / F^{p+1} H^n \to F^p H'^n / F^{p+1} H'^n$ correspond under the isomorphisms $E_\infty^{pq} \cong F^p H^{p+q} / F^{p+1} H^{p+q}$ and $E_\infty'^{pq} \cong F^p H'^{p+q} / F^{p+1} H'^{p+q}$ to $f_\infty^{pq}: E_\infty^{pq} \to E_\infty'^{pq}$.
   \end{enumerate}
\end{definition}
