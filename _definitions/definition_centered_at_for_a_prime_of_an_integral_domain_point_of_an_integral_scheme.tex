\begin{definition} \label{definition:centered_at_for_a_prime_of_an_integral_domain_point_of_an_integral_scheme}
    \begin{enumerate}
        \item Let $R$ be an \CrefAndHyperrefIfExist{definition:domain_integral_domain}{integral domain}. Let $\mathfrak{p}$ be a \CrefAndHyperrefIfExist{definition:prime_and_maximal_ideal_of_a_ring}{prime ideal of $R$}. Let $v$ be a \CrefAndHyperrefIfExist{definition:archimedean_absolute_value_on_a_field}{non-archimedean valuation} on the \CrefAndHyperrefIfExist{definition:field_of_fractions_of_an_integral_domain}{fraction field} of $R$.

        We say that $v$ is \hldef{centered at $\mathfrak{p}$} if the \CrefAndHyperrefIfExist{definition:valuation_ring_of_a_nonarchimedean_valued_field}{valuation ring} $\mathcal{O}_v$ \CrefAndHyperrefIfExist{definition:dominates_a_local_ring_for_a_local_ring_in_a_field}{dominates} $R_\mathfrak{p}$\CrefIfExists{definition:localization_of_a_commutative_ring_by_a_multiplicative_subset}. 

        \item Let $X$ be an \CrefAndHyperrefIfExist{definition:integral_scheme}{integral scheme}. Let $x \in X$ be a point. Let $v$ be a \CrefAndHyperrefIfExist{definition:valuation_on_a_field}{valuation} on the \CrefAndHyperrefIfExist{definition:function_field_over_a_field}{function field} of $X$. 
        We say that $v$ is \hldef{centered at $x$} if the \CrefAndHyperrefIfExist{definition:valuation_ring_of_a_nonarchimedean_valued_field}{valuation ring} $\mathcal{O}_v$ \CrefAndHyperrefIfExist{definition:dominates_a_local_ring_for_a_local_ring_in_a_field}{dominates} \CrefAndHyperrefIfExist{definition:stalk_of_a_presheaf_on_a_topological_space_at_a_point}{$\mathcal{O}_{X,x}$}. 
    \end{enumerate}
    Note that for an integral domain $R$, prime ideal $\mathfrak{p}$, and a valuation $v$ of $\operatorname{Frac}(R)$, the valuation $v$ is centered at $\mathfrak{p}$ if and only if it is centered at the point of $\operatorname{Spec} R$ \CrefIfExists{definition:affine_scheme} corresponding to $\mathfrak{p}$.
\end{definition}
