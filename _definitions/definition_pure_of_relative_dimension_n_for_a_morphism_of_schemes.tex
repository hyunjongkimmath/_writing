\begin{definition}[pure of relative dimension $n$ for a morphism of schemes] \label{definition:pure_of_relative_dimension_n_for_a_morphism_of_schemes}
    Let $f\colon X \to Y$ be a \CrefAndHyperrefIfExist{definition:morphism_of_schemes}{morphism of schemes} that is \CrefAndHyperrefIfExist{definition:locally_of_finite_type_for_a_morphism_of_schemes}{locally of finite type}, and let $n \geq 0$ be an integer. We say that $f$ is \hldef{pure of relative dimension $n$} if for every $y \in Y$, the fiber $X_y = X \times_Y \Spec(\kappa(y))$ is a \CrefAndHyperrefIfExist{definition:pure_dimensional_scheme}{pure-dimensional scheme} of dimension $n$, i.e., every \CrefAndHyperrefIfExist{definition:irreducible_component_of_a_topological_space}{irreducible component} of $X_y$ has dimension $n$.
\end{definition}
