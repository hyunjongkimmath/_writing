\begin{definition}[see {\cite[7.3.7]{knus_qhfr}}] \label{definition:augmentation_trace_on_a_group_ring_over_an_involution_ring}

    Let $(R, \sigma_R)$ be a \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{ring with involution}. Let $G$ be a \CrefAndHyperrefIfExist{definition:group}{group}. 

    The \hldef{augmentation trace} or \hldef{standard trace} \hl{$\tau: R[G] \to R$} is defined as
    $$\tau\left(\sum_{g \in G} a_g g \right) = a_1$$
    where $1$ is the \CrefAndHyperrefIfExist{definition:group}{identity element of $G$}. 
    
    Assuming that $R$ is commutative and that $G$ is finite, this is an \CrefAndHyperrefIfExist{definition:involution_trace_map_for_a_morphism_of_involution_rings}{involution trace} with respect to the \CrefAndHyperrefIfExist{definition:canonical_involution_on_a_group_ring_over_a_ring_with_involution}{canonical involution} of $R[G]$
    and is also a \CrefAndHyperrefIfExist{definition:symmetric_algebra_over_a_rnig}{symmetric trace}, and hence $R[G]$ is a \CrefAndHyperrefIfExist{definition:symmetric_algebra_over_a_ring_with_involution}{symmetric algebra with involution} over $R$.
\end{definition}
