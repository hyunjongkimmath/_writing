\begin{definition}[{\cite[Before Definition 2.4.]{scholze_lag}}] \label{definition:solidification_of_polynomial_ring_of_profinite_set}
    Let $S = \varprojlim_i S_i$ \CrefIfExists{definition:projective_and_inductive_limits_in_categories} be a \CrefAndHyperrefIfExist{definition:profinite_set}{profinite set} with $S_i$ \CrefAndHyperrefIfExist{definition:countable_finite_uncountable_sets}{finite} \CrefAndHyperrefIfExist{definition:discrete_topological_space_of_a_set}{discrete sets}. 

    Define 
    $$\hlin{\mathbb{Z}[S]^\blacksquare \coloneq \varprojlim \mathbb{Z}[S_i]}$$
    
    where $\bbZ[S_i]$ is the free ring generated by $S_i$ and the morphisms among these is the natural morphism between the free rings induced by the morphisms of finite sets.
