\begin{definition} \label{definition:connected_component_of_a_point_of_a_topological_space}
Let $X$ be a \CrefAndHyperrefIfExist{definition:category_of_topological_spaces}{topological space} and $x$ a point in $X$. The \hldef{connected component of $x$} is the union of all \CrefAndHyperrefIfExist{definition:disconnected_and_connected_topological_spaces}{connected} \CrefAndHyperrefIfExist{definition:subspace_of_a_topological_space}{subspaces} of $X$ that contain $x$.
\end{definition}
