\begin{definition} \label{definition:subgroup_of_profinite_group_topologically_generated_by_elements}
    Let $G$ be a profinite group. Let $S \subseteq G$ be a subset.
    The \hldef{subgroup of $G$ topologically/profinitely generated by $S$} refers to the \CrefAndHyperrefIfExist{definition:closure_of_a_subspace_of_a_topological_space}{closure in $G$} of the \CrefAndHyperrefIfExist{definition:subgroup_of_a_group_generated_by_elements}{subgroup algebraically generated by $S$}. 
\end{definition}
