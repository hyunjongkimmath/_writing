\begin{definition}[Cumulative distribution function] \label{definition:cumulative_distribution_function_of_a_real_valued_random_variable_on_a_probability_measure_space}
Let $(\Omega, \mathcal{F}, P)$ be a \CrefAndHyperref{definition:probability_measure_space_for_a_measure_on_a_measurable_space}{probability measure space}. Let $X: \Omega \to (\mathbb{R}, \mathcal{B}(\mathbb{R}))$ be a \CrefAndHyperrefIfExist{definition:random_variable_on_a_probability_measure_space}{random variable}, where $\mathcal{B}(\mathbb{R})$ is the Borel $\sigma$-algebra on $\mathbb{R}$. Let $\mu_X$ be the \CrefAndHyperrefIfExist{definition:distribution_of_a_random_variable_on_a_probability_measure_space}{distribution of $X$}. The \hldef{cumulative distribution function of $X$}, denoted by \hl{$F_X$}, is the function $F_X: \mathbb{R} \to [0, 1]$ defined by
$$F_X(t) = \mu_X((-\infty, t]) = P(X^{-1}((-\infty, t])).$$
\end{definition}
