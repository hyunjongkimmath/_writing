\begin{definition} \label{definition:galois_action_of_a_galois_group_of_a_galois_field_extension_on_a_set}
    Let $K$ be a \CrefAndHyperrefIfExist{definition:field}{field}. Let $X$ be a \CrefAndHyperrefIfExist{definition:topological_space}{topological space}.

    \begin{enumerate}
        \item Let $L/K$ be a \CrefAndHyperrefIfExist{definition:galois_extension_of_fields}{Galois field extension}. A \hldef{Galois action of $\Gal(L/K)$ on $X$} refers to a \CrefAndHyperrefIfExist{definition:continuous_action_on_a_topological_space_by_a_topological_group}{continuous action} of $\Gal(L/K)$ on $X$ where $\Gal(L/K)$ is equipped with its \CrefAndHyperrefIfExist{definition:profinite_group}{profinite topology}.

        \item Let $L/K$ be a \CrefAndHyperrefIfExist{definition:galois_extension_of_fields}{Galois field extension}. A \hldef{Galois action of $K$ on $X$} refers to a Galois action of $\Gal(\overline{K} / K)$\CrefIfExists{definition:absolute_galois_group_of_a_field} on $X$.
    \end{enumerate}
    When $X$ is merely a set (not equipped with a topology), one speaks of a Galois action by equipping $X$ with the \CrefAndHyperrefIfExist{definition:discrete_topological_space_of_a_set}{discrete topology}.
\end{definition}
