\begin{definition}[Dualizing complex for Verdier duality on a noetherian scheme with rational coefficients] \label{definition:dualizing_complex_for_verdier_duality_in_the_bdd_constr_derived_category_of_adic_sheaves_with_rational_coefficients_on_a_finite_type_scheme_over_a_noetherian_scheme}
    Let $(R, \lambda)$ be a \CrefAndHyperrefIfExist{definition:complete_ring_with_respect_to_an_ideal}{complete} \CrefAndHyperrefIfExist{definition:discrete_valuation_ring}{DVR} such that $R/\lambda^n$ is finite for all $n \geq 1$. Let $E$ be the \CrefAndHyperrefIfExist{definition:field_of_fractions_of_an_integral_domain}{fraction field} of $R$. Let $S$ be a \CrefAndHyperrefIfExist{definition:locally_noetherian_and_noetherian_scheme}{noetherian scheme}. Consider sheaves on the (small) \'etale site of $S$.

    \begin{enumerate}
        \item The \hldef{dualizing complex for Verdier duality on $S$ with rational coefficients}, denoted by \hl{$\mathbb{D}_S^\bullet(E)$}, is the object of \CrefAndHyperrefIfExist{definition:derived_category_of_cohomologically_bounded_constructible_sheaves_with_rational_adic_coefficients_on_a_noetherian_scheme}{$\Dbc(S,E)$} given as the \CrefAndHyperrefIfExist{definition:dualizing_complex_for_verdier_duality_in_the_bdd_constr_derived_category_of_adic_sheaves_with_integral_coefficients_on_a_finite_type_scheme_over_a_noetherian_scheme}{dualizing complex} $\mathbb{D}_S^\bullet(R)$ in \CrefAndHyperrefIfExist{definition:derived_category_of_bounded_constructible_complexes_of_adic_sheaves_with_integral_coefficients_on_a_noetherian_scheme}{$\Dbc(S,R)$}.
        
        % is the constant inverse system $(E)_n$ regarded concentrated in degree $0$, where each $E \cong R[\lambda^{-1}]$.

        \item The \hldef{(relative) dualizing complex for Verdier duality on $X/S$ with rational coefficients}, denoted by \hl{$\mathbb{D}_{X/S}^\bullet(E)$}, is the object of $\Dbc(X,E)$ defined by
        $$\mathbb{D}_{X/S}^\bullet(E) = Rf^! \, \mathbb{D}_S^\bullet(E).$$
        \CrefIfExists{definition:exceptional_inverse_image_functor_on_bdd_const_derived_category_of_adic_sheaves_with_rational_coefficients_on_schemes_of_finite_type_over_a_a_noetherian_scheme_via_a_compactifiable_morphism}
        Equivalently, as an object of $\Dbc(X,R)$, it is given as the \CrefAndHyperrefIfExist{definition:dualizing_complex_for_verdier_duality_in_the_bdd_constr_derived_category_of_adic_sheaves_with_integral_coefficients_on_a_finite_type_scheme_over_a_noetherian_scheme}{dualizing complex} $\mathbb{D}_{X/S}^\bullet(R)$
    \end{enumerate}
\end{definition}
