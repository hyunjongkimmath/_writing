\begin{definition} \label{definition:splitting_type_of_a_prime_in_extension_of_dedekind_domains}

    Assume the setting of \Cref{definition:prime_decomposition_in_extension_of_dedekind_domains}. Let $\mathfrak{p}$ be a prime of $R$ with factorization $\mathfrak{p}\calO_L = \prod_{i=1}^{g} \mathfrak{P}_i^{e_i}$ and inertial degrees $f_i = f(\mathfrak{P}_i/\mathfrak{p})$.

    \begin{itemize}
        \item $\mathfrak{p}$ is \hldef{unramified in $L$} if $e_i = 1$ for all $i$ and each residue field extension $\kappa(\mathfrak{P}_i)/\kappa(\mathfrak{p})$ is \CrefAndHyperref{definition:separable_field_extension_general}{separable}. Otherwise $\mathfrak{p}$ is \hldef{ramified in $L$}.
        \item $\mathfrak{p}$ \hldef{splits completely in $L$} if $g = [L:K]$ (equivalently $e_i = f_i = 1$ for all $i$).
        \item $\mathfrak{p}$ is \hldef{inert in $L$} if $g = 1$ and $e_1 = 1$, so that $\mathfrak{p}\calO_L = \mathfrak{P}$ is itself a prime of $\calO_L$ and $f_1 = [L:K]$.
        \item $\mathfrak{p}$ is \hldef{totally ramified in $L$} if $g = 1$ and $f_1 = 1$, so that $\mathfrak{p}\calO_L = \mathfrak{P}^{[L:K]}$.
    \end{itemize}
\end{definition}
