\begin{definition} \label{definition:stone_space}
A \CrefAndHyperrefIfExist{definition:category_of_topological_spaces}{topological space} $X$ is a \hldef{Stone space} if it is \CrefAndHyperrefIfExist{definition:compact_topological_space}{compact}, \CrefAndHyperrefIfExist{definition:separation_axioms_of_topology}{Hausdorff}, and \CrefAndHyperrefIfExist{definition:totally_disconnected_topological_space}{totally disconnected}.
\end{definition}
