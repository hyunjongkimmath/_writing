\begin{definition}[Convergence of a sequence in a topological space] \label{definition:convergence_and_limit_of_sequence_in_a_topological_space}
    Let $(X, \mathcal{T})$ be a \CrefAndHyperrefIfExist{definition:topological_space}{topological space}. Let $(x_n)_{n=1}^\infty$ be a sequence in $X$, and let $x \in X$.

    The sequence $(x_n)$ \hldef{converges to $x$} if for every \CrefAndHyperrefIfExist{definition:neighborhood_and_neighborhood_basis_of_a_point_in_a_topological_space}{open neighborhood} $U$ of $x$, there exists an integer $N \in \mathbb{N}$ such that for all $n \geq N$, $x_n \in U$. Otherwise, we say that $(x_n)$ \hldef{diverges}.

    In that case, $x$ is called the \hldef{limit of the sequence $(x_n)$}, and we write
    $$\hlin{\lim_{n \to \infty} x_n = x \quad \text{or} \quad x_n \to x.}$$

    \TextIfExists{definition:infinite_series_and_convergence_of_infinite_series_in_a_topological_abelian_semigroup}{
        A notable application is the \CrefAndHyperrefIfExist{definition:infinite_series_and_convergence_of_infinite_series_in_a_topological_abelian_semigroup}{convergence of an infinite series}, which is defined as the topological convergence of its sequence of partial sums.
    }
\end{definition}
