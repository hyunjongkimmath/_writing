\begin{definition}[Real and imaginary parts of a complex number] \label{definition:real_and_imaginary_parts_of_a_complex_number}
    Let $\bbC$ denote the \CrefAndHyperrefIfExist{definition:field_of_complex_numbers}{field of complex numbers}. Every $z \in \bbC$ can be written uniquely as $z = x + iy$ with $x, y \in \bbR$, where $i^2 = -1$. The \hldef{real part of $z$}, denoted \hl{$\operatorname{Re}(z)$}, is the unique real number $x$, and the \hldef{imaginary part of $z$}, denoted \hl{$\operatorname{Im}(z)$}, is the unique real number $y$.
\end{definition}
