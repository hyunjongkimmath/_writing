\begin{definition} \label{definition:equivalent_places_of_a_field}
    Let $K$ be a \CrefAndHyperrefIfExist{definition:field}{field}. Two places $P_1: K \to L_1 \cup \{\infty\}$ and $P_2: K \to L_2 \cup \{\infty\}$ are said to be equivalent if there exists a field isomorphism $\sigma: \operatorname{im}(P_1) \cap L_1  \to \operatorname{im}(P_2) \cap L_2$ such that $P_2 = \sigma \circ P_1$, where $\sigma$ is extended to $\operatorname{im}(P_1) \cup \{\infty\}$ by $\sigma(\infty) = \infty$.
\end{definition}
