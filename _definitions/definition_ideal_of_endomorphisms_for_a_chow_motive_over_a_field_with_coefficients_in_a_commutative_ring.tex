\begin{definition} \label{definition:ideal_of_endomorphisms_for_a_chow_motive_over_a_field_with_coefficients_in_a_commutative_ring}
Let $k$ be a \CrefAndHyperrefIfExist{definition:field}{field}, $R$ a \CrefAndHyperrefIfExist{definition:ring}{commutative ring}, and $\mathfrak{Chow}(k, R)$ the \CrefAndHyperrefIfExist{definition:category_of_relative_chow_motives_over_a_noetherian_base_and_coefficients_in_a_commutative_ring}{category of Chow motives over $k$ with coefficients in $R$}. Let $M$ be a Chow motive over $k$.

For any field extension $E/k$, the kernel of the restriction map 
$$\mathrm{res}_{E/k}: \mathrm{End}_k(M) \to \mathrm{End}_E(M_E)$$ 
is denoted by \hl{$\mathcal{I}_{E/k}(M)$}. We say $\mathcal{I}_{E/k}(M)$ is the \hldef{ideal of endomorphisms that vanish over $E$}.
\end{definition}
