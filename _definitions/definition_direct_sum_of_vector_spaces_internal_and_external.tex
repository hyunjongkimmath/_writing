\begin{definition} \label{definition:direct_sum_of_vector_spaces_internal_and_external}
    Let $F$ be a \CrefAndHyperrefIfExist{definition:field}{field}, and let $V$ be an \CrefAndHyperrefIfExist{definition:vector_space_over_a_field}{$F$-vector space}.

    \textbf{(Internal direct sum.)} Let $W_1, \dots, W_n$ be subspaces of $V$. Their sum $W_1 + \cdots + W_n = \{ w_1 + \cdots + w_n : w_i \in W_i \}$ is called the \hldef{internal direct sum} of $W_1, \dots, W_n$ if every $v \in W_1 + \cdots + W_n$ has a unique expression $v = w_1 + \cdots + w_n$ with $w_i \in W_i$. In this case we write
    $$\hlin{W_1 \oplus \cdots \oplus W_n}.$$

    \textbf{(External direct sum.)} Let $V_1, \dots, V_n$ be $F$-vector spaces. Their \hldef{external direct sum} is the Cartesian product $V_1 \times \cdots \times V_n$ equipped with component-wise addition and scalar multiplication, denoted
    $$\hlin{V_1 \oplus \cdots \oplus V_n}.$$

\TextIfExists{definition:product_of_modules_of_rings}{The external direct sum of vector spaces is the special case of \Cref{definition:product_of_modules_of_rings} for $F$-modules. For finite families, it coincides with \CrefAndHyperrefIfExist{definition:coproduct_of_modules_of_rings}{the coproduct of modules}.}

\end{definition}
