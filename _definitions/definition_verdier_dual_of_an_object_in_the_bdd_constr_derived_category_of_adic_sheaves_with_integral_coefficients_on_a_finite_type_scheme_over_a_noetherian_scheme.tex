\begin{definition}[Verdier dual of a constructible complex with integral coefficients] \label{definition:verdier_dual_of_an_object_in_the_bdd_constr_derived_category_of_adic_sheaves_with_integral_coefficients_on_a_finite_type_scheme_over_a_noetherian_scheme}
    Let $(R, \lambda)$ be a \CrefAndHyperrefIfExist{definition:complete_orthogonal_idempotents_of_a_ring}{complete} \CrefAndHyperrefIfExist{definition:discrete_valuation_ring}{DVR} such that $R/\lambda^n$ is finite for all $n \geq 1$. Let $S$ be a \CrefAndHyperrefIfExist{definition:locally_noetherian_and_noetherian_scheme}{noetherian scheme}. Let $X$ be an $S$-scheme of \CrefAndHyperrefIfExist{definition:finite_type_morphism_of_schemes}{finite type}.  Consider sheaves on the \CrefAndHyperrefIfExist{definition:small_etale_site_of_a_scheme}{(small) \'etale sites} of $X$ and $S$.

    For any complex $M \in \Dbc(X, R)$\CrefIfExists{definition:derived_category_of_bounded_constructible_complexes_of_adic_sheaves_with_integral_coefficients_on_a_noetherian_scheme}, the \hldef{Verdier dual of $M$ on $X$ relative to $S$ with integral coefficients} is the object of $\Dbc(X,R)$ defined by
    $$\hlin{\mathbb{D}_X(M) := R\mathscr{H}om_R(M, \mathbb{D}_{X/S}^\bullet(R))}$$
    \CrefIfExists{definition:derived_internal_hom_on_bdd_constructible_complexes_with_integral_coefficients_on_a_noetherian_scheme} \CrefIfExists{definition:dualizing_complex_on_a_finite_type_scheme_relative_to_a_noetherian_base_with_integral_coefficients}
\end{definition}
