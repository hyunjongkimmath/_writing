\begin{definition} \label{definition:category_of_filtered_objects_in_a_category}
Let $\mathcal{C}$ be a category. 
\begin{enumerate}
    \item The \hldef{category of increasing filtered objects in $\calC$} is the category \hl{$\text{Filt}^{\text{inc}}(\mathcal{C})$} whose objects are pairs $(M, F)$ consisting of an object $M \in \mathcal{C}$ and an \CrefAndHyperrefIfExist{definition:filtration_on_an_object_of_a_category}{increasing filtration} $F$, and whose morphisms are \CrefAndHyperrefIfExist{definition:morphism_of_filtered_objects_in_a_category}{morphisms of filtered objects}.

    \item The \hldef{category of decreasing filtered objects in $\calC$} is the category \hl{$\text{Filt}^{\text{dec}}(\mathcal{C})$} whose objects are pairs $(M, F)$ consisting of an object $M \in \mathcal{C}$ and an \CrefAndHyperrefIfExist{definition:filtration_on_an_object_of_a_category}{increasing filtration} $F$, and whose morphisms are \CrefAndHyperrefIfExist{definition:morphism_of_filtered_objects_in_a_category}{morphisms of filtered objects}.
\end{enumerate}
When no ambiguity is present, one may simply write \hl{$\text{Filt}(\mathcal{C})$} for either category.
\end{definition}
