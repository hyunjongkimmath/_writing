\begin{definition} \label{definition:ramification_index_of_extension_of_valuation_fields}
    Assume \Cref{context:extension_of_valuation_fields}. The \hldef{ramification index of $w$ over $v$}, denoted \hl{$e(w/v)$}, is the index of the value group $\Gamma_v$ in $\Gamma_w$:
    $$e(w/v) = (\Gamma_w: \Gamma_v)$$
\end{definition}
