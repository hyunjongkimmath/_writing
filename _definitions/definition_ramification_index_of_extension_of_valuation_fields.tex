\begin{definition} \label{definition:ramification_index_of_extension_of_valuation_fields}
    Assume \Cref{context:extension_of_valuation_fields}. The \hldef{ramification index of $w$ over $v$}, denoted \hl{$e(w/v)$}, is the \CrefAndHyperrefIfExist{definition:index_of_a_subgroup_in_a_group}{index} of the value group $\Gamma_v$ in $\Gamma_w$:
    $$e(w/v) = [\Gamma_w: \Gamma_v]$$
\end{definition}
