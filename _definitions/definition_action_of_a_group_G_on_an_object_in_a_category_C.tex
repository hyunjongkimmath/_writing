\begin{definition}[action of a group on an object in a category]
\label{definition:action_of_a_group_G_on_an_object_in_a_category_C}
Let $G$ be a \CrefAndHyperrefIfExist{definition:group}{group}, let $\mathcal{C}$ be a \CrefAndHyperrefIfExist{definition:category}{category}, and let $c$ be an object of $\mathcal{C}$. An \hldef{action of $G$ on $c$ in $\mathcal{C}$} is a family $(\alpha(g))_{g \in G}$ where, for each $g \in G$, the morphism $\alpha(g): c \to c$ is an \CrefAndHyperrefIfExist{definition:automorphism_of_an_object_of_a_category_automorphism_group_of_an_object_of_a_locally_small_category}{automorphism of $c$} such that
$$\alpha(e) = \operatorname{id}_c \quad \text{and} \quad \alpha(gh) = \alpha(g) \circ \alpha(h) \quad \text{for all } g, h \in G,$$
where $e$ denotes the identity element of $G$. If $\mathcal{C}$ is a \CrefAndHyperrefIfExist{definition:locally_small_category}{locally small category}, an action of $G$ on $c$ is equivalently given by a \CrefAndHyperrefIfExist{definition:group_homomorphism}{group homomorphism} $\alpha: G \to \operatorname{Aut}_{\mathcal{C}}(c)$ to the \CrefAndHyperrefIfExist{definition:automorphism_of_an_object_of_a_category_automorphism_group_of_an_object_of_a_locally_small_category}{automorphism group of $c$}.
\end{definition}
