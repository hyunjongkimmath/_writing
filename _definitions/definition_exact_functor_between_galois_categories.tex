\begin{definition}[{\cite[Expos\'e V, Proposition 6.1.]{SGA1}}] \label{definition:exact_functor_between_galois_categories}
    Let $\calC$ and $\calC'$ be \CrefAndHyperrefIfExist{definition:pre_galois_category}{pre-Galois categories}, and let $F'$ be a \CrefAndHyperrefIfExist{definition:fiber_functor_for_a_functor_from_a_galois_category_to_the_category_of_finite_sets}{fiber functor} of $\calC'$. Let $H: \calC \to \calC'$ be a covariant functor. Let $F = F' \circ H$. We say that $H$ is an \hldef{exact functor $(\calC, F) \to (\calC', F')$} if the following equivalent conditions hold:
    \begin{enumerate}
        \item $H$ is \CrefAndHyperrefIfExist{definition:left_right_exact_functor_between_categories}{exact} as a functor.

        \item $H$ is \CrefAndHyperrefIfExist{definition:left_right_exact_functor_between_categories}{left exact}, transforms finite \CrefAndHyperrefIfExist{definition:product_and_coproduct_of_objects_in_a_category}{coproducts} to finite coproducts, and \CrefAndHyperrefIfExist{definition:monomorphism_and_epimorphism_in_categories}{epimorphisms} into epimorphisms.

        \item $F$ is a fiber functor of $\calC$.
    \end{enumerate}
\end{definition}
