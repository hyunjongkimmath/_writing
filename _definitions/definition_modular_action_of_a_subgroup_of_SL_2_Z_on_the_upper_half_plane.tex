\begin{definition}[Modular group action] \label{definition:modular_action_of_a_subgroup_of_SL_2_Z_on_the_upper_half_plane}
    Let $\Gamma \subseteq \mathrm{SL}_2(\mathbb{Z})$ be a \CrefAndHyperrefIfExist{definition:subgroup_of_a_group}{subgroup}. 
    For $\gamma = \begin{pmatrix} a & b \\ c & d \end{pmatrix}\in \Gamma$ and $\tau \in \mathbb{H} := \{ z \in \mathbb{C} \mid \operatorname{Im}(z) > 0 \}$, define the action
    $$\hlin{\gamma \cdot \tau := \frac{a\tau + b}{c\tau + d}.}$$
    This is called the \hldef{modular action of $\Gamma$ on the upper half-plane $\mathbb{H}$}.
\end{definition}
