\begin{definition} \label{definition:small_site_locally_small_site_U_site_U_small_site_U_topology}
    \begin{enumerate}
        \item An \hldef{small site} is a \CrefAndHyperrefIfExist{definition:grothendieck_topology_on_a_category_site_covering_sieve_topologically_generating_family}{site} whose underlying category is \CrefAndHyperrefIfExist{definition:locally_small_category}{small}.

        \TextIfExists{definition:essentially_small_site}{cf. the notion of an \CrefAndHyperrefIfExist{definition:essentially_small_site}{essentially small site}.}

        \item A \hldef{locally small site} is a site whose underlying category is \CrefAndHyperrefIfExist{definition:locally_small_category}{locally small}.

        \item Given a \CrefAndHyperrefIfExist{definition:grothendieck_universe}{Grothendieck universe} $\mathscr{U}$, a \hldef{$\mathscr{U}$-site} is a \CrefAndHyperrefIfExist{definition:grothendieck_topology_on_a_category_site_covering_sieve_topologically_generating_family}{site} whose underlying \CrefAndHyperrefIfExist{definition:category}{category} is a \CrefAndHyperrefIfExist{definition:locally_small_category}{$\mathscr{U}$-(locally small) category} and that has a \CrefAndHyperrefIfExist{definition:grothendieck_topology_on_a_category_site_covering_sieve_topologically_generating_family}{$\mathscr{U}$-small topologically generating family}; a Grothendieck topology making the site a $\mathscr{U}$-site is called a \hldef{$\mathscr{U}$-topology}.

        A $\mathscr{U}$-site is called \hldef{$\mathscr{U}$-small} if its underlying category is \CrefAndHyperrefIfExist{definition:locally_small_category}{$\mathscr{U}$-small}. 
    \end{enumerate}
\end{definition}
