    Let $T$ be an \CrefAndHyperrefIfExist{definition:ordered_commutative_monoid}{ordered commutative monoid}, let $T^\infty$ be the \CrefAndHyperrefIfExist{definition:ordered_monoid_with_adjoined_infinity}{ordered monoid with adjoined infinity} obtained from $T$, let $X$ be a set, and let $(Y, d)$ be an \CrefAndHyperrefIfExist{definition:extended_metric_on_a_set_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity}{extended metric space valued in $T^\infty$}.
    A sequence of functions $f_n : X \to Y$ is said to be \hldef{uniformly Cauchy} if for every $\varepsilon \in T_{>0}$, there exists $N \in \mathbb{N}$ such that for all $n, m \geq N$ and all $x \in X$, one has
    $$d(f_n(x), f_m(x)) < \varepsilon.$$
    Equivalently, the sequence $(f_n)$ is uniformly Cauchy if
    $$\lim_{n,m \to \infty} \sup_{x \in X} d(f_n(x), f_m(x)) = 0_T.$$

    If $X$ is a singleton, then a sequence of functions from $X$ to $Y$ is uniformly Cauchy if and only if the corresponding sequence in $Y$ is \CrefAndHyperrefIfExist{definition:cauchy_sequence_in_an_extended_metric_space_valued_in_ordered_commutative_monoid_with_adjoined_infinity}{Cauchy}.
\end{definition}
