    Let $\mathcal{C}$ be a \CrefAndHyperrefIfExist{definition:category}{category} and $X$ an object of $\mathcal{C}$.
    \begin{enumerate}
        \item An \hldef{endomorphism of $X$} is a morphism $f: X \to X$, i.e. an element of $\operatorname{Hom}_{\mathcal{C}}(X,X)$.

        \item Assume that $\mathcal{C}$ is a \CrefAndHyperrefIfExist{definition:locally_small_category}{locally small category}. The \hldef{endomorphism monoid of $X$}, denoted by \hl{$\operatorname{End}_{\mathcal{C}}(X)$}, is the set $\operatorname{Hom}_{\mathcal{C}}(X,X)$. It is a \CrefAndHyperrefIfExist{definition:monoid}{monoid} under composition of morphisms in $\mathcal{C}$, with identity element $\operatorname{id}_X$.
    \end{enumerate}
\end{definition}
