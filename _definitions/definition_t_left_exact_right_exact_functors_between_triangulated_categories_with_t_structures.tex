\begin{definition} \label{definition:t_left_exact_right_exact_functors_between_triangulated_categories_with_t_structures}
    Let $T: D_1 \to D_2$ be an \CrefAndHyperrefIfExist{definition:exact_functor_between_triangulated_categories}{exact functor} between \CrefAndHyperrefIfExist{definition:triangulated_category}{triangulated categories} with \CrefAndHyperrefIfExist{definition:t_structure_on_a_triangulated_category}{$t$-structure}. 
    Such a functor is called \hldef{$t$-left exact} if $T(\Dleqzero_1) \subset \Dleqzero_2$, \hldef{$t$-right exact} if $T(\Dgeqzero_1) \subset \Dgeqzero_2$, and \hldef{$t$-exact} if $T$ is both $t$-left and $t$-right exact.
\end{definition}
