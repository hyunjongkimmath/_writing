\begin{definition}[Arithmetic progression and arithmetic series in a commutative semiring] \label{definition:arithmetic_progression_and_arithmetic_series_in_a_commutative_semiring}
    \TODO{AI generated, verify}
    Let $S$ be a \CrefAndHyperrefIfExist{definition:semiring}{commutative semiring}. Let $a \in S$ and $d \in S$. For each integer $n \geq 1$, let
    $$a_n = a + (n-1)d,$$
    where $(n-1)d$ denotes the element $d$ added to itself $n-1$ times in $S$\CrefIfExists{definition:power_of_a_group_element_by_an_integer_multiple_of_an_abelian_group_element_by_an_integer}. The sequence $(a_n)_{n=1}^\infty$ is called an \hldef{arithmetic progression in $S$} with \hldef{first term} $a$ and \hldef{common difference} $d$.

    For each integer $N \geq 1$, the finite series $\sum_{n=1}^N a_n = \sum_{n=1}^N (a + (n-1)d)$ is called an \hldef{arithmetic series of length $N$ in $S$}.

    If $S$ carries a topology making addition continuous, then the infinite series $\sum_{n=1}^\infty (a + (n-1)d)$ is called an \hldef{infinite arithmetic series in $S$}, understood as the limit of its partial sums in the topology of $S$\CrefIfExists{definition:infinite_series_and_convergence_of_infinite_series_in_a_topological_abelian_semigroup}.
\end{definition}
