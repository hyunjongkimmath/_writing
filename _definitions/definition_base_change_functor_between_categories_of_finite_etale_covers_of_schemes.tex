\begin{definition}\label{definition:base_change_functor_between_categories_of_finite_etale_covers_of_schemes}
    Let $f: X \to Y$ be a \CrefAndHyperrefIfExist{definition:morphism_of_schemes}{morphism of schemes}. 
    The \hldef{base change functor on $\FEt_Y$} is the functor 
    $$\FEt_Y \to \FEt_X, \quad Z \mapsto Z \times_Y X$$
    \CrefIfExists{definition:category_of_finite_etale_covers_of_a_scheme}
    \CrefIfExists{definition:cartesian_product_of_two_objects_in_a_category_over_an_object}.
\end{definition}
