\begin{definition} \label{definition:tame_inertia_subgroup_of_etale_fundamental_group_of_connected_scheme_at_geometric_point}
    Let $X$ be a \CrefAndHyperrefIfExist{definition:disconnected_and_connected_topological_spaces}{connected} \CrefAndHyperrefIfExist{definition:scheme}{scheme}, let $x \in X$ be a point, and let $\overline{x}$ be a \CrefAndHyperrefIfExist{definition:geometric_point_of_a_scheme}{geometric point} of $X$ whose image is $x$. Let $p$ be the characteristic of the residue field $k(x)$.

    Let $I_{\overline{x}}$ be the \CrefAndHyperrefIfExist{definition:inertia_group_of_the_etale_fundamental_group_of_a_connected_scheme_at_a_geometric_point}{inertia subgroup} of $\pioneet(X, \overline{x})$ and let $P_{\overline{x}}$ be the \CrefAndHyperrefIfExist{definition:wild_inertia_subgroup_of_etale_fundamental_group_of_connected_scheme_at_geometric_point}{wild inertia subgroup}.
    The \hldef{tame inertia subgroup} \hl{$I_{\overline{x}}^{\text{tame}}$} is the quotient
    $$\hlin{I_{\overline{x}}^{\text{tame}} = I_{\overline{x}} / P_{\overline{x}}}$$
    It is a procyclic pro-prime-to-$p$ group.
\end{definition}
