\begin{definition}[Convergence of a monotone sequence in an ordered semiring with adjoined infinity] \label{definition:convergence_of_a_monotone_sequence_in_a_partially_ordered_set_with_adjoined_infinity}
    Let $T$ be a \CrefAndHyperrefIfExist{definition:partially_ordered_set}{partially ordered set}, and let \hl{$T^\infty$} be the \CrefAndHyperrefIfExist{definition_partially_ordered_set_with_adjoined_infinity.tex}{partially ordered set with adjoined infinity} obtained from $T$. Let $(s_N)_{N=1}^\infty$ be a \CrefAndHyperrefIfExist{definition:monotone_sequence_of_elements_in_a_partially_ordered_set}{monotone sequence} in $T^\infty$.

    \begin{enumerate}
        \item If $(s_N)_{N=1}^\infty$ is \CrefAndHyperrefIfExist{definition:monotone_sequence_of_elements_in_a_partially_ordered_set}{nondecreasing}, then is said to \hldef{converge to an element $s \in T$} if $s$ is a \CrefAndHyperrefIfExist{definition:supremum_infimum_of_a_subset_of_an_ordered_set}{supremum} of the sequence.

        If such a supremum does not exist in $T$, then we say that the sequence \hldef{diverges}.

        \item If $(s_N)_{N=1}^\infty$ is \CrefAndHyperrefIfExist{definition:monotone_sequence_of_elements_in_a_partially_ordered_set}{nonincreasing}, then is said to \hldef{converge to an element $s \in T$} if $s$ is a \CrefAndHyperrefIfExist{definition:supremum_infimum_of_a_subset_of_an_ordered_set}{infimum} of the sequence.

        If such a supremum does not exist in $T^\infty$, then we say that the sequence \hldef{diverges}.
    \end{enumerate}

    In the case that $s_N = \infty$ for some $N$, the sequence converges to $\infty$.

    \TextIfExists{proposition:convergence_of_a_monotone_sequence_to_a_supremum_or_infimum_in_a_totally_ordered_set_coincides_with_topological_convergence_in_the_order_topology}{
        When $T$ is a \CrefAndHyperrefIfExist{definition:total_order_linear_order_on_a_set}{totally ordered set} equipped with the \CrefAndHyperrefIfExist{definition:order_topology_on_a_partially_ordered_set}{order topology}, this order-theoretic notion of convergence of a monotone sequence to a supremum or infimum coincides with topological \CrefAndHyperrefIfExist{definition:convergence_and_limit_of_sequence_in_a_topological_space}{convergence of a sequence in a topological space} whenever the relevant supremum or infimum exists.
    }
\end{definition}
