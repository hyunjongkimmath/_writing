\begin{definition}[Measure valued in an ordered semiring] \label{definition:measure_on_a_measurable_space_valued_in_an_ordered_semiring_and_measure_space_with_such_a_measure}
    \TODO{AI generated, verify}
Let $K$ be an \CrefAndHyperrefIfExist{definition:ordered_semiring}{ordered semiring} such that every countable subset of $K_{\geq 0}$ bounded above has a supremum in $K$. For instance, this holds when $K = [0,\infty]$ or more generally when $K$ is an \CrefAndHyperrefIfExist{definition:ordered_semiring_with_adjoined_infinity}{ordered semiring with adjoined infinity}.

Let $(X,\mathcal{A})$ be a \CrefAndHyperrefIfExist{definition:sigma_algebra_on_a_set_and_measurable_space}{measurable space}. A function $\mu : \mathcal{A} \to K$ is called a \hldef{$K$-valued measure on $(X,\mathcal{A})$} if:
\begin{enumerate}
    \item $\mu(\varnothing) = 0_K$,
    \item for every sequence $(A_n)_{n\geq 1}$ of pairwise disjoint sets in $\mathcal{A}$, one has
    $$\mu\Bigl(\bigcup_{n=1}^\infty A_n\Bigr) = \sum_{n=1}^\infty \mu(A_n),$$
    where the infinite sum on the right-hand side is defined as the supremum of the finite partial sums:
    $$\sum_{n=1}^\infty \mu(A_n) := \sup_{m \geq 1} \sum_{n=1}^m \mu(A_n),$$
    which exists because the sequence of partial sums is non-decreasing and bounded above by $\mu(\bigcup_{n=1}^\infty A_n)$.
\end{enumerate}

The triple $(X,\mathcal{A},\mu)$ is then called a \hldef{$K$-valued measure space}.

\CrefIfExists{definition:measure_on_a_measurable_space_valued_in_the_extended_real_line_and_measure_space}{
    When $K = [0,\infty]$, this recovers the classical notion of a \CrefAndHyperrefIfExist{definition:measure_on_a_measurable_space_valued_in_the_extended_real_line_and_measure_space}{measure on $(X,\mathcal{A})$}.
}

\CrefIfExists{definition:sigma_complete_lattice_ordered_semimodule_over_an_ordered_semiring}{
    When $K$ is itself a $\sigma$-complete \CrefAndHyperrefIfExist{definition:lattice_ordered_semimodule_over_an_ordered_semiring}{lattice-ordered semimodule} over some ordered semiring, the condition that countable bounded-above sums have suprema follows from the $\sigma$-completeness of $K$.
}
\end{definition}
