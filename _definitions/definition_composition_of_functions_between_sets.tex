\begin{definition} \label{definition:composition_of_functions_between_sets}
    Let $f: X \to Y$ and $g: Y \to Z$ be \CrefAndHyperrefIfExist{definition:function_of_sets}{functions} between \CrefAndHyperrefIfExist{definition:zermelo_fraenkel_set_theory}{sets}. 
    The composition \hl{$g \circ f$} is the function $g \circ f: X \to Z$ given as follows: for all $x \in X$, $(g \circ f)(x) = g(f(x))$.
    Equivalently, $g \circ f$ is the function \CrefAndHyperrefIfExist{definition:function_of_sets}{corresponding to} the \CrefAndHyperrefIfExist{definition:n_ary_relation_on_sets}{binary relation} that is the \CrefAndHyperrefIfExist{definition:composition_of_binary_relations_on_sets}{composition} of the binary relations corresponding to $g$ and $f$.
\end{definition}
