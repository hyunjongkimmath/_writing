\begin{definition} \label{definition:conjugate_transpose_sesquilinear_form_of_a_sesquilinear_form_on_a_module_over_a_ring_with_involution}
    Let $A$ be a \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{ring with involution} denoted $\overline{\cdot}$ and $M$ a \CrefAndHyperrefIfExist{definition:bimodule_over_rings_module_over_a_ring}{right $A$-module} equipped with a \CrefAndHyperrefIfExist{definition:sesquilinear_form_on_a_module_over_a_ring_with_involution}{sesquilinear form} $b: M \times M \to A$. 

    Define a sesquilinear form \hl{$b^*: M \times M \to A$}, which may be called by names such as the \hldef{conjugate transpose of $b$} or the \hldef{adjoint sesquilinear form of $b$} (not to be confused with the adjoint map $M \to M^*$ of \Cref{definition:adjoint_map_of_a_sesquilinear_form_on_a_module_over_a_ring_with_involution}), by 
    $$b^{*}(x, y)=\overline{b(y, x)}, \quad x, y \in M.$$

\end{definition}
