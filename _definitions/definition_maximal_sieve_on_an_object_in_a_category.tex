\begin{definition} \label{definition:maximal_sieve_on_an_object_in_a_category}
    Let $C$ be a \CrefAndHyperrefIfExist{definition:category}{category} and let $U$ be an object of $C$. 
    The \hldef{maximal sieve on $U$} is the collection of all morphisms with target $U$; clearly, it is a \CrefAndHyperrefIfExist{definition:sieve_on_an_object_in_a_category}{sieve} on $U$.
\end{definition}
