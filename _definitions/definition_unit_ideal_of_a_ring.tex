\begin{definition} \label{definition:unit_ideal_of_a_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{ring}. 
    The abelian group $R$ is itself an \CrefAndHyperrefIfExist{definition:ideal_of_a_ring}{$R$-ideal} --- it is referred to as the \hldef{unit ideal of $R$}. 
\end{definition}
