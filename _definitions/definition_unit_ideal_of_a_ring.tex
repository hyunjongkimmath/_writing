\begin{definition} \label{definition:unit_ideal_of_a_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{ring}. 
    The abelian group $R$ is itself an \CrefAndHyperrefIfExist{definition:ideal_of_a_ring}{$R$-ideal} --- it is referred to as the \hldef{unit ideal of $R$}. 
    
    Equivalently, it is the \CrefAndHyperrefIfExist{definition:ideal_of_a_ring}{(two-sided) ideal} \CrefAndHyperrefIfExist{definition:ideal_generated_by_a_subset_of_a_ring}{generated by} $1$ or by any other \CrefAndHyperrefIfExist{definition:unit_of_a_ring}{unit} of $R$.
\end{definition}
