\begin{definition} \label{definition:decomposition_group_of_a_finite_galois_extension_of_valuation_fields}
    Assume \Cref{context:extension_of_valuation_fields} and further assume that $L/K$ is a \CrefAndHyperrefIfExist{definition:degree_of_a_field_extension}{finite} \CrefAndHyperrefIfExist{definition:galois_extension_of_fields}{Galois extension}.

    The \hldef{decomposition group of $L/K$} is the subgroup 
    $$\hlin{D_w = \{\sigma \in \Gal(L/K) : w \circ \sigma = w\}}$$
    \CrefIfExists{definition:galois_extension_of_fields}
\end{definition}
