\begin{definition} \label{definition:separable_closure_of_a_field_in_a_field_extension}
    Let $L/K$ be a \CrefAndHyperrefIfExist{definition:extension_of_a_field}{field extension}. 
    The \hldef{separable closure of $K$ in $L$} is the \CrefAndHyperrefIfExist{definition:extension_of_a_field}{subfield} of $L$ consisting of all \CrefAndHyperrefIfExist{definition:algebraic_element_minimal_polynomial_of_an_algebraic_element_algebraic_field_extension_transcendental_element_field_extension}{algebraic} elements of $L$ over $K$.
\end{definition}
