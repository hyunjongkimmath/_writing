\begin{definition} \label{definition:good_open_cover_of_a_topological_space}
    Let $X$ be a \CrefAndHyperrefIfExist{definition:topological_space}{topological space}. 
    An \CrefAndHyperrefIfExist{definition:open_covering_of_a_topological_space}{open cover} $\{U_i\}_{i \in I}$ is said to be a \hldef{good cover} if every finite intersection $\bigcap_{j=1}^n U_{i_j}$ is empty or \CrefAndHyperrefIfExist{definition:contractible_topological_space}{contractible}.
\end{definition}
