\begin{definition} \label{definition:tamely_and_wildly_ramified_extension_of_valuation_fields}
    Assume \Cref{context:extension_of_valuation_fields} and let $p$ be the \CrefAndHyperrefIfExist{definition:characteristic_of_a_ring}{characteristic} of the \CrefAndHyperrefIfExist{definition:residue_field_of_a_ring_at_a_point}{residue field} $k_v$.
    \begin{enumerate}
        \item Further assume that $L/K$ is \CrefAndHyperrefIfExist{definition:finite_field_extension}{finite}. 
        \begin{enumerate}
            \item The extension $(L,w)/(K,v)$ is \hldef{tamely ramified} if:
            \begin{enumerate}
                \item The \CrefAndHyperrefIfExist{definition:residue_field_of_a_ring_at_a_point}{residue field} \CrefAndHyperrefIfExist{definition:extension_of_a_field}{extension} $k_w/k_v$ is \CrefAndHyperrefIfExist{definition:separable_algebraic_field_extension}{separable};
                \item If $p > 0$, then $p \nmid e(w/v)$;
                \item The extension is defectless, i.e., $[L:K] = e(w/v)f(w/v)$.
            \end{enumerate}
            \item The extension $(L,w)/(K,v)$ is \hldef{wildly ramified} if it is not tamely ramified. 
            \begin{itemize}
                \item \textit{Note:} For \CrefAndHyperrefIfExist{definition:nonarchimedean_absolute_value_induced_by_a_discrete_valuation_on_a_field}{non-archimedean} \CrefAndHyperrefIfExist{definition:local_field}{local fields}, this occurs if and only if $p \mid e(w/v)$.
            \end{itemize}
        \end{enumerate}

        \item Removing the assumption that $L/K$ is finite, the extension $(L,w)/(K,v)$ is \hldef{tamely ramified} (resp. \hldef{wildly ramified}) if for every (resp. for some) finite subextension $K \subseteq E \subseteq L$, $(E, w|_E) /(K,v)$ is tamely ramified (resp. wildly ramified).
    \end{enumerate}
\end{definition}
