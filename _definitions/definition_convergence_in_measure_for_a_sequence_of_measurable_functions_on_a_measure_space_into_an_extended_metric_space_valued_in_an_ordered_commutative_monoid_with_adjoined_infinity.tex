\begin{definition}[Convergence in measure] \label{definition:convergence_in_measure_for_a_sequence_of_measurable_functions_on_a_measure_space_into_an_extended_metric_space_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity}
Let $(X, \mathcal{M}, \mu)$ be a \CrefAndHyperref{definition:measure_on_a_measurable_space_valued_in_the_extended_real_line_and_measure_space}{measure space}. Let $T$ be an \CrefAndHyperrefIfExist{definition:ordered_commutative_monoid}{ordered commutative monoid}, let $T^\infty$ be the \CrefAndHyperrefIfExist{definition:ordered_monoid_with_adjoined_infinity}{ordered monoid with adjoined infinity} obtained from $T$, and let $(Y, d)$ be a \CrefAndHyperref{definition:extended_metric_on_a_set_valued_in_an_ordered_commutative_monoid_with_adjoined_infinity}{metric space valued in $T^\infty$}. Let $\{f_n\}_{n=1}^\infty$ be a sequence of \CrefAndHyperref{definition:measurable_function_between_measure_spaces}{measurable functions} $f_n: X \to Y$, and let $f: X \to Y$ be a measurable function.

The sequence $\{f_n\}$ is said to \hldef{converge in measure (on $(X, \mathcal{M}, \mu)$ into $(Y, d)$) to $f$} if for every $\varepsilon \in T_{>0}$,
$$\mu(\{x \in X : d(f_n(x), f(x)) \geq \varepsilon\}) \to 0 \quad \text{as } n \to \infty.$$
That is, for every $\varepsilon \in T_{>0}$ and every $\delta > 0$ in $\bbR$, there exists $N \in \mathbb{N}$ such that for all $n \geq N$,
$$\mu(\{x \in X : d(f_n(x), f(x)) \geq \varepsilon\}) < \delta.$$
\end{definition}
