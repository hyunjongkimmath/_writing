\begin{definition}  \label{definition:strong_rost_nilpotence_for_a_motive_over_a_field_with_coefficients_in_a_commutative_ring}
Let $k$ be a \CrefAndHyperrefIfExist{definition:field}{field}, $R$ a \CrefAndHyperrefIfExist{definition:commutative_ring}{commutative ring}, and $\mathfrak{Chow}(k, R)$ the \CrefAndHyperrefIfExist{definition:category_of_relative_chow_motives_over_a_noetherian_base_and_coefficients_in_a_commutative_ring}{category of Chow motives over $k$ with coefficients in $R$}.

A motive $M \in \mathfrak{Chow}(k, R)$ satisfies \hldef{strong Rost nilpotence} if for every \CrefAndHyperrefIfExist{definition:extension_of_a_field}{field extension} $E/k$, the ideal $\mathcal{I}_{E/k}(M)$ (\Cref{definition:ideal_of_endomorphisms_for_a_chow_motive_over_a_field_with_coefficients_in_a_commutative_ring}) is a \CrefAndHyperrefIfExist{definition:nilpotent_ideal_of_a_ring}{nilpotent ideal}.
\end{definition}
