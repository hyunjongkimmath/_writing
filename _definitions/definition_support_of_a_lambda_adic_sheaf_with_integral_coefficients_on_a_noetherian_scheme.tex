\begin{definition}[Support of a $\lambda$-adic sheaf with integral coefficients on a noetherian scheme] \label{definition:support_of_a_lambda_adic_sheaf_with_integral_coefficients_on_a_noetherian_scheme}
\TODO{AI generated, verify}
    Let $(R, \lambda)$ be a \CrefAndHyperrefIfExist{definition:complete_ring_with_respect_to_an_ideal}{complete} \CrefAndHyperrefIfExist{definition:discrete_valuation_ring}{DVR} such that $R/\lambda^n$ is finite for all $n \geq 1$. Let $X$ be a \CrefAndHyperrefIfExist{definition:locally_noetherian_and_noetherian_scheme}{noetherian scheme}, and consider sheaves on the \CrefAndHyperrefIfExist{definition:small_etale_site_of_a_scheme}{(small) \'etale site} of $X$. Let $\calF = (\calF_n, u_n)_{n \geq 1}$ be a \CrefAndHyperrefIfExist{definition:lambda_adic_sheaf_on_the_small_etale_site_of_a_scheme_for_an_ideal_of_a_commutative_ring}{$\lambda$-adic sheaf} on $X$.

    The \hldef{support of $\calF$} is the union of the supports of its level-wise reductions:
    $$\hlin{\Supp(\calF) := \bigcup_{n \geq 1} \Supp(\calF_n)}$$
    (\Cref{definition:support_of_an_etale_sheaf_of_abelian_groups_on_a_scheme}) Each $\Supp(\calF_n)$ is a closed subset of $X$. When $\calF$ satisfies the \CrefAndHyperrefIfExist{definition:artin_rees_mittag_leffler_condition_on_a_tower_in_a_category_with_images}{Artin--Rees--Mittag--Leffler condition}, the sequence of supports stabilizes for sufficiently large $n$, and the union is a closed subset of $X$.
\end{definition}
