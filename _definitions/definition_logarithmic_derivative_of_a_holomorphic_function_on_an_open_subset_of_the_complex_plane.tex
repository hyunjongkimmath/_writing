\begin{definition}[Logarithmic derivative of a holomorphic function] \label{definition:logarithmic_derivative_of_a_holomorphic_function_on_an_open_subset_of_the_complex_plane}
    \TODO{AI generated, verify}
Let $U \subseteq \bbC$ be an open set (or more generally $U \subseteq \bbC^n$ open) and let $f : U \to \bbC$ be a \CrefAndHyperrefIfExist{definition:holomorphic_function_from_a_subset_of_Cn_to_Cm_and_derivative_of_a_holomorphic_function_at_a_point}{holomorphic function} that is not identically zero.

The \hldef{logarithmic derivative of $f$} is the \CrefAndHyperrefIfExist{definition:meromorphic_function_on_an_open_subset_of_the_complex_plane}{meromorphic function}
$$
\frac{f'}{f} : U \setminus f^{-1}(0) \to \bbC, \qquad z \mapsto \frac{f'(z)}{f(z)},
$$
\CrefIfExists{definition:holomorphic_function_from_a_subset_of_Cn_to_Cm}

At a point $z_0 \in U$ where $f(z_0) \neq 0$, the logarithmic derivative is simply the derivative of a local \CrefAndHyperrefIfExist{definition:branch_of_the_complex_logarithm_on_an_open_subset}{branch} of $\log f(z)$. At a zero $z_0$ of $f$ of order $m \ge 1$, the logarithmic derivative has a simple pole\CrefIfExists{definition:isolated_singularity_of_a_holomorphic_function_from_an_open_subset_of_the_complex_plane_to_the_complex_plane_classified_as_removable_pole_or_essential} with \hldef{residue} $m$\CrefIfExists{definition:residue_of_a_holomorphic_function_at_an_isolated_singularity_defined_as_the_coefficient_of_z_minus_z_0_to_the_negative_1_in_its_laurent_expansion}; that is, in a neighborhood of $z_0$,
$$
\frac{f'(z)}{f(z)} = \frac{m}{z - z_0} + h(z),
$$
where $h$ is holomorphic near $z_0$.

If $f$ and $g$ are holomorphic and not identically zero, then
$$
\frac{(fg)'}{fg} = \frac{f'}{f} + \frac{g'}{g},
$$
which generalizes the real-variable formula. For an infinite product $f(z) = \prod_{n=1}^\infty f_n(z)$ converging absolutely and uniformly on compact subsets, with each $f_n$ holomorphic and not identically zero, the logarithmic derivative is
$$
\frac{f'(z)}{f(z)} = \sum_{n=1}^\infty \frac{f_n'(z)}{f_n(z)},
$$
where the sum converges uniformly on compact subsets avoiding the zeros of $f$ and each $f_n$.
\end{definition}
