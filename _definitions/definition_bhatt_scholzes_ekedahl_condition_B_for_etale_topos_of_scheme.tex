\begin{definition}[{\cite[Definition 5.5.1.]{bhatt_scholze_pfgs}}] \label{definition:bhatt_scholzes_ekedahl_condition_B_for_etale_topos_of_scheme}
    \TODO{complete ring with respect to ideal}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:commutative_ring}{commutative} \CrefAndHyperrefIfExist{definition:noetherian_ring}{noetherian ring} complete for the \CrefAndHyperrefIfExist{definition:topological_space}{topology} defined by an \CrefAndHyperrefIfExist{definition:ideal_of_a_ring}{ideal} $\mathfrak{m} \subset R$. 

    We say that $(R, \mathfrak{m})$ satisfy \hldef{Bhatt and Scholze's Ekedahl's condition B} if the following holds:
    \begin{center}
        \TODO{regular ideal of a ring}
        $\mathfrak{m}$ is a regular ideal of $R$ and the $R/\mathfrak{m}$-module $\mathfrak{m}^n/\mathfrak{m}^{n+1}$ has finite flat dimension bounded independently of $n$. 
    \end{center}

    When considering $R$ and $\mathfrak{m}$ as \CrefAndHyperrefIfExist{definition:constant_sheaf_on_a_site_with_sheafification}{constant} \CrefAndHyperrefIfExist{definition:sheaf_on_a_site}{sheaves} on \CrefAndHyperrefIfExist{definition:small_etale_site_of_a_scheme}{$X_{\text{\'et}}$}, this condition may be slightly stronger than \Cref{definition:ekedahls_condition_A_and_B}{Ekedahl's condition B}.
\end{definition}
