\begin{definition} \label{definition:exponentiation_in_a_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{ring}, and let $a \in R$. For each integer $n \in \mathbb{Z}$, the \hldef{$n$th power of $a$ in $R$}, denoted by \hl{$a^n$}, is defined as follows:
    \begin{itemize}
        \item If $n \geq 0$, then $a^n$ is the \CrefAndHyperrefIfExist{definition:exponentiation_in_a_semiring}{$n$th power} of $a$ with respect to the underlying \CrefAndHyperrefIfExist{definition:semiring}{multiplicative semiring structure} $(R, +, \cdot, 0_R, 1_R)$.
        \item If $n < 0$ and $a$ is a \CrefAndHyperrefIfExist{definition:unit_of_a_ring}{unit of $R$}, then $a^n := (a^{-1})^{-n}$, where $a^{-1}$ is the \CrefAndHyperrefIfExist{definition:unit_of_a_ring}{multiplicative inverse} of $a$, and $(a^{-1})^{-n}$ is the power by the nonnegative integer $-n$ as in the first case.
    \end{itemize}
\end{definition}
