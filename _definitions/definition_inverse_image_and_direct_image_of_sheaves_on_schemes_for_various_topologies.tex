\begin{definition}  \label{definition:inverse_image_and_direct_image_of_sheaves_on_schemes_for_various_topologies}

Let $f : X \to Y$ be a morphism of schemes, and let $\mathbf{C}(X)$ and $\mathbf{C}(Y)$ be \CrefAndHyperrefIfExist{definition:big_site_on_the_category_of_schemes_over_a_scheme_and_small_site}{small sites} associated to $X$ and $Y$ respectively, equipped with a Grothendieck topologies for which the \CrefAndHyperrefIfExist{definition:continuous_functor_of_sites}{continuous functor} \CrefAndHyperrefIfExist{definition:continuous_functors_on_sites_on_schemes_induced_by_scheme_morphism}{$f^{-1} : \mathbf{C}(Y) \to \mathbf{C}(X)$} on sites underlies a \CrefAndHyperrefIfExist{definition:morphism_of_sites}{site morphism} $f: \mathbf{C}(X) \to \mathbf{C}(Y)$. For example, the common Grothendieck topologies such as \CrefAndHyperrefIfExist{definition:small_zariski_site_of_a_schem}{Zariski}, \CrefAndHyperrefIfExist{definition:small_etale_site_of_a_scheme}{étale}, \CrefAndHyperrefIfExist{definition:small_nisnevich_site_of_a_schem}{Nisnevich}, \CrefAndHyperrefIfExist{definition:small_fppf_site_of_a_scheme}{fppf}, \CrefAndHyperrefIfExist{definition:small_fpqc_site_of_a_schem}{fpqc}, or crystalline satisfy this hypothesis\TODO{cite these}. 

\begin{enumerate}

    \item 
    Let $\mathcal{A}$ be a (large) category.
    Given a sheaf $\calF \in \operatorname{Sh}(\mathbf{C}(X); \mathcal{A})$, the \hldef{direct image along $f$} is the sheaf \hl{$f_* \calF$} defined as the \CrefAndHyperrefIfExist{definition:direct_image_of_a_sheaf_on_a_site_under_a_continuous_functor_of_sites_or_a_site_morphism}{direct image} of $\calF$ under site morphism $f: \mathbf{C}(X) \to \mathbf{C}(Y)$.
    % the \CrefAndHyperrefIfExist{definition:continuous_functor_of_sites}{continuous functor} \CrefAndHyperrefIfExist{definition:continuous_functors_on_sites_on_schemes_induced_by_scheme_morphism}{$f^{-1} : \mathbf{C}(Y) \to \mathbf{C}(X)$} on sites. 
    In particular, it is an object of $\operatorname{Sh}(\mathbf{C}(Y); \mathcal{A})$ if it exists. If $f_* \calF$ exists for all sheaves $\calF \in \operatorname{Sh}(\mathbf{C}(X); \mathcal{A})$, then $f_*$ yields a functor
    $$f_* : \operatorname{Sh}(\mathbf{C}(X); \mathcal{A}) \to \operatorname{Sh}(\mathbf{C}(Y); \mathcal{A}).$$

    \item We have (\Cref{definition:direct_image_of_a_sheaf_of_abelian_groups_on_a_site_under_a_continuous_functor_of_sites_or_a_site_morphism}) a direct image functor 
    $$\hlin{f_* : \operatorname{Sh}(\mathbf{C}(X); \Ab) \to \operatorname{Sh}(\mathbf{C}(Y); \Ab).}$$
    In fact, if $f$ yields a ringed site morphism $f: (\mathbf{C}(X), \calO_X) \to (\mathbf{C}(Y), \calO_Y)$, then this induces (\Cref{definition:direct_image_of_a_module_on_a_ringed_site_via_a_morphism_of_ringed_sites}) a functor
    $$\hlin{f_* : \Mod(\calO_X) \to \Mod(\calO_Y).}$$

    \item 
    Let $\mathcal{A}$ be a (large) category such that the \CrefAndHyperrefIfExist{definition:presheaf_on_a_category}{presheaf category} $\operatorname{PreSh}(\calD,K;\mathcal{A})$ has \CrefAndHyperrefIfExist{definition:sheafification_functor_on_a_site}{sheafification}.
    Given a sheaf $\calG \in \operatorname{Sh}(\mathbf{C}(Y); \mathcal{A})$, the \hldef{inverse image of $\calG$ along $f$} is the sheaf \hl{$f^* \calG$} defined as the \CrefAndHyperrefIfExist{definition:inverse_image_of_a_sheaf_under_a_continuous_functor_of_sites_or_a_site_morphism}{inverse image} of $\calG$ under the \CrefAndHyperrefIfExist{definition:continuous_functor_of_sites}{continuous functor} \CrefAndHyperrefIfExist{definition:continuous_functors_on_sites_on_schemes_induced_by_scheme_morphism}{$f^{-1} : \mathbf{C}(Y) \to \mathbf{C}(X)$} on sites. In particular, it is an object of $\operatorname{Sh}(\mathbf{C}(X); \mathcal{A})$ and $f^*$ yields a functor
    $$f^* : \operatorname{Sh}(\mathbf{C}(Y); \mathcal{A}) \to \operatorname{Sh}(\mathbf{C}(X); \mathcal{A}),$$

    \item We have (\Cref{definition:inverse_image_of_a_sheaf_of_abelian_groups_on_a_site_under_a_continuous_functor_of_sites_or_a_site_morphism}) an inverse image functor 
    $$\hlin{f^* : \operatorname{Sh}(\mathbf{C}(Y); \Ab) \to \operatorname{Sh}(\mathbf{C}(X); \Ab).}$$
    In fact, if $f$ yields a ringed site morphism $f: (\mathbf{C}(X), \calO_X) \to (\mathbf{C}(Y), \calO_Y)$, then this induces (\Cref{definition:inverse_image_of_a_module_on_a_ringed_site_via_a_morphism_of_ringed_sites}) a functor
    $$\hlin{f^* : \Mod(\calO_Y) \to \Mod(\calO_X).}$$
\end{enumerate}
\end{definition}
