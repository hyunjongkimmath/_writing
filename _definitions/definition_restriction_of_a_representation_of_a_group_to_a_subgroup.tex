    Let $(G, \cdot)$ be a \CrefAndHyperrefIfExist{definition:group}{group}, $H \leq G$ a subgroup, and $(\rho, V)$ a \CrefAndHyperrefIfExist{definition:representation_of_a_group_on_a_vector_space_over_a_field}{representation} of $G$ over a field $k$.

    The \hldef{restriction of $\rho$ to $H$} is the representation
    $$\hlin{\mathrm{Res}_H^G(\rho) = \rho|_H : H \to \mathrm{GL}(V)}$$
    obtained by composing $\rho$ with the inclusion map $H \hookrightarrow G$. That is, $\mathrm{Res}_H^G(\rho)(h) = \rho(h)$ for all $h \in H$. The pair $(\mathrm{Res}_H^G(\rho), V)$ is called the \hldef{restricted representation of $V$ from $G$ to $H$}.
\end{definition}
