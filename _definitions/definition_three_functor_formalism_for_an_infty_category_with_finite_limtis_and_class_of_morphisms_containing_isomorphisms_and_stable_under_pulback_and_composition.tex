    As the name suggests, $3$ classes of functors are of particular interest in a $3$-functor formalism:
    \begin{enumerate}
        \item For all objects $X \in C$, \hldef{internal tensor product functors} \hl{$\otimes: D(X) \times D(X) \to D(X)$}, defined as the composition of htel ax symmetric monoidal structure map $D(X) \times D(X) \to D(X \times X)$ followed by the pullback $\Delta^*$ along the diagonal $\Delta: X \to X \times X$.
        \item For all morphisms $f: X \to Y$ in $C$, \hldef{inverse image}/\hldef{pullback functors} \hl{$f^*: D(Y) \to D(X)$}
        \item For all morphisms $f: X \to Y$ in $E$, \hldef{``compactly supported'' direct image/pushforward functors} \hl{$f_!: D(X) \to D(Y)$}
    \end{enumerate}
\end{definition}
