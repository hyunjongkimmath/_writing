\begin{definition}[see {\cite[Chapter I, 5.1]{knus_qhfr}}] \label{definition:unitary_algebra_over_an_involution_ring}
    Let $R$ be an \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{involution ring}. 
    A \hldef{unitary $R$-algebra} is a \CrefAndHyperrefIfExist{definition:unitary_ring}{unitary ring} $(A, \varepsilon, \Lambda)$ such that $A$ is an $R$-algebra with an \CrefAndHyperrefIfExist{definition:algebra_with_involution_over_a_ring_with_involution}{$R$-linear involution} and $\Lambda$ is an $R$-submodule of $A$ such that $S_{-\varepsilon}(A) \subseteq \lambda \subseteq S^{-\varepsilon}(A)$ and $a \cdot x \cdot \overline{a} \in \Lambda$ for all $x \in \Lambda$ and $a \in \lambda$.

\end{definition}
