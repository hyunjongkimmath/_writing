\begin{definition} \label{definition:unibranch_geometircally_unibranch_scheme}
    Let $X$ be a scheme.
    \begin{enumerate}
        \item We say that $X$ is \hldef{unibranch at a point $x \in X$} if the \CrefAndHyperrefIfExist{definition:local_ring}{local ring} \CrefAndHyperrefIfExist{definition:stalk_of_a_presheaf_on_a_topological_space_at_a_point}{$\mathcal{O}_{X,x}$} is \CrefAndHyperrefIfExist{definition:unibranch_geometrically_unibranch_local_ring}{unibranch}.

        \item We say that $X$ is \hldef{unibranch} if it is unibranch at all points.

        \item We say that $X$ is \hldef{geometrically unibranch at a point $x \in X$} if the \CrefAndHyperrefIfExist{definition:local_ring}{local ring} \CrefAndHyperrefIfExist{definition:stalk_of_a_presheaf_on_a_topological_space_at_a_point}{$\mathcal{O}_{X,x}$} is \CrefAndHyperrefIfExist{definition:unibranch_geometrically_unibranch_local_ring}{geometrically unibranch}.

        \item We say that $X$ is \hldef{geometrically unibranch} if it is geometrically unibranch at all points.
    \end{enumerate}
\end{definition}
