\begin{definition} \label{definition:category_with_images_coimages}
    Let $\calC$ be a category. 
    \begin{enumerate}
        \item $\calC$ is said to \hldef{have images} if every morphism $f: A \to B$ in $\calC$ has an \CrefAndHyperrefIfExist{definition:image_coimage_of_a_morphism_in_a_category}{image}.
        \item $\calC$ is said to \hldef{have coimages} if every morphism $f: A \to B$ in $\calC$ has a \CrefAndHyperrefIfExist{definition:image_coimage_of_a_morphism_in_a_category}{coimage}.
    \end{enumerate}
\end{definition}
