\begin{definition}[Coset Space / Quotient Set] \label{definition:coset_space_of_a_subgroup_of_a_group}

Let $G$ be a \CrefAndHyperrefIfExist{definition:group}{group} and $H \le G$ a \CrefAndHyperrefIfExist{definition:subgroup_of_a_group}{subgroup}.
\begin{enumerate}
    \item The \hldef{left coset space of $G$ by $H$} (also called the \hldef{left quotient set of $G$ by $H$}) is the set of all \CrefAndHyperrefIfExist{definition:coset_of_a_subgroup_of_a_group_by_an_element}{left cosets} of $H$ in $G$:
    $$\hlin{ G/H := \{ gH \mid g \in G \}. }$$
    By definition, the \CrefAndHyperrefIfExist{definition:cardinal_number_and_cardinality_of_a_set}{cardinality} of $G/H$ is the \CrefAndHyperrefIfExist{definition:index_of_a_subgroup_in_a_group}{index} of $H$ in $G$, denoted $[G:H]$.

    If $H \trianglelefteq G$ is a \CrefAndHyperrefIfExist{definition:normal_subgroup_of_a_group}{normal subgroup}, then $G/H$ carries a natural group structure and is called the \CrefAndHyperrefIfExist{definition:quotient_of_a_group_by_a_normal_subgroup}{quotient group} of $G$ by $H$.

    \item The \hldef{right coset space of $G$ by $H$} (also called the \hldef{right quotient set of $G$ by $H$}) is the set of all \CrefAndHyperrefIfExist{definition:coset_of_a_subgroup_of_a_group_by_an_element}{right cosets} of $H$ in $G$:
    $$\hlin{ H \backslash G := \{ Hg \mid g \in G \}. }$$
    The cardinality of $G/H$ is the \CrefAndHyperrefIfExist{definition:index_of_a_subgroup_in_a_group}{index} of $H$ in $G$, denoted $[G:H]$.

\end{enumerate}
\end{definition}
