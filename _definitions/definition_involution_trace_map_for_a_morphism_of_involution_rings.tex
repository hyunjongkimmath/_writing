\begin{definition}[{\cite[Chapter I, (7.2.4) Proposition]{knus_qhfr}}] \label{definition:involution_trace_map_for_a_morphism_of_involution_rings}
    Let $\eta: A \to A'$ be a \CrefAndHyperrefIfExist{definition:homomorphism_of_involution_rings}{homomorphism of rings with involution}. An \hldef{involution trace of $\eta$} is a morphism $\tau: A' \to A$ of $A$-\CrefAndHyperrefIfExist{definition:module_over_a_sheaf_of_rings_on_a_site}{bimodules} such that  
    \begin{itemize}
        \item $\tau(ayb) = a\tau(y)b$ for all $a,b \in A$ and $y \in A'_{(A)}$, i.e. $\tau: A'_{(A)} \to A$ is a bimodule homomorphism. 

        \item $$\tau(\overline{a}) = \overline{\tau(a)}$$
            for all $a \in A'$

        \item The $1$-hermitian form $t_{A'/A}: A' \times A' \to A$ given by 
        $$t_{A'/A}(x,y) = \tau(\overline{x} y)$$
        is \CrefAndHyperrefIfExist{definition:nondegenerate_nonsingular_regular_sesquilinear_form_on_a_module_over_an_involution_ring}{nonsingular}. 
    \end{itemize}

\end{definition}
