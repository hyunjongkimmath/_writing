\begin{definition} \label{definition:cancellative_binary_operation_on_a_set}
    Let $\cdot : S \times S \to S$ be a \CrefAndHyperrefIfExist{definition:n_ary_operation_binary_operation_ternary_operation_on_a_set}{binary operation} on a \CrefAndHyperrefIfExist{definition:zermelo_fraenkel_set_theory}{set} $S$.
    \begin{enumerate}
        \item The operation $\cdot$ is \hldef{left cancellative} if for all $a,b,c \in S$,
        $$a \cdot b = a \cdot c \implies b = c.$$
        \item The operation $\cdot$ is \hldef{right cancellative} if for all $a,b,c \in S$,
        $$b \cdot a = c \cdot a \implies b = c.$$
        \item The operation $\cdot$ is \hldef{cancellative} (or \hldef{two-sided cancellative}) if it is both left cancellative and right cancellative.
    \end{enumerate}
\end{definition}
