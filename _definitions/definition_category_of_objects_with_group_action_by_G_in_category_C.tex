\begin{definition}[category of objects with $G$-action in $\mathcal{C}$]
\label{definition:category_of_objects_with_group_action_by_G_in_category_C}
Let $G$ be a \CrefAndHyperrefIfExist{definition:group}{group} and let $\mathcal{C}$ be a \CrefAndHyperrefIfExist{definition:category}{category}. The \hldef{category of objects with $G$-action in $\mathcal{C}$}, denoted by \hl{$G\text{-}\mathcal{C}$}, is defined as follows:
\begin{enumerate}
    \item An object is a pair $(c, \alpha)$ where $c$ is an object of $\mathcal{C}$ and $\alpha$ is an \CrefAndHyperrefIfExist{definition:action_of_a_group_G_on_an_object_in_a_category_C}{action of $G$ on $c$ in $\mathcal{C}$}.
    \item A morphism from $(c, \alpha)$ to $(d, \beta)$ is an \CrefAndHyperrefIfExist{definition:equivariant_morphism_between_objects_with_G_action_in_category_C}{equivariant morphism from $(c, \alpha)$ to $(d, \beta)$ in $\mathcal{C}$}.
\end{enumerate}
\end{definition}
