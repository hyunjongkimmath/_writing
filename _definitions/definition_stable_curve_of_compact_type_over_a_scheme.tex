\begin{definition}[Stable curve of compact type] \label{definition:stable_curve_of_compact_type_over_a_scheme}
    \TODO{dual graph of a graph, tree}
    Let $\pi\colon \mathcal{C} \to S$ be a stable curve over a scheme $S$. We say that $\mathcal{C}$ is of \hldef{compact type} if for every \CrefAndHyperrefIfExist{definition:geometric_point_of_a_scheme}{geometric point} $\bar{s} \to S$, the dual graph of the fiber $\mathcal{C}_{\bar{s}}$ is a tree.

    \TODO{semiabelian group scheme}
    Equivalently, $\mathcal{C}$ is of compact type if and only if the semi-abelian group scheme $\mathrm{Pic}^0_{\mathcal{C}/S}$ is an \CrefAndHyperrefIfExist{definition:abelian_scheme_over_a_scheme}{abelian scheme} over $S$.

    \TODO{normalization of a scheme}
    When $X$ is a stable curve over a field $k$ that is of compact type, its Jacobian $\mathrm{Jac}(X) = \mathrm{Pic}^0_{X/k}$ is an \CrefAndHyperrefIfExist{definition:abelian_variety_over_a_field}{abelian variety} (as opposed to a general semi-abelian variety). If $X$ is \CrefAndHyperrefIfExist{definition:irreducible_topological_space}{reducible} with \CrefAndHyperrefIfExist{definition:irreducible_component_of_a_topological_space}{irreducible components} $C_1, \dots, C_n$, then
    $$\mathrm{Jac}(X) \cong \prod_{i=1}^n \mathrm{Jac}(\widetilde{C}_i),$$
    where $\widetilde{C}_i$ is the normalization of $C_i$, and the \CrefAndHyperrefIfExist{definition:polarization_of_an_abelian_variety_over_a_field}{polarization} on $\mathrm{Jac}(X)$ is the product of the canonical polarizations on each factor.
\end{definition}
