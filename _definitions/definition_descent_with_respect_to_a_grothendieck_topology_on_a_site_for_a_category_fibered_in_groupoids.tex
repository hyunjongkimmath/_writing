\begin{definition}
\label{definition:descent_with_respect_to_a_grothendieck_topology_on_a_site_for_a_category_fibered_in_groupoids}
Let $(\mathcal{C}, J)$ be a \CrefAndHyperrefIfExist{definition:grothendieck_topology_on_a_category_site_covering_sieve_topologically_generating_family}{site}. Let $p: \mathcal{F} \to \mathcal{C}$ be a \CrefAndHyperrefIfExist{definition:category_fibered_in_groupoids_over_a_category}{category fibered in groupoids over $\mathcal{C}$}, equipped with a fixed \CrefAndHyperrefIfExist{definition:cleavage_splitting_for_a_category_fibered_in_groupoids}{cleavage}.

We say that $p$ \hldef{satisfies descent with respect to the topology $J$} (or simply that $\mathcal{F}$ has \hldef{$J$-descent}) if for every object $U \in \operatorname{Ob}(\mathcal{C})$ and every \CrefAndHyperrefIfExist{definition:grothendieck_topology_on_a_category_site_covering_sieve_topologically_generating_family}{covering sieve} $S \in J(U)$, every \CrefAndHyperrefIfExist{definition:descent_datum_for_a_category_fibered_in_groupoids_over_a_category_relative_to_a_sieve_on_an_object_of_the_base_category}{descent datum for $\mathcal{F}$ relative to the sieve $S$} is \CrefAndHyperrefIfExist{definition:effective_descent_datum_for_a_category_fibered_in_groupoids_over_a_category_relative_to_a_sieve_on_an_object_of_the_base_category}{effective}.
\end{definition}
