\begin{definition} \label{definition:bounded_object_bounded_triangulated_category_with_respect_to_a_t_structure}
    Given a \CrefAndHyperrefIfExist{definition:triangulated_category}{triangulated category} $D$ with a \CrefAndHyperrefIfExist{definition:t_structure_on_a_triangulated_category}{$t$-structure} $(D^{\leq 0}, D^{\geq 0})$, call an object $M \in D$ \hldef{bounded} if $M \in \Dgeqn[a] \cap \Dleqn[b]$ for some integers $a$ and $b$. Say that $D$ is \hldef{bounded with respect to the $t$-structure} if every object of $D$ is bounded.
\end{definition}
