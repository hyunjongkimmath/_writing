\begin{definition}[Dual abelian variety] \label{definition:dual_abelian_variety_of_an_abelian_variety_over_a_field}
    Let $A$ be an \CrefAndHyperrefIfExist{definition:abelian_variety_over_a_field}{abelian variety} over a field $k$. The \hldef{dual abelian variety of $A$}, denoted by \hl{$A^*$}, is the abelian variety representing the Picard functor $\mathrm{Pic}_{A/k}$, i.e.,
    \TODO{picard functor}
    $$A^* = \mathrm{Pic}^0(A),$$
    where $\mathrm{Pic}^0(A)$ is the connected component of the identity in the Picard scheme of $A$. For any commutative $k$-algebra $R$, the group $A^*(R)$ classifies line bundles on $A \times_k \Spec R$ that have degree zero on every fiber over $\Spec R$.
\end{definition}
