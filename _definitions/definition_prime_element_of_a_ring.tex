\begin{definition} \label{definition:prime_element_of_a_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{ring}.
    A \hldef{prime element of $R$} is an element $p \in R$ that is not zero, not a unit, and $p \mid ab$ implies that $p \mid a$\CrefIfExists{definition:divisibility_of_elements_of_rings} or $p \mid b$.

    \TextIfExists{definition:prime_and_maximal_ideal_of_a_ring}{
        If $R$ is \CrefAndHyperrefIfExist{definition:commutative_ring}{commutative}, then $p \in R$ is prime if and only if the \CrefAndHyperrefIfExist{definition:ideal_generated_by_a_subset_of_a_ring}{principal ideal} $(p)$ \CrefAndHyperrefIfExist{definition:ideal_generated_by_a_subset_of_a_ring}{generated by} $p$ is a \CrefAndHyperrefIfExist{definition:prime_and_maximal_ideal_of_a_ring}{prime ideal}.
    }
\end{definition}
