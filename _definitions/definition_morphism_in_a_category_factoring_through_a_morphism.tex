\begin{definition} \label{definition:morphism_in_a_category_factoring_through_a_morphism}
    Let $\calC$ be a category. Let $f: X \to Y$ be a morphism in $\calC$. Let $Z$ be an object of $\calC$.
    \begin{enumerate}
        \item We say that $f$ \hldef{factors through a morphism $g: Z \to Y$} if there exists a morphism $h: X \to Z$ such that $f = g \circ h$.
        \item We say that $f$ \hldef{factors through a morphism $h: X \to Z$} if there exists a morphism $g: Z \to Y$ such that $f = g \circ h$.
    \end{enumerate}
\end{definition}
