\begin{definition}[{\cite[5.2.11]{weibel}}, cf. {\cite[012U]{stacks-project}}] \label{definition:weak_convergence_to_a_graded_object_for_a_spectral_sequence_in_an_abelian_category}
    Let $\mathcal{A}$ be an \CrefAndHyperrefIfExist{definition:abelian_category}{abelian category}.

    \begin{enumerate}
        \item Let $E = \{E_{p,q}^r, d^r\}_{r \ge r_0}$ be a \CrefAndHyperrefIfExist{definition:spectral_sequence_in_an_abelian_category}{homological spectral sequence}, and let $H = \{H_n\}_{n \in \mathbb{Z}}$ be a \CrefAndHyperrefIfExist{definition:graded_object_in_a_category}{graded object} in $\mathcal{A}$.
        We say that $E$ \hldef{weakly converges to $H$}, denoted \hl{$E_{p,q}^r \Rightarrow H_{p+q}$}, if:
        \begin{enumerate}
            \item For every pair $(p,q)$, the term $E_{p,q}^r$ \CrefAndHyperrefIfExist{definition:stabilize_for_a_spectral_sequence_in_an_abelian_category_and_infinity_page}{stabilizes} to a limit $E_{p,q}^\infty$.
            \item Each object $H_n$ is equipped with an \CrefAndHyperrefIfExist{definition:filtration_on_an_object_of_a_category}{increasing filtration}
            $$F: \cdots \subseteq F_{p-1} H_n \subseteq F_p H_n \subseteq F_{p+1} H_n \subseteq \cdots \subseteq H_n$$
            such that there is an isomorphism:
            $$E_{p,q}^\infty \cong \text{Gr}_p H_{p+q} := \frac{F_p H_{p+q}}{F_{p-1} H_{p+q}}.$$
            (\Cref{definition:bigraded_object_associated_to_a_filtered_graded_object_in_an_abelian_category})
        \end{enumerate}

        \item Let $E = \{E_r^{p,q}, d_r\}_{r \ge r_0}$ be a \CrefAndHyperrefIfExist{definition:spectral_sequence_in_an_abelian_category}{cohomological spectral sequence}, and let $H = \{H^n\}_{n \in \mathbb{Z}}$ be a \CrefAndHyperrefIfExist{definition:graded_object_in_a_category}{graded object} in $\mathcal{A}$.

        We say that $E$ \hldef{weakly converges to $H$}, denoted \hl{$E_r^{p,q} \Rightarrow H^{p+q}$}, if:
        \begin{enumerate}
            \item For every pair $(p,q)$, the term $E_r^{p,q}$ \CrefAndHyperrefIfExist{definition:stabilize_for_a_spectral_sequence_in_an_abelian_category_and_infinity_page}{stabilizes} to a limit $E_\infty^{p,q}$.
            \item Each object $H^n$ is equipped with a \CrefAndHyperrefIfExist{definition:filtration_on_an_object_of_a_category}{decreasing filtration} 
            $$F: H^n \supseteq \cdots \supseteq F^{p-1} H^n \supseteq F^p H^n \supseteq F^{p+1} H^n \supseteq \cdots$$
            such that there is an isomorphism:
            $$E_\infty^{p,q} \cong \text{Gr}^p H^{p+q} := \frac{F^p H^{p+q}}{F^{p+1} H^{p+q}}.$$
            (\Cref{definition:bigraded_object_associated_to_a_filtered_graded_object_in_an_abelian_category})
        \end{enumerate}
    \end{enumerate}

    It should be noted that terminology for these notions may be very inconsistent in the literature.
\end{definition}
