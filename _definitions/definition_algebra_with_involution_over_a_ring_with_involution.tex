\begin{definition} \label{definition:algebra_with_involution_over_a_ring_with_involution}
    \begin{enumerate}
        \item 

        Let $(R, \sigma_R)$ be a \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{ring with involution}. 
        An \hldef{$R$-algebra with involution} is an \CrefAndHyperrefIfExist{definition:algebra_of_a_ring}{$R$-algebra $A$} 
        equipped with an \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{involution} $\sigma_A$, which may be called an \hldef{$R$-linear involution}, such that:
        \begin{equation*}
            \sigma_A(ra) = \sigma_A(a)\sigma_R(r) \quad \text{for all } r \in R, a \in A.
        \end{equation*}
        (\CrefAndHyperrefIfExist{theorem:equivalent_conditions_for_a_short_exact_sequence_in_an_additive_category_to_be_split}{Equivalently}, $\sigma_A(ar) = \sigma_R(r)\sigma_A(a)$).

        \item Let $R$ be a ring.

        An \hldef{$R$-algebra with $R$-linear involution} or simply an \hldef{$R$-algebra with involution} usually refers to an $R$-algebra with involution in the above sense where $R$ is considered as an involution ring equipped with the \CrefAndHyperrefIfExist{definition:trivial_involution_on_a_ring}{trivial involution}.
    \end{enumerate}


\end{definition}
