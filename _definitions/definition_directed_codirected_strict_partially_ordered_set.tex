\begin{definition} \label{definition:directed_codirected_strict_partially_ordered_set}
    Let $(P,<)$ be a \CrefAndHyperrefIfExist{definition:strict_partially_ordered_set}{strict partially ordered set}. 
    \begin{enumerate}
        \item $(P, <)$ is called a \hldef{directed partially ordered set} if for every pair $a,b \in P$, there exists $c \in P$ such that
        \[
        a < c \quad \text{and} \quad b < c.
        \]

        \item $(P, \leq)$ is called a \hldef{codirected partially ordered set} (or \hldef{downward directed poset}) if for every pair $a,b \in P$, there exists $d \in P$ such that
        \[
        d < a \quad \text{and} \quad d < b.
        \]
    \end{enumerate}

\end{definition}
