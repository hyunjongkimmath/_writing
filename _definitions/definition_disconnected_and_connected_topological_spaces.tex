\begin{definition} \label{definition:disconnected_and_connected_topological_spaces}
A \CrefAndHyperrefIfExist{definition:category_of_topological_spaces}{topological space} $X$ is \hldef{disconnected} if there exist two disjoint, non-empty \CrefAndHyperrefIfExist{definition:topological_space}{open sets} $U$ and $V$ such that $X = U \cup V$. The space $X$ is \hldef{connected} if it is not disconnected.
\end{definition}
