\begin{definition} \label{definition:piecewise_continuously_differentiable_path_from_a_closed_real_interval_to_the_complex_plane}
Let $[a, b] \subset \mathbb{R}$ be a \CrefAndHyperrefIfExist{definition:interval_in_the_real_line}{closed real interval}. 
A \CrefAndHyperrefIfExist{definition:path_and_loop_in_a_topological_space}{continuous path} $\gamma : [a, b] \to \mathbb{C}$ is called a \hldef{piecewise continuously differentiable path} if there exists a \CrefAndHyperrefIfExist{definition:partition_of_a_closed_interval_in_the_real_line}{partition} $a = t_0 < t_1 < \cdots < t_k = b$ such that:
\begin{enumerate}
  \item $\gamma$ is \CrefAndHyperrefIfExist{definition:C_k_morphism_between_C_k_manifolds}{continuously differentiable} on each open subinterval $(t_{i-1}, t_i)$ for $i = 1, \ldots, k$, and
  \item the \CrefAndHyperrefIfExist{definition:right_limit_and_left_limit_of_a_function_in_an_ordered_extended_metric_space}{one-sided derivatives} $\lim_{t \to t_{i-1}^+} \gamma'(t)$ and $\lim_{t \to t_i^-} \gamma'(t)$ exist (and are finite) for each $i = 1, \ldots, k$.
\end{enumerate}
The \CrefAndHyperrefIfExist{definition:derivative_of_a_real_valued_function_on_a_real_interval}{derivative} $\gamma'(t)$ is defined for all $t \in [a, b] \setminus \{t_1, \ldots, t_{k-1}\}$.
\end{definition}
