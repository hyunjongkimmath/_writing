\begin{definition}[{\cite[Definition 2.4.]{scholze_lag}}] \label{definition:solid_condensed_abelian_group}
    A \CrefAndHyperrefIfExist{definition:condensed_object_of_a_category}{condensed abelian group} is solid if for any \CrefAndHyperrefIfExist{definition:profinite_set}{profinite set} $S$ with a map $f: S \to M$, there is a unique map \hl{$\widetilde{f}: \mathbb{Z}[S]^\blacksquare \to M$} (\Cref{definition:solidification_of_polynomial_ring_of_profinite_set}) such that the composition
    \TODO{talk about yoneda embedding so that $S$ makes sense as a condensed set}
    $$S \to \mathbb{Z}[S]^\blacksquare \xrightarrow{\widetilde{f}} M$$
    is $f$.

    The \CrefAndHyperrefIfExist{definition:full_subcategory_of_a_category}{full subcategory} of \CrefAndHyperrefIfExist{definition:condensed_object_of_a_category}{$\operatorname{Cond}(\operatorname{Ab})$} consisting of solid objects may be denoted by \hl{$\operatorname{Solid}$} or \hl{$\operatorname{Sol}$}.
\end{definition}
