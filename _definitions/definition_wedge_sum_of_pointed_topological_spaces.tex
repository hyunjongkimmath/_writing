\begin{definition}[Wedge sum] \label{definition:wedge_sum_of_pointed_topological_spaces}
    Let $\{(X_i, x_i)\}_{i \in I}$ be a collection of \CrefAndHyperrefIfExist{definition:pointed_topological_space}{pointed topological spaces}. The \hldef{wedge sum of the collection}, denoted by \hl{$\bigvee_{i \in I} X_i$}, is the \CrefAndHyperrefIfExist{definition:quotient_of_a_topological_space_with_respect_to_an_equivalence_relations}{quotient} of the \CrefAndHyperrefIfExist{definition:disjoint_union_topology_of_topological_spaces}{disjoint union} $\coprod_{i \in I} X_i$ by the \CrefAndHyperrefIfExist{definition:equivalence_relation}{equivalence relation} that identifies all base points $\{x_i\}_{i \in I}$ to a single point.
    
    For a finite collection $(X_1, x_1), \dots, (X_n, x_n)$, the wedge sum is denoted \hl{$X_1 \vee \dots \vee X_n$}. It is the \CrefAndHyperrefIfExist{definition:product_and_coproduct_of_objects_in_a_category}{coproduct} in the \CrefAndHyperrefIfExist{definition:pointed_topological_space}{category of pointed topological spaces}.
\end{definition}
