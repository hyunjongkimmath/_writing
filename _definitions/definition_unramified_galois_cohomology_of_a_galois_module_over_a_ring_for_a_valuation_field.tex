\begin{definition} \label{definition:unramified_galois_cohomology_of_a_galois_module_over_a_ring_for_a_valuation_field}
    Let $(K,v)$ be a \CrefAndHyperrefIfExist{definition:valuation_on_a_field}{valuation field}. Let $K^{\text{sep}}$ be a \CrefAndHyperrefIfExist{definition:separable_closure_of_a_field}{separable closure} of $K$, and let $w$ be an \CrefAndHyperrefIfExist{definition:extension_of_a_field}{extension} of $v$ to $K^{\text{sep}}$. Write $G_K$ for $\Gal(K^{\text{sep}} / K)$ and $I_w$ for the \CrefAndHyperrefIfExist{definition:inertia_group_of_a_general_galois_extension_of_valuation_fields}{inertia subgroup} of $G_K$ for $(K^{\text{sep}},w)/(K,v)$.

    Let $R$ be a \CrefAndHyperrefIfExist{definition:topological_ring}{topological ring} and let $M$ be a \CrefAndHyperrefIfExist{definition:galois_module_over_a_topological_ring_of_a_field}{Galois module} for $K$ over $R$.

    The \hldef{unramified Galois cohomology groups}, denoted by notations such as \hl{$H^n_{\text{ur}}(K,M)$}, \hl{$H^n_{\text{unr}}(K,M)$}, \hl{$H^n_{\text{nr}}(K,M)$}, or \hl{$H^n_{\text{ur}}(K,M)_v$}, is defined as the kernel of the \CrefAndHyperrefIfExist{definition:restriction_map_of_continuous_group_cohomology_of_a_topological_group_on_a_topological_module}{restriction map}
    $$\operatorname{res}: H^n(G_K, M) \to H^n(I_w, M)$$
    \CrefIfExists{definition:galois_cohomology_of_a_galois_field_extension_with_coefficients_in_a_topological_galois_module_over_a_topological_ring}
    
    
\end{definition}
