\begin{definition} \label{definition:left_right_T_nilpotent_subset_of_a_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{ring}, not assumed commutative.  
    A subset $I \subseteq R$ is 
    \begin{enumerate}
        \item \hldef{left T-nilpotent} if for every sequence $\{a_1, a_2, a_3, \dots\}$ of elements in $I$, there exists an integer $n \geq 1$ such that the product $a_1 a_2 \dots a_n = 0$.
        \item \hldef{right T-nilpotent} if for every sequence $\{a_1, a_2, a_3, \dots\}$ of elements in $I$, there exists an integer $n \geq 1$ such that the product $a_n a_{n-1} \dots a_1 = 0$.
    \end{enumerate}
\end{definition}
