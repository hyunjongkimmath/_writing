\begin{definition}[Dualizing complex for Verdier duality on a noetherian scheme with integral coefficients] \label{definition:dualizing_complex_for_verdier_duality_in_the_bdd_constr_derived_category_of_adic_sheaves_with_integral_coefficients_on_a_finite_type_scheme_over_a_noetherian_scheme}
    Let $(R, \lambda)$ be a \CrefAndHyperrefIfExist{definition:complete_ring_with_respect_to_an_ideal}{complete} \CrefAndHyperrefIfExist{definition:discrete_valuation_ring}{DVR} such that $R/\lambda^n$ is finite for all $n \geq 1$. Let $S$ be a \CrefAndHyperrefIfExist{definition:locally_noetherian_and_noetherian_scheme}{noetherian scheme}. Consider sheaves on the (small) \'etale site of $S$.

    \begin{enumerate}
        \item The \hldef{dualizing complex for Verdier duality on $S$ with integral coefficients}, denoted by \hl{$\mathbb{D}_S^\bullet(R)$}, is the object of \CrefAndHyperrefIfExist{definition:derived_category_of_bounded_constructible_complexes_of_adic_sheaves_with_integral_coefficients_on_a_noetherian_scheme}{$\Dbc(X, R)$} given as the constant inverse system $(R/\lambda^n)_n$ where each $R/\lambda^n$ is regarded as an object of \CrefAndHyperrefIfExist{derived_category_of_cohomologically_constructible_complexes_of_sheaves_of_modules_of_a_sheaf_of_rings_on_a_topological_space_or_scheme}{$\Dbc(X, R/\lambda^n)$} concentrated in degree $0$.
        
        \item The \hldef{(relative) dualizing complex for Verdier duality on $X/S$ with integral coefficients}, denoted by \hl{$\mathbb{D}_{X/S}^\bullet(R)$}, is defined by
        $$\mathbb{D}_{X/S}^\bullet(R) = Rf^! \, \mathbb{D}_S^\bullet(R)$$
        \CrefIfExists{definition:exceptional_inverse_image_functor_on_bdd_const_derived_category_of_adic_sheaves_with_integral_coefficients_on_schemes_of_finite_type_over_a_a_noetherian_scheme_via_a_compactifiable_morphism}
    \end{enumerate}
\end{definition}
