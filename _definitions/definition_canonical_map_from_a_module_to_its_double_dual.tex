\begin{definition} \label{definition:canonical_map_from_a_module_to_its_double_dual}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{(not necessarily commutative) ring}. Let $M$ be a (left/right) $R$-module. 

    The \hldef{canonical (evaluation) map from $M$ to its double dual} refers to the morphism
    $$M \to (M^\vee)^{\vee} = \Hom_R(M^\vee, R) = \Hom_R(\Hom_R(M, R), R), \quad m \mapsto (f \mapsto f(m)).$$
\end{definition}
