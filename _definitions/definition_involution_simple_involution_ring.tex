\begin{definition} \label{definition:involution_simple_involution_ring}
    Let $(R,\sigma)$ be an \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{involution ring}. 
    It is said to be \hldef{involution simple}, \hldef{$*$-simple}, or \hldef{$\sigma$-simple} if the only \CrefAndHyperrefIfExist{definition:involution_stable_ideal_of_an_involution_ring}{involution stable two-sided ideals of $R$} are $\{0\}$ and $R$ itself.
\end{definition}
