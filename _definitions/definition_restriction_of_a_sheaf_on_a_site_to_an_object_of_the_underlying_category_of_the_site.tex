\begin{definition} \label{definition:restriction_of_a_sheaf_on_a_site_to_an_object_of_the_underlying_category_of_the_site}
    Let $(\calC, J)$ be a \CrefAndHyperrefIfExist{definition:grothendieck_topology_on_a_category_site_covering_sieve_topologically_generating_family}{site}, let $\calF$ be a \CrefAndHyperrefIfExist{definition:presheaf_on_a_category}{presheaf} on $\calC$ valued in some category, and let $U \in \Ob(\calC)$. 

    The \hldef{restriction of $\calF$ to $U$} is the sheaf \hl{$\calF|_U$} on the \CrefAndHyperrefIfExist{definition:site_induced_by_a_site_on_an_over_category}{induced site $(\calC/U, J/U)$} defined by 
    $$\calF|_U(f: V \to U) = \calF(V).$$
    More precisely,
    $$\calF|_U  = \calF \circ j_U^{\op}$$
    where $j_U: \calC/U \to \calC$ is the forgetful functor, which is a \CrefAndHyperrefIfExist{definition:continuous_functor_of_sites}{continuous} and \CrefAndHyperrefIfExist{definition:flat_functor_between_categories}{flat functor},  and $j_U^{\op}: (\calC/U)^{\op} \to \calC^{\op}$\CrefIfExists{definition:opposite_category_of_a_category} is the opposite functor.

    If $\calF$ is a sheaf on $(\calC, J)$, then $\calF|_U$ is a sheaf on $(\calC/U, J/U)$. 
\end{definition}
