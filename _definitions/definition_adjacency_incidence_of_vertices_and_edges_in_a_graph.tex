\begin{definition} \label{definition:adjacency_incidence_of_vertices_and_edges_in_a_graph}

Let $G$ be a graph as in \CrefAndHyperrefIfExist{definition:simple_graph_directed_graph_multigraph_weighted_graph}{simple graph or directed graph}.

\begin{enumerate}
    \item If $G$ is a \CrefAndHyperrefIfExist{definition:simple_graph_directed_graph_multigraph_weighted_graph}{simple graph}, then two vertices $u, v \in V(G)$ are called \hldef{adjacent in $G$} if $\{u,v\} \in E(G)$. A vertex $v \in V(G)$ and an edge $e \in E(G)$ are called \hldef{incident in $G$} if $v \in e$, that is, if $e = \{v,w\}$ for some $w \in V(G)$.

    \item If $D$ is a \CrefAndHyperrefIfExist{definition:simple_graph_directed_graph_multigraph_weighted_graph}{directed graph}, then two vertices $u, v \in V(D)$ are called \hldef{adjacent in $D$} if $(u,v) \in E(D)$ or $(v,u) \in E(D)$. A vertex $v$ is an \hldef{in-neighbor of $u$} if $(v,u) \in E(D)$, and an \hldef{out-neighbor of $u$} if $(u,v) \in E(D)$.
\end{enumerate}
\end{definition}
