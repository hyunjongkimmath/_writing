\begin{definition}[{\cite[Section 1]{ekedahl_af}}]  \label{definition:ekedahls_condition_A_and_B}
    Assume \Cref{context:ekedahl_af_topos_morphisms_to_and_from_Pro_N_S_to_S}.
   \begin{enumerate}
    \item We may say that $(S,R)$ \hldef{satisfies Ekedahl's condition $A$} if the following holds: there exists a class \hl{$S^{\text{gen}}$} of \CrefAndHyperrefIfExist{definition:generator_of_a_category}{generators of $S$} and an integer $N$ such that for all $T \in S^{\text{gen}}$, all $R/m$-\CrefAndHyperrefIfExist{definition:module_over_a_sheaf_of_rings_on_a_site}{modules} $M$, and $i > N$, we have $H^i(T,M) = 0$ (\Cref{definition:sheaf_cohomology_group_of_a_sheaf_of_modules_over_a_sheaf_of_rings_on_a_site}).

    \item We may say that $(S,R)$ \hldef{satisfies Ekedahl's condition $B$} if the following hold: 
    \begin{itemize}
        \item there is an integer $N$ such that for every $n \geq 1$ and locally on $(\calC, J)$, there is a \CrefAndHyperrefIfExist{definition:left_right_resolution_of_an_object_by_a_class_of_objects_in_an_abelian_category}{left resolution} $F_n^\bullet \to R/m^n$ of $R/m^n$ by a complex of (right) $R$-modules of finitely generated free objects such that $\pi^* F_n^\bullet \to \pi^*(R/m^n)$ is a resolution modulo \CrefAndHyperrefIfExist{definition:essentially_zero_inverse_sequence_essentially_bounded_complex_of_sheaves_of_abelian_groups_on_a_site}{essentially zero systems}, where $\pi: (\mathrm{Pro}_{\bbN}(S), R_\bullet) \to (S,R)$ is considered as a morphism of ringed topoi. 

        \item $m^n/m^{n+1}$ is, locally on $(\calC, J)$, of finite Tor-dimension over $R/m$. \TODO{finite Tor dimension}
    \end{itemize}

    % \item We may say that $(S,R)$ \hldef{satisfies Bhatt-Scholze's Ekedahl condition $B$} if the following hold:
    % \begin{itemize}
    %     \item $m$ 
    % \end{itemize}
    
   \end{enumerate}  

   Here are sufficient conditions for condition $A$ to be fulfilled:
   \begin{enumerate}
    \item When the topos is that of the \CrefAndHyperrefIfExist{definition:small_etale_site_of_a_scheme}{small \'etale site} $X_{\text{\'et}}$, where $X$ is the spectrum of an \CrefAndHyperrefIfExist{definition:algebraically_closed_field}{algebraically closed field}, a \CrefAndHyperrefIfExist{definition:local_field}{local field}, a \CrefAndHyperrefIfExist{definition:global_field}{global field}, or $\bbZ$, and $R$ is a \CrefAndHyperrefIfExist{definition:constant_sheaf_on_a_site_with_sheafification}{constant ring} $X_{\text{\'et}}$.

    \item Whenever condition $A$ holds for $X_{\text{\'et}}$ for a \CrefAndHyperrefIfExist{definition:scheme}{scheme} $X$, a constant ring $R$ on $X_{\text{'et}}$, and a two sided ideal $m \subseteq R$, it also holds for $(Y_{\text{\'et}}, R, m)$ for any scheme $Y$ of \CrefAndHyperrefIfExist{definition:finite_type_morphism_of_schemes}{finite type} over $X$.
   \end{enumerate}

   Here are sufficient conditions for condition $B$ to be fulfilled:
   \begin{enumerate}
    \item $R$ is a constant ring that is \CrefAndHyperrefIfExist{definition:noetherian_ring}{noetherian}, \CrefAndHyperrefIfExist{definition:commutative_ring}{commutative}, and \CrefAndHyperrefIfExist{definition:local_ring}{local} of \CrefAndHyperrefIfExist{definition:left_right_global_dimension_of_a_ring}{finite global dimension}.
    \item Whenever condition $A$ holds for $(S, R, m)$, it also holds for $(S, R[G], m[G])$ for a \CrefAndHyperrefIfExist{definition:group_object_in_a_category_with_a_final_object}{group} $G$ on $S$.
   \end{enumerate}

\end{definition}
