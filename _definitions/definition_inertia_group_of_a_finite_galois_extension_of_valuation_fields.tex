\begin{definition} \label{definition:inertia_group_of_a_finite_galois_extension_of_valuation_fields}
    Assume \Cref{context:extension_of_valuation_fields} and further assume that $L/K$ is a \CrefAndHyperrefIfExist{definition:degree_of_a_field_extension}{finite} \CrefAndHyperrefIfExist{definition:galois_extension_of_fields}{Galois extension}.

    There is a natural surjective homomorphism 
    $$D_w \to \Gal(k_w / k_v)$$
    on the residue fields. The \hldef{inertia group of $L/K$} is its kernel: 
    $$\hlin{I_w = \{\sigma \in D_w : \sigma(x) \equiv x \pmod{\mathfrak{m}_w} \text{ for all } x \in \mathcal{O}_w\}}$$
    \CrefIfExists{definition:decomposition_group_of_a_finite_galois_extension_of_valuation_fields}
\end{definition}
