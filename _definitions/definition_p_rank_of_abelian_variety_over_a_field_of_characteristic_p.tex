\begin{definition}[p-rank of an abelian variety over a field of characteristic p] \label{definition:p_rank_of_abelian_variety_over_a_field_of_characteristic_p}
    Let $A$ be an abelian variety of dimension $g$ over a field $k$ of characteristic $p > 0$. The \hldef{$p$-rank} (or \hldef{$f$-invariant}) of $A$ is the integer
    $$f(A) = \dim_{\mathbb{F}_p} A[p](\bar{k}),$$
    where $\bar{k}$ is an algebraic closure of $k$.

    The $p$-rank satisfies $0 \leq f(A) \leq g$. An abelian variety is ordinary if and only if $f(A) = g$, and it has maximal $p$-rank deficiency if $f(A) = 0$.
\end{definition}
