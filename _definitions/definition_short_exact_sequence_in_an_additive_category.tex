\begin{definition} \label{definition:short_exact_sequence_in_an_additive_category}
Let $\mathcal{A}$ be an \CrefAndHyperrefIfExist{definition:additive_category_preadditive_category}{additive category}. A sequence 
$$0 \to A \xrightarrow{f} B \xrightarrow{g} C \to 0$$ 
of morphisms in $\mathcal{A}$ is called a \hldef{short exact sequence} if the morphisms satisfy:

\begin{itemize}
  \item $f : A \to B$ is a \CrefAndHyperrefIfExist{definition:kernel_and_cokernel_of_a_morphism_in_a_pointed_category}{kernel of $g$},
  \item $g : B \to C$ is a \CrefAndHyperrefIfExist{definition:kernel_and_cokernel_of_a_morphism_in_a_pointed_category}{cokernel of $f$},
  % \item the sequence is \CrefAndHyperrefIfExist{definition:acyclic_complex_of_objects_in_an_abelian_category}{exact at} $B$, meaning \CrefAndHyperrefIfExist{definition:image_coimage_of_a_morphism_in_a_category}{$\mathrm{Im}(f) = \mathrm{Ker}(g)$}.
\end{itemize}

%   \item $g : B \to C$ is an \CrefAndHyperrefIfExist{definition:monomorphism_and_epimorphism_in_categories}{epimorphism} and is the \CrefAndHyperrefIfExist{definition:kernel_and_cokernel_of_a_morphism_in_a_pointed_category}{cokernel of $f$},
%   \item the sequence is \CrefAndHyperrefIfExist{definition:acyclic_complex_of_objects_in_an_abelian_category}{exact at} $B$, meaning \CrefAndHyperrefIfExist{definition:image_coimage_of_a_morphism_in_a_category}{$\mathrm{Im}(f) = \mathrm{Ker}(g)$}.
% \end{itemize}
% This means the sequence starts and ends with the zero object and is exact everywhere.
In other words, the sequences $0 \to A \xrightarrow{f} B \xrightarrow{g} C$ and $A \xrightarrow{f} B \xrightarrow{g} C \to 0$ are \CrefAndHyperrefIfExist{definition:left_right_exact_sequences_in_an_additive_category}{left and right exact} respectively.

%and is \CrefAndHyperrefIfExist{definition:acyclic_complex_of_objects_in_an_abelian_category}{exact everywhere}.
\end{definition}
