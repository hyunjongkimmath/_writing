    Let $(A,\sigma)$ be a \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{ring with involution}, and let $H \in M_n(A)$ be an \CrefAndHyperrefIfExist{definition:epsilon_hermitian_matrix_over_a_ring_with_involution}{$\varepsilon$-Hermitian matrix} for some $\varepsilon \in Z(A)$ with $\varepsilon \cdot \overline{\varepsilon} = 1$.

    An invertible matrix $U \in \operatorname{GL}_n(A)$\CrefIfExists{definition:general_linear_group_over_a_ring} is called a \hldef{unitary transformation for $H$} (or simply \hldef{$H$-unitary}) if
    $$U^* H U = H.$$
    Equivalently, $U$ is unitary for $H$ if and only if $U$ preserves $h_H$. 

    The set of all such matrices forms a group
    $$\hlin{U(H) = \{ U \in \operatorname{GL}_n(A) : U^* H U = H \}},$$
    called the \hldef{unitary group of $H$}. It is the subgroup of $\operatorname{GL}_n(A)$ that fixes $H$ under the \CrefAndHyperrefIfExist{definition:star_congruence_of_matrices_over_a_ring_with_involution}{$*$-congruence} action.
\end{definition}
