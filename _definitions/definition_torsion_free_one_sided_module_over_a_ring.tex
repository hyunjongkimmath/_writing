\begin{definition} \label{definition:torsion_free_one_sided_module_over_a_ring}
Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{ring}.
\begin{enumerate}
    \item A \CrefAndHyperrefIfExist{definition:bimodule_over_rings_module_over_a_ring}{left $R$-module} $M$ is said to be \hldef{(left) $R$-torsion-free} if for every \CrefAndHyperrefIfExist{definition:zero_divisor_of_a_ring}{regular} element $r \in R$ and every $m \in M$, $rm = 0$ implies $m = 0$.

    \item A \CrefAndHyperrefIfExist{definition:bimodule_over_rings_module_over_a_ring}{right $R$-module} $M$ is said to be \hldef{(right) $R$-torsion-free} if for every regular element $r \in R$ and every $m \in M$, $mr = 0$ implies $m = 0$.
\end{enumerate}
\end{definition}
