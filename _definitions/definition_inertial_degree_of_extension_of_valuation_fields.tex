\begin{definition} \label{definition:inertial_degree_of_extension_of_valuation_fields}
    Assume \Cref{context:extension_of_valuation_fields}. The \hldef{inertial degree index of $w$ over $v$}, denoted \hl{$f(w/v)$}, is the \CrefAndHyperrefIfExist{definition:degree_of_a_field_extension}{degree} of the \CrefAndHyperrefIfExist{definition:residue_field_of_a_ring_at_a_point}{residue field} \CrefAndHyperrefIfExist{definition:extension_of_a_field}{extension}:
    $$f(w/v) = [k_w: k_v]$$
\end{definition}
