\begin{definition} \label{definition:global_field}
A \hldef{global field} is a field $K$ that is either:
\begin{itemize}
    \item a \CrefAndHyperrefIfExist{definition:number_field}{number field}, i.e. a \CrefAndHyperrefIfExist{definition:finite_field_extension}{finite extension} of \CrefAndHyperrefIfExist{definition:field_of_rational_numbers}{$\bbQ$}, or
    \item a \CrefAndHyperrefIfExist{definition:global_function_field}{global function field}, i.e. a finite extension of the \CrefAndHyperref{definition:function_field_over_an_integral_domain_in_finitely_many_variables}{field of rational functions} $\mathbb{F}_q(t)$ in one variable over a \CrefAndHyperrefIfExist{definition:finite_field}{finite field} $\mathbb{F}_q$.
\end{itemize}
\end{definition}
