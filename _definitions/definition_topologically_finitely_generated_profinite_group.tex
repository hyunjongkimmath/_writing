\begin{definition} \label{definition:topologically_finitely_generated_profinite_group}
    \TODO{dense}
    Let $G$ be a \CrefAndHyperrefIfExist{definition:profinite_group}{profinite group}. 
    It is said to be \hldef{topologically finite generated} if there exists a finite subset $S \subseteq G$ that topologically generates $G$, i.e. $\overline{\langle S \rangle} = G$ or equivalently such that the \CrefAndHyperrefIfExist{definition:subgroup_of_a_group_generated_by_elements}{subgroup algebraically generated by $S$} is dense in $G$.
\end{definition}
