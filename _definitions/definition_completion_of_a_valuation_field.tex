\begin{definition} \label{definition:completion_of_a_valuation_field}
    Let $(K,v)$ be a \CrefAndHyperrefIfExist{definition:valuation_on_a_field}{valuation field}. 

    The \hldef{completion of $K$ at $v$}, often denoted \hl{$K_v$}, is the \CrefAndHyperrefIfExist{definition:completion_of_an_extended_metric_space_valued_in_ordered_commutative_monoid_with_adjoined_infinity}{completion of $K$ with respect to} the \CrefAndHyperrefIfExist{definition:metric_induced_by_an_absolute_value_on_a_commutative_ring}{metric induced by $|\cdot|_v$}. It is a \CrefAndHyperrefIfExist{definition:complete_valuation_field}{complete valued field}.
\end{definition}
