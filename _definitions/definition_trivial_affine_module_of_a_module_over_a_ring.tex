\begin{definition} \label{definition:trivial_affine_module_of_a_module_over_a_ring}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:ring}{ring}. Let $A$ be an $R$-module. 
    The \hldef{trivial affine module} \hl{$A$} is the \CrefAndHyperrefIfExist{definition:affine_module_over_a_ring}{affine module} whose underlying set is $A$ and whose affine module structure is given as follows: for an element $a$ of $A$ as an affine module and for an element $b$ of $B$ as a module, the element $a+b$ of $A$ as an affine module is the element $a+b$ of $A$ as a module.
\end{definition}
