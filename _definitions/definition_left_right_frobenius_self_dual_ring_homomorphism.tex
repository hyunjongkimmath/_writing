\begin{definition} \label{definition:left_right_frobenius_self_dual_ring_homomorphism}
    Let $\varphi: R \to A$ be a \CrefAndHyperrefIfExist{definition:ring_homomorphism}{ring homomorphism}. 
    We say that $\varphi$ is a \hldef{left (resp. right) Frobenius ring homomorphism} or \hldef{left self-dual} if there exists a left (resp. right) $A$-module isomorphism
    $$\psi: A \xrightarrow{\cong} A^\vee = \Hom_{{}_R \Mod}(A,R).$$
    (resp.
    $$\psi: A \xrightarrow{\cong} A^\vee = \Hom_{\Mod_{R}}(A,R).$$
    )
    \CrefIfExists{definition:dual_of_a_left_right_two_sided_module} 
\end{definition}
