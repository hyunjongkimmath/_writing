\begin{definition} \label{definition:factorial_of_a_nonnegative_integer}
    For each nonnegative integer $n \in \mathbb{Z}_{\geq 0}$, the \hldef{factorial of $n$}, denoted by \hl{$n!$}, is the element of $\mathbb{Z}$ \CrefAndHyperrefIfExist{definition:recursive_function_on_naturals}{defined recursively} by
    \begin{align*}
        0! &:= 1, \\
        n! &:= n \cdot (n-1)! \quad\text{for } n \geq 1.
    \end{align*}
    Equivalently, $n! = 1 \cdot 2 \cdots n$ for $n \geq 1$.
\end{definition}
