\begin{definition}[{\cite[Definition 2.3.]{clausen_scholze_cmcg}}] \label{definition:kappa_small_profinite_set_for_a_cardinal}
    Let $\kappa$ be a \CrefAndHyperrefIfExist{definition:cardinal_number_and_cardinality_of_a_set}{cardinal}.
    A \CrefAndHyperrefIfExist{definition:profinite_set}{profinite set} $S$ is called \hldef{$\kappa$-small} if the set of \CrefAndHyperrefIfExist{definition:clopen_subset_of_a_topological_space}{clopen subsets} of $S$ has cardinality $< \kappa$. The \CrefAndHyperrefIfExist{definition:full_subcategory_of_a_category}{full subcategory} of $\kappa$-small profinite sets is denoted \hl{$\operatorname{Prof}_{\kappa} \subset \text { Prof }$} (\Cref{definition:category_of_profinite_sets}), and this \CrefAndHyperrefIfExist{definition:category}{category} is equipped with the \CrefAndHyperrefIfExist{definition:grothendieck_topology_on_a_category_site_covering_sieve_topologically_generating_family}{Grothendieck topology} which is the \CrefAndHyperrefIfExist{}{restriction} of \CrefAndHyperrefIfExist{definition:grothendieck_topology_on_the_category_of_profinite_sets}{the one} on $\operatorname{Prof}$ to $\operatorname{Prof}_{\kappa}$.

    It should be noted that $\operatorname{Prof}_{\kappa}$ is an \CrefAndHyperrefIfExist{definition:small_site}{essentially small site}, unlike $\operatorname{Prof}$.
\end{definition}
