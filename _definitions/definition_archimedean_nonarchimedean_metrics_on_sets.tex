% ARCHIMEDEAN AND NONARCHIMEDEAN METRICS - Definitions of archimedean and non-archimedean metrics
\begin{definition}\label{definition:archimedean_and_nonarchimedean_metrics_on_sets}
    Let $X$ be a set.

    \begin{enumerate}
        \item A \CrefAndHyperrefIfExist{definition:extended_metric_on_a_set}{metric} $d$ on $X$ is called a \hldef{non-Archimedean metric} (or \hldef{ultrametric}) if it satisfies the \hldef{ultrametric inequality}: for all $x, y, z \in X$,
        $$
            d(x, z) \leq \max\{d(x, y), d(y, z)\}.
        $$
        A metric space $(X, d)$ equipped with a non-Archimedean metric is called a \hldef{non-Archimedean metric space} (or \hldef{ultrametric space}).
    \end{enumerate}

\end{definition}
