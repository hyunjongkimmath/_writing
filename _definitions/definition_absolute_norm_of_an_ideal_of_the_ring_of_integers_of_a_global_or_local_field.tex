\begin{definition} \label{definition:absolute_norm_of_an_ideal_of_the_ring_of_integers_of_a_global_or_local_field}
    Let $K$ be a \CrefAndHyperrefIfExist{definition:global_field}{global field} or a \CrefAndHyperrefIfExist{definition:local_field}{local field}, let $\calO_K$ be its \CrefAndHyperrefIfExist{definition:ring_of_integers_of_a_global_or_local_ring}{ring of integers}, and let $I \subseteq \calO_K$ be a nonzero \CrefAndHyperrefIfExist{definition:ideal_of_a_ring}{ideal}.  
    The \hldef{(absolute) norm of $I$} is defined as the \CrefAndHyperrefIfExist{definition:cardinal_number_and_cardinality_of_a_set}{cardinality} of $R/I$, which is always \CrefAndHyperrefIfExist{definition:countable_finite_uncountable_sets}{finite}.
\end{definition}
