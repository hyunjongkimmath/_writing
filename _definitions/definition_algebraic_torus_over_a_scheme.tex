\begin{definition} \label{definition:algebraic_torus_over_a_scheme}
    Let $S$ be a \CrefAndHyperrefIfExist{definition:scheme}{scheme}. An \hldef{(algebraic) torus over $S$} is an \CrefAndHyperrefIfExist{definition:algebraic_group_scheme_over_a_scheme}{algebraic group scheme} $\pi\colon T \to S$ such that there exists a \CrefAndHyperrefIfExist{definition:full_and_faithful_functor_between_locally_small_categories}{faithfully flat} morphism $S' \to S$ for which the \CrefAndHyperrefIfExist{definition:base_change_of_a_morphism_in_a_category_by_a_morphism}{base change} $T_{S'} = T \times_S S'$ is isomorphic as an \CrefAndHyperrefIfExist{definition:algebraic_group_scheme_over_a_scheme}{algebraic group} over $S'$ to $(\mathbb{G}_{m,S'})^n$ for some nonnegative integer $n$. The integer $n$ is called the \hldef{rank of $T$}.
\end{definition}
