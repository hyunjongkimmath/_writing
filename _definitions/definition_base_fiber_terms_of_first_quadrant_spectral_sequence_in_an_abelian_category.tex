\begin{definition} \label{definition:base_fiber_terms_of_first_quadrant_spectral_sequence_in_an_abelian_category}
Let $\mathcal{A}$ be an \CrefAndHyperrefIfExist{definition:abelian_category}{abelian category}. 

\begin{enumerate}
    \item Let $E = \{E^r, d^r\}$ be a \CrefAndHyperrefIfExist{definition:first_quadrant_spectral_sequence}{first quadrant} \CrefAndHyperrefIfExist{definition:spectral_sequence_in_an_abelian_category}{homological spectral sequence} (meaning $E^r_{p,q} = 0$ if $p < 0$ or $q < 0$).
    \begin{enumerate}
        \item The objects $\{E^r_{p,0}\}_{p \ge 0}$ are called the \hldef{base terms of the spectral sequence}.
        \item The objects $\{E^r_{0,q}\}_{q \ge 0}$ are called the \hldef{fiber terms of the spectral sequence}.
    \end{enumerate}

    \item Let $E = \{E_r, d_r\}$ be a first quadrant \CrefAndHyperrefIfExist{definition:spectral_sequence_in_an_abelian_category}{cohomological spectral sequence} (meaning $E_r^{p,q} = 0$ if $p < 0$ or $q < 0$).
    \begin{enumerate}
        \item The objects $\{E_r^{p,0}\}_{p \ge 0}$ are called the \hldef{base terms of the spectral sequence}.
        \item The objects $\{E_r^{0,q}\}_{q \ge 0}$ are called the \hldef{fiber terms of the spectral sequence}.
    \end{enumerate}
\end{enumerate}
\end{definition}
