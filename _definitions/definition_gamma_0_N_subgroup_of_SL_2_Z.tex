\begin{definition}[Congruence subgroup $\Gamma_0(N)$ of $\mathrm{SL}_2(\mathbb{Z})$] \label{definition:gamma_0_N_subgroup_of_SL_2_Z}
    Let $N$ be a positive integer. The \hldef{congruence subgroup $\Gamma_0(N)$} is the \CrefAndHyperrefIfExist{definition:subgroup_of_a_group}{subgroup} of $\mathrm{SL}_2(\mathbb{Z})$ defined by
    $$\hlin{\Gamma_0(N) = \left\{ \begin{pmatrix} a & b \\ c & d \end{pmatrix} \in \mathrm{SL}_2(\mathbb{Z}) \;\middle|\; c \equiv 0 \pmod{N} \right\}.}$$

    The subgroup $\Gamma_0(N)$ contains \CrefAndHyperref{definition:gamma_1_N_subgroup_of_SL_2_Z}{$\Gamma_1(N)$} as a subgroup. There is a surjective homomorphism
    $$\Gamma_0(N) / \Gamma_1(N) \twoheadrightarrow (\mathbb{Z}/N\mathbb{Z})^\times$$
    given by the map sending $\begin{pmatrix} a & b \\ c & d \end{pmatrix}$ to $a \pmod{N}$.

    % The inclusion chain of congruence subgroups is
    % $$$\Gamma(N)$ \subseteq \Gamma_1(N) \subseteq \Gamma_0(N) \subseteq \mathrm{SL}_2(\mathbb{Z}),$$
    % where $\Gamma(N)$ is the principal congruence subgroup of level $N$.

    % The subgroup $\Gamma_0(N)$ is used in the construction of the modular curve $X_0(N)$.
\end{definition}
