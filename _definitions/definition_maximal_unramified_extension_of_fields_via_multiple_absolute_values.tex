\begin{definition} \label{definition:maximal_unramified_extension_of_fields_via_multiple_absolute_values}
    Let $K$ be a \CrefAndHyperrefIfExist{definition:field}{field}. Fix an \CrefAndHyperrefIfExist{definition:algebraic_element_minimal_polynomial_of_an_algebraic_element_algebraic_field_extension_transcendental_element_field_extension}{algebraic closure} $\overline{K}$ of $K$.

    \begin{enumerate}
        \item Let $S$ be a set of (\CrefAndHyperrefIfExist{definition:equivalent_absolute_values_on_a_field}{equivalence classes} of) \CrefAndHyperrefIfExist{definition:absolute_value_on_a_ring_valued_in_an_ordered_semiring}{absolute values} on $K$. 
        The \hldef{maximal unramified extension of $K$ over $S$ in $\overline{K}$} is the subextension denoted by notations such as \hl{$K^{\text{unr, S}}$}, \hl{$K^{\text{ur}, S}$} which is the compositum of all subextensions $L$ in $\overline{K}$ of finite degree over $K$ such that $L$ is \CrefAndHyperrefIfExist{definition:unramified_extension_of_absolute_valued_fields_general}{unramified over $S$}.

        \item The \hldef{maximal unramified extension of $K$ in $\overline{K}$} is the maximal unramified extension of $K$ over $S$ in $\overline{K}$ in the above sense where $S$ is the set of all \CrefAndHyperrefIfExist{definition:archimedean_and_nonarchimedean_metrics_on_sets}{nonarchimedean} \CrefAndHyperrefIfExist{definition:absolute_value_on_a_field}{absolute values} of $K$.
    \end{enumerate}
\end{definition}
