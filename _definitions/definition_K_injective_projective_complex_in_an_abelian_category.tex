\begin{definition} \label{definition:K_injective_projective_complex_in_an_abelian_category}
    Let $\calA$ be an abelian caetgory. 
    A chain complex $A^\bullet$ of objects in $\calA$ is called \hldef{$K$-projective} (resp. \hldef{$K$-injective}) if for every \CrefAndHyperrefIfExist{definition:acyclic_complex_of_objects_in_an_abelian_category}{acyclic complex} $X^\bullet$ of objects in $\calA$, the complex of abelian groups \CrefAndHyperref{definition:internal_hom_complex_of_chain_complexes_in_an_additive_category}{$\Hom^\bullet(A^\bullet, X^\bullet)$} (resp. $\Hom^\bullet(X^\bullet, A^\bullet)$) is acyclic.

    \TextIfExists{definition:K_flat_complex_in_an_abelian_category_with_respect_to_a_biadditive_functor}{
    Equivalently, $A^\bullet$ is $K$-injective if and only if it is \CrefAndHyperref{definition:K_flat_complex_in_an_abelian_category_with_respect_to_a_biadditive_functor}{$K$-flat with respect to} $\Hom$ on the left for product total complexes and $A^\bullet$ is $K$-projective if and only if it is $K$-flat with respect to $\Hom$ on the right for product total complexes.
    }
\end{definition}
