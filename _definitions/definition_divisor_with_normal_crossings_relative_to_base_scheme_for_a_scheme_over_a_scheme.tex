\begin{definition}[Divisor with normal crossings relative to base scheme for a scheme over a scheme] \label{definition:divisor_with_normal_crossings_relative_to_base_scheme_for_a_scheme_over_a_scheme}
Let $S$ be a \CrefAndHyperrefIfExist{definition:scheme}{base scheme}, let $X$ be an $S$-scheme, and let $D$ be an effective divisor on $X$.
\TODO{effective divisor in what sense}

\textbf{(Strictly with normal crossings.)} We say that $D$ is \hldef{strictly with normal crossings relative to the base scheme $S$ for the $S$-scheme $X$} if there exists a finite family $(f_i)_{i \in I}$ of global sections in $\Gamma(X, \mathcal{O}_X)$ such that
$$D = \sum_{i \in I} \operatorname{div}(f_i)$$
and the following condition holds: for every point $x$ in the support $\Supp(D)$ of $D$, let $I(x) = \{ i \in I \mid f_i(x) = 0 \}$ be the set of indices vanishing at $x$. Then:
\begin{enumerate}
    \item $X$ is \CrefAndHyperrefIfExist{definition:smooth_morphism_of_schemes}{smooth over $S$} at $x$, and
    \item the closed subscheme $V((f_i)_{i \in I(x)})$ defined by the \CrefAndHyperrefIfExist{definition:ideal_generated_by_a_subset_of_a_ring}{ideal generated by} $\{ f_i \}_{i \in I(x)}$ is an \CrefAndHyperrefIfExist{definition:open_and_closed_subscheme_of_a_scheme}{closed subscheme} of $X$ that is smooth over $S$ of \CrefAndHyperrefIfExist{definition:codimension_of_an_irreducible_closed_subset_in_a_topological_space}{codimension} $\operatorname{card}(I(x))$.
\end{enumerate}

\textbf{(With normal crossings.)} We say that $D$ has \hldef{normal crossings relative to the base scheme $S$ for the $S$-scheme $X$} if, locally on $X$ for the étale topology, it is strictly with normal crossings relative to the base scheme $S$ for the $S$-scheme $X$. More precisely: for every point $x \in X$, there exists an étale morphism $U \to X$ from a scheme $U$ and a point $\tilde{x} \in U$ mapping to $x$ such that the pullback of $D$ to $U$ is strictly with normal crossings relative to the base scheme $S$ for the $S$-scheme $X$.
\end{definition}
