\begin{example}
    Here is an example of a ``boring'' category:
    \begin{enumerate}
        \item There is only one object, say $X$.
        \item There is only one morphism, the identity $\id_X: X \to X$.
    \end{enumerate}
    The composed morphism $\id_X \circ \id_X$ is then just $\id_X$, and associativity automatically holds.
\end{example}

\begin{example}
    Any \CrefAndHyperrefIfExist{definition:partially_ordered_set}{poset} $(P,\leq)$ induces a \CrefAndHyperrefIfExist{definition:category}{category} --- let the objects be the elements of the set $P$, and let the morphisms/arrows be given as follows: there is a unique arrow $a \to b$ whenever $a \leq b$. Composition of arrows works as follows: given arrows $a \to b$ and $b \to c$, the composed arrow $a \to b \to c$ will be the unique arrow $a \to c$ corresponding to $a \leq c$. See \Cref{lemma:posets_correspond_to_small_filtered_thin_categories}
\end{example}

    \begin{enumerate}

        \item ``forgetful functors''; some \CrefAndHyperrefIfExist{definition:forgetful_functor_between_categories}{forgetful functors} include 
        \begin{enumerate}
            \item The forgetful functor $F:\mathbf{Grp} \to \mathbf{Sets}$\CrefIfExists{definition:category_of_groups_of_abelian_groups}\CrefIfExists{definition:category_of_sets} sending a \CrefAndHyperrefIfExist{definition:group}{group} $G$ to the underlying set of $G$, and sending a \CrefAndHyperrefIfExist{definition:group_homomorphism}{group homomorphism} $G_1 \to G_2$ to the \CrefAndHyperrefIfExist{definition:function_of_sets}{set function} $G_1 \to G_2$.             \end{align*}

        \item The forgetful functor $F: \mathbf{Ab} \to \mathbf{Grp}$ \CrefIfExists{definition:category_of_groups_of_abelian_groups} sending an abelian group $A$ to itself, and sending a group homomorphism $A_1 \to A_2$ to itself.
        \end{enumerate}

    \item ``Free'' functors
    \begin{enumerate}
        \item There is a functor $\Sets \to \mathbf{Grp}$ sending a set $S$ to the \CrefAndHyperrefIfExist{definition:free_group_generated_by_a_set}{free group} $\langle S \rangle$ generated by $S$. The functor sends the morphism $f:S_1 \to S_2$ of sets, to the unique \CrefAndHyperrefIfExist{definition:group_homomorphism}{group homomorphism} $\langle S_1 \rangle \to \langle S_2 \rangle$ given by sending $s \in S_1$ to $f(s) \in S_2$.

        \item Similarly, there is a functor $\Sets \to \mathbf{Ab}$ sending a set $S$ to the \CrefAndHyperrefIfExist{definition:free_abelian_group_generated_by_a_set}{free abelian group} $\bbZ S$ generated by $S$. The functor sends the morphism $f:S_1 \to S_2$ of sets, to the unique \CrefAndHyperrefIfExist{definition:group_homomorphism}{group homomorphism} $\bbZ S_1 \to \bbZ S_2$ given by sending $s \in S_1$ to $f(s) \in S_2$.

    \end{enumerate}

    \item The fundamental group functor $\Top_\bullet \to \mathbf{Grp}$\CrefIfExists{definition:pointed_topological_space}; given a \CrefAndHyperrefIfExist{definition:pointed_topological_space}{pointed topological space} $(X,x)$, its associated \CrefAndHyperrefIfExist{definition:homotopy_groups_of_a_pointed_topological_space}{fundamental group} $\pi_1(X,x)$ is the group of homotopy classes of loops $\gamma: [0,1] \to X$. Given a morphism $(X,x) \to (Y,y)$ of pointed topological spaces, there is an (functorially) induced morphism $f: \pi_1(X,x) \to \pi_1(Y,Y)$ sending the homotopy class $[\gamma]$ of the loop $\gamma: [0,1] \to X$ to the homotopy class $[f \circ \gamma]$.
    \end{enumerate}
\end{example}

\begin{example} \label{example:examples_of_contravariant_functors}
    Here are some examples of \CrefAndHyperrefIfExist{definition:functor_between_categories}{contravariant functors}.
    \begin{enumerate}
        \item The dual of a vector space: given a \CrefAndHyperrefIfExist{definition:vector_space_over_a_field}{vector space} $V$ over a field $k$, the \CrefAndHyperrefIfExist{definition:dual_of_a_left_right_two_sided_module}{dual} is defined as $V^\vee \coloneq \Hom_k(V, k)$. The assignment $V \mapsto V^\vee$ specifies a contravariant functor $\VectorSpaces_k \to \VectorSpaces_k$ --- given a \CrefAndHyperrefIfExist{definition:morphism_of_vector_spaces}{linear map} $f:V_1 \to V_2$, there is an induced linear map $V_2^\vee \to V_1^\vee$ given by sending the linear map $\phi: V_2 \to k$, which is an element of $V_2^\vee$, to the linear map $V_2 \circ f V_1 \to k$, which is an element of $V_1^\vee$. 


        \item A \CrefAndHyperrefIfExist{definition:representable_functor_on_a_locally_small_category}{representable functor}: given any \CrefAndHyperrefIfExist{definition:locally_small_category}{locally small category} $\calC$ and any object $X$ of $\calC$, there is a contravariant functor $h_X: \calC^{\op} \to \Sets, T \mapsto \Hom_{\calC}(T,X)$; given a morphism $f:T_1 \to T_2$ in $\calC$, there is an induced map $h_X(T_2) \to h_X(T_1)$ of sets, i.e. a map $\Hom_\calC(T_2, X) \to \Hom_\calC(T_1, X)$, given by $\phi \mapsto \phi \circ f$.

        \item There is the contravariant power set functor $\calP: \Sets^{\op} \to \Sets$ that sends a set $S$ to its \CrefAndHyperrefIfExist{definition:power_set_of_a_set}{power set} $\calP(S)$ and that sends a set morphism $f: S \to T$ to the set morphism $f^*: \calP(T) \to \calP(S)$ given by $B \mapsto f^{-1}(B)$. 
        
        % Given a \CrefAndHyperrefIfExist{definition:continuous_map_of_topological_spaces}{continuous map} $f: X \to Y$ of \CrefAndHyperrefIfExist{definition:topological_space}{topological spaces}, there is a ``preimage functor'' $f^{-1}: \operatorname{Top}(Y) \to \operatorname{Top}

        \item The functor of continuous functions: let $\Top$ be the category of topological spaces and $\mathbb{R}\text{-}\operatorname{Alg}$ the category of \CrefAndHyperrefIfExist{definition:algebra_of_a_ring}{$\mathbb{R}$-algebras}. The assignment $X \mapsto C(X, \mathbb{R})$ of a space to its algebra of continuous real-valued functions is a contravariant functor. For any continuous map $g: X \to Y$, the induced algebra homomorphism $g^*: C(Y, \mathbb{R}) \to C(X, \mathbb{R})$ is given by the pullback $g^*(\phi) = \phi \circ g$ for $\phi \in C(Y, \mathbb{R})$.


    \end{enumerate}
\end{example}





\subsection{Natural transformation}

One overarching philosophy in various categories is that we only really care about objects ``up to equivalence'';  intuitively, we consider objects to be equivalent when they are \CrefAndHyperrefIfExist{definition:isomorphism_in_a_category}{isomorphic}. Similarly, we only really care about categories ``up to equivalence'' as well; there is a notion of \CrefAndHyperrefIfExist{definition:equivalence_of_categories}{equivalence} between categories. To define it, we first need to define the notion of \CrefAndHyperrefIfExist{definition:natural_transformation_between_functors_between_categories}{natural transformations} between functors --- a natural transformation is like a ``morphism'' between functors in a sense. 

\input{../_definitions/definition_natural_transformation_between_functors_between_categories.tex}

\begin{example}
    Let $k$ be a \CrefAndHyperrefIfExist{definition:field}{field}. Note that there is a ``double dual'' functor $\left((-)^{\vee} \right)^\vee:\VectorSpaces_k \to \VectorSpaces_k$ given by $V \mapsto (V^{\vee})^\vee$\CrefIfExists{definition:dual_of_a_left_right_two_sided_module}. \CrefAndHyperref{example:examples_of_contravariant_functors}{Recall that} $V \mapsto V^\vee$ is a \CrefAndHyperrefIfExist{definition:functor_between_categories}{contravariant functor}, so the double dual functor is covariant. Recall that there is also the \CrefAndHyperrefIfExist{definition:identity_functor_on_a_category}{identity functor} $\id_{\VectorSpaces_k}: \VectorSpaces_k \to \VectorSpaces_k$ given by $V \mapsto V$.  

    For general vector spaces $V$, there is an injective \CrefAndHyperrefIfExist{definition:morphism_of_vector_spaces}{$k$-linear map}
    $$\alpha_V: V \to (V^{\vee})^\vee, \quad v \mapsto v^*, $$
    where $v^*: V^\vee \to k$ is the $k$-linear map given by $\phi \mapsto \phi(v)$. We note that if $V$ is \CrefAndHyperrefIfExist{definition:basis_and_dimension_of_a_vector_space_over_a_field}{finite dimensional}, then $\alpha_V$ is an isomorphism. In fact, $\alpha$ is a natural transformation $\id_{\VectorSpaces_k} \Rightarrow \left( (-)^\vee \right)^\vee$ --- one should verify that, for every $k$-linear map $f: V_1 \to V_2$, the following diagram commutes:
    \begin{center}
    \begin{tikzcd}
        V_1 \arrow[r, "f"] \arrow[d, "\alpha_{V_1}"']
        & V_2 \arrow[d, "\alpha_{V_2}"] \\
        (V_1^\vee)^\vee \arrow[r, "(f^\vee)^\vee"']
        & (V_2^\vee)^\vee 
    \end{tikzcd}
    \end{center}
\end{example}

\begin{example}
    Let $\operatorname{CRing}$ be the \CrefAndHyperrefIfExist{definition:category_of_rings}{category of commutative rings} and $\operatorname{Group}$ the \CrefAndHyperrefIfExist{definition:category_of_groups_of_abelian_groups}{category of groups}. For a fixed $n \ge 1$, we have a \CrefAndHyperrefIfExist{definition:functor_between_categories}{covariant functor} $\GL_n: \operatorname{CRing} \to \operatorname{Group}$ assigning a ring $R$ to the general linear group $\GL_n(R)$, and a functor $(\cdot)^\times: \operatorname{CRing} \to \operatorname{Group}$ assigning a ring to its \CrefAndHyperrefIfExist{definition:unit_of_a_ring}{group of units}. 
    
    The determinant $\det: \GL_n \Rightarrow (\cdot)^\times$ is a \CrefAndHyperrefIfExist{definition:natural_transformation_between_functors_between_categories}{natural transformation}. Its component at a ring $R$ is the group homomorphism $\det_R: \GL_n(R) \to R^\times$. For any ring homomorphism $f: R \to S$, the following diagram commutes:
    \begin{center}
    \begin{tikzcd}
        \GL_n(R) \arrow[r, "\GL_n(f)"] \arrow[d, "\det_R"'] & \GL_n(S) \arrow[d, "\det_S"] \\
        R^\times \arrow[r, "f|_{R^\times}"'] & S^\times
    \end{tikzcd}
    \end{center}
    This commutativity expresses that the determinant is defined by the same universal formula (a polynomial in the matrix entries) regardless of the ring $R$, and is thus preserved by the "change of scalars" $f$.
\end{example}

\begin{example} \label{example:properties_of_functors}
    Below are examples illustrating various properties of functors:
    \begin{enumerate}
        \item \textbf{Faithful but not full:} The \CrefAndHyperrefIfExist{definition:forgetful_functor_between_categories}{forgetful functor} $U: \operatorname{Group} \to \operatorname{Set}$. It is \CrefAndHyperrefIfExist{definition:full_and_faithful_functor_between_locally_small_categories}{faithful} because group homomorphisms are distinct if they are distinct as functions. It is not full because not every function between groups is a group homomorphism (e.g., the constant function $x \mapsto g$ for $g \neq e$).

        \item \textbf{Full but not faithful:} The canonical functor $H: \mathsf{Top} \to h\mathsf{Top}$ from the \CrefAndHyperrefIfExist{definition:category_of_topological_spaces}{category of topological spaces} to the \CrefAndHyperrefIfExist{definition:homotopy_category_of_topological_spaces}{homotopy category} of topological spaces. It is the identity on objects and sends a continuous map $f$ to its homotopy class $[f]$. This is \CrefAndHyperrefIfExist{definition:full_and_faithful_functor_between_locally_small_categories}{full} by the definition of morphisms in $h\mathsf{Top}$, but not \CrefAndHyperrefIfExist{definition:full_and_faithful_functor_between_locally_small_categories}{faithful} because it identifies distinct but \CrefAndHyperrefIfExist{definition:homotopy_of_maps_of_topological_spaces_relative_to_a_subset}{homotopic maps} (e.g., any two paths in $\mathbb{R}^n$ with the same endpoints).

        \item \textbf{Fully faithful:} The inclusion functor $\iota: \operatorname{Ab} \to \operatorname{Group}$. It is \CrefAndHyperrefIfExist{definition:full_and_faithful_functor_between_locally_small_categories}{faithful} (it is an embedding) and it is \CrefAndHyperrefIfExist{definition:full_and_faithful_functor_between_locally_small_categories}{full} because any group homomorphism between two abelian groups is, by definition, a morphism in $\operatorname{Ab}$.

        \item \textbf{Essentially surjective:} Let $\mathbf{S}$ be the category whose objects are the standard sets $\underline{n} = \{0, \dots, n-1\}$ for each $n \in \mathbb{N}$ (and whose objects are the set functions between these sets), and $\mathbf{FinSet}$ be the category of all finite sets. The inclusion functor $I: \mathbf{S} \to \mathbf{FinSet}$ is essentially surjective because every finite set $X$ is isomorphic to $\underline{n}$ where $n$ is the cardinality of $X$.

        % \item \textbf{Essentially surjective:} Let $\operatorname{Vect}_k^{\text{fin}}$ be the category of finite-dimensional vector spaces and $\operatorname{Mat}_k$ the category whose objects are natural numbers $n \in \mathbb{N}$ and morphisms are $m \times n$ matrices. The functor $F: \operatorname{Vect}_k^{\text{fin}} \to \operatorname{Mat}_k$ sending $V$ to $\dim(V)$ is essentially surjective because every $n \in \mathbb{N}$ is the dimension of some vector space (namely $k^n$).
    \end{enumerate}
\end{example}

\subsubsection{Yoneda lemma}

The Yoneda lemma basically expresses the idea that an object of a (locally small) category is essentially determined by its morphisms to other objects.

\begin{example} 
The algebra of \CrefAndHyperrefIfExist{definition:C_k_morphism_between_C_k_manifolds}{smooth functions} $C^\infty(M)$ on a \CrefAndHyperrefIfExist{definition:C_k_manifold}{smooth manifold} $M$ is represented by the real line $\mathbb{R}$ in the category of smooth manifolds $\mathsf{Man}$.
\begin{enumerate}
    \item \textbf{The Functor:} Consider the covariant \CrefAndHyperrefIfExist{definition:representable_functor_on_a_locally_small_category}{representable functor} $h^{\mathbb{R}} = \operatorname{Hom}_{\mathsf{Man}}(-, \mathbb{R})$. For any manifold $M$, we have 
    $$\operatorname{Hom}_{\mathsf{Man}}(M, \mathbb{R}) = C^\infty(M),$$ 
    i.e. the representable functor $h^{\bbR}$ assigns $M$ to the set of all smooth maps $f: M \to \mathbb{R}$, which is exactly $C^\infty(M)$.
    
    \item \textbf{The Yoneda Correspondence:} By the \CrefAndHyperrefIfExist{theorem:yoneda_lemma_on_a_locally_small_category}{Yoneda Lemma}, there is a natural bijection between the points of $M$ and the natural transformations from $h^M$ to $h^{\mathbb{R}}$:
    \[ \operatorname{Nat}(h^M, h^{\mathbb{R}}) \cong h^{\mathbb{R}}(M) = C^\infty(M) \]
    
    \item \textbf{Geometric Interpretation:} This identifies a smooth function $\psi \in C^\infty(M)$ with a "natural" way to turn $M$-valued points of any other smooth manifold $X$ into real-valued points. Specifically, if we have a "probe" $\alpha: X \to M$, the function $\psi$ induces a map:
    \[ \alpha \mapsto \psi \circ \alpha \]
    mapping $\operatorname{Hom}(X, M)$ to $\operatorname{Hom}(X, \mathbb{R})$.
\end{enumerate}
This illustrates that the "global" algebraic data of a manifold (its functions) is entirely captured by its relationship to the "simplest" non-trivial manifold, $\mathbb{R}$.
\end{example}

\begin{example}[Geometric and Algebraic Limits/Colimits]
Many fundamental constructions in topology and algebra are characterized by universal properties of limits and colimits.

\begin{enumerate}
    \item \textbf{The $p$-adic Integers (Inverse Limit):} 
    The ring of $p$-adic integers $\mathbb{Z}_p$ is the limit of the inverse system of finite cyclic rings $\mathbb{Z}/p^n\mathbb{Z}$ with quotient maps:
    \[ \mathbb{Z}_p \cong \varprojlim_{n} \mathbb{Z}/p^n\mathbb{Z} = \{ (a_n)_{n \in \mathbb{N}} \in \prod \mathbb{Z}/p^n\mathbb{Z} \mid a_{n+1} \equiv a_n \pmod{p^n} \} \]
    

    \item \textbf{Absolute Galois Groups (Inverse Limit):} 
    The absolute Galois group $G_K = \operatorname{Gal}(\overline{K}/K)$ of a field $K$ is the inverse limit of the Galois groups of all finite Galois extensions $L/K$ contained in $\overline{K}$:
    \[ \operatorname{Gal}(\overline{K}/K) \cong \varprojlim_{L/K \text{ finite}} \operatorname{Gal}(L/K) \]
    This gives $G_K$ the structure of a profinite group.

    \item \textbf{The Cone of a Space (Pushout):} 
    Think of $X \times [0,1]$ as a cylinder. The map $X \to X \times [0,1]$ picks out the "top lid." The map $X \to \{*\}$ sends that entire lid to a single point. 
    The **pushout** $CX$ is the result of taking the cylinder and pinching the entire top lid into a single vertex.
    \begin{center}
    \begin{tikzcd}
    X \arrow[r, "{(x,1)}"] \arrow[d] & X \times [0,1] \arrow[d] \\
    \{*\} \arrow[r] & CX = (X \times [0,1]) / (X \times \{1\})
    \end{tikzcd}
    \end{center}

\end{enumerate}
\end{example}

\subsection{Homology and cohomology of a chain complex}

\input{../_definitions/definition_boundary_cycle_coboundary_cocyble_of_a_chain_cochain_complex.tex}
\input{../_definitions/definition_homology_and_cohomology_objects_for_a_chain_complex_in_an_abelian_category.tex}

\begin{example}[The Singular Complex of a Point]
The simplest non-zero geometric example is the chain complex of a single point $X = \{*\}$. In singular homology, there is exactly one map $\sigma_n: \Delta^n \to X$ for each $n \geq 0$. Thus, $C_n(X) \cong \mathbb{Z}$ for all $n \geq 0$. The boundary map $d_n: C_n \to C_{n-1}$ is given by the alternating sum of face maps, which results in:
\[
\dots \xrightarrow{\text{id}} \mathbb{Z} \xrightarrow{0} \mathbb{Z} \xrightarrow{\text{id}} \mathbb{Z} \xrightarrow{0} \mathbb{Z} \to 0
\]
The homology groups are $H_0(X) \cong \mathbb{Z}$ and $H_n(X) = 0$ for $n > 0$.
\end{example}

\begin{example}[The Algebraic Complex of a Ring Element]
Let $R$ be a ring and $x \in R$ be an element. We can define a chain complex $C_\bullet$ of $R$-modules concentrated in degrees 1 and 0:
\[
0 \to R \xrightarrow{d_1} R \to 0
\]
where $d_1(r) = xr$. Since $d_0 = 0$, the condition $d_0 \circ d_1 = 0$ is trivially satisfied. The homology groups are:
\begin{itemize}
    \item $H_1(C_\bullet) = \text{ker}(\cdot x) = \{r \in R \mid xr = 0\}$, the $x$-torsion of $R$.
    \item $H_0(C_\bullet) = \text{coker}(\cdot x) = R/xR$.
\end{itemize}
\end{example}

\begin{example}[Acyclic Complexes from Exact Sequences]
A chain complex $C_\bullet$ is said to be \textit{acyclic} if $H_n(C_\bullet) = 0$ for all $n$. Any short exact sequence $0 \to A \xrightarrow{f} B \xrightarrow{g} C \to 0$ in an abelian category $\calA$ can be viewed as an acyclic chain complex:
\[
\dots \to 0 \to A \xrightarrow{f} B \xrightarrow{g} C \to 0 \to \dots
\]
The condition $g \circ f = 0$ is guaranteed by the fact that $\text{im}(f) = \text{ker}(g)$, which simultaneously ensures that the homology at object $B$ is zero.
\end{example}

\begin{example}[The Simplicial Triangle]
Let $K$ be a 2-simplex (a triangle) with vertices $\{v_0, v_1, v_2\}$. There is an associated chain complex
$$\cdots \to 0 \to C_2 \to C_1 \to C_0 \to 0 \to \cdots$$
of abelian groups. The chain groups are:
\begin{itemize}
    \item $C_0 = \mathbb{Z}\langle v_0, v_1, v_2 \rangle$ (rank 3)
    \item $C_1 = \mathbb{Z}\langle [v_0,v_1], [v_1,v_2], [v_0,v_2] \rangle$ (rank 3)
    \item $C_2 = \mathbb{Z}\langle [v_0,v_1,v_2] \rangle$ (rank 1)
\end{itemize}
The boundary map $d_2: C_2 \to C_1$ is $d_2([v_0,v_1,v_2]) = [v_1,v_2] - [v_0,v_2] + [v_0,v_1]$. One can verify that $d_1 \circ d_2 = 0$ by calculating the boundaries of the resulting edges.
\end{example}

\begin{definition} 
An \hldef{abstract simplicial complex} $K$ is a collection of finite non-empty subsets of a set of vertices $V$, called \hldef{simplices}, such that for every $\sigma \in K$, every non-empty subset $\tau \subseteq \sigma$ (called a \hldef{face}) is also in $K$. An $n$-simplex is a simplex consisting of $n+1$ vertices. The \hldef{geometric realization} $|K|$ is the union of the corresponding geometric simplices in $\mathbb{R}^V$ equipped with the \hldef{weak topology} (the finest topology such that the inclusion of each simplex is continuous).
\end{definition}

\begin{definition} 
A \hldef{abstract simplicial pair $(K, L)$} consists of an abstract simplicial complex $K$ and a \hldef{subcomplex} $L \subseteq K$, which is a subset of $K$ that is itself an abstract simplicial complex. 
\end{definition}

\begin{definition} 
An \hldef{oriented simplex} is a simplex $\sigma \in K$ with a chosen ordering of its vertices, where two orderings are considered equivalent if they differ by an even permutation. We denote an oriented simplex by $[v_0, \dots, v_n]$.

Let $G$ be an abelian group. The \hldef{group of simplicial $n$-chains} $C_n^{\Delta}(K; G)$ is the free abelian group generated by the oriented $n$-simplices of $K$ with coefficients in $G$, subject to the relation $[v_{\sigma(0)}, \dots, v_{\sigma(n)}] = \text{sgn}(\sigma) [v_0, \dots, v_n]$.

For a abstract simplicial pair $(K, L)$, the \hldef{relative group of simplicial $n$-chains} \hl{$C_n^{\Delta}(K, L; G)$} is the quotient group:
$$C_n^{\Delta}(K, L; G) := C_n^{\Delta}(K; G) / C_n^{\Delta}(L; G)$$
\end{definition}

\begin{lemma} 
For an abstract simplicial complex $K$, the group $C_n^{\Delta}(K; G)$ is isomorphic to the cellular chain group of its geometric realization:
$$C_n^{\Delta}(K; G) \cong H_n(|K|^n, |K|^{n-1}; G)$$
where $|K|^n$ denotes the $n$-skeleton of the geometric realization.
\end{lemma}

\begin{definition} 
The \hldef{simplicial boundary map} \hl{$\partial_n: C_n^{\Delta}(K, L; G) \to C_{n-1}^{\Delta}(K, L; G)$} is the map induced on the quotient by:
$$\partial_n([v_0, \dots, v_n]) = \sum_{i=0}^n (-1)^i [v_0, \dots, \hat{v}_i, \dots, v_n]$$
This map satisfies $\partial_n \circ \partial_{n+1} = 0$, forming the \hldef{simplicial chain complex} $(C_{\bullet}^{\Delta}(K, L; G), \partial_{\bullet})$.
\end{definition}

\begin{definition} 
The \hldef{relative simplicial homology groups} \hl{$H_n^{\Delta}(K, L; G)$} are defined as:
$$H_n^{\Delta}(K, L; G) = \frac{\ker(\partial_n)}{\text{im}(\partial_{n+1})}$$
\end{definition}

\begin{theorem} 
For any abstract simplicial pair $(K, L)$ where $K$ is an abstract simplicial complex, there is a natural isomorphism:
$$H_n^{\Delta}(K, L; G) \cong H_n(|K|, |L|; G)$$
\end{theorem}

\begin{example}
The $n$-simplex $\Delta^n$ is an abstract simplicial complex consisting of $n+1$ vertices and all their subsets. Its boundary $\partial \Delta^n$ is the subcomplex of all proper subsets. The pair $(\Delta^n, \partial \Delta^n)$ provides a triangulation of the disk $D^n$ relative to its boundary $S^{n-1}$. The group $C_n^{\Delta}(\Delta^n, \partial \Delta^n; \mathbb{Z})$ is $\mathbb{Z}$, generated by the single $n$-simplex. Since all $(n-1)$-faces are in the subcomplex $\partial \Delta^n$, they are zero in the quotient, making the boundary map zero. Thus $H_n^{\Delta}(\Delta^n, \partial \Delta^n; \mathbb{Z}) \cong \mathbb{Z}$.
\end{example}

\begin{example}
The 2-torus $T^2$ can be triangulated as an abstract simplicial complex $K$. While $C_n^{\Delta}$ contains more generators than a CW complex (requiring at least 14 triangles and 21 edges), the resulting homology groups are identical:
$$H_1^{\Delta}(K; \mathbb{Z}) \cong \mathbb{Z}^2, \quad H_2^{\Delta}(K; \mathbb{Z}) \cong \mathbb{Z}$$
\end{example}

\begin{example}
The $n$-sphere $S^n$ for $n > 0$ can be given a CW structure with one 0-cell $e^0$ and one $n$-cell $e^n$. The characteristic map for $e^n$ maps $\partial D^n$ to $e^0$.
The cellular chain groups are:
$$C_k^{CW}(S^n; \mathbb{Z}) \cong 
\begin{cases} 
\mathbb{Z} & \text{if } k = 0, n \\
0 & \text{otherwise}
\end{cases}$$
If $n > 1$, the boundary maps $d_k$ must be zero because there are no adjacent non-zero chain groups. Thus:
$$H_k(S^n; \mathbb{Z}) \cong 
\begin{cases} 
\mathbb{Z} & \text{if } k = 0, n \\
0 & \text{otherwise}
\end{cases}$$
\end{example}

\begin{example}
The Real Projective Plane $\mathbb{RP}^2$ has a CW structure with one 0-cell $e^0$, one 1-cell $e^1$, and one 2-cell $e^2$. The 1-skeleton is $S^1$. The 2-cell is attached to the 1-skeleton by a map of degree 2 (the map $z \mapsto z^2$ on the unit circle).
The cellular chain complex is:
$$0 \to \mathbb{Z} \xrightarrow{d_2} \mathbb{Z} \xrightarrow{d_1} \mathbb{Z} \to 0$$
The map $d_1$ is the zero map because it maps the 1-cell to $e^0 - e^0$. The map $d_2$ is multiplication by the degree of the attaching map, which is $2$.
Therefore:
$$H_0(\mathbb{RP}^2; \mathbb{Z}) \cong \mathbb{Z}, \quad H_1(\mathbb{RP}^2; \mathbb{Z}) \cong \mathbb{Z}/2\mathbb{Z}, \quad H_2(\mathbb{RP}^2; \mathbb{Z}) \cong 0$$
\end{example}

\begin{example}
The complex projective space $\mathbb{CP}^n$ has a CW structure with exactly one cell in each even dimension $0, 2, 4, \dots, 2n$. 
Because there are no cells in odd dimensions, all boundary maps $d_k$ in the cellular chain complex must be zero (as either the domain or codomain is 0). 
Consequently, the homology groups are:
$$H_k(\mathbb{CP}^n; \mathbb{Z}) \cong 
\begin{cases} 
\mathbb{Z} & \text{if } k \in \{0, 2, \dots, 2n\} \\
0 & \text{otherwise}
\end{cases}$$
\end{example}

\begin{example}
The 2-torus $T^2$ admits a CW structure with:
\begin{itemize}
    \item One 0-cell: $e^0$ (the vertex where all corners of the square meet).
    \item Two 1-cells: $a$ and $b$ (the horizontal and vertical edges).
    \item One 2-cell: $f$ (the interior of the square).
\end{itemize}
The cellular chain groups with integer coefficients are:
$$C_2^{CW}(T^2) \cong \mathbb{Z} \langle f \rangle, \quad C_1^{CW}(T^2) \cong \mathbb{Z} \langle a, b \rangle, \quad C_0^{CW}(T^2) \cong \mathbb{Z} \langle e^0 \rangle$$
The boundary maps are computed as follows:
\begin{itemize}
    \item $d_1(a) = e^0 - e^0 = 0$ and $d_1(b) = e^0 - e^0 = 0$. Thus, $d_1$ is the zero map.
    \item The boundary of the 2-cell $f$ follows the word $aba^{-1}b^{-1}$. In the abelian group $C_1^{CW}$, this becomes $a + b - a - b = 0$. Thus, $d_2$ is the zero map.
\end{itemize}
Since all boundary maps are zero, the homology groups are isomorphic to the chain groups:
$$H_0(T^2; \mathbb{Z}) \cong \mathbb{Z}, \quad H_1(T^2; \mathbb{Z}) \cong \mathbb{Z}^2, \quad H_2(T^2; \mathbb{Z}) \cong \mathbb{Z}$$
\end{example}

\subsection{Sheaves on more general sites}

Note that on a topological space $X$, it is natural to talk about open sets $\{U_i\}$ which cover an open set $U$ of $X$. One can express more general notions of ``coverings'' of objects in a category through the notion of a \CrefAndHyperrefIfExist{definition:grothendieck_topology_on_a_category_site_covering_sieve_topologically_generating_family}{``site''}, which consists of a category and a \CrefAndHyperrefIfExist{definition:grothendieck_topology_on_a_category_site_covering_sieve_topologically_generating_family}{``Grothendieck topology''} on the category. The notion of \CrefAndHyperrefIfExist{definition:sheaf_on_a_topological_space_valued_in_a_category_with_a_terminal_object}{``sheaf on a topological space''} above is a specialization of this more general notion.

A \CrefAndHyperrefIfExist{definition:grothendieck_topology_on_a_category_site_covering_sieve_topologically_generating_family}{site} is a \CrefAndHyperrefIfExist{definition:category}{category} equipped with a \CrefAndHyperrefIfExist{definition:grothendieck_topology_on_a_category_site_covering_sieve_topologically_generating_family}{Grothendieck topology}, which is a notion that generalizes the idea for an object to have ``covers''. One then has a general notion of \CrefAndHyperrefIfExist{definition:sheaf_on_a_site}{sheaves} on a site; a \CrefAndHyperrefIfExist{definition:sheaf_on_a_topological_space_valued_in_a_category_with_a_terminal_object}{sheaf on a topological space $X$} is then simply a sheaf on the category \CrefAndHyperrefIfExist{definition:category_of_opens_of_a_topological_space}{$\mathbf{Open}(X)$} equipped with \CrefAndHyperrefIfExist{definition:site_of_opens_on_a_topological_space}{its canonical Grothendieck topology}. It can therefore be fruitful to gain intuition about sheaves in general by studying sheaves on topological spaces and to have the understanding that many notions discussed for sheaves on topological spaces are special cases of those for sheaves on general sites. We may at times state definitions and concepts in terms of sheaves on somewhat general sites (e.g. sites that are \CrefAndHyperrefIfExist{definition:small_site_locally_small_site_U_site_U_small_site_U_topology}{small}/are \CrefAndHyperrefIfExist{definition:essentially_small_site}{essentially small}/have a small \CrefAndHyperrefIfExist{definition:grothendieck_topology_on_a_category_site_covering_sieve_topologically_generating_family}{topologically generating family}), but it should be understood that they specialize to sheaves on topological spaces (valued in sets).

A \CrefAndHyperrefIfExist{definition:topos}{(Grothendieck) topos} is a category that is \CrefAndHyperrefIfExist{definition:equivalence_of_categories}{equivalent to} a category of sheaves valued in sets on some \CrefAndHyperrefIfExist{definition:big_site_on_the_category_of_schemes_over_a_scheme_and_small_site}{small site}, or equivalently a category that satisfies Giraud's axioms. The theory of topoi is quite robust and appears in many contexts. Moreover, all \CrefAndHyperrefIfExist{definition:locally_small_category}{locally small categories} are basically \CrefAndHyperrefIfExist{definition:full_subcategory_of_a_category}{full subcategories} of \CrefAndHyperrefIfExist{definition:topos}{topoi} essentially due to \CrefAndHyperrefIfExist{corollary:yoneda_embedding_on_a_locally_small_category}{Yoneda embedding}.

See \Cref{section:appendix_grothendieck_topologies}.

\begin{remark} \label{remark:cech_cohomology_with_respect_to_arbitrary_open_cover_may_not_have_long_exact_sequence}
    Let $X$ be a \CrefAndHyperrefIfExist{definition:topological_space}{topological space}. Let $\calA$ be an \CrefAndHyperrefIfExist{definition:grothendiecks_additional_axioms_for_abelian_categories}{$AB3$ category}. Assume that a \CrefAndHyperrefIfExist{definition:sheafification_of_a_presheaf_on_a_topological_space}{sheafification functor}
    $$a: \PreShv(X, \calA) \to \Shv(X, \calA) $$
    exists so that $\Sh(X, \calA)$ is an abelian caetgory (\Cref{proposition:circumstances_under_which_categories_of_sheaves_and_presheaves_on_a_topological_space_valued_in_an_abelian_category_are_abelian}).

    Given an arbitrary \CrefAndHyperrefIfExist{definition:open_covering_of_a_topological_space}{open cover} $\calU$, the \CrefAndHyperrefIfExist{definition:functor_between_categories}{functors} 
    $$\{\check{H}^p(\calU,-): \Sh(X, \calA) \to \calA \}_{p \geq 0}$$
    \CrefIfExists{definition:sheaf_on_a_topological_space_valued_in_a_category_with_a_terminal_object}\CrefIfExists{definition:category_of_groups_of_abelian_groups}\CrefIfExists{definition:cech_cohomology_of_a_sheaf_of_abelian_groups_on_a_topological_space}
    may not form a \CrefAndHyperrefIfExist{definition:homological_cohomological_delta_functor_between_abelian_categories}{cohomological $\delta$-functor}. In other words, \CrefAndHyperrefIfExist{definition:short_exact_sequence_in_an_additive_category}{short exact sequences}
    $$0 \to \calF' \to \calF \to \calF'' \to 0$$
    in $\Sh(X, \calA)$ may not yield \CrefAndHyperrefIfExist{theorem:long_exact_sequence_in_homology_cohomology_of_a_short_exact_sequence_of_chain_complexes_in_an_abelian_category}{long exact sequences} of the form
    \begin{align*}
    0 &\to \check{H}^0(\calU, \calF') \to  \check{H}^0(\calU, \calF) \to \check{H}^0(\calU, \calF'')  \\
    &\to \check{H}^1(\calU, \calF') \to \check{H}^1(\calU, \calF) \to \check{H}^1(\calU, \calF'') \\ 
    &\to \cdots.
    \end{align*}
    This is because, for general $\calU$, the sequence 
    $$0 \to C^\bullet(\calU, \calF') \to C^\bullet(\calU, \calF) \to C^\bullet(\calU, \calF'') \to 0$$
    is generally only \CrefAndHyperrefIfExist{definition:exact_functor_between_abelian_categories}{left exact} because sections are left exact \CrefIfExists{proposition:sections_functors_on_presheaves_on_a_topological_space_valued_in_an_abelian_category_are_left_exact}. 

    On the other hand, at least in the case that $\calA = \Ab$ and given any open cover $\calU$ of $X$, there is a long exact sequence
    \begin{align*}
    0 &\to \check{H}^0(X, \calF') \to  \check{H}^0(X, \calF) \to \check{H}^0(X, \calF'')  \\
    &\to \check{H}^1(X, \calF') \to \check{H}^1(X, \calF) \to \check{H}^1(X, \calF'') \\
    &\to \cdots.
    \end{align*}
\end{remark}

\begin{example}
    Let $r \in \bbQ^\times$ and let $\calF_r$ be the \CrefAndHyperrefIfExist{definition:locally_constant_sheaf}{locally constant sheaf} of $\bbQ$-vector spaces on $S^1$ with ``monodromy'' $r$ --- by this, we mean that $\calF_r$ is the (isomorphism class of) sheaf on $S^1$ corresponding to (\Cref{theorem:loc_const_shvs_of_R_mods_on_a_conn_loc_path_conn_semi_loc_simp_conn_are_eq_to_rep_of_fund_grp}) the homomorphism $\pi_1(S^1,x) \to \Aut(\bbQ) \cong \bbQ^\times$ given by sending a generator of $\pi_1(S^1, x)$ to $r \in \bbQ^\times$. 
    
    Consider the \CrefAndHyperrefIfExist{definition:open_covering_of_a_topological_space}{open cover} $\calU = \{U_0, U_1\}$ consisting of two overlapping semicircles. The intersection $U_0 \cap U_1 = V_a \sqcup V_b$ is a disjoint union of two \CrefAndHyperrefIfExist{definition:contractible_topological_space}{contractible} arcs, making $\calU$ a \CrefAndHyperrefIfExist{definition:semi_good_open_cover_of_a_topological_space}{semi-good cover}. 

    As the specification of $\calF_r$ by monodromy only determines its isomorphism class, we provide an explicit description of a representative element. Since each $U_i$ is \CrefAndHyperrefIfExist{definition:contractible_topological_space}{contractible} and locally contractible, the restriction $\calF_r|_{U_i}$ is a \CrefAndHyperrefIfExist{definition:constant_sheaf_on_a_topological_space_valued_in_category_admitting_sheafification}{constant sheaf} with value $\bbQ$. Thus, we have isomorphisms $\phi_i: \Gamma(U_i, \calF_r) \xrightarrow{\sim} \bbQ_i$, where $\bbQ_0, \bbQ_1$ are two copies of $\bbQ$.

    Note that $\Gamma(U_0 \cap U_1, \calF_r) \cong \Gamma(V_a, \calF_r) \oplus \Gamma(V_b, \calF_r)$. Since $V_a, V_b \subset U_i$ are inclusions of contractible sets into a larger contractible set, the restriction maps $\Gamma(U_i, \calF_r) \to \Gamma(V_j, \calF_r)$ are isomorphisms. By identifying each $\Gamma(V_j, \calF_r)$ with $\Gamma(U_i, \calF_r)$ via these internal restrictions, the map $\Gamma(U_i, \calF_r) \to \Gamma(U_0 \cap U_1, \calF_r)$ is identified with the diagonal:
    $$s \mapsto (s, s) \in \Gamma(V_a, \calF_r) \oplus \Gamma(V_b, \calF_r).$$

    The monodromy $r$ is encoded in the transition between the choices of basis $1 \in \bbQ_0$ and $1 \in \bbQ_1$. We construct $\calF_r$ such that:
    \begin{enumerate}
        \item On the overlap $V_a$, the identification $\bbQ_0 \cong \Gamma(V_a, \calF_r) \cong \bbQ_1$ is the identity ($1 \mapsto 1$).
        \item On the overlap $V_b$, the identification $\bbQ_0 \cong \Gamma(V_b, \calF_r) \cong \bbQ_1$ is multiplication by $r$ ($1 \mapsto r$).
    \end{enumerate}

    With these identifications, an element $(c_0, c_1) \in \bbQ_0 \oplus \bbQ_1$ maps via the Čech differential $d^0$ to:
    $$d^0(c_0, c_1) = (c_1 - c_0, c_1 - r c_0) \in \bbQ \oplus \bbQ.$$
    For $r \neq 1$, the only solution to $d^0(c_0, c_1) = (0, 0)$ is $c_0 = c_1 = 0$, implying $\check{H}^0(S^1, \calF_r) = 0$. 

    To compute $\check{H}^1(S^1, \calF_r) = \text{coker } d^0$, we observe that for any target vector $(a, b) \in \bbQ \oplus \bbQ$, we seek $(c_0, c_1)$ such that:
    $$ \begin{cases} c_1 - c_0 = a \\ c_1 - r c_0 = b \end{cases} $$
    Subtracting the first equation from the second yields $(1-r)c_0 = b - a$. Since $r \neq 1$, we can solve for $c_0 = \frac{b-a}{1-r}$ and subsequently $c_1 = a + c_0$. Thus, $d^0$ is \CrefAndHyperrefIfExist{definition:surjective_function}{surjective}, and the quotient group $\check{H}^1(S^1, \calF_r)$ is zero.
    
\end{example}

    \begin{equation} \label{equation:four_term_exact_sequence_in_cech_cohomology_for_holomorphic_exponential_short_exact_sequence_with_respect_to_nice_cover}

    \begin{tikzcd}
        0 \ar[r] & C^0(\calU, \underline{\bbZ}) \ar[r] \ar[d] & C^0(\calU, \calO_X) \ar[r] \ar[d] & C^0(\calU, \calO_X^*) \ar[r] \ar[d] & 0 \\
        0 \ar[r] & C^1(\calU, \underline{\bbZ}) \ar[r] \ar[d] & C^1(\calU, \calO_X) \ar[r] \ar[d] & C^1(\calU, \calO_X^*) \ar[r] \ar[d] & 0 \\
                 & \vdots                                     & \vdots                            & \vdots                              & 
        \end{tikzcd}
    \end{center}
    is identified with 
    \begin{center}
        \begin{tikzcd}
            0 \ar[r] & \Gamma(U_0, \underline{\bbZ}) \oplus \Gamma(U_1, \underline{\bbZ}) \ar[r] \ar[d, "d^0"] & \Gamma(U_0, \calO_X) \oplus \Gamma(U_1, \calO_X) \ar[r] \ar[d, "d^0"] & \Gamma(U_0, \calO_X^*) \oplus \Gamma(U_1, \calO_X^*) \ar[r] \ar[d, "d^0"] & 0 \\
            0 \ar[r] & \Gamma(U_0 \cap U_1, \underline{\bbZ}) \ar[r] \ar[d]                                    & \Gamma(U_0 \cap U_1, \calO_X) \ar[r] \ar[d]                           & \Gamma(U_0 \cap U_1, \calO_X^*) \ar[r] \ar[d]                            & 0 \\
                    & 0                                                                                       & 0                                                                     & 0                                                                        & 
        \end{tikzcd}
    \end{center}
    In the nonzero upper right corner, $z \in \check{H}^0(X, \calO_X^*) \cong \check{H}^0(\calU, \calO_X^*)$ is represented by $(z,z) \in \Gamma(U_0, \calO_X^*) \oplus \Gamma(U_1, \calO_X^*)$. Lift $(z,z)$ to $f \in \Gamma(U_0, \calO_X) \oplus \Gamma(U_1, \calO_X)$ by letting $f = (f_0,f_1)$ for local lifts $f_i \in \Gamma(U_i, \calO_X)$ of $z \in \Gamma(U_i, \calO_X^*)$; in other words, $f_i$ is a local logarithm $\log z$ on $U_i$. The differential $d(f) \in C^1(\calU, \calO_X) = \Gamma(U_0 \cap U_1, \calO_X)$ is $f_1 - f_0$, as a function on $U_0 \cap U\Gamma(U_0 \cap U_1, \calO_X)_1$. Note that the image of this in $\Gamma(U_0 \cap U_1, \calO_X^*)$ is $0$, so there exists a unique element $c_{01} \in \Gamma(U_0 \cap U_1, \underline{Z})$ whose image in $\Gamma(U_0 \cap U_1, \calO_X)$ is $f_1 - f_0$. By the logic of the Snake lemma, this $c_{01}$ is well defined, no matter how $f$ was chosen.

    Explicitly, let $f_0(z) = \log |z| + i \arg(z)$ with $\arg(z) \in (-\pi, \pi)$ and let $f_1(z) = \log |z| + i \arg(z)$ with $\arg(z) \in (0, 2 \pi)$. We note that 
    $$f_1(z) - f_0(z) = \begin{cases} 0 &\text{if } z \in V_{\text{up}} \\ 2\pi i &\text{if } z \in V_{\text{low}} \end{cases}.$$
    Therefore, $c_{01} \in \Gamma(U_0 \cap U_1, \underline{Z})$ is the section on $U_0 \cap U_1$ given by $0$ on $V_{\text{up}}$ and $1$ on $V_{\text{low}}$. 

    Now the differential map 
    $$d^0: \Gamma(U_0, \underline{\bbZ}) \oplus \Gamma(U_1, \underline{\bbZ}) \to \Gamma(U_0 \cap U_1, \underline{\bbZ})$$
    is given by
    $$\bbZ \oplus \bbZ \to \bbZ \oplus \bbZ, \quad (a,b) \mapsto (a-b, a-b).$$
    The cokernel of this map, which is exactly $\check{H}^1(\calU, \underline{\bbZ})$ by definition, is thus the quotient of $\bbZ \oplus \bbZ$ by pairs of the form $(a-b, a-b)$ for $a,b \in \bbZ$. This quotient is isomorphic to $\bbZ$ and the element $c_{01} = (0,1)$ is can be identified with $1$ in this $\bbZ$. Therefore, $\delta(z)$ is a generator of $\check{H}^1(\calU, \underline{\bbZ}) \cong \check{H}^1(X, \underline{\bbZ}) \cong \bbZ$.
    
\end{example}

\begin{proof}
    \begin{enumerate}
        \item By the \CrefAndHyperrefIfExist{definition:weak_convergence_to_a_graded_object_for_a_spectral_sequence_in_an_abelian_category}{weak convergence}, there are isomorphisms
        $$E_{p,q}^\infty \cong \text{Gr}_p H_{p+q} := \frac{F_p H_{p+q}}{F_{p-1} H_{p+q}}$$
        for all $p,q$. In particular, for a fixed $n$, we have $\text{Gr}_p H_n \cong E_{p, n-p}^\infty$. Since the spectral sequence is first quadrant, $E_{p,q}^r = 0$ if $p < 0$ or $q < 0$. This implies $E_{p, n-p}^\infty = 0$ whenever $p < 0$ or $p > n$.

        Consequently, $\text{Gr}_p H_n = 0$ for $p < 0$ and $p > n$. This vanishing of the graded pieces means that the filtration stabilizes:
        \begin{align*}
            \cdots = F_{-2} H_n = F_{-1} H_n \quad \text{and} \quad F_n H_n = F_{n+1} H_n = \cdots
        \end{align*}
        Since the filtration is \CrefAndHyperrefIfExist{definition:exhaustive_separated_filtrations_on_an_abelian_category}{exhaustive and separated}, we have $F_{-1} H_n = 0$ and $F_n H_n = H_n$. This yields the desired finite filtration:
        $$0 = F_{-1} H_n \subseteq F_0 H_n \subseteq F_1 H_n \subseteq \cdots \subseteq F_n H_n = H_n.$$

        \item By the \CrefAndHyperrefIfExist{definition:weak_convergence_to_a_graded_object_for_a_spectral_sequence_in_an_abelian_category}{weak convergence}, there are isomorphisms
        $$E_\infty^{p,q} \cong \text{Gr}^p H^{p+q} := \frac{F^p H^{p+q}}{F^{p+1} H^{p+q}}$$
        for all $p,q$. In particular, for a fixed $n$, we have $\text{Gr}^p H^n \cong E_\infty^{p, n-p}$. Since the spectral sequence is first quadrant, $E_r^{p,q} = 0$ if $p < 0$ or $q < 0$. This implies $E_\infty^{p, n-p} = 0$ whenever $p < 0$ or $p > n$.

        Consequently, $\text{Gr}^p H^n = 0$ for $p < 0$ and $p > n$. This vanishing of the graded pieces means that the filtration stabilizes:
        \begin{align*}
            \cdots = F^{-1} H^n = F^0 H^n \quad \text{and} \quad F^n H^n = F^{n+1} H^n = \cdots
        \end{align*}
        Since the filtration is \CrefAndHyperrefIfExist{definition:exhaustive_separated_filtrations_on_an_abelian_category}{exhaustive and separated}, we have $F^0 H^n = H^n$ and $F^{n+1} H^n = 0$. This yields the desired finite filtration:
        $$H^n = F^0 H^n \supseteq F^1 H^n \supseteq F^2 H^n \supseteq \cdots \supseteq F^n H^n \supseteq F^{n+1} H^n = 0.$$
    \end{enumerate}
    
\end{proof}

    The filtration on $H^n = H^n(D_m, \bbZ)$ is of the form 
    $$H^n = F^0 H^n \supseteq F^1 H^n \supseteq \cdots \cdots F^{n+1} H^n = 0.$$
    by \Cref{lemma:filtered_graded_abutment_of_a_first_quadrant_spectral_sequence_in_an_abelian_category_has_finite_filtration}. Moreover, the \CrefAndHyperrefIfExist{definition:weak_convergence_to_a_graded_object_for_a_spectral_sequence_in_an_abelian_category}{weak convergence} of the spectral sequence means that we have 
    $$E^{p,q}_\infty \cong F^{p} H^{p+q} / F^{p+1} H^{p+q}$$
    For $n = 0$, the filtration is 
    $$H^0 = F^0 H^0 \supseteq F^1 H^0 = 0$$
    and $F^0 H^{0} / F^1 H^0 \cong E^{0,0}_\infty = \bbZ$, so $H^0 \cong \bbZ$. Moreover, for $n = 2$, the filtration is 
    $$H^2 = F^0 H^2 \supseteq F^1 H^2 \supseteq F^2 H^2 \supseteq F^3 H^2 = 0$$
    and 
    \begin{align*}
    F^0 H^2 / F^1 H^2 \cong E_\infty^{0,2} &= 0 \\ 
    F^1 H^2 / F^2 H^2 \cong E_\infty^{1,1} &= 0 \\ 
    F^2 H^2 / F^3 H^2 \cong E_\infty^{2,0} &= \bbZ/2\bbZ.
    \end{align*}
    Therefore, 
    $$H^2 \cong \bbZ/2\bbZ.$$
    Similarly analysis shows that $H^n \cong 0$ for $n$ odd and that $H^n \cong \bbZ/2\bbZ$ for $n \geq 2$ even. 
    To conclude, 
    $$H^n(D_m, \bbZ) = \begin{cases} \bbZ &\text{if } n = 0 \\ \bbZ / 2 \bbZ &\text{if } n \geq 2 \text{ is even} \\ 0 &\text{otherwise} \end{cases}$$

\end{example}

        \item $U_2 = \{ [x:y:z] \in \bbR\bbP^2 \mid z \neq 0 \}$
    \end{itemize}

        \item Show that $\Lambda$ is a \CrefAndHyperrefIfExist{definition:locally_constant_sheaf_on_a_topological_space_valued_in_category_admitting_sheafification}{locally constant sheaf}.

\end{problem}
