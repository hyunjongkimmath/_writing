\begin{proposition} \label{proposition:finite_field_extension_is_separable_iff_there_are_degree_many_base_embeddings_of_extension_into_algebraic_closure}
Let $L/K$ be a \CrefAndHyperrefIfExist{definition:degree_of_a_field_extension}{finite extension} of \CrefAndHyperrefIfExist{definition:field}{fields} of \CrefAndHyperrefIfExist{definition:degree_of_a_field_extension}{degree} $n$. Let $\bar{K}$ be an \CrefAndHyperrefIfExist{definition:algebraic_element_minimal_polynomial_of_an_algebraic_element_algebraic_field_extension_transcendental_element_field_extension}{algebraic closure} of $K$. The \CrefAndHyperrefIfExist{definition:extension_of_a_field}{extension} $L/K$ is \CrefAndHyperrefIfExist{definition:separable_algebraic_field_extension}{separable} if and only if the number of distinct \CrefAndHyperrefIfExist{definition:field_homomorphism}{$K$-embeddings} of $L$ into $\bar{K}$ is equal to $n$.
\end{proposition}

\begin{proposition} \label{proposition:absolute_frobenius_endomorphism_on_a_finite_field_is_a_field_automorphism}
Let $\mathbb{F}_q$ be a \CrefAndHyperrefIfExist{definition:finite_field}{finite field} of \CrefAndHyperrefIfExist{definition:characteristic_of_a_ring}{characteristic $p$}.
The \CrefAndHyperrefIfExist{definition:frobenius_endomorphism_on_truncated_witt_vectors_of_length_n_over_a_commutative_ring}{Frobenius endomorphism} $\Phi: \mathbb{F}_q \to \mathbb{F}_q$ is a \CrefAndHyperrefIfExist{definition:field}{field} \CrefAndHyperrefIfExist{definition:field_automorphism}{automorphism}.
\end{proposition}

\begin{lemma} \label{lemma:irreducible_poly_in_char_p_field_is_insep_iff_formal_derivative_is_zero}
Let $K$ be a \CrefAndHyperrefIfExist{definition:field}{field} of \CrefAndHyperrefIfExist{definition:characteristic_a_prime_for_a_scheme}{characteristic $p$}.
\TODO{irred poly, separable polynomial, formal derivative}
An irreducible polynomial $f \in K[x]$ is inseparable if and only if its formal derivative $f'$ is the zero polynomial, which occurs if and only if there exists a polynomial $g \in K[x]$ such that $f(x) = g(x^p)$.
\end{lemma}

\begin{theorem} \label{theorem:finite_fields_are_perfect}
Every \CrefAndHyperrefIfExist{definition:finite_field}{finite field} $\mathbb{F}_q$ is a \CrefAndHyperrefIfExist{definition:perfect_field}{perfect field}.

In other words, every finite extension of $\bbF_q$ is a \CrefAndHyperrefIfExist{definition:separable_algebraic_field_extension}{separable extension}.
\end{theorem}
