\section{Condensed sets}

\begin{definition} \label{definition:profinite_set}
A \hldef{profinite set} is a set which is the \CrefAndHyperrefIfExist{definition:projective_and_inductive_limits_in_categories}{projective limit} of a \CrefAndHyperrefIfExist{definition:system_in_a_category_indexed_by_a_directed_poset}{(small) directed system} of \CrefAndHyperrefIfExist{definition:countable_finite_uncountable_sets}{finite sets}. 

A profinite set $X = \varprojlim_i X_i$ has a natural topology, referred to as the \hldef{profinite topology} or \hldef{limit topology}. It can be formulated in the following equivalent ways:

\TODO{define inverse limit topology}
\begin{enumerate}
    \item As the \CrefAndHyperrefIfExist{definition:projective_and_inductive_limits_in_categories}{inverse limit} topology with respect to the family of projection maps $\pi_i : X \to X_i$ onto the finite sets $X_i$, each equipped with the \CrefAndHyperrefIfExist{definition:discrete_topological_space_of_a_set}{discrete topology}, in the system.

    \item As the \CrefAndHyperrefIfExist{definition:subspace_topology}{subspace topology} on $\varprojlim_i X_i$ inherited from the \CrefAndHyperrefIfExist{definition:product_topology}{product topology} on $\prod_i X_i$.

    \item As the topology generated by the \CrefAndHyperrefIfExist{definition:basis_for_a_topology}{basis} of "finite index" sets (specifically, the clopen sets) of the form $\pi_i^{-1}(V)$ for some index $i$ and some subset $V \subseteq X_i$.
\end{enumerate}
These topologies do not depend on the inverse system representing $X$. 
\end{definition}

\begin{proposition} \label{proposition:profinite_set_is_same_as_compact_hausdorff_totally_disconnected_set_and_stone_space}
Let $X$ be a \CrefAndHyperrefIfExist{definition:category_of_topological_spaces}{topological space}. The following conditions are equivalent:
\begin{itemize}
        \item $X$ is a \CrefAndHyperrefIfExist{definition:profinite_set}{profinite set}.
        \item $X$ is \CrefAndHyperrefIfExist{definition:compact_topological_space}{compact}, \CrefAndHyperrefIfExist{definition:separation_axioms_of_topology}{Hausdorff}, and \CrefAndHyperrefIfExist{definition:totally_disconnected_topological_space}{totally disconnected}.
        \item $X$ is a \CrefAndHyperrefIfExist{definition:stone_space}{Stone space}.
\end{itemize}
\end{proposition}

\begin{definition} \label{definition:category_of_profinite_sets}
    The \hldef{category of profinite sets}, denoted by notations such as \hl{$\operatorname{Prof}$}, is the category whose objects are \CrefAndHyperrefIfExist{definition:profinite_set}{profinite sets} and whose morphisms are \CrefAndHyperrefIfExist{definition:morphism_of_profinite_sets}{morphisms of profinite sets}.

    This category is \CrefAndHyperrefIfExist{definition:equivalence_of_categories}{equivalent to} the category of \CrefAndHyperrefIfExist{definition:compact_topological_space}{compact} \CrefAndHyperrefIfExist{definition:separation_axioms_of_topology}{Hausdorff} \CrefAndHyperrefIfExist{definition:totally_disconnected_topological_space}{totally disconnected} \CrefAndHyperrefIfExist{definition:topological_spaces}{topological spaces} whose morphsims are \CrefAndHyperrefIfExist{definition:continuous_map_of_topological_spaces}{continuous maps}. 

    This category is also equivalent to the \CrefAndHyperrefIfExist{definition:ind_pro_category_of_a_locally_small_category}{pro-category} of the category of finite sets \TODO{category of finite sets}.

    $\operatorname{Prof}$ admits all \CrefAndHyperrefIfExist{definition:small_and_finite_limits_and_colimits_in_a_category}{small limits}.

\end{definition}

\begin{definition}[{\cite[Definition 2.2.]{clausen_scholze_cmcg}}, {\cite[Definition 1.1.]{scholze_lag}}] \label{definition:grothendieck_topology_on_the_category_of_profinite_sets}
    Define the following \CrefAndHyperrefIfExist{definition:grothendieck_topology_on_a_category_site_covering_sieve_topologically_generating_family}{Grothendieck topology} on the category \CrefAndHyperrefIfExist{definition:profinite_set}{$\operatorname{Prof}$} of \CrefAndHyperrefIfExist{definition:profinite_set}{profinite sets}: a finite collection $(f_i : S_i \to S)_{i \in I}$ of maps in $\operatorname{Prof}$ is a covering if the maps are jointly surjective, i.e. $\coprod_{i \in I} S_i \to S$ is surjective.

\begin{definition}[{\cite[Definition 2.4.]{clausen_scholze_cmcg}}] \label{definition:kappa_condensed_set_for_a_cardinal}
    Let $\kappa$ be a \CrefAndHyperrefIfExist{definition:cardinal_number_and_cardinality_of_a_set}{cardinal}. 
    A \hldef{$\kappa$-condensed set} is a \CrefAndHyperrefIfExist{definition:sheaf_on_a_site}{sheaf of sets} on \CrefAndHyperrefIfExist{definition:kappa_small_profinite_set_for_a_cardinal}{$\operatorname{Prof}_{\kappa}$}.

    The category of $\kappa$-condensed sets is denoted by \hl{$\operatorname{CondSet}_{\kappa}$}.

    Explicitly, a $\kappa$-condensed set $X$ is a \CrefAndHyperrefIfExist{definition:functor_between_categories}{functor}
    $$ X: \operatorname{Prof}_{\kappa}^{o p} \rightarrow \text { Set } $$
    such that

    \begin{enumerate}
        \item If $\left(S_{i}\right)_{i \in I}$ is a finite collection of objects of $\operatorname{Prof}_{\kappa}^{o p}$, then

        $$ X\left(\sqcup_{i \in I} S_{i}\right) \xrightarrow{\sim} \prod_{i \in I} X\left(S_{i}\right) ; $$

        \item If $f: T \rightarrow S$ is a surjective map in $\operatorname{Prof}_{\kappa}$, then

        $$ X(S) \stackrel{f^{*}}{\hookrightarrow} X(T) $$

        is injective with image equal to the set of those $x \in X(T)$ for which $p_{1}^{*} x=p_{2}^{*} x$, where $p_{1}, p_{2}: T \times_{S} T \rightarrow T$ are the two projections.
    \end{enumerate}

\begin{definition} \label{definition:kappa_condensed_object_of_a_category_for_a_cardinal}
    Let $\calA$ be a \CrefAndHyperrefIfExist{definition:category}{category}. Let $\kappa$ be a \CrefAndHyperrefIfExist{definition:cardinal_number_and_cardinality_of_a_set}{cardinal}

\end{definition}

\begin{theorem}[{\cite[Theorem 5.8.(i), Theorem 6.2.(i)]{}}, see also {\cite[Theorem 2.5.]{scholze_lag}}]  \label{theorem:solidification_of_condensed_abelian_group}
    \begin{enumerate}
        \item The category $\operatorname{Solid}$ of solid abelian groups is closed under small limits, small colimits, and extensions as a subcategory of the category \CrefAndHyperrefIfExist{definition:condensed_object_of_a_category}{$\operatorname{Cond}(\operatorname{Ab})$}.

        \item The inclusion functor $\operatorname{Solid} \hookrightarrow \operatorname{Cond}(\operatorname{Ab})$ admits a \CrefAndHyperrefIfExist{definition:adjoint_functors_between_categories_unit_counit_of_adjoint_functors}{left adjoint}, denoted by \hl{$M \mapsto M^\bullet$} and called \hldef{solidification}.
