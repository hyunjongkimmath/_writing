\begin{notation}[{\cite[Chapter I, 2.3]{knus_qhfr}}] \label{notation:spaces_of_varepsilon_hermitian_forms_and_even_varepsilon_hermitian_forms_on_module_over_a_ring_with_involution}
    Let $A$ be a \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{ring with involution} denoted by $\overline{\cdot}$, let $\varepsilon \in Z(A)$ satisfy $\varepsilon \cdot \overline{\varepsilon} = 1$, and let $M$ be a \CrefAndHyperrefIfExist{definition:bimodule_over_rings_module_over_a_ring}{(right) $A$-module}. 

    Write 
    $$\hlin{ \begin{aligned} &\operatorname{Sesq}_{\varepsilon}(M)\coloneq \operatorname{Im}\left(S_{\varepsilon}\right) \subset \operatorname{Sesq}_{A}(M) \\ &\operatorname{Sesq}^{\varepsilon}(M) \coloneq \operatorname{Ker}\left(S_{-\varepsilon}\right) \subset \operatorname{Sesq}_{A}(M) \end{aligned}.}$$
    (\Cref{definition:form_and_operator_symmetrizer_maps_for_sesquilinear_forms_on_modules_over_rings_with_involutions}); in general, $\Sesq_\varepsilon(M) \subset \Sesq^\varepsilon(M)$.  

    The groups $\operatorname{Sesq}^{\varepsilon}(M)$ and $\operatorname{Sesq}_{\varepsilon}(M)$ are the groups of \CrefAndHyperrefIfExist{definition:varepsilon_hermitian_form_on_a_module_over_a_ring_with_involution}{$\varepsilon$-Hermitian forms on $M$} and of even $\varepsilon$-Hermitian forms on $M$ respectively.
    
\end{notation}
