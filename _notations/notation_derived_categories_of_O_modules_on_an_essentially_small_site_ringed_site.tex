\begin{notation} \label{notation:derived_categories_of_O_modules_on_an_essentially_small_site_ringed_site}
    Let $(\calC, J, \calO)$ be a \CrefAndHyperrefIfExist{definition:ringed_site}{ringed site} where the underlying site $(\calC, J)$ is \CrefAndHyperrefIfExist{definition:essentially_small_site}{essentially small}. In particular, $\Mod_\calO$ is an \CrefAndHyperrefIfExist{definition:abelian_category}{abelian category}.
    
    Denote by \hl{$D(\calO)$}, \hl{$D^+(\calO)$}, \hl{$D^-(\calO)$}, \hl{$D^b(\calO)$} the derived categories $D(\calA)$, $D^+(\calA)$, $D^-(\calA)$, $D^b(\calA)$ (\Cref{definition:derived_category_of_an_abelian_category}) where $\calA$ is either the abelian category $_{\calO} \Mod$ or $\Mod_{\calO}$.
\end{notation}
