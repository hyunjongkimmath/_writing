\begin{notation} \label{notation:derived_category_of_sheaves_of_abelian_groups_on_small_site_of_a_scehem}
    Let $X/S$ be a scheme over a base scheme $S$, and choose some \CrefAndHyperref{definition:big_site_on_the_category_of_schemes_over_a_scheme_and_small_site}{small site} $X_{\text{site}}$ on $X$ associated to a big site on $(\Sch/S)$. Let $\calO$ be a sheaf of rings on $X_{\text{site}}$. It is standard to write the following:
    \begin{itemize}
        \item \hl{$D(X_{\text{site}}, \calO)$}, or simply \hl{$D(X, \calO)$} if the small site $X_{\text{site}}$ is understood, for the \CrefAndHyperrefIfExist{definition:derived_category_of_an_abelian_category}{derived category} of the \CrefAndHyperrefIfExist{definition:module_over_a_sheaf_of_rings_on_a_site}{category of sheaves of $\calO$-modules}.
        \item \hl{$D_{\mathrm{tor}}(X_{\text{site}}, \calO)$} or \hl{$D_{\mathrm{tors}}(X_{\text{site}}, \calO)$} for the full subcategory of $D(X)$ of complexes whose \CrefAndHyperrefIfExist{definition:homology_and_cohomology_objects_for_a_chain_complex_in_an_abelian_category}{cohomology objects} are \CrefAndHyperrefIfExist{definition:torsion_sheaf_of_abelian_groups_on_a_site}{torsion sheaves} 
        \TODO{continue}
        \item In the case that $\calO = \bbZ$ (\Cref{definition:constant_sheaf_on_a_site_with_sheafification}), i.e. for derived categories of categories of sheaves of abelian groups, one may simply write $D(X_{\text{site}})$, etc.
    \end{itemize}
    Superscripts of $+$, $-$, $b$ may also be used accordingly --- for example, for $? \in \{+,-,b\}$, it makes sense to write \hl{$D^?(X_{\text{site}}, \calO)$}, see \Cref{definition:derived_category_of_an_abelian_category}. 
    \TextIfExists{notation:derived_categories_of_O_modules_on_an_essentially_small_site_ringed_site}{
    Compare with \Cref{notation:derived_categories_of_O_modules_on_an_essentially_small_site_ringed_site}.
    }
\end{notation}
