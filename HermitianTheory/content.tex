\begin{lemma} \label{lemma:hermitian_modules_over_product_of_involution_rings_split_along_involution_ring_factors}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{ring with involution}. Suppose that there is an isomorphism $R \cong \prod_{i=1}^n R_i$ of involution rings and write $e_1,\ldots,e_n$ for the \CrefAndHyperrefIfExist{definition:complete_orthogonal_idempotents_of_a_ring}{complete} \CrefAndHyperrefIfExist{definition:center_of_a_ring}{central} orthogonal idempotents \CrefAndHyperrefIfExist{proposition:ring_is_product_ring_if_and_ony_if_it_has_central_orthogonal_idempotents}{corresponding to} this decomposition. Let $\varepsilon \in Z(R)$ satisfy $\varepsilon \cdot \overline{\varepsilon} = 1$ and write $\varepsilon_i \in R_i$ for the component of $\varepsilon$ in $R_i$. 
    \TODO{orthogonal sum}
    \begin{enumerate}
        \item Every \CrefAndHyperrefIfExist{definition:varepsilon_hermitian_form_on_a_module_over_a_ring_with_involution}{$\varepsilon$-Hermitian form} $H: V \times V \to R$ on a (right) $R$-module $V$ uniquely decomposes as an orthogonal sum $H = \perp_{i=1}^n H_i$ where each $H_i: V_i \times V_i \to R_i$ is an $\varepsilon_i$-Hermitian form on the $R_i$-module $V_i = e_i V$. In fact, $H_i$ is constructed as $H|_{V_i}: V_i \times V_i \to R$ and has values in $R_i$.

        \item If $V$ is a \CrefAndHyperrefIfExist{definition:finitely_generated_modules_over_rings}{finitely generated} \CrefAndHyperrefIfExist{definition:projective_bimodule_over_rings}{projective} $R$-module (equivalently each $V_i$ is a finitely generated projective $R_i$-module), then $H$ is \CrefAndHyperrefIfExist{definition:nondegenerate_nonsingular_regular_sesquilinear_form_on_a_module_over_an_involution_ring}{nonsingular} if and only if each $H_i$ is nonsingular.
    \end{enumerate}
\end{lemma}

\subsection{Base change of Sesquilinear forms, Hermitian forms, and Unitary modules}

\begin{lemma}[{\cite[Chapter I, 7.1]{weibel}}]
Let $\eta \to A \to A'$ be a \CrefAndHyperrefIfExist{definition:homomorphism_of_involution_rings}{homomorphism of rings with involution}. 
Given a \CrefAndHyperrefIfExist{definition:module_of_a_ring}{right $A$-module} $M$ and there is a right $A'$\CrefAndHyperrefIfExist{definition:homomorphism_of_modules_over_a_ring}{-module homomorphism} 
$$j_M: M^* \otimes_A A' \to (M \otimes_A A')^* = \Hom_{A'}(M', A')$$
(\Cref{definition:twisted_dual_module_of_a_module_over_a_ring_with_involution}) (\Cref{definition:extension_of_scalars_base_change_induction_for_modules_along_ring_homomorphisms}) (\Cref{definition:hom_of_left_right_bi_modules_of_rings}) given by 
$$j_M(f \otimes a')(m \otimes b') = \overline{a}' \eta(f(m)) b'$$
for $f \in M^*$, and $a',b' \in A'$. The maps $j_M$ specify a \CrefAndHyperrefIfExist{definition:natural_transformation_between_functors_between_categories}{natural transformation} in the \CrefAndHyperrefIfExist{definition:category_of_modules_and_bimodules_over_rings}{category of right $A'$-modules}.

If $M$ is a \CrefAndHyperrefIfExist{definition:projective_bimodule_over_rings}{projective} right $A$-module, then $j_M$ is an isomorphism.
\end{lemma}

\begin{lemma}[see {\cite[Chapter I, 7.2]{knus_qhfr}}]
    Let $\eta: A \to A'$ be a \CrefAndHyperrefIfExist{definition:homomorphism_of_involution_rings}{homomorphism of rings with involution}. 
    Given a right $A'$-module $M'$, write $M'_{(A)}$ for the \CrefAndHyperrefIfExist{definition:restriction_of_scalars_of_a_left_or_right_module_along_a_ring_homomorphism_from_the_base_ring}{restriction} of $M'$ as a right $A$-module (via $\eta$).Let $\tau: A' \to A$ be a \CrefAndHyperrefIfExist{definition:homomorphism_of_modules_over_a_ring}{morphism of $A$-bimodules} and assume that 
    $$\tau(\overline{a}) = \overline{\tau(a)}$$
    for all $a \in A'$

    The \CrefAndHyperrefIfExist{definition:scharlau_transfer_of_sesquilinear_forms_on_modules_under_involution_ring_homomorphism}{Scharlau transfer}

    $$\tau: \Sesq^{\varepsilon'}(M') \to \Sesq^{\varepsilon}(M'_{(A)})$$
    $$\tau: \Sesq_{-\varepsilon'}(M') \to \Sesq_{-\varepsilon}(M'_{(A)})$$
    (\Cref{notation:spaces_of_varepsilon_hermitian_forms_and_even_varepsilon_hermitian_forms_on_module_over_a_ring_with_involution})
\end{lemma}
