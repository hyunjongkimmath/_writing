\begin{lemma} \label{lemma:sesquilinear_form_associated_to_a_homomorphsim_from_a_module_to_its_twisted_dual_is_indeeed_sesquilinear}
    Let $A$ be an \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{involution ring}, and let $M$ be a \CrefAndHyperref{definition:module_of_a_ring}{left/right $A$-module}. 
    Let $h: M \to M^*$ be a left/right $A$-module homomorphism where $M^*$ is the \CrefAndHyperref{definition:twisted_dual_module_of_a_module_over_a_ring_with_involution}{twisted dual module of $M$}. The \CrefAndHyperrefIfExist{definition:sesquilinear_form_associated_to_a_homomorphism_from_a_module_to_its_twisted_dual}{sesquilinear form $b_h$ associated to $h$} is indeed a \CrefAndHyperrefIfExist{definition:sesquilinear_form_on_a_module_over_a_ring_with_involution}{sesquilinear form} over $A$.
\end{lemma}
\begin{proof}
    Suppose that $M$ is a left $A$-module. Since $h$ is a left $A$-module homomorphism, $h(av) = a \cdot h(v)$ for all $a \in A$ and $v \in M$. Thus, 
    $$b_h(u,av) = h(av)(u) = (ah(v))(u) = h(v)(u) \cdot \overline{a}$$
    for all $u \in M$.
    Moreover, for all $a \in A$ and $u \in M$, since $h(v) \in M^* = \Hom_{A}(M, A)_\sigma$ is a left $A$-module homomorphism, 
    $$b_h(au,v) = h(v)(au) = a \cdot (h(v)(u)) = a \cdot b_h(u,v).$$
    Therefore, $b_h$ is $A$-linear in the first argument and conjugate linear in the second.

    Suppose that $M$ is a right $A$-module. Since $h$ is a right $A$-module homomorphism, $h(ua) = h(u) \cdot a$ for all $a \in A$ and $u \in M$. Thus, 
    $$b_h(ua,v) = h(ua)(v) = (h(u)a)(v) = \overline{a} \cdot h(u)(v)$$
    for all $u \in M$.
    Moreover, for all $a \in A$ and $u \in M$, since $h(u) \in M^* = \Hom_{A}(M, A)_\sigma$ is a right $A$-module homomorphism, 
    $$b_h(u,va) = h(u)(va) = (h(u)(v)) \cdot a =  b_h(u,v) \cdot a.$$
    Therefore, $b_h$ is $A$-linear in the second argument and conjugate linear in the first.
\end{proof}

\begin{corollary} \label{corollary:seqsquilinear_forms_on_a_module_over_a_ring_with_involution_correspond_to_bimodule_maps_from_modules_tensored_by_opposites_to_ring}
    Let $(A,\sigma)$ be a \CrefAndHyperref{definition:involution_on_a_ring}{ring with involution}. 
    \begin{enumerate}
        \item Let $M$ be a right $A$-module. There is a natural abelian group isomorphism
        $$\Sesq_A(M) \cong \Hom_{A-A}(\overline{M} \otimes_{\bbZ} M, A)$$
        \CrefIfExists{definition:opposite_module_of_a_module_over_a_ring_with_involution}\CrefIfExists{definition:tensor_product_of_bimodules_of_rings}\CrefIfExists{definition:hom_of_left_right_bi_modules_of_rings}

        \item Let $M$ be a left $A$-module. There is a natural abelian group isomorphism
        $$\Sesq_A(M) \cong \Hom_{A-A}(M \otimes_{\bbZ} \overline{M}, A)$$
        \CrefIfExists{definition:opposite_module_of_a_module_over_a_ring_with_involution}\CrefIfExists{definition:tensor_product_of_bimodules_of_rings}\CrefIfExists{definition:hom_of_left_right_bi_modules_of_rings}

\end{corollary}
\begin{proof}

        \item This holds similarly.
    \end{enumerate}
\end{proof}

\begin{lemma} \label{lemma:varepsilon_hermitian_form_induced_by_a_hermitian_matrix_over_a_ring_with_involution_is_a_hermitian_form}
    Let $(A,\sigma)$ be a \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{ring with involution}, and let $\varepsilon \in Z(A)$\CrefIfExists{definition:center_of_a_ring} satisfy $\varepsilon \cdot \overline{\varepsilon} = 1$.
    Let $H \in M_n(A)$\CrefIfExists{definition:matrix_over_a_ring} be an \CrefAndHyperrefIfExist{definition:epsilon_hermitian_matrix_over_a_ring_with_involution}{$\varepsilon$-Hermitian matrix} in the sense that
    $$H = \overline{\varepsilon} \cdot H^*.$$
    Let $M = A^n$ be the free (left/right) $A$-module of \CrefAndHyperrefIfExist{definition:row_vector_and_column_vector_over_a_ring}{column vectors} with standard basis $\mathcal{E} = (e_1,\ldots,e_n)$.

    The \CrefAndHyperrefIfExist{definition:epsilon_hermitian_form_induced_by_an_epsilon_hermitian_matrix}{$\varepsilon$-Hermitian form $h_H$ induced by $H$} is indeed a \CrefAndHyperrefIfExist{definition:varepsilon_hermitian_form_on_a_module_over_a_ring_with_involution}{$\varepsilon$-Hermitian form} on $M$; in particular, 
        $$h_H(u,v) = \varepsilon \cdot \overline{h_H(v,u)}$$
    for all $u,v \in M$. 

    % in the following sense:
    % \begin{enumerate}
    %     \item if we regard $M$ as a right $A$-module so that 
    %     $$h_H(u,v) = u^* H v$$
    %     and if $H$ is a $\varepsilon$-Hermitian matrix in the sense that
    %     $$H^* = \overline{\varepsilon} \cdot H,$$
    %     then 
    %     $$h_H(u,v) = \varepsilon \cdot \overline{h_H(v,u)}$$
    %     for all $u,v \in M$. 
    %     \item if we regard $M$ as a left $A$-module so that 
    %     $$h_H(u,v) = u^\top H \overline{v}$$
    %     and if $H$ is a $\varepsilon$-Hermitian matrix in the sense that
    %     $$H^* = \varepsilon \cdot H,$$
    %     then 
    %     $$h_H(u,v) = \varepsilon \cdot \overline{h_H(v,u)}$$
    %     for all $u,v \in M$. 
    % \end{enumerate}

\end{lemma}
\begin{proof}
    Suppose that $M$ is a right $A$-module and hence $h_H$ is defined as 
    $$h_H: M \times M \to A, \quad h_H(u,v) = u^* H v.$$
    We show that $h_H$ is conjugate linear in the first variable and linear in the second --- given $a,b \in A$, we have 
    \begin{align*}
        h_H(ua, vb) = (ua)^* H (vb) = \overline{a} \cdot u^* H v \cdot b = \overline{a} \cdot h_H(u,v) \cdot b.
    \end{align*}
    Moreover, assuming that $H$ is $\varepsilon$-Hermitian in the sense that 
    $$H^* = \overline{\varepsilon} H,$$
    we have 
    $$h_H(v,u) = v^* H u = (u^* H^* v)^*.$$
    By \Cref{lemma:conjugate_transpose_of_product_of_matrices_over_ring_with_involution_is_anti_product_of_conjugate_transposes}, $(u^* H^* v)^* = v^* H u$, and since this is a $1 \times 1$ matrix, it equals its own transpose, so
    $$\overline{u^* H^* v} = \overline{u^* \cdot (\overline{\varepsilon} H) \cdot v}  = \varepsilon \cdot \overline{u^*  H v} = \varepsilon \cdot \overline{h_H(u,v)}$$

    Similarly, if $M$ is a left $A$-module, then $h_H$ is linear in the first variable and conjugate linear in the second. Moreover, assuming that $H$ is $\varepsilon$-Hermitian in the sense that 
    $$H^* = \varepsilon H,$$
    we have 
    $$h_H(v,u) = v^\top H \overline{u} = (u^\top H^* \overline{v})^*.$$
    As this is a $1 \times 1$ matrix, it equals its own transpose 
    $$\overline{u^\top H^* \overline{v}} = \overline{u^\top \cdot (\overline{\varepsilon} H) \cdot \overline{v}}  = \varepsilon \cdot \overline{u^*  H v} = \varepsilon \cdot \overline{h_H(u,v)}$$
\end{proof}

\begin{lemma} \label{lemma:star_conjugate_transpose_congruent_square_matrices_over_a_ring_with_involution_are_varepsilon_hermitian_if_and_only_if_the_other_is}
    Let $(A,\sigma)$ be a \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{ring with involution}.

    Let $H,K \in M_n(A)$ be \CrefAndHyperrefIfExist{definition:star_congruence_of_matrices_over_a_ring_with_involution}{conjugate-transpose congruent}. If $H$ is \CrefAndHyperrefIfExist{definition:epsilon_hermitian_matrix_over_a_ring_with_involution}{$\varepsilon$-Hermitian}, then so is $K$.
\end{lemma}
\begin{proof}

    $$K^* = (P^* H P)^* = P^* H^* P = P^* (\overline{\varepsilon} H) P = \overline{\varepsilon} (P^* H P) = \overline{\varepsilon} K.$$
\end{proof}

\begin{lemma} \label{lemma:scharlau_transfer_of_sesquilinear_forms_on_modules_under_involution_ring_homomorphism_is_indeed_an_additive_map_between_groups_of_sesquilinear_forms}
    Let $\eta: A \to A'$ be a \CrefAndHyperrefIfExist{definition:homomorphism_of_involution_rings}{homomorphism of rings with involution}. 
    Given a (right) $A'$-module $M'$, write $M'_{(A)}$ for the \CrefAndHyperrefIfExist{definition:restriction_of_scalars_of_a_left_or_right_module_along_a_ring_homomorphism_from_the_base_ring}{restriction} of $M'$ as a right $A$-module (via $\eta$). Let $\tau: A' \to A$ be a \CrefAndHyperrefIfExist{definition:homomorphism_of_modules_over_a_ring}{morphism of $A$-bimodules}.

    The Scharlau transfer 
    $$b' \mapsto (\tau(b'): (x,y) \mapsto \tau(b'(x,y)))$$
    is indeed an additive map 
    $$\tau: \Sesq_{A'}(M') \to \Sesq_A(M'_{(A)}).$$
    (\Cref{definition:sesquilinear_form_on_a_module_over_a_ring_with_involution})
\end{lemma}
\begin{proof}
    For all $a_1,a_2 \in A$ and $m,n \in M'$, we have 
    $$\tau(b')(m a_1,n a_2) = \tau(b'(m a_1,n a_2)) = \tau( \overline{a}_1 b'(m, n) a_2) = \overline{a}_1 \tau(b'(m,n)) a_2 = \overline{a}_1 \cdot \tau(b')(m,n) \cdot a_2,$$
    Therefore, $\tau(b')$ is conjugate linear in the first argument and linear in the second. 
\end{proof}

\begin{lemma}[see {\cite[Chapter I, 7.2]{knus_qhfr}}]
    Let $\eta: A \to A'$ be a \CrefAndHyperrefIfExist{definition:homomorphism_of_involution_rings}{homomorphism of rings with involution}. 
    Given a right $A'$-module $M'$, write $M'_{(A)}$ for the \CrefAndHyperrefIfExist{definition:restriction_of_scalars_of_a_left_or_right_module_along_a_ring_homomorphism_from_the_base_ring}{restriction} of $M'$ as a right $A$-module (via $\eta$). Let $\tau: A' \to A$ be a \CrefAndHyperrefIfExist{definition:homomorphism_of_modules_over_a_ring}{morphism of $A$-bimodules} and assume that

\begin{proposition} \label{proposition:finite_etale_algebras_over_commutative_rings_are_symmetric_algebras}

In particular, the relative trace map $\Tr_{A/R}: A \to R$ is a symmetric trace.
\end{proposition}

The map

\begin{proposition} \label{proposition:group_algebra_of_a_finite_group_over_a_commutative_ring_is_a_symmetric_algebra_with_involution}
    Let $R$ be a \CrefAndHyperrefIfExist{definition:commutative_ring}{commutative ring} (regarded as an \CrefAndHyperrefIfExist{definition:involution_on_a_ring}{involution ring} with the \CrefAndHyperrefIfExist{definition:trivial_involution_on_a_ring}{trivial involution}). Let $G$ be a \CrefAndHyperrefIfExist{definition:cardinal_number_and_cardinality_of_a_set}{finite} \CrefAndHyperrefIfExist{definition:group}{group}. 
    The \CrefAndHyperrefIfExist{definition:group_ring_of_a_group_over_a_ring}{group algebra} $R[G]$ is a \CrefAndHyperrefIfExist{definition:symmetric_algebra_over_a_ring_with_involution}{symmetric involution algebra} over $R$ with the \CrefAndHyperrefIfExist{definition:canonical_involution_on_a_group_ring_over_a_ring_with_involution}{canonical involution}; the \CrefAndHyperrefIfExist{definition:augmentation_trace_on_a_group_ring_over_an_involution_ring}{augmentation trace} $\tau: R[G] \to R$ is an \CrefAndHyperrefIfExist{definition:involution_trace_map_for_a_morphism_of_involution_rings}{involution trace}.
\end{proposition}
\begin{proof}
    By \Cref{proposition:group_algebra_of_a_finite_group_over_a_commutative_ring_is_a_symmetric_algebra_with_symmetric_trace_given_by_the_augmentation_trace}, $R[G]$ is a symmetric $R$-algebra with $\tau$ as a \CrefAndHyperrefIfExist{definition:symmetrically_self_dual_ring_over_a_ring}{symmetric trace}. It remains to show that $\tau$ is an involution trace. It is clearly an $R$-bimodule homomorphism, we have 
    $$\tau\left(\overline{\sum_{g \in G} a_g g}\right) = \tau\left(\sum_{g \in G} a_g g^{-1}\right) = a_1 = \tau\left( \sum_{g \in G} a_g g \right),$$
    (recall that $R$ is equipped with the trivial involution) and that $R[G] \times R[G] \to R, \quad (x,y) \mapsto \tau(xy)$ is nonsingular \TODO{}. Thus, $\tau$ is indeed an involution trace.
\end{proof}
